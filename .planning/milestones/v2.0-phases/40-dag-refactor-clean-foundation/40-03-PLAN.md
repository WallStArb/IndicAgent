```python
#!/usr/bin/env python3
"""Create Redpanda topics for DAG pipeline stages."""

from __future__ import annotations

import subprocess
import sys

from src.config.settings import Settings
from src.core.stream_keys import (
    topic_attribution,
    topic_calibrated,
    topic_data_quality,
    topic_quality_gated,
    topic_ranked,
    topic_regime_gated,
    theme_tod_adjusted,
    topic_winner,
)

# 7-day retention in milliseconds
RETENTION_MS = 604800000

# All stage topics
STAGE_TOPICS = [
    ("quality_gated", topic_quality_gated),
    ("regime_gated", topic_regime_gated),
    ("tod_adjusted", topic_tod_adjusted),
    ("calibrated", topic_calibrated),
    ("ranked", topic_ranked),
    ("winner", topic_winner),
    ("attribution", topic_attribution),
    ("data_quality", topic_data_quality),
]


def create_topic(topic_name: str, settings: Settings) -> bool:
    """Create a single Redpanda topic with 7-day retention."""
    cmd = [
        "docker",
        "exec",
        "redpanda",
        "rpk",
        "topic",
        "create",
        topic_name,
        "--set",
        f"retention.ms={RETENTION_MS}",
    ]

    try:
        result = subprocess.run(cmd, capture_output=True, text=True, check=True)
        print(f"✓ Created topic: {topic_name}")
        return True

    except subprocess.CalledProcessError as e:
        if "already exists" in e.stderr:
            print(f"⊙ Topic already exists: {topic_name}")
            return True
        else:
            print(f"✗ Failed to create topic {topic_name}: {e.stderr}")
            return False


def main():
    """Create all stage topics."""
    settings = Settings()

    print("Creating DAG pipeline stage topics...")
    print(f"Environment prefix: {settings.env_prefix}")
    print(f"Retention: {RETENTION_MS}ms (7 days)")
    print()

    success_count = 0
    for name, topic_builder in STAGE_TOPICS:
        topic_name = topic_builder(settings.env_prefix)
        if create_topic(topic_name, settings):
            success_count += 1

    print()
    print(f"Created {success_count}/{len(STAGE_TOPICS)} topics")

    if success_count != len(STAGE_TOPICS):
        sys.exit(1)


if __name__ == "__main__":
    main()
```

Key implementation points:
- Creates all 8 topics with 7-day retention (604800000ms)
- Uses topic builders from stream_keys.py
- Gracefully handles "topic already exists"
- Returns exit code 1 on any failure
- Topic names include env_prefix (development/production)
