    ```python
    from datetime import date, timedelta
    import calendar as cal_mod

    def get_expiry_date(base_symbol: str, expiry_month: int, expiry_year: int) -> date:
        """Approximate expiry date per contract family.
        - Quarterly (equity index, rates, VIX): third Friday of expiry month
        - Monthly energy/metals: last business day of month PRIOR to expiry month
        - Monthly grain: Friday closest to 15th of expiry month
        - Default: 25th of expiry month (conservative)
        """
        if base_symbol in _QUARTERLY_SYMBOLS:
            # Third Friday of expiry month
            first_day = date(expiry_year, expiry_month, 1)
            weekday_of_first = first_day.weekday()  # 0=Mon, 4=Fri
            first_friday = first_day + timedelta(days=(4 - weekday_of_first) % 7)
            return first_friday + timedelta(weeks=2)

        if base_symbol in _ENERGY_METALS_SYMBOLS:
            # Last business day of month PRIOR to expiry month
            prior_month = expiry_month - 1 if expiry_month > 1 else 12
            prior_year = expiry_year if expiry_month > 1 else expiry_year - 1
            last_day = date(prior_year, prior_month, cal_mod.monthrange(prior_year, prior_month)[1])
            # Walk back to weekday
            while last_day.weekday() >= 5:  # Sat=5, Sun=6
                last_day -= timedelta(days=1)
            return last_day

        if base_symbol in _GRAIN_SYMBOLS:
            # Friday closest to 15th of expiry month
            mid = date(expiry_year, expiry_month, 15)
            days_to_friday = (4 - mid.weekday()) % 7
            if days_to_friday > 3:
                days_to_friday -= 7
            return mid + timedelta(days=days_to_friday)

        # Conservative default: 25th of expiry month (per context.md specifics)
        day = min(25, cal_mod.monthrange(expiry_year, expiry_month)[1])
        return date(expiry_year, expiry_month, day)
    ```

    ```python
    def get_roll_window(base_symbol: str, ref_date: date) -> tuple[date, date] | None:
        """Return (roll_start, roll_end) if a roll is upcoming within 21 days, else None.

        Roll window: starts ~10 trading days before expiry, ends ~2 trading days before.
        Uses calendar days as approximation (14 cal days before to 3 cal days before).
        """
        chain = derive_roll_chain(base_symbol)
        if not chain:
            return None

        for entry in chain:
            expiry = get_expiry_date(
                base_symbol, entry["expiry_month"], entry["expiry_year"]
            )
            roll_start = expiry - timedelta(days=14)  # ~10 trading days before
            roll_end = expiry - timedelta(days=3)  # ~2 trading days before

            if roll_start <= ref_date <= roll_end:
                return (roll_start, roll_end)
            # Check if upcoming within 21 days (look ahead)
            if roll_start > ref_date and (roll_start - ref_date).days <= 21:
                return (roll_start, roll_end)

        return None
    ```

    **Step 1 — Create `production/scripts/validate_roll_detection.py`:**

    The script:
    1. Queries `market_data_5m` view (per D-21: 5m bars give cleaner volume signal than 1m for session-level trends) for all active futures symbols over the past 6-12 months.
    2. Defines known historical roll dates per contract family (quarterly rolls happen in Mar/Jun/Sep/Dec; energy/metals roll monthly; grains per their schedule). Use `derive_roll_chain()` to reconstruct expected roll dates.
    3. For each symbol, replays volume bars through the `get_roll_window()` + z-score detection logic (same algorithm as RollMonitor.check_roll but offline — instantiate a RollMonitor, feed historical volume bars, check if it fires during the expected roll windows).
    4. Computes:
       - **Detection rate**: (roll events correctly detected / total known roll events) — must be >= 0.90
       - **False positive rate**: (spurious detections outside any roll window / total non-roll-window bars checked) — must be < 0.10
    5. Prints a clear PASS/FAIL summary with per-symbol breakdown.
    6. Exits with code 0 on PASS, code 1 on FAIL.

    **Step 2 — Script structure:**
    ```python
    #!/usr/bin/env python3
    """Offline validation of calendar + z-score roll detection against historical data.

    Per D-21: accuracy gate >= 90% detection rate, < 10% false positive rate.
    Uses market_data_5m view for cleaner volume signal.

    Usage:
        .venv/bin/python production/scripts/validate_roll_detection.py
    """
    import asyncio
    import sys
    from collections import defaultdict
    from datetime import date, timedelta, timezone

    from src.config.contracts import derive_roll_chain, get_expiry_date, get_roll_window
    from src.config.settings import Settings, get_active_contracts
    from src.core.database_manager import DatabaseManager

    DETECTION_RATE_GATE = 0.90
    FALSE_POSITIVE_RATE_GATE = 0.10
    LOOKBACK_DAYS = 365  # 1 year of history

    async def main():
        settings = Settings()
        db = DatabaseManager(settings.database_url)
        await db.initialize()

        contracts = get_active_contracts()
        futures_symbols = [c.symbol for c in contracts if c.asset_class == "future"]

        results = {}
        for symbol in futures_symbols:
            # Get expected roll dates from derive_roll_chain
            chain = derive_roll_chain(symbol)
            expected_rolls = []
            for entry in chain:
                exp_date = get_expiry_date(symbol, entry["expiry_month"], entry["expiry_year"])
                if exp_date > date.today() - timedelta(days=LOOKBACK_DAYS):
                    expected_rolls.append(exp_date)

            if not expected_rolls:
                continue

            # Query 5m volume bars
            rows = await db.execute_query(
                """
                SELECT timestamp, volume FROM market_data_5m
                WHERE symbol = $1 AND timestamp > NOW() - INTERVAL '%s days'
                ORDER BY timestamp ASC
                """ % LOOKBACK_DAYS,
                symbol,
            )

            if not rows or len(rows) < 100:
                print(f"  {symbol}: insufficient data ({len(rows) if rows else 0} bars), skipping")
                continue

            # Replay through RollMonitor logic
            from collections import deque
            import numpy as np

            window_size = settings.roll_monitor_window_size
            vol_window = deque(maxlen=window_size)
            confirmation_count = 0
            detections = []

            for row in rows:
                ts = row["timestamp"]
                vol = float(row["volume"])
                vol_window.append(vol)

                ref_date = ts.date() if hasattr(ts, 'date') else ts
                roll_win = get_roll_window(symbol, ref_date)

                if roll_win is None or len(vol_window) < 20:
                    confirmation_count = 0
                    continue

                arr = np.array(vol_window)
                mean_v = arr[:-1].mean()
                std_v = arr[:-1].std()
                if std_v < 1e-9:
                    confirmation_count = 0
                    continue

                z = (arr[-1] - mean_v) / std_v
                if z < -2.0:
                    confirmation_count += 1
                else:
                    confirmation_count = 0

                if confirmation_count >= 3:
                    detections.append(ref_date)
                    confirmation_count = 0  # Reset after detection

            # Score: detection rate and false positive rate
            detected_rolls = 0
            for exp_date in expected_rolls:
                roll_start = exp_date - timedelta(days=14)
                roll_end = exp_date - timedelta(days=3)
                if any(roll_start <= d <= roll_end for d in detections):
                    detected_rolls += 1

            detection_rate = detected_rolls / len(expected_rolls) if expected_rolls else 0

            # False positives: detections outside any roll window
            false_positives = 0
            for d in detections:
                in_any_window = False
                for exp_date in expected_rolls:
                    roll_start = exp_date - timedelta(days=14)
                    roll_end = exp_date - timedelta(days=3)
                    if roll_start <= d <= roll_end:
                        in_any_window = True
                        break
                if not in_any_window:
                    false_positives += 1

            fp_rate = false_positives / max(1, len(detections))

            results[symbol] = {
                "expected_rolls": len(expected_rolls),
                "detected": detected_rolls,
                "detection_rate": detection_rate,
                "total_detections": len(detections),
                "false_positives": false_positives,
                "fp_rate": fp_rate,
    ```
