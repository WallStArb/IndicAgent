**Core functions (pure, extractable for testing):**

```python
def generate_rth_timestamps(date_et: date) -> list[datetime]:
    """Generate expected 1m bar timestamps for one RTH session (390 bars).

    All returned timestamps are UTC (converted from ET).
    Skips weekends (returns empty list for Saturday/Sunday).
    """
    if date_et.weekday() >= 5:  # Saturday=5, Sunday=6
        return []
    open_et = datetime(date_et.year, date_et.month, date_et.day,
                       RTH_OPEN_HOUR, RTH_OPEN_MIN, tzinfo=ET)
    close_et = datetime(date_et.year, date_et.month, date_et.day,
                        RTH_CLOSE_HOUR, RTH_CLOSE_MIN, tzinfo=ET)
    result = []
    ts = open_et
    while ts < close_et:
        result.append(ts.astimezone(UTC))
        ts += timedelta(minutes=1)
    return result  # 390 bars per session


def detect_gaps(expected: list[datetime], actual: set[datetime]) -> list[datetime]:
    """Return timestamps present in expected but missing from actual."""
    return sorted([ts for ts in expected if ts not in actual])
```

**Async functions:**

```python
async def get_actual_timestamps(pool, symbol: str, start_utc: datetime, end_utc: datetime) -> set[datetime]:
    """Query market_data_ohlcv for actual 1m bar timestamps in the window."""
    rows = await pool.fetch(
        """SELECT timestamp FROM market_data_ohlcv
           WHERE symbol = $1 AND timeframe = '1m'
           AND timestamp >= $2 AND timestamp < $3""",
        symbol, start_utc, end_utc,
    )
    return {row["timestamp"] for row in rows}


async def insert_bars(pool, bars: list[dict]) -> int:
    """Insert fetched bars with ON CONFLICT DO NOTHING. Returns count inserted."""
    if not bars:
        return 0
    result = await pool.executemany(
        """INSERT INTO market_data_ohlcv (symbol, timeframe, timestamp, open, high, low, close, volume)
           VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
           ON CONFLICT DO NOTHING""",
        [(b["symbol"], "1m", b["timestamp"], b["open"], b["high"], b["low"], b["close"], b["volume"])
         for b in bars],
    )
    return len(bars)  # executemany doesn't return affected rows; count is approximate


async def fill_gaps_for_symbol(pool, symbol: str, lookback_days: int, logger, dry_run: bool = False):
    """Detect and fill gaps for one symbol over the lookback window."""
    now_et = datetime.now(ET)
    total_missing = 0

    for day_offset in range(lookback_days):
        check_date = (now_et - timedelta(days=day_offset + 1)).date()
        expected = generate_rth_timestamps(check_date)
        if not expected:
            continue  # Weekend

        actual = await get_actual_timestamps(pool, symbol, expected[0], expected[-1] + timedelta(minutes=1))
        missing = detect_gaps(expected, actual)

        if missing:
            total_missing += len(missing)
            logger.info("gaps_found", symbol=symbol, date=str(check_date), missing_count=len(missing))

            if not dry_run:
                # Fetch from IBKR and insert
                # Use the existing historical_backfill fetch pattern
                try:
                    bars = await fetch_bars_from_ibkr(symbol, missing, logger)
                    inserted = await insert_bars(pool, bars)
                    GAP_FILL_BARS_FETCHED.labels(symbol=symbol).inc(inserted)
                    logger.info("bars_fetched", symbol=symbol, date=str(check_date), count=inserted)
                except Exception as e:
                    GAP_FILL_FETCH_FAILED.labels(symbol=symbol).inc(1)
                    logger.error("fetch_failed", symbol=symbol, date=str(check_date), error=str(e))

    GAP_FILL_GAPS_DETECTED.labels(symbol=symbol).inc(total_missing)

    if total_missing > CRITICAL_GAP_THRESHOLD:
        logger.critical(
            "systemic_gap_detected",
            symbol=symbol,
            gap_count=total_missing,
            msg=f"CRITICAL: {total_missing} missing bars detected for {symbol} — possible systemic data collection failure",
        )

    return total_missing
```

**main():**
```python
async def run(args):
    settings = Settings()
    logger = setup_service_logging("gap-fill")
    start_http_server(METRICS_PORT)

    pool = await asyncpg.create_pool(settings.database_url)
    symbols = get_active_symbols()

    logger.info("gap_fill_start", symbols=len(symbols), lookback_days=args.lookback_days)

    total_gaps = 0
    for symbol in symbols:
        gaps = await fill_gaps_for_symbol(pool, symbol, args.lookback_days, logger, args.dry_run)
        total_gaps += gaps

    logger.info("gap_fill_complete", total_gaps=total_gaps, symbols_checked=len(symbols))
    await pool.close()


def main():
    parser = argparse.ArgumentParser(description="Gap-fill missing 1m RTH bars")
    parser.add_argument("--lookback-days", type=int, default=5, help="Days to check (default: 5)")
    parser.add_argument("--dry-run", action="store_true", help="Detect gaps only, no fetch")
    args = parser.parse_args()
    asyncio.run(run(args))
```

import pytest
from datetime import date, datetime, timezone, timedelta
from services.gap_fill_service import generate_rth_timestamps, detect_gaps

UTC = timezone.utc

class TestRTHWindowGeneration:
    def test_weekday_produces_390_bars(self):
        """Standard weekday RTH session has exactly 390 1-minute bars."""
        # Monday 2026-03-16
        timestamps = generate_rth_timestamps(date(2026, 3, 16))
        assert len(timestamps) == 390

    def test_first_bar_is_0930_et(self):
        """First bar timestamp corresponds to 09:30 ET in UTC."""
        timestamps = generate_rth_timestamps(date(2026, 3, 16))
        # 2026-03-16 is EDT (UTC-4), so 09:30 ET = 13:30 UTC
        assert timestamps[0].hour == 13
        assert timestamps[0].minute == 30

    def test_last_bar_is_1559_et(self):
        """Last bar timestamp corresponds to 15:59 ET in UTC."""
        timestamps = generate_rth_timestamps(date(2026, 3, 16))
        assert timestamps[-1].hour == 19
        assert timestamps[-1].minute == 59

    def test_saturday_returns_empty(self):
        """Saturday produces no RTH timestamps."""
        timestamps = generate_rth_timestamps(date(2026, 3, 14))  # Saturday
        assert timestamps == []

    def test_sunday_returns_empty(self):
        """Sunday produces no RTH timestamps."""
        timestamps = generate_rth_timestamps(date(2026, 3, 15))  # Sunday
        assert timestamps == []

    def test_all_timestamps_are_utc(self):
        """All returned timestamps are timezone-aware UTC."""
        timestamps = generate_rth_timestamps(date(2026, 3, 16))
        for ts in timestamps:
            assert ts.tzinfo is not None
            assert ts.tzinfo == UTC or ts.utcoffset() == timedelta(0)


class TestGapDetection:
    def test_no_gaps_returns_empty(self):
        """When all expected bars exist, no gaps detected."""
        expected = generate_rth_timestamps(date(2026, 3, 16))
        actual = set(expected)
        assert detect_gaps(expected, actual) == []

    def test_missing_bars_detected(self):
        """Missing bars are correctly identified."""
        expected = generate_rth_timestamps(date(2026, 3, 16))
        actual = set(expected[5:])  # Remove first 5 bars
        gaps = detect_gaps(expected, actual)
        assert len(gaps) == 5
        assert gaps == expected[:5]

    def test_all_missing_returns_full_list(self):
        """When no actual bars exist, all expected are gaps."""
        expected = generate_rth_timestamps(date(2026, 3, 16))
        gaps = detect_gaps(expected, set())
        assert len(gaps) == 390

Create `production/systemd/indicagent-gap-fill.timer`:

```ini
[Unit]
Description=Daily gap-fill for missing 1m RTH bars
After=network-online.target

[Timer]
# 09:20 ET = 13:20 UTC (EDT) or 14:20 UTC (EST)
# Use America/New_York timezone for automatic DST handling
OnCalendar=*-*-* 13:20:00 UTC
Persistent=true

[Install]
WantedBy=timers.target
```

Note: `Persistent=true` means if the system was off at the scheduled time, the timer fires on next boot. Using UTC time (13:20 = 09:20 EDT). During EST (winter), this runs at 08:20 ET — still before market open. If exact ET timing is critical, the service itself can check and skip non-trading-day runs.
