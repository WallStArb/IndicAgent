from datetime import datetime
from pathlib import Path

import pandas as pd
from src.config.settings import Settings

)
from src.intelligence.confluence.cross_tf_sr_confluence import (
    CrossTFSRConfluencePlugin,
)
from tools.backtest_i6_plugin import backtest_i6_plugin
from tools.validate_i6_backtest import validate_backtest_results


async def backtest_all_cross_tf_plugins(
    start_date: datetime,
    end_date: datetime,
    output_dir: str | Path = "/tmp",
) -> dict[str, pd.DataFrame]:
    """Backtest all 5 cross-TF plugins.

    Args:
        start_date: Backtest start date
        end_date: Backtest end date
        output_dir: Directory for CSV outputs

    Returns:
        Dict mapping plugin name -> backtest DataFrame
    """
    plugins = {
        "CrossTFMomentumDivergence": (CrossTFMomentumDivergencePlugin, "ctf_momentum_divergence"),
        "CrossTFSRConfluence": (CrossTFSRConfluencePlugin, "ctf_sr_confluence"),
        "CrossTFRegimeAgreement": (CrossTFRegimeAgreementPlugin, "ctf_regime_agreement"),
        "CrossTFSqueezeExpansion": (CrossTFSqueezeExpansionPlugin, "ctf_squeeze_expansion"),
        "CrossTFOrderFlowAlignment": (CrossTFOrderFlowAlignmentPlugin, "ctf_orderflow_alignment"),
    }

    results = {}
    output_path = Path(output_dir)

    for name, (plugin_class, field_name) in plugins.items():
        print(f"\n{'='*60}")
        print(f"Backtesting {name}...")
        print(f"{'='*60}")

        try:
            df = backtest_i6_plugin(
                plugin_class=plugin_class,
                start_date=start_date,
                end_date=end_date,
            )

            # Save to CSV
            csv_file = output_path / f"{name.lower()}_backtest.csv"
            df.to_csv(csv_file, index=False)
            print(f"Saved: {csv_file}")
            print(f"Rows: {len(df)}")

            # Validate
            validation = validate_backtest_results(df, field_name)
            print(f"IC = {validation.ic:.4f}, p = {validation.p_value:.4f}, N = {validation.n}")
            print(f"Status: {'PASS' if validation.passed else 'FAIL'}")

            results[name] = {
                "df": df,
                "validation": validation,
            }

        except Exception as e:
            print(f"ERROR backtesting {name}: {e}")
            continue

    return results


async def main():
    """Main entry point."""
    import asyncio

    settings = Settings()

    # Backtest period: 6 months
    start_date = datetime(2025, 10, 1)
    end_date = datetime(2026, 4, 1)

    print("Phase 64: Cross-TF Plugin Backtest Validation")
    print(f"Period: {start_date.date()} to {end_date.date()}")

    # Run backtests
    results = await backtest_all_cross_tf_plugins(start_date, end_date)

    # Summary
    print("\n" + "="*60)
    print("SUMMARY")
    print("="*60)
    passed = sum(1 for r in results.values() if r["validation"].passed)
    total = len(results)
    print(f"Passed: {passed}/{total}")

    # Feature selection
    print("\nFeature Selection:")
    for name, res in results.items():
        ic = res["validation"].ic
        if ic > 0.05:
            decision = "KEEP → Deploy to shadow mode"
        elif ic > 0.02:
            decision = "TWEAK → Adjust parameters, re-test"
        else:
            decision = "KILL → No predictive value"
        print(f"  {name}: {decision} (IC={ic:.4f})")

    return 0 if passed == total else 1


if __name__ == "__main__":
    import sys
    sys.exit(asyncio.run(main()))

"""Backtest macro factors (yield curve, flight-to-quality) on historical data."""

from datetime import datetime
from pathlib import Path

import asyncpg
import pandas as pd
from src.config.settings import Settings
from src.intelligence.macro.flight_to_quality import compute_flight_to_quality
from src.intelligence.macro.yield_curve import compute_yield_curve_slope
from tools.validate_i6_backtest import validate_backtest_results


async def backtest_yield_curve(
    start_date: datetime,
    end_date: datetime,
) -> pd.DataFrame:
    """Backtest yield curve factor on historical rate futures data."""
    settings = Settings()

    conn = await asyncpg.connect(settings.database_url)

    try:
        # Load rate futures bars from market_data_ohlcv
        rows = await conn.fetch("""
            SELECT ts, symbol, tf, open, high, low, close, volume
            FROM market_data_ohlcv
            WHERE ts BETWEEN $1 AND $2
              AND symbol IN ('ZT', 'ZN', 'ZB', 'ZF')
            ORDER BY ts, symbol
        """, start_date, end_date)
    finally:
        await conn.close()

    if not rows:
        print(f"No data found for {start_date} to {end_date}")
        return pd.DataFrame()

    # Convert to DataFrame
    df = pd.DataFrame(rows, columns=["ts", "symbol", "tf", "open", "high", "low", "close", "volume"])

    # Group by timestamp, build windows per symbol
    results = []
    window_bars = 10

    # Sliding window backtest (FIXED: use only past data, no look-ahead)
    for i in range(window_bars, len(df)):
        # FIXED: Slice only past data to match lookback parameter
        window_df = df.iloc[i - window_bars : i]

        # Build bars dict by symbol
        bars = {}
        for symbol, group in window_df.groupby("symbol"):
            from collections import deque
            records = group.to_dict("records")
            bars[symbol] = deque(records, maxlen=window_bars + 1)

        if not bars or len(bars.get("ZT", [])) < window_bars:
            continue

        # Compute yield curve
        yc_result = compute_yield_curve_slope(bars, lookback=window_bars)

        # FIXED: Get timestamp from end of window (most recent bar)
        ts = window_df.iloc[-1]["ts"]

        results.append({
            "ts": ts,
            "yield_curve_slope": yc_result["yield_curve_slope"],
            "yield_curve_regime": yc_result["yield_curve_regime"],
        })

    result_df = pd.DataFrame(results)

    # Join with signal_ledger for pnl_r outcomes
    conn = await asyncpg.connect(settings.database_url)
    try:
        ledger_rows = await conn.fetch("""
            SELECT
                sl.feature_ts,
                sl.feature_tf,
                sl.pnl_r,
                ic.hmm_regime
            FROM signal_ledger sl
            JOIN intelligence_features ic ON (
                sl.symbol = ic.symbol AND
                sl.feature_ts = ic.ts AND
                sl.feature_tf = ic.tf
            )
            WHERE sl.feature_ts BETWEEN $1 AND $2
              AND sl.exit_at IS NOT NULL
            ORDER BY sl.feature_ts
        """, start_date, end_date)
    finally:
        await conn.close()

    if ledger_rows:
        ledger_df = pd.DataFrame(ledger_rows, columns=["feature_ts", "feature_tf", "pnl_r", "hmm_regime"])
        # Merge on timestamp (approximate join)
        result_df["ts"] = pd.to_datetime(result_df["ts"])
        ledger_df["feature_ts"] = pd.to_datetime(ledger_df["feature_ts"])

        # Find closest match within 1 minute
        merged = pd.merge_asof(
            result_df.sort_values("ts"),
            ledger_df.sort_values("feature_ts"),
            left_on="ts",
            right_on="feature_ts",
            direction="nearest",
            tolerance=pd.Timedelta("1min"),
        )
        return merged.dropna(subset=["pnl_r"])

    return result_df
