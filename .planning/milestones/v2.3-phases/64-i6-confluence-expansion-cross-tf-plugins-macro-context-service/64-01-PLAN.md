```python
"""Cross-timeframe momentum divergence plugin.

Computes continuous gradient [-1, +1] measuring momentum divergence
between higher (HTF) and lower (LTF) timeframes.

Outputs:
    ctf_momentum_divergence: float [-1, +1]
        Positive: HTF and LTF momentum aligned (both bullish)
        Negative: HTF and LTF momentum opposed (divergence)
        Near 0: Weak or conflicting signals
    ctf_momentum_regime: str
        aligned_htf_bull: HTF bullish, LTF bullish
        aligned_htf_bear: HTF bearish, LTF bearish
        pullback_bull: HTF bullish, LTF bearish (bullish pullback)
        pullback_bear: HTF bearish, LTF bullish (bearish pullback)
        mixed: Conflicting signals
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from ..plugins import InputSpec
from ..utils.gradient_utils import linear_ramp, sigmoid_score, z_score_to_score
from .confluence_weights import _TF_MINUTES, extract_trend_sign, get_recency_weight


@dataclass
class CrossTFMomentumDivergencePlugin:
    """Cross-TF momentum divergence plugin.

    Reads momentum indicators (RSI, MACD) from cached cross-TF intelligence
    and computes a continuous gradient measuring alignment/divergence.
    """

    name: str = "i6_CrossTFMomentumDivergence"
    outputs: frozenset[str] = frozenset({
        "ctf_momentum_divergence",
        "ctf_momentum_regime",
    })
    min_lookback: int = 1
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"confluence", "cross_tf"})
    inputs: list[InputSpec] = ()
    _state: dict = field(default_factory=dict)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """Compute momentum divergence across timeframes."""
        features = frames.get("features") or {}
        current_tf = frames.get("timeframe", "")

        # Collect available cross-TF intelligence dicts
        other_intel: dict[str, dict[str, Any]] = {}
        for key, val in frames.items():
            if key.startswith("intel_") and isinstance(val, dict):
                tf = key[6:]  # "intel_5m" → "5m"
                other_intel[tf] = val

        if not other_intel:
            return {
                "ctf_momentum_divergence": 0.0,
                "ctf_momentum_regime": "mixed",
            }

        # Determine HTF/LTF relationship
        current_tf_min = _TF_MINUTES.get(current_tf, 1)
        htf_intel: dict[str, Any] = {}
        ltf_intel: dict[str, Any] = {}

        # Separate intelligence into HTF and LTF groups
        for tf, intel in other_intel.items():
            tf_min = _TF_MINUTES.get(tf, 1)
            if tf_min > current_tf_min:
                htf_intel[tf] = intel
            elif tf_min < current_tf_min:
                ltf_intel[tf] = intel

        # If no HTF/LTF data available, return neutral
        if not htf_intel or not ltf_intel:
            return {
                "ctf_momentum_divergence": 0.0,
                "ctf_momentum_regime": "mixed",
            }

        # Compute weighted momentum scores for HTF and LTF
        htf_momentum = self._compute_momentum_score(htf_intel, frames, current_tf)
        ltf_momentum = self._compute_momentum_score(ltf_intel, frames, current_tf)

        # Compute divergence: 1.0 = aligned, -1.0 = opposed
        # Use product to preserve sign: (+1 * +1) = +1 (aligned bullish)
        #                              (+1 * -1) = -1 (divergence)
        divergence = htf_momentum * ltf_momentum

        # Classify regime based on HTF/LTF combination
        regime = self._classify_regime(htf_momentum, ltf_momentum)

        return {
            "ctf_momentum_divergence": round(divergence, 4),
            "ctf_momentum_regime": regime,
        }

    def _compute_momentum_score(
        self, intel_group: dict[str, dict[str, Any]], frames: dict[str, Any], current_tf: str
    ) -> float:
        """Compute weighted momentum score from a group of timeframes."""
        total_weight = 0.0
        weighted_score = 0.0

        for tf, intel in intel_group.items():
            # Recency weight: stale intel contributes less
            weight = get_recency_weight(frames, tf)
            if weight <= 0:
                continue

            # Extract momentum from RSI and MACD if available
            rsi = intel.get("rsi")
            macd_hist = intel.get("macd_histogram")

            # Convert RSI to momentum score [-1, +1]
            # RSI > 50 = bullish, RSI < 50 = bearish
            if rsi is not None:
                rsi_score = linear_ramp(rsi, 30, 70, -1.0, 1.0)
                momentum = rsi_score
            elif macd_hist is not None:
                # MACD histogram: positive = bullish, negative = bearish
                # Normalize via z-score for continuous gradient
                momentum = z_score_to_score(macd_hist, sigma_scale=2.0)
                momentum = linear_ramp(momentum, -1.0, 1.0, -1.0, 1.0)
            else:
                # Fall back to trend sign
                momentum = float(extract_trend_sign(intel))

            weighted_score += weight * momentum
            total_weight += weight

        if total_weight == 0:
            return 0.0

        return weighted_score / total_weight

    def _classify_regime(self, htf_momentum: float, ltf_momentum: float) -> str:
        """Classify regime based on HTF/LTF momentum combination."""
        # Thresholds for regime classification
        threshold = 0.3  # Require clear signal strength

        htf_bull = htf_momentum > threshold
        htf_bear = htf_momentum < -threshold
        ltf_bull = ltf_momentum > threshold
        ltf_bear = ltf_momentum < -threshold

        if htf_bull and ltf_bull:
            return "aligned_htf_bull"
        elif htf_bear and ltf_bear:
            return "aligned_htf_bear"
        elif htf_bull and ltf_bear:
            return "pullback_bull"  # HTF bullish, LTF bearish = bullish pullback
        elif htf_bear and ltf_bull:
            return "pullback_bear"  # HTF bearish, LTF bullish = bearish pullback
        else:
            return "mixed"

    def compute_next(self, windows: dict[str, Any]) -> dict[str, Any]:
        return self.compute_full(windows)
```

Key implementation points:
- Uses gradient_utils functions: linear_ramp(), z_score_to_score()
- Reads from frames["intel_*"] for cross-TF data
- Computes weighted momentum score per TF group (HTF/LTF)
- Returns continuous gradient [-1, +1] for ctf_momentum_divergence
- Categorical regime label for ML segmentation
- Degrades gracefully when data missing

"""Unit tests for CrossTFMomentumDivergence plugin."""

import pytest

from src.intelligence.confluence.cross_tf_momentum_divergence import (
    CrossTFMomentumDivergencePlugin,
)


class TestCrossTFMomentumDivergencePlugin:
    """Test cross-TF momentum divergence computation."""

    def test_plugin_metadata(self):
        """Plugin has correct metadata."""
        plugin = CrossTFMomentumDivergencePlugin()
        assert plugin.name == "i6_CrossTFMomentumDivergence"
        assert "ctf_momentum_divergence" in plugin.outputs
        assert "ctf_momentum_regime" in plugin.outputs
        assert plugin.min_lookback == 1
        assert not plugin.supports_incremental

    def test_aligned_bullish_momentum(self):
        """HTF and LTF both bullish → aligned_htf_bull regime."""
        plugin = CrossTFMomentumDivergencePlugin()
        frames = {
            "features": {"rsi": 65},  # Current TF bullish
            "timeframe": "5m",
            "intel_1h": {
                "rsi": 70,  # HTF bullish
                "macd_histogram": 0.5,
            },
            "intel_1m": {
                "rsi": 60,  # LTF bullish
                "macd_histogram": 0.3,
            },
        }

        result = plugin.compute_full(frames)

        assert result["ctf_momentum_divergence"] > 0.5  # Strong alignment
        assert result["ctf_momentum_regime"] == "aligned_htf_bull"

    def test_aligned_bearish_momentum(self):
        """HTF and LTF both bearish → aligned_htf_bear regime."""
