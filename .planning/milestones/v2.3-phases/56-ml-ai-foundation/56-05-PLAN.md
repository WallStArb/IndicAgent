**Goal:** Replace `services/ai_narrative_service.py` (1,327 lines) with a thin `ai_narrative_agent.py` (~200 lines) that delegates all narrative logic to `src/intelligence/narrative/` (Plan 56-02). Uses `LLMProviderChain` from `src/core/llm/` (Plan 56-01).

**Architecture:** Agent is Kafka I/O + lifecycle only. It creates `LLMProviderChain(call_type="narrative")` and `NarrativeOrchestrator`, subscribes to `topic_intelligence_record()`, calls `orchestrator.generate()`, and publishes the result. All prompt logic, parsing, and synthesis live in the narrative module.

**Tech Stack:** Python 3.11, asyncio, `KafkaConsumerClient`/`KafkaProducerClient`, structlog, `src/core/agent/base.BaseAgent`, `LLMProviderChain`, `NarrativeOrchestrator`

```python
"""AINarrativeComputeAgent — thin Kafka wrapper around NarrativeOrchestrator.

Subscribes to intelligence record topic, generates LLM narrative per signal,
publishes to narratives topic. All narrative logic is in src/intelligence/narrative/.
"""
from __future__ import annotations

import asyncio
import signal
from typing import Any

import structlog

from src.config.settings import Settings
from src.core.agent.base import BaseAgent
from src.core.kafka_utils import KafkaConsumerClient, KafkaProducerClient
from src.core.llm.chain import LLMProviderChain
from src.core.service_utils import setup_service_logging
from src.core.stream_keys import topic_intelligence_record, topic_narratives
from src.intelligence.narrative.orchestrator import NarrativeOrchestrator

logger = structlog.get_logger(__name__)


class AINarrativeComputeAgent(BaseAgent):
    """Thin agent: consume BarIntelligenceRecord → generate narrative → publish."""

    def __init__(self, settings: Settings) -> None:
        super().__init__()
        self._settings = settings
        chain = LLMProviderChain(call_type="narrative", settings=settings)
        self._orchestrator = NarrativeOrchestrator(chain=chain, max_tokens=200, timeout=30.0)
        self._consumer = KafkaConsumerClient(
            topics=[topic_intelligence_record(settings.env_name)],
            group_id="ai_narrative_consumer",
            bootstrap_servers=settings.kafka_bootstrap_servers,
        )
        self._producer = KafkaProducerClient(
            bootstrap_servers=settings.kafka_bootstrap_servers,
        )

    async def run(self) -> None:
        """Main loop: consume records, generate narratives, publish."""
        logger.info("ai_narrative_agent.starting")
        async for message in self._consumer.consume():
            try:
                record = message.value  # already deserialized BarIntelligenceRecord
                await self._process_bar(record)
            except Exception as exc:
                logger.exception("ai_narrative_agent.consume_error", error=str(exc))

    async def _process_bar(self, record: Any) -> None:
        """Generate and publish a narrative for one BarIntelligenceRecord."""
        try:
            narrative = await self._orchestrator.generate(record)
            if narrative is None:
                return

            intel = record.intelligence
            topic = topic_narratives(self._settings.env_name)
            await self._producer.publish(
                topic,
                {
                    "symbol": intel.symbol,
                    "tf": intel.tf,
                    "ts": intel.ts.isoformat(),
                    "narrative": narrative,
                    "record_id": getattr(record, "record_id", None),
                },
            )
        except Exception as exc:
            logger.exception("ai_narrative_agent.process_error", error=str(exc))


def main() -> None:
    setup_service_logging("logs/ai_narrative_agent.log")
    settings = Settings()
    agent = AINarrativeComputeAgent(settings)

    loop = asyncio.get_event_loop()

    def _sigterm(_sig, _frame):
        logger.info("ai_narrative_agent.sigterm_received")
        loop.stop()

    signal.signal(signal.SIGTERM, _sigterm)
    signal.signal(signal.SIGINT, _sigterm)

    try:
        loop.run_until_complete(agent.run())
    finally:
        loop.close()


if __name__ == "__main__":
    main()
```
