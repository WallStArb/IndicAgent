    ```python
    """CI gate: assert zero binary scoring patterns remain in plugin source.

    Imports the scanner module directly (no subprocess). If ANY non-exempt
    binary violation is found, the test fails with file:line details.
    """
    import os
    import pytest
    from tools.scan_binary_patterns import scan_all_plugins, format_violations


    def test_zero_binary_violations():
        """Assert the binary pattern scanner finds zero non-exempt violations."""
        root = os.path.join(os.path.dirname(__file__), "..", "..", "..", "src", "intelligence")
        root = os.path.normpath(root)
        violations = scan_all_plugins(root)
        # Filter to non-exempt only
        real_violations = [v for v in violations if not v.get("is_exempt", False)]
        if real_violations:
            summary = format_violations(real_violations, verbose=True)
            pytest.fail(
                f"Binary pattern scanner found {len(real_violations)} violation(s):\n{summary}"
            )


    def test_scanner_runs_without_error():
        """Smoke test: scanner processes all plugin files without crashing."""
        root = os.path.join(os.path.dirname(__file__), "..", "..", "..", "src", "intelligence")
        root = os.path.normpath(root)
        violations = scan_all_plugins(root)
        assert isinstance(violations, list)
        # Pre-fix baseline was ~50; post-fix should be 0 non-exempt
        # (exempt violations are expected -- direction encoders, categorical flags, etc.)
    ```
