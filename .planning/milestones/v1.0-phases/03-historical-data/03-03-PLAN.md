<task type="auto">
  <name>Task 1: Run SQL validation audit against both tables</name>
  <files></files>
  <action>
    Run the following SQL validation queries against the live indicagent database. Execute each block and capture the output. No writes are performed — all queries are SELECT only (except the orphan check which uses NOT EXISTS).

    ```bash
    psql postgresql://postgres:postgres@localhost:5432/indicagent <<'SQL'

    -- === BLOCK 1: signal_ledger row counts ===
    SELECT 'signal_ledger total' AS check_name, count(*) AS result
    FROM signal_ledger;

    SELECT 'signal_ledger with feature_ts set' AS check_name, count(*) AS result
    FROM signal_ledger
    WHERE feature_ts IS NOT NULL;

    SELECT 'signal_ledger by timeframe' AS check_name, timeframe, count(*) AS signals
    FROM signal_ledger
    GROUP BY timeframe
    ORDER BY timeframe;

    SELECT 'signal_ledger date range' AS check_name,
           min(timestamp)::date AS from_dt,
           max(timestamp)::date AS to_dt,
           (max(timestamp)::date - min(timestamp)::date) AS span_days
    FROM signal_ledger;

    -- === BLOCK 2: intelligence_features row counts ===
    SELECT 'intelligence_features total (backfill)' AS check_name, count(*) AS result
    FROM intelligence_features
    WHERE source = 'backfill';

    SELECT 'intelligence_features by symbol and tf' AS check_name,
           symbol, tf, count(*) AS feature_rows,
           min(ts)::date AS from_dt, max(ts)::date AS to_dt
    FROM intelligence_features
    WHERE source = 'backfill'
    GROUP BY symbol, tf
    ORDER BY symbol, tf;

    -- === BLOCK 3: JOIN integrity — orphaned signals ===
    SELECT 'orphaned_signals (must be 0)' AS check_name, count(*) AS result
    FROM signal_ledger sl
    WHERE sl.feature_ts IS NOT NULL
      AND NOT EXISTS (
          SELECT 1 FROM intelligence_features inf
          WHERE inf.ts = sl.feature_ts
            AND inf.symbol = sl.symbol
            AND inf.tf = sl.feature_tf
      );

    -- === BLOCK 4: JSONB structure validation (sample rows) ===
    SELECT 'intelligence_features JSONB sample' AS check_name,
           ts, symbol, tf, source,
           (bar->>'c')::float AS close_price,
           (bar->>'o')::float AS open_price,
           (bar->>'v')::int   AS volume,
           (i1->>'rsi_14')::float AS rsi_14,
           jsonb_typeof(i3) AS i3_type,
           jsonb_typeof(i4) AS i4_type,
           jsonb_typeof(i1) AS i1_type
    FROM intelligence_features
    WHERE source = 'backfill'
    ORDER BY ts DESC
    LIMIT 5;

    -- === BLOCK 5: Empty JSONB column check ===
    SELECT 'rows with empty bar JSONB (anomaly)' AS check_name, count(*) AS result
    FROM intelligence_features
    WHERE source = 'backfill'
      AND bar = '{}'::jsonb;

    SELECT 'rows with empty i1 JSONB (anomaly)' AS check_name, count(*) AS result
    FROM intelligence_features
    WHERE source = 'backfill'
      AND i1 = '{}'::jsonb;

    -- === BLOCK 6: Date coverage comparison ===
    SELECT 'date coverage gap check' AS check_name,
           sl_range.from_dt AS sl_from,
           sl_range.to_dt   AS sl_to,
           inf_range.from_dt AS inf_from,
           inf_range.to_dt   AS inf_to
    FROM
      (SELECT min(timestamp)::date AS from_dt, max(timestamp)::date AS to_dt FROM signal_ledger) AS sl_range,
      (SELECT min(ts)::date AS from_dt, max(ts)::date AS to_dt FROM intelligence_features WHERE source = 'backfill') AS inf_range;

    SQL
    ```

    Capture all output. After running, evaluate each block against the expected thresholds defined in the verify section below.
  </action>
