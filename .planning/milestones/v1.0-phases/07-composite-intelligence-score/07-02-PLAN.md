# dt_db_pattern direction * dt_db_confidence (0.40) +
# hs_pattern direction * hs_confidence (0.30) +
# tri_breakout_bias * tri_confidence (0.20) +

# CIS = sum(weights[b] * bucket_scores[b] for b in BUCKET_NAMES)

    ```python
    # src/intelligence/trading/cis_scorer.py
    from __future__ import annotations
    from dataclasses import dataclass
    from typing import Any

    BUCKET_NAMES = ("trend", "momentum", "structure", "pattern", "institutional", "regime")

    BOOTSTRAP_WEIGHTS: dict[str, float] = {
        "trend": 0.20, "momentum": 0.20, "structure": 0.15,
        "pattern": 0.05, "institutional": 0.25, "regime": 0.15,
    }

    @dataclass
    class CISResult:
        cis_score: float          # clamp(sum(w*s), -1.0, 1.0)
        direction: int            # -1, 0, +1
        bucket_scores: dict[str, float]
        weights_version: int      # 0 = bootstrap
        buckets_agreeing: int

    class CISScorer:
        CIS_THRESHOLD = 0.35
        AGREE_THRESHOLD = 3
        BUCKET_SCORE_NOISE = 0.1  # minimum absolute bucket score to count as "agreeing"

        def __init__(self, weights: dict[str, float] | None = None, weights_version: int = 0):
            self._weights = weights or BOOTSTRAP_WEIGHTS
            self._weights_version = weights_version

        def score(self, features: dict[str, Any], plugin_outputs: dict[str, dict]) -> CISResult:
            bucket_scores = {
                "trend": self._trend(features),
                "momentum": self._momentum(features, plugin_outputs),
                "structure": self._structure(features, plugin_outputs),
                "pattern": self._pattern(features, plugin_outputs),
                "institutional": self._institutional(features, plugin_outputs),
                "regime": self._regime(features, plugin_outputs),
            }
            cis_raw = sum(self._weights[b] * bucket_scores[b] for b in BUCKET_NAMES)
            cis_score = max(-1.0, min(1.0, cis_raw))
            direction = 0
            if abs(cis_score) > self.CIS_THRESHOLD:
                direction = 1 if cis_score > 0 else -1
            agreeing = sum(
                1 for b in BUCKET_NAMES
                if bucket_scores[b] * (cis_score if cis_score != 0 else 1) > self.BUCKET_SCORE_NOISE
            )
            if agreeing < self.AGREE_THRESHOLD:
                direction = 0
            return CISResult(
                cis_score=round(cis_score, 4),
                direction=direction,
                bucket_scores={k: round(v, 4) for k, v in bucket_scores.items()},
                weights_version=self._weights_version,
                buckets_agreeing=agreeing,
            )

        def _fval(self, features: dict, key: str, default: float = 0.0) -> float:
            v = features.get(key)
            if v is None: return default
            try: return float(v)
            except (TypeError, ValueError): return default

        def _plug(self, plugin_outputs: dict, name: str) -> tuple[int, float]:
            """Get (direction, confidence) from plugin output dict."""
            sig = plugin_outputs.get(name, {})
            d = sig.get("direction", 0)
            c = sig.get("confidence", 0.0)
            if not isinstance(d, (int, float)): d = 0
            if not isinstance(c, (int, float)): c = 0.0
            return int(d), float(c)

        def _trend(self, f: dict) -> float:
            score = (
                0.35 * max(-1.0, min(1.0, self._fval(f, "trend_regime")))
                + 0.20 * (1.0 if self._fval(f, "kalman_slope") > 0 else (-1.0 if self._fval(f, "kalman_slope") < 0 else 0.0))
                + 0.25 * max(-1.0, min(1.0, self._fval(f, "smc_trend_direction")))
                + 0.10 * max(-1.0, min(1.0, self._fval(f, "ctf_trend_alignment")))
                + 0.10 * max(-1.0, min(1.0, self._fval(f, "trend_confluence_score")))
            )
            return max(-1.0, min(1.0, score))

        def _momentum(self, f: dict, po: dict) -> float:
            rsi = self._fval(f, "rsi_14")
            rsi_dir = (rsi - 50.0) / 50.0  # maps [0,100] → [-1,+1]
            macd = self._fval(f, "macd_histogram_12_26_9")
            macd_dir = 1.0 if macd > 0 else (-1.0 if macd < 0 else 0.0)
            roc = self._fval(f, "roc_14")
            roc_dir = 1.0 if roc > 0 else (-1.0 if roc < 0 else 0.0)
            d, c = self._plug(po, "trad_DivergenceStack")
            return max(-1.0, min(1.0,
                0.30 * rsi_dir + 0.25 * macd_dir + 0.20 * roc_dir
                + 0.15 * max(-1.0, min(1.0, self._fval(f, "momentum_bias")))
                + 0.10 * d * c
            ))

        def _structure(self, f: dict, po: dict) -> float:
            d, c = self._plug(po, "trad_CHoCHReversal")
            bos = self._fval(f, "bos_detected") * self._fval(f, "bos_direction")
            choch = self._fval(f, "choch_detected") * self._fval(f, "choch_direction")
            return max(-1.0, min(1.0,
                0.30 * max(-1.0, min(1.0, self._fval(f, "swing_pattern")))
                + 0.25 * max(-1.0, min(1.0, bos))
                + 0.25 * max(-1.0, min(1.0, choch))
                + 0.20 * d * c
            ))

        def _pattern(self, f: dict, po: dict) -> float:
            dt_dir = -1.0 if self._fval(f, "dt_db_pattern") == 1.0 else (1.0 if self._fval(f, "dt_db_pattern") == 2.0 else 0.0)
            hs_dir = -1.0 if self._fval(f, "hs_pattern") == 1.0 else (1.0 if self._fval(f, "hs_pattern") == 2.0 else 0.0)
            d, c = self._plug(po, "trad_PatternCompletion")
            return max(-1.0, min(1.0,
                0.40 * dt_dir * self._fval(f, "dt_db_confidence")
                + 0.30 * hs_dir * self._fval(f, "hs_confidence")
                + 0.20 * self._fval(f, "tri_breakout_bias") * self._fval(f, "tri_confidence")
                + 0.10 * d * c
            ))

        def _institutional(self, f: dict, po: dict) -> float:
            zone = self._fval(f, "in_demand_zone") - self._fval(f, "in_supply_zone")
            fvg_t = self._fval(f, "fvg_type")
            fvg_active = 1.0 if self._fval(f, "fvg_open_count") > 0 else 0.0
            fd, fc = self._plug(po, "trad_FVGFill")
            sd_d, sd_c = self._plug(po, "trad_SupplyDemandSetup")
            return max(-1.0, min(1.0,
                0.25 * max(-1.0, min(1.0, self._fval(f, "ob_type") * self._fval(f, "ob_strength")))
                + 0.15 * max(-1.0, min(1.0, fvg_t * fvg_active))
                + 0.20 * max(-1.0, min(1.0, zone))
                + 0.20 * fd * fc
                + 0.20 * sd_d * sd_c
            ))

        def _regime(self, f: dict, po: dict) -> float:
            hmm_dir = self._fval(f, "hmm_prob_trending_up") - self._fval(f, "hmm_prob_trending_down")
            cp = self._fval(f, "cp_probability")
            cp_dir = 1.0 if cp > 0.5 else 0.0  # changepoint = directional uncertainty → neutral-ish
            d, c = self._plug(po, "trad_RegimeTransition")
            return max(-1.0, min(1.0,
                0.35 * max(-1.0, min(1.0, hmm_dir))
                + 0.15 * cp_dir
                + 0.20 * max(-1.0, min(1.0, self._fval(f, "ctf_regime_agreement")))
                + 0.20 * max(-1.0, min(1.0, self._fval(f, "vol_regime") * -1.0))  # high vol = bearish for regime
                + 0.10 * d * c
            ))
    ```
