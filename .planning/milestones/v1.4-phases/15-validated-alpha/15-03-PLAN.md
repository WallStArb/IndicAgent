```python
# pp = df.iloc[-3]; extract pp_o, pp_h, pp_l, pp_c
# Helper: body_size = abs(close - open), range_size = high - low

# Three White Soldiers (bullish, output: "three_white_soldiers")
# 3 consecutive bullish, each opens within prior body, each closes near high
# pp_c > pp_o AND p_c > p_o AND c_c > c_o
# each body > 0.5 * range (strong bars, not dojis)
# p_o between pp_o and pp_c (opens within prior body)
# c_o between p_o and p_c (opens within prior body)
# each upper wick < 0.25 * body (closes near high)

# Three Black Crows (bearish, output: "three_black_crows") — mirror
# pp_c < pp_o AND p_c < p_o AND c_c < c_o
# each body > 0.5 * range
# p_o between pp_c and pp_o
# c_o between p_c and p_o
# each lower wick < 0.25 * body

# Morning Star (bullish, output: "morning_star")
# pp = large bearish: body > 0.6 * range
# p = small body (star): body < 0.3 * range (indecision)
# c = large bullish: body > 0.6 * range AND c_c > (pp_o + pp_c) / 2

# Evening Star (bearish, output: "evening_star") — mirror of Morning Star
# pp = large bullish: body > 0.6 * range
# p = small body (star): body < 0.3 * range (indecision)
# c = large bearish: body > 0.6 * range AND c_c < (pp_o + pp_c) / 2

# Three Inside Up (bullish, output: "three_inside_up")
# pp = large bearish bar: body > 0.5 * range
# p = bullish, body inside pp body (harami): p_o > pp_c AND p_c < pp_o AND p_c > p_o
# c = bullish, c_c > pp_o (closes above pp open = full reversal)

# Three Inside Down (bearish, output: "three_inside_down") — mirror
# pp = large bullish: body > 0.5 * range
# p = bearish, body inside pp body: p_o < pp_c AND p_c > pp_o AND p_c < p_o
# c = bearish, c_c < pp_o

# Harami Cross (reversal, output: "harami_cross")
# pp = large body (either direction): body > 0.6 * range
# p = doji inside pp body: body / range < 0.10 AND p_h < max(pp_o, pp_c) AND p_l > min(pp_o, pp_c)
# Direction stored as +1 (bearish reversal after bullish pp) or -1 (bullish reversal after bearish pp)
# For binary output use: harami_cross = 1.0 if condition else 0.0 (direction implicit from context)

# Dark Cloud Cover (bearish, output: "dark_cloud_cover")
# pp = bullish bar
# c_o > pp_h (gaps above pp high)
# c_c < (pp_o + pp_c) / 2 (closes below pp midpoint — engulfs upper half bearishly)
# c_c > pp_o (doesn't close fully below pp — that would be bearish engulfing)

# Piercing Line (bullish, output: "piercing_line")
# pp = bearish bar
# c_o < pp_l (gaps below pp low)
# c_c > (pp_o + pp_c) / 2 (closes above pp midpoint)
# c_c < pp_o (doesn't close fully above pp — that would be bullish engulfing)
```
