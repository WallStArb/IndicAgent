    ```python
        def _parse_payload(self, payload: dict) -> list | None:
            """Parse llm.calls topic message into INSERT param tuple."""
            parsed = _parse_llm_call_fields(payload)
            if parsed is None:
                return None  # Route to DLQ
            return [self._parsed_to_insert_tuple(parsed)]

        async def _flush_batch(self, batch: list) -> None:
            """Write batch of llm_calls to TimescaleDB."""
            if not self.db_manager:
                raise RuntimeError("No database connection")

            # Report lag before flush
            PERSISTENCE_CONSUMER_LAG.labels(agent_id=self.name).set(len(batch))

            # Execute batch insert
            with self._batch_latency.time():
                await self.db_manager.execute_batch(_INSERT_LLM_CALL_SQL, batch)

            # Clear and report success
            PERSISTENCE_CONSUMER_LAG.labels(agent_id=self.name).set(0)
            self.batch_writes_total.inc()
            self._total_batches += 1
    ```
