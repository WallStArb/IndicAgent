       ```python
       """SwarmLedgerWriterAgent — writer-owned projection of swarm aggregate adjustments into signal_ledger (Phase 80, D-07)."""
       from __future__ import annotations

       import asyncio
       import signal
       from typing import Any

       import asyncpg
       import structlog

       from src.config.settings import Settings
       from src.core.agent.base import BaseAgent
       from src.core.kafka.consumer import KafkaConsumerClient
       from src.core.service_utils import setup_service_logging
       from src.core.stream_keys import topic_swarm_alpha
       from src.observability.metrics import SWARM_SIGNAL_LEDGER_UPDATE_TOTAL

       logger = structlog.get_logger(__name__)

       _RETRY_BACKOFF_S = (0.1, 0.25, 0.5, 1.0, 2.0)


       class SwarmLedgerWriterAgent(BaseAgent):
           agent_id = "swarm_ledger_writer"

           def __init__(self, **kwargs: Any) -> None:
               super().__init__(name=self.__class__.__name__, **kwargs)
               self.settings = Settings()
               self._pool: asyncpg.Pool | None = None
               self._consumer: KafkaConsumerClient | None = None
               self._stop = asyncio.Event()

           async def start(self) -> None:
               setup_service_logging("logs/swarm_ledger_writer_agent.log")
               self._pool = await asyncpg.create_pool(self.settings.database_url, min_size=2, max_size=8)
               self._consumer = KafkaConsumerClient(
                   topics=[topic_swarm_alpha(self.settings.env_name)],
                   group_id="swarm_ledger_writer_consumer",
                   bootstrap_servers=self.settings.kafka_bootstrap,
               )
               await self._consumer.start()
               for sig in (signal.SIGINT, signal.SIGTERM):
                   asyncio.get_running_loop().add_signal_handler(sig, self._stop.set)
               try:
                   await self._consume_loop()
               finally:
                   await self._consumer.stop()
                   if self._pool is not None:
                       await self._pool.close()

           async def _consume_loop(self) -> None:
               assert self._consumer is not None
               async for msg in self._consumer:
                   if self._stop.is_set():
                       break
                   try:
                       await self._handle_event(msg.value)
                   except Exception as exc:
                       logger.warning("swarm_ledger_writer.handle_failed", error=str(exc))

           async def _handle_event(self, payload: dict) -> None:
               signal_id = payload.get("signal_id")
               sm = payload.get("swarm_multiplier")
               ac = payload.get("adjusted_confidence")
               sac = payload.get("swarm_agent_count")
               if not signal_id or sm is None or ac is None:
                   logger.warning("swarm_ledger_writer.invalid_event", payload=payload)
                   return
               await self._apply_projection(signal_id, sm, ac, sac)

           async def _apply_projection(
               self, signal_id: str, swarm_multiplier: float,
               adjusted_confidence: float, swarm_agent_count: int | None,
           ) -> None:
               assert self._pool is not None
               for attempt, delay in enumerate(_RETRY_BACKOFF_S):
                   async with self._pool.acquire() as conn:
                       result = await conn.execute(
                           """
                           UPDATE signal_ledger
                              SET adjusted_confidence = $2,
                                  swarm_multiplier = $3,
                                  swarm_agent_count = $4
                            WHERE signal_id = $1
                           """,
                           signal_id, adjusted_confidence, swarm_multiplier, swarm_agent_count,
                       )
                   # asyncpg returns "UPDATE N" — parse N
                   try:
                       rowcount = int(result.split()[-1])
                   except (ValueError, IndexError):
                       rowcount = 0
                   if rowcount > 0:
                       SWARM_SIGNAL_LEDGER_UPDATE_TOTAL.labels(status="success").inc()
                       return
                   SWARM_SIGNAL_LEDGER_UPDATE_TOTAL.labels(status="retry").inc()
                   await asyncio.sleep(delay)
               # Exhausted retries — row never appeared
               SWARM_SIGNAL_LEDGER_UPDATE_TOTAL.labels(status="miss").inc()
               logger.warning(
                   "swarm_ledger_writer.row_missing",
                   signal_id=signal_id,
                   attempts=len(_RETRY_BACKOFF_S),
               )


       def main() -> None:
           import asyncio as _aio
           _aio.run(SwarmLedgerWriterAgent().start())


       if __name__ == "__main__":
           main()
       ```
