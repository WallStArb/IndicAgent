async def shadow_registry_ensure(conn: Any, component_name: str, component_type: str) -> None:
    """Idempotent enrollment of a component into shadow_registry.

    Uses ON CONFLICT DO NOTHING so custom gate parameters tuned directly in DB

2. Add async method `_load_shadow_cache(self) -> None`:
```python
async def _load_shadow_cache(self) -> None:
    """Load shadow state from shadow_registry into _shadow_cache.
    
    Called at startup and every 5 minutes by background refresh task.
    Uses asyncio.Lock to prevent races during dict swap.
    """
    try:
        rows = await self._db_manager.execute_query(
            "SELECT component_name, is_shadow FROM shadow_registry"
        )
        new_cache = {row["component_name"]: bool(row["is_shadow"]) for row in rows}
        async with self._shadow_cache_lock:
            self._shadow_cache = new_cache
        logger.debug("shadow_cache_loaded", count=len(new_cache))
    except Exception as exc:
        logger.warning("shadow_cache_load_failed", error=str(exc))
        # Keep existing cache on failure — do not clear
```

3. Add method `_is_shadow(self, plugin_name: str) -> bool`:

def _is_shadow(self, plugin_name: str) -> bool:

    Not-found means not enrolled, which should not happen after enroll_all_plugins().
    Default to False (live) to avoid accidentally suppressing unenrolled plugins.
    """
    return self._shadow_cache.get(plugin_name, False)

| T-75-03-02 | Denial of Service | _shadow_cache_refresh_loop | mitigate | On DB failure, keep existing cache (do not clear); log warning; never crash pipeline |
