```python
# src/intelligence/trading/gap_analysis_setup.py
@dataclass
class GapAnalysisSetupPlugin:
    name: str = "trad_GapAnalysisSetup"
    outputs: frozenset[str] = frozenset({
        "signal_type",       # "gap_fade_long" | "gap_fade_short" | "gap_cont_long" | "gap_cont_short" | "none"
        "direction",         # 1 (bullish), -1 (bearish), 0 (no signal)
        "bias",              # "fade" | "continuation" | ""
        "gap_size_atr",      # gap magnitude as ATR multiple (float)
        "confidence",        # 0.0–0.95 (float)
        "entry_type",        # "at_limit" (fade) | "at_pullback" (continuation)
        "entry_price",       # float
        "stop_loss",         # float — 1.0*ATR beyond open for fade, 1.5*ATR for continuation
        "targets",           # list[float], at least 1 element
        "regime_context",    # "gap_open"
        "supporting_factors",# list[str]
    })
    min_lookback: int = 50
    supports_incremental: bool = False
    min_gap_atr_mult: float = 0.3          # minimum gap/ATR to fire
    continuation_atr_mult: float = 1.0     # gap/ATR threshold for continuation bias
    volume_confirm_ratio: float = 1.5      # vol[-1] / mean(vol[-21:-1]) for continuation
    stop_atr_fade: float = 1.0
    stop_atr_cont: float = 1.5
    target_atr_cont: float = 2.0

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """
        frames["main"]     = OHLCV DataFrame (min 50 rows)
        frames["features"] = dict, at minimum {"atr_14": float}

        GAP-01 Detection:
          gap_size = df["open"].iloc[-1] - df["close"].iloc[-2]  # prior close vs current open
          direction = 1 if gap_size > 0, -1 if gap_size < 0, 0 if == 0
          gate: abs(gap_size) < min_gap_atr_mult * atr → _no_signal()

        GAP-02 Classification:
          gap_size_atr = abs(gap_size) / atr
          vol_ratio = vol[-1] / mean(vol[-21:-1])
          high_volume = vol_ratio >= volume_confirm_ratio  # 1.5x threshold
          bias = "continuation" if gap_size_atr >= continuation_atr_mult AND high_volume else "fade"

        GAP-03 Signal fields:
          fade:         entry_type="at_limit", entry = open_[-1]  # current session open
          continuation: entry_type="at_pullback", entry = open_[-1] - direction * 0.25 * atr
          stop (fade):  open_[-1] - direction * stop_atr_fade * atr
          stop (cont):  open_[-1] - direction * stop_atr_cont * atr
          targets (fade): [prior_close, prior_close + direction * 0.5 * atr]
          targets (cont): [open[-1] + direction * 2.0 * atr, open[-1] + direction * 3.0 * atr]
          confidence: base = min(1.0, gap_size_atr / 2.0); +0.15 if high_volume; clamp [0.05, 0.95]
        """
```

CRITICAL pitfall — gap injection:
  make_ohlcv() generates open from close with a random seed. The last bar's open is near close[-1]
  (same bar), not close[-2] (prior bar). You MUST explicitly overwrite open after calling make_ohlcv()
  to create a controlled gap. Do not rely on the generated open value.

CRITICAL pitfall — time gating in unit tests:
  When "bars_since_session_start" is absent from features, the plugin fires regardless.
  Tests that don't inject this feature must still see signals for valid gaps.
  Only inject "bars_since_session_start" in tests that specifically test the time gate.

CRITICAL pitfall — volume_confirm_ratio threshold:
  Default is 1.5. For high-volume tests, set the last bar's volume to mean * 2.5 to reliably exceed it.
  For normal volume, use uniform volume=np.full(n, 1000.0) — ratio will be 1.0 (below threshold).
