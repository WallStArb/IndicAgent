```python
# src/intelligence/trading/gap_analysis_setup.py
@dataclass
class GapAnalysisSetupPlugin:
    name: str = "trad_GapAnalysisSetup"
    outputs: frozenset[str] = frozenset({
        "signal_type",   # "gap_fade_long" | "gap_fade_short" | "gap_cont_long" | "gap_cont_short" | "none"
        "direction",     # 1, -1, or 0
        "bias",          # "fade" | "continuation" | ""
        "gap_size_atr",  # gap size as multiple of ATR (float)
        "confidence",    # 0.0–1.0
        "entry_type",    # "at_limit" | "at_pullback"
        "entry_price",   # float
        "stop_loss",     # float (stop_price)
        "targets",       # list[float] — at least one element
        "regime_context",
        "supporting_factors",
    })
    inputs: tuple[InputSpec, ...] = (InputSpec(symbol=".*", timeframe="1m", lookback=100),)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """
        frames["main"] = OHLCV DataFrame (min 50 rows)
        frames["features"] = dict with at least "atr_14" and optionally "volume_regime"

        Gap detection: gap_size = df["open"].iloc[-1] - df["close"].iloc[-2]
        Direction: gap_size > 0 → bullish (direction=1), gap_size < 0 → bearish (direction=-1)
        Gate: abs(gap_size) < 0.3 * atr → no signal (gap too small)

        Bias logic (GAP-02):
        - gap_size_atr >= 1.0 AND volume confirms (last volume > 1.2 * mean volume) → continuation
        - Otherwise → fade (mean-reversion toward the gap)

        Entry (GAP-03):
        - fade:         entry_type="at_limit", entry = prior_close (gap fill target area)
        - continuation: entry_type="at_pullback", entry = open ± 0.25 * atr (pullback entry)

        Stop (GAP-03):
        - fade long:    stop = open - 1.0 * atr (below the gap open)
        - fade short:   stop = open + 1.0 * atr (above the gap open)
        - cont long:    stop = open - 1.5 * atr
        - cont short:   stop = open + 1.5 * atr

        Target (GAP-03):
        - fade long:    target = prior_close (gap fill)
        - fade short:   target = prior_close (gap fill)
        - cont long:    target = open + 2.0 * atr
        - cont short:   target = open - 2.0 * atr
        """
```
