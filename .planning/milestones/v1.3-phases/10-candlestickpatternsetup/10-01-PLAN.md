```python
@dataclass
class CandlestickPatternSetupPlugin:
    name: str = "trad_CandlestickPatternSetup"
    outputs: frozenset[str] = frozenset({
        "signal_type",        # "candlestick_{pattern}_{long|short}" | "none"
        "direction",          # 1 (bullish), -1 (bearish), 0 (no signal)
        "entry_price",        # float — close[-1]
        "stop_loss",          # float — 1.5 * atr_14 beyond entry
        "targets",            # list[float] — [2.0, 3.5, 5.0] * atr_14
        "confidence",         # float — base per pattern + volume/S/R boosts, clamped [0.10, 0.90]
        "confluence_score",   # float — count of confirming factors (trend + optional vol + optional S/R)
        "regime_context",     # "bullish" | "bearish"
        "supporting_factors", # list[str]
    })
    min_lookback: int = 20
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"trading", "pattern", "structure"})
    inputs: tuple[InputSpec, ...] = (InputSpec(symbol=".*", timeframe="1m", lookback=50),)
    regime_threshold: float = 0.5    # abs(trend_regime) must meet this
    volume_boost_ratio: float = 1.3  # vol[-1] / volume_sma_20 for volume confirm
    sr_proximity_atr: float = 0.3    # nearest S/R within this * atr_14 to confirm
    atr_stop_mult: float = 1.5
    atr_target_mults: tuple = (2.0, 3.5, 5.0)
    _state: dict = field(default_factory=dict)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """
        frames["main"]     = OHLCV DataFrame (min 20 rows)
        frames["features"] = dict with I4/I5/I1/I3 feature values

        CNDL-01 — I5 consumption (no re-detection):
          Pattern signals read from features dict:
            engulfing_bull, engulfing_bear  — 1.0 or 0.0
            pin_bar_bull, pin_bar_bear      — 1.0 or 0.0
            hammer_detected                 — 1.0 or 0.0  (bullish)
            shooting_star_detected          — 1.0 or 0.0  (bearish)

          Priority: hammer/shooting_star (rank 0) > engulfing (rank 1) > pin_bar (rank 2)
          When multiple directional patterns fire on same bar → highest priority wins

        CNDL-02 — Confluence gating:
          Mandatory: abs(trend_regime) >= regime_threshold (0.5) → else _no_signal()
          Direction agreement: bullish patterns require trend_regime > 0
                               bearish patterns require trend_regime < 0 → else _no_signal()
          At least ONE additional factor must confirm:
            - volume: df["volume"].iloc[-1] > volume_sma_20 * volume_boost_ratio (1.3x)
            - S/R proximity: nearest_support or nearest_resistance within sr_proximity_atr * atr_14
          Exception: hammer_detected and shooting_star_detected satisfy S/R factor automatically
          If neither volume nor S/R confirms and pattern is not hammer/shooting_star → _no_signal()

        CNDL-03 — Signal fields:
          entry_price = float(close[-1])
          stop (long)  = entry - atr * atr_stop_mult
          stop (short) = entry + atr * atr_stop_mult
          targets (long)  = [entry + atr * m for m in atr_target_mults]
          targets (short) = [entry - atr * m for m in atr_target_mults]
          confidence base by pattern:
            hammer / shooting_star: 0.65
            engulfing: 0.55
            pin_bar: 0.45
          +0.10 if volume confirms, +0.10 if S/R confirms
          clamp to [0.10, 0.90]
          confluence_score = number of confirming factors present (1 mandatory + up to 2 optional)
        """
```
