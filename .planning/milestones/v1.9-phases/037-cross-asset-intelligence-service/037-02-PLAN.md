```python
from __future__ import annotations
from dataclasses import dataclass
from typing import Any

from src.intelligence.plugins import InputSpec
from src.intelligence.trading.trade_framer import frame_trade

_EQ_INDEX_BASES = frozenset({"ES", "NQ", "RTY", "YM"})
_FIRE_THRESHOLD = 2.0

@dataclass
class CrossAssetDivergencePlugin:
    name: str = "trad_CrossAssetDivergence"
    regime_type: str = "any"
    outputs: frozenset[str] = frozenset({
        "signal_type", "direction", "confidence", "entry", "stop",
        "target_1", "target_2", "target_full", "setup_variant",
        "supporting_factors", "stop_basis",
    })
    inputs: tuple[InputSpec, ...] = ()
    min_lookback: int = 1
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"trading", "cross_asset", "divergence"})

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        features = frames.get("features", {})

        # Guard: symbol must be in EQ_INDEX group
        symbol = features.get("symbol", "")
        base = None
        for b in _EQ_INDEX_BASES:
            if symbol.startswith(b) and len(symbol) > len(b):
                base = b
                break
        if base is None:
            return self._no_signal()

        # Guard: cross_asset data must be ready
        xa = frames.get("cross_asset", {})
        if not xa.get("ready"):
            return self._no_signal()

        # Guard: low_vol_flag means signal unreliable
        if xa.get("low_vol_flag", False):
            return self._no_signal()

        # Determine active pair and its spread_z
        active_pair = xa.get("active_pair", "ES_NQ")
        if active_pair == "ES_NQ":
            spread_z = xa.get("es_nq_spread_z", 0.0)
        else:
            spread_z = xa.get("es_rty_spread_z", 0.0)

        # Guard: threshold gate
        if abs(spread_z) <= _FIRE_THRESHOLD:
            return self._no_signal()

        # Direction logic — regime-biased
        hmm_regime = features.get("hmm_regime")
        hmm_regime_prob = features.get("hmm_regime_prob", 0.0) or 0.0
        spread_positive = spread_z > 0  # ES outperformed

        if hmm_regime == 0:
            # Ranging: mean reversion — fade the outperformer
            direction = -1 if spread_positive else 1
            setup_variant = "cross_asset_reversion"
        elif hmm_regime == 1:
            # Trending up: continuation — follow the leader
            direction = 1 if spread_positive else -1
            setup_variant = "cross_asset_continuation"
        elif hmm_regime == 2:
            # Trending down: continuation — follow the leader down
            direction = -1 if spread_positive else 1
            setup_variant = "cross_asset_continuation"
        else:
            # Unknown/unavailable regime: default to reversion (conservative)
            direction = -1 if spread_positive else 1
            setup_variant = "cross_asset_reversion"

        # Confidence formula
        base_conf = 0.55 + (abs(spread_z) - _FIRE_THRESHOLD) * 0.05
        pairs_confirming = xa.get("pairs_confirming", 0)
        if pairs_confirming == 2:
            base_conf *= 1.2

        # Multi-TF confirmation
        xa_5m = frames.get("cross_asset_5m", {})
        if xa_5m.get("ready"):
            # Check if 5m spread_z agrees in sign with current direction
            z5m_key = "es_nq_spread_z" if active_pair == "ES_NQ" else "es_rty_spread_z"
            z5m = xa_5m.get(z5m_key, 0.0)
            if (z5m > 0) == spread_positive:
                base_conf *= 1.2

        eq_vol_imbalance = xa.get("eq_vol_imbalance", 1.0)
        if eq_vol_imbalance > 1.5:
            base_conf += 0.05

        if hmm_regime_prob >= 0.75:
            base_conf += 0.10

        confidence = min(max(base_conf, 0.0), 1.0)

        # Frame trade for stop/target
        df = frames.get("main")
        if df is None or len(df) < 2:
            return self._no_signal()

        tf = frame_trade(df, direction, "cross_asset_divergence", features=features)

        supporting_factors = {
            "active_pair": active_pair,
            "es_nq_spread_z": xa.get("es_nq_spread_z"),
            "es_rty_spread_z": xa.get("es_rty_spread_z"),
            "eq_corr_break": xa.get("eq_corr_break"),
            "eq_vol_imbalance": eq_vol_imbalance,
            "pairs_confirming": pairs_confirming,
            "data_quality_score": xa.get("data_quality_score"),
            "hmm_regime": hmm_regime,
            "hmm_regime_prob": hmm_regime_prob,
        }

        return {
            "signal_type": "cross_asset_divergence",
            "direction": direction,
            "confidence": float(confidence),
            "entry": float(tf.entry),
            "stop": float(tf.stop),
            "target_1": float(tf.target_1),
            "target_2": float(tf.target_2),
            "target_full": float(tf.target_full),
            "stop_basis": tf.stop_basis,
            "setup_variant": setup_variant,
            "supporting_factors": supporting_factors,
        }

    def compute_next(self, windows: dict[str, Any]) -> dict[str, Any]:
        return self.compute_full(windows)

    @staticmethod
    def _no_signal() -> dict[str, Any]:
        return {"signal_type": "none", "direction": 0, "confidence": 0.0}

plugin = CrossAssetDivergencePlugin()
```
