       Add an `es.onerror` handler that triggers token refresh + EventSource reconnect on auth failure. When the EventSource receives an error (which happens silently when the 15-min access_token expires), the handler should:
       - Track failures and implement exponential backoff (avoid refresh spam on transient errors)
       - Call `POST /api/auth/refresh` with `credentials: "include"` only when backoff allows
       - If refresh succeeds, close the old EventSource and explicitly trigger reconnection
       - If refresh fails (401 — refresh_token also expired), redirect to `/login`
       ```typescript
       let failureCount = 0;
       let lastRefreshAttempt = 0;
       const MIN_REFRESH_INTERVAL = 5000; // 5 seconds minimum between refresh attempts

       es.onerror = async () => {
         const now = Date.now();
         failureCount++;

         // Exponential backoff: wait 2^failureCount seconds before attempting refresh
         const backoffMs = Math.min(1000 * Math.pow(2, failureCount), 60000); // Max 60s

         // Only attempt refresh if enough time has passed since last attempt
         if (now - lastRefreshAttempt < MIN_REFRESH_INTERVAL) {
           return; // Skip this error, will retry after backoff
         }

         lastRefreshAttempt = now;
         await new Promise(resolve => setTimeout(resolve, backoffMs));

         // Attempt token refresh before giving up
         try {
           const refreshRes = await fetch(`${getApiBase()}/api/auth/refresh`, {
             method: "POST",
             credentials: "include",
           });
           if (refreshRes.ok) {
             // Token refreshed — reset counters and reconnect
             failureCount = 0;
             es.close();
             // Explicitly trigger reconnection (re-run useEffect or call reconnect function)
             reconnectEventSource();
             return;
           }
         } catch { /* network error — will retry with backoff */ }

         // Refresh failed (401 or network) — redirect to login
         window.location.href = "/login";
       };
       ```

    3. Create `dashboard/src/hooks/use-token-refresh.ts` — proactive refresh hook that refreshes the access_token every ~12 minutes (before the 15-min expiry):
       ```typescript
       "use client";
       import { useEffect, useCallback } from "react";
       import { getApiBase } from "@/lib/api";

       const REFRESH_INTERVAL_MS = 12 * 60 * 1000; // 12 minutes (before 15-min expiry)

       export function useTokenRefresh() {
         const refresh = useCallback(async () => {
           try {
             const res = await fetch(`${getApiBase()}/api/auth/refresh`, {
               method: "POST",
               credentials: "include",
             });
             if (!res.ok) {
               // Refresh token expired — redirect to login
               window.location.href = "/login";
             }
           } catch {
             // Network error — will retry on next interval
           }
         }, []);

         useEffect(() => {
           const id = setInterval(refresh, REFRESH_INTERVAL_MS);
           return () => clearInterval(id);
         }, [refresh]);
       }
       ```

       "use client";
       import { useState, FormEvent } from "react";
       import { useRouter } from "next/navigation";
       import { getApiBase } from "@/lib/api";

       export default function LoginPage() {
         const [username, setUsername] = useState("");
         const [password, setPassword] = useState("");
         const [error, setError] = useState<string | null>(null);
         const [loading, setLoading] = useState(false);
         const router = useRouter();

         async function handleSubmit(e: FormEvent) {
           e.preventDefault();
           setError(null);
           setLoading(true);
           try {
             const res = await fetch(`${getApiBase()}/api/auth/login`, {
               method: "POST",
               headers: { "Content-Type": "application/json" },
               credentials: "include",
               body: JSON.stringify({ username, password }),
             });
             if (!res.ok) {
               const body = await res.json().catch(() => ({}));
               setError(body.detail || "Login failed");
               return;
             }
             router.push("/");
           } catch (err) {
             setError("Network error");
           } finally {
             setLoading(false);
           }
         }

         return (
           <div className="min-h-screen flex items-center justify-center bg-zinc-950">
             <form onSubmit={handleSubmit} className="bg-zinc-900 p-8 rounded-lg shadow-lg w-full max-w-sm space-y-4">

    5. Create `dashboard/src/components/auth-guard.tsx`:
       ```tsx
       "use client";
       import { useEffect, useState, ReactNode } from "react";
       import { usePathname, useRouter } from "next/navigation";
       import { getApiBase } from "@/lib/api";
       import { useTokenRefresh } from "@/hooks/use-token-refresh";

       export function AuthGuard({ children }: { children: ReactNode }) {
         const [authenticated, setAuthenticated] = useState<boolean | null>(null);
         const pathname = usePathname();
         const router = useRouter();

         // Proactive token refresh — only when not on login page (no token to refresh yet)
         if (pathname !== "/login") {
           useTokenRefresh();
         }

         useEffect(() => {
           if (pathname === "/login") {
             setAuthenticated(true); // Don't guard the login page itself
             return;
           }
           const checkAuth = async () => {
             try {
               const res = await fetch(`${getApiBase()}/api/auth/me`, { credentials: "include" });
               if (res.ok) {
                 setAuthenticated(true);
               } else {
                 router.push("/login");
               }
             } catch {
               router.push("/login");
             }
           };
           checkAuth();
         }, [pathname, router]);

         if (authenticated === null) {
           return <div className="min-h-screen flex items-center justify-center bg-zinc-950">
             <p className="text-zinc-500">Loading...</p>
           </div>;
         }

         return <>{children}</>;
       }
       ```
