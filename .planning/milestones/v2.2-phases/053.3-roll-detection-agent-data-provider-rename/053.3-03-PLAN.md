    ```python
    class RollComputeAgent(BaseAgent):
        def __init__(self) -> None:
            settings = Settings()
            metrics_port = 9122  # config-before-super
            super().__init__(name="roll_compute_agent", metrics_port=metrics_port)
            self.env_name = settings.env_name or ""
            self._kafka_bootstrap = settings.kafka_bootstrap_servers
            self._roll_monitor = RollMonitor(settings)
            self._kafka_producer: KafkaProducerClient | None = None
            self._kafka_consumer: KafkaConsumerClient | None = None
            self._symbol_to_base: dict[str, str] = {
                c.symbol: getattr(c, "base", c.symbol)
                for c in get_active_contracts(settings)
            }
            # Golden Signals (D-17)
            self._events_consumed = Counter(
                "events_consumed_total", "Bars consumed", ["agent"]
            )
            self._rolls_detected = Counter(
                "rolls_detected_total", "Roll events detected", ["agent"]
            )
            self._detection_latency = Histogram(
                "detection_latency_seconds", "Roll detection latency", ["agent"]
            )
            self._detection_errors = Counter(
                "detection_errors_total", "Roll detection errors", ["agent"]
            )

        @property
        def topics_consumed(self) -> list[str]:
            return [topic_market_bars(self.env_name)]

        @property
        def topics_produced(self) -> list[str]:
            return [topic_roll_events(self.env_name)]

        async def _setup(self) -> None:
            self._kafka_producer = KafkaProducerClient(bootstrap_servers=self._kafka_bootstrap)
            await self._kafka_producer.start()
            self._kafka_consumer = KafkaConsumerClient(
                bootstrap_servers=self._kafka_bootstrap,
                group_id="roll_compute_consumer",
                topics=[topic_market_bars(self.env_name)],
            )
            await self._kafka_consumer.start()

        async def _teardown(self) -> None:
            if self._kafka_consumer:
                await self._kafka_consumer.stop()
            if self._kafka_producer:
                await self._kafka_producer.stop()

        async def _run(self) -> None:
            roll_topic = topic_roll_events(self.env_name)
            async for msg in self._kafka_consumer:
                if not self.running:
                    break
                try:
                    payload = msg.value
                    symbol = payload.get("symbol", "")
                    base_symbol = self._symbol_to_base.get(symbol, symbol)
                    volume = float(payload.get("volume", 0))
                    bar_ts_str = payload.get("timestamp", "")

                    self._events_consumed.labels(agent=self.name).inc()
                    self._roll_monitor.update_volume(base_symbol, volume)

                    bar_utc = datetime.fromisoformat(bar_ts_str) if bar_ts_str else datetime.now(UTC)

                    with self._detection_latency.labels(agent=self.name).time():
                        rolled = self._roll_monitor.check_roll(base_symbol, bar_utc)

                    if rolled:
                        # Build typed RollEvent (D-11, D-13)
                        contracts = get_active_contracts(Settings())
                        old_contract = symbol
                        new_contract = symbol  # placeholder — derive from roll chain
                        # Attempt to get new contract from derive_roll_chain
                        try:
                            chain = derive_roll_chain(base_symbol)
                            if chain and len(chain) >= 2:
                                new_contract = chain[-1].symbol
                                old_contract = chain[-2].symbol
                        except Exception:
                            pass

                        roll_event = RollEvent(
                            symbol=base_symbol,
                            old_contract=old_contract,
                            new_contract=new_contract,
                            roll_gap_price=0.0,  # computed from price diff
                            roll_gap_pct=0.0,
                            detection_ts=bar_utc,
                            volume_zscore=self._roll_monitor._last_volume_zscore,
                            confirmation_count=self._roll_monitor._last_confirmation_count,
                        )
                        await self._kafka_producer.send(
                            roll_topic, roll_event.model_dump_json().encode()
                        )
                        self._rolls_detected.labels(agent=self.name).inc()
                        self.logger.info(
                            "roll_detected",
                            symbol=base_symbol,
                            old_contract=old_contract,
                            new_contract=new_contract,
                            volume_zscore=roll_event.volume_zscore,
                            confirmation_count=roll_event.confirmation_count,
                        )
                except Exception as exc:
                    self._detection_errors.labels(agent=self.name).inc()
                    self.logger.error("roll_detection_error", error=str(exc), symbol=symbol)


    async def main() -> None:
        agent = RollComputeAgent()
        await agent.start()


    if __name__ == "__main__":
        from src.observability.otel import init_tracing
        init_tracing(service_name="roll_compute_agent")
        asyncio.run(main())
    ```
