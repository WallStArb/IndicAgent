```python
@dataclass
class MeanReversionPlugin:
    name: str = "trad_MeanReversion"
    outputs: frozenset[str] = frozenset({"signal_type", "direction", "entry_price", "stop_loss", "targets", "confidence", "regime_context", "supporting_factors"})
    min_lookback: int = 50
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"trading", "mean_reversion"})
    inputs: tuple[InputSpec, ...] = (InputSpec(symbol=".*", timeframe=".*", lookback=100),)
    regime_type: str = "mean_reversion"
    _state: dict = field(default_factory=dict)
    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]: ...
    def compute_next(self, windows): return self.compute_full(windows)
    @staticmethod
    def _no_signal(): return {"signal_type": "none", "direction": 0, "confidence": 0.0}
plugin = MeanReversionPlugin()
```

       ```python
       session_vwap = features.get("session_vwap")
       if session_vwap is None:
           return self._no_signal()
       session_vwap = float(session_vwap)
       close = float(features.get("close") or df["close"].iloc[-1])
       above_now = close > session_vwap

       key = (frames.get("symbol", ""), frames.get("timeframe", ""))
       state = self._state.get(key, {})
       above_prev = state.get("above_vwap_prev", None)
       bars_wrong_side = state.get("bars_wrong_side", 0)

       # Update counter
       if not above_now:
           bars_wrong_side = min(bars_wrong_side + 1, 20)
       else:
           # Reset if we're back above (for long setup tracking)
           # For short: track bars above VWAP symmetrically
           pass
       # Need symmetric tracking for both directions
       bars_below = state.get("bars_below", 0)
       bars_above = state.get("bars_above", 0)
       if not above_now:
           bars_below = min(bars_below + 1, 20)
           bars_above = 0
       else:
           bars_above = min(bars_above + 1, 20)
           bars_below = 0
       state["above_vwap_prev"] = above_now
       state["bars_below"] = bars_below
       state["bars_above"] = bars_above
       self._state[key] = state
       ```
     - Gate: reclaim from below (long) = `above_prev == False and above_now == True and bars_below > 0`
     - Gate: reclaim from above (short) = `above_prev == True and above_now == False and bars_above > 0`

       ```python
       rel_volume = features.get("rel_volume")
       if rel_volume is not None and isinstance(rel_volume, (int, float)):
           vol_ok = float(rel_volume) >= 1.2
       else:
           bar_vol = float(df["volume"].iloc[-1])
           avg_vol = float(df["volume"].mean())
           vol_ok = avg_vol > 0 and bar_vol >= 1.2 * avg_vol
       if not vol_ok:
           return self._no_signal()
       ```
