```python
async def _record_swarm_result(
    self, signal_id: str, enriched: SwarmContext, result: AgentResult,
) -> None:
    # 1. ShadowRecorder write (existing alpha_multiplier_shadow path)
    await self._recorder.record(...)  # existing args, unchanged
    # 2. TransformRecorder dual-write (new in Task 1)
    mapping = _SWARM_AGENT_TO_TRANSFORM.get(result.agent_id)
    if mapping is not None and self._transform_recorder is not None:
        transform_id, dag_order = mapping
        seg = f"{enriched.hmm_regime}.{enriched.timeframe}"
        await self._transform_recorder.record(
            signal_id=signal_id, transform_id=transform_id,
            dag_order=dag_order, multiplier=result.multiplier,
            segment_key=seg, metadata=result.metadata,
        )
    elif mapping is None:
        self.logger.warning(
            "swarm_dispatch.unmapped_agent_for_transform_log",
            agent_id=result.agent_id,
        )
```
