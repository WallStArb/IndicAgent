    - /etc/systemd/system/indicagent-api.service (reference for unit file patterns)

  <action>
    1. Update `dashboard/next.config.ts` to enable standalone output:
       ```typescript
       import type { NextConfig } from "next";

       const nextConfig: NextConfig = {
         output: "standalone",
         allowedDevOrigins: ["dash.indicagent.com", "www.indicagent.com", "192.168.1.158"],
       };

       export default nextConfig;
       ```

    2. Build the dashboard:
       ```bash
       cd /home/bg/dev/indicagent/dashboard && npm run build
       ```
       Verify `.next/standalone/server.js` exists after build.

    3. Copy static assets into standalone output (Next.js standalone does not include static files):
       ```bash
       cp -r /home/bg/dev/indicagent/dashboard/.next/static /home/bg/dev/indicagent/dashboard/.next/standalone/.next/static
       cp -r /home/bg/dev/indicagent/dashboard/public /home/bg/dev/indicagent/dashboard/.next/standalone/public 2>/dev/null || true
       ```

    4. Update systemd unit file. Write to `/tmp/indicagent-dashboard.service`:
       ```ini
       [Unit]
       Description=IndicAgent Dashboard — Next.js standalone production server
       After=network-online.target indicagent-api.service
       Wants=network-online.target

       [Service]
       Type=simple
       User=bg
       WorkingDirectory=/home/bg/dev/indicagent/dashboard/.next/standalone
       Environment=NODE_ENV=production
       Environment=PORT=3000
       Environment=HOSTNAME=0.0.0.0
       ExecStart=/usr/bin/node server.js
       Restart=always
       RestartSec=5
       TimeoutStopSec=30
       StandardOutput=journal
       StandardError=journal
       SyslogIdentifier=indicagent-dashboard

       [Install]
       WantedBy=multi-user.target
       ```
       Then install: `sudo cp /tmp/indicagent-dashboard.service /etc/systemd/system/indicagent-dashboard.service && sudo systemctl daemon-reload && sudo systemctl restart indicagent-dashboard`

    5. Verify the dashboard is serving:
       ```bash
       sleep 3 && curl -s -o /dev/null -w "%{http_code}" http://localhost:3000
       ```
       Expected: 200

    1. Update `/etc/cloudflared/config.yml` to add `disableChunkedEncoding: true` on the API ingress rule to prevent SSE buffering. Write to `/tmp/cloudflared-config.yml`:
       ```yaml

       tunnel: abcb8d19-f3dd-4897-8091-d090a8facb94

       ingress:
         # Backend API + SSE stream — disable chunked encoding to prevent SSE buffering
         - hostname: api.indicagent.com
           service: http://localhost:8000

           service: http://localhost:3000
         - hostname: www.indicagent.com
           service: http://localhost:3000
         - hostname: dash.indicagent.com
           service: http://localhost:3000
         - service: http_status:404
       ```
