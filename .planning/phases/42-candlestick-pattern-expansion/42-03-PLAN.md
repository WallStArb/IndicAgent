    async def _load_pattern_weights(self, db: "DatabaseManager") -> dict[str, float]:
        """Load pattern confidence weights from pattern_reliability table.

        Returns cached values if within TTL, otherwise queries DB and updates cache.
        Bootstrap priors (is_bootstrap=true) are included; calibrated weights
        replace priors once sample_size >= 30 and p_value < 0.05.
        """
        now = datetime.now(UTC)
        if (
            self._cache_ts is not None
            and (now - self._cache_ts).total_seconds() < self._cache_ttl_sec
            and self._pattern_weights_cache is not None
        ):
            return self._pattern_weights_cache

        rows = await db.execute_query("""
            SELECT pattern_name, base_confidence
            FROM pattern_reliability
            WHERE is_bootstrap = true OR sample_size >= 30
        """)

        weights = {r["pattern_name"]: float(r["base_confidence"]) for r in rows}
        self._pattern_weights_cache = weights
        self._cache_ts = now
        return weights
