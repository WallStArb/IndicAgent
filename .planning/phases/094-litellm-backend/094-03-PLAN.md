<objective>
Add Instructor-based structured output to the LiteLLM stack. Create typed `BaseModel` result classes for the 4 swarm agents that currently use JSON (skeptic, correlation, counterfactual, regime_coherence), delete the per-agent `_validate_*_fields` boilerplate they replace, and measure parse failure rate before and after.

**Scope boundary:** Narrative agent is explicitly excluded — it returns prose text, not JSON; Instructor adds no value there.

**Latency constraint:** Instructor default retries = 3. With gemma4:e4b at ~50s/call, 3 retries = 150s — over the 120s `latency_budget_ms` of all swarm agents. The instructor client MUST use `max_retries=1`.
</objective>

<threat_model>
ASVS L1 threats for this plan:

- **Latency blowout (mitigated):** Default instructor retries=3 × ~50s = 150s > 120s budget. Mitigated by passing `max_retries=1` to every `generate_structured()` call. Documented in code comment.

- **Narrative agent accidental migration (mitigated):** Explicit exclusion in scope. Narrative agent remains on `_llm_generate()` / `response.strip()` path — no changes to `narrative_agent.py`.

   ```python
   async def generate_structured(
       self,
       prompt: str,
       system: str,
       response_model: type[BaseModel],
       max_tokens: int = 500,
       timeout: float = 120.0,
   ) -> BaseModel | None:
       """Structured output via instructor with Pydantic validation.

       Uses max_retries=1 per call — NOT the instructor default of 3.
       With gemma4:e4b at ~50s/call, 3 retries = 150s > the 120s latency_budget_ms
       of all swarm agents.

       Sets last_provider_id and last_instructor_retries as side effects.
       Returns None when instructor exhausts retries or the model is unavailable.
       """
       if not self.providers:
           self.last_provider_id = None
           self.last_instructor_retries = 0
           return None

       provider = self.providers[0]  # primary provider for structured output
       cb = self._circuit_breaker_for(provider)

       if not cb.allow_request():
           self.last_provider_id = None
           self.last_instructor_retries = 0
           return None

       messages = [
           {"role": "system", "content": system},
           {"role": "user", "content": prompt},
       ]
       extra_kwargs = self._build_extra_kwargs(provider)

       try:
           result = await self._instructor_client.chat.completions.create(
               model=provider,
               response_model=response_model,
               messages=messages,
               max_tokens=max_tokens,
               timeout=timeout,
               max_retries=1,
               **extra_kwargs,
           )
           cb.record_success()
           self.last_provider_id = provider
           self.last_instructor_retries = 0
           return result
       except Exception:
           cb.record_failure()
           self.last_provider_id = None
           self.last_instructor_retries = 0
           return None
   ```

```python
async def generate_structured(
    self,
    prompt: str,
    system: str,
    response_model: type[BaseModel],
    max_tokens: int = 500,
    timeout: float = 120.0,
    audit_context: dict | None = None,
) -> BaseModel | None:
    """Structured output via instructor. Returns validated BaseModel or None on failure."""
    from pydantic import BaseModel as _BaseModel  # local import avoids circular at module level

    with _tracer.start_as_current_span(
        "llm.generate_structured",
        attributes={"call_type": self._call_type},
    ) as span:
        t0 = time.monotonic()
        result = await self._inner.generate_structured(
            prompt=prompt,
            system=system,
            response_model=response_model,
            max_tokens=max_tokens,
            timeout=timeout,
        )
        latency_s = time.monotonic() - t0
        provider_id = self._inner.last_provider_id or "unknown"
        retries = self._inner.last_instructor_retries
        span.set_attribute("llm.provider", provider_id)
        span.set_attribute("llm.latency_ms", round(latency_s * 1000, 1))
        span.set_attribute("llm.instructor_retries", retries)

        if result is None:
            record_llm_call(provider_id, self._call_type, latency_s, status="failure")
            span.set_attribute("llm.status", "failure")
            return None

        response_str = result.model_dump_json()
        token_usage = self._inner.last_token_usage
        actual_total = token_usage.get("total_tokens") if isinstance(token_usage, dict) else None
        tokens = (
            actual_total
            if (actual_total and actual_total > 0)
            else max(1, len(prompt) // 4 + len(response_str) // 4)
        )
        span.set_attribute("llm.tokens", tokens)
        record_llm_call(provider_id, self._call_type, latency_s, tokens=tokens)
        await self._publish_audit(
            audit_context, provider_id, latency_s, tokens, response_str, "instructor"
        )
        return result
```

```python
async def _llm_generate_structured(
    self,
    context: AIContext,
    prompt: str,
    system: str,
    response_model: type[BaseModel],
    max_tokens: int,
    timeout: float,
) -> tuple[BaseModel | None, str]:
    """Generate structured output via instructor with automatic audit trail.

    Mirrors _llm_generate() — same audit context, same span setup, same call_id
    returned for _report_parse_failure(). Use this instead of calling
    self._llm.generate_structured() directly.

    Returns (result, call_id). result is None when instructor exhausts retries.
    Caller must invoke _report_parse_failure(call_id) on None result.
    """
    from pydantic import BaseModel as _BaseModel

    with self.tracer.start_as_current_span(
        "agent.llm_generate_structured",
        attributes={
            ATTR_AGENT_ID: self.agent_id,
            ATTR_SYMBOL: context.symbol,
            ATTR_TF: context.timeframe,
        },
    ) as span:
        try:
            call_id = str(uuid4())
            audit_context: dict[str, Any] = {
                "call_id": call_id,
                "called_at": format_iso_ts(datetime.now(UTC)),
                "symbol": context.symbol,
                "signal_id": str(context.signal_id) if context.signal_id else None,
                "group_name": self.group,
                "agent_id": self.agent_id,
                "prompt_version": self.prompt_version,
                "timeframe": context.timeframe,
                "prompt": prompt,
                "succeeded": True,
                "parse_success": True,
            }

            if context.smc is not None:
                regime = getattr(context.smc, "hmm_regime", None)
                if regime is not None:
                    audit_context["regime"] = str(int(regime))

            result = await self._llm.generate_structured(
                prompt=prompt,
                system=system,
                response_model=response_model,
                max_tokens=max_tokens,
                timeout=timeout,
                audit_context=audit_context,
            )
            if result is None:
                span.set_attribute("llm.empty_response", True)
                LLM_EMPTY_RESPONSES.add(1, self._agent_labels)
            return result, call_id
        except Exception as exc:
            span.set_status(StatusCode.ERROR, str(exc))
            span.record_exception(exc)
            raise
```
