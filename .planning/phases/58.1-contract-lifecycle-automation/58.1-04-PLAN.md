    ```python
    #!/usr/bin/env python3
    """Validate RollComputeAgent detection accuracy against known historical rolls.

    Runs against live TimescaleDB. Checks:
    1. Known rolls detected within expected window
    2. Zero false positives in pre-window period
    3. Cooldown prevents double-fire

    Usage:
        .venv/bin/python production/scripts/roll_backtest.py
    
    Exit codes:
        0 = all checks passed
        1 = one or more checks failed
    """
    import asyncio
    import sys
    from datetime import UTC, datetime, timedelta
    from pathlib import Path

    project_root = Path(__file__).parent.parent.parent
    sys.path.insert(0, str(project_root))

    import asyncpg
    from src.config.settings import Settings
    ```

    **Known rolls to validate (hardcoded test data):**
    ```python
    KNOWN_ROLLS = [
        {
            "base_symbol": "ES",
            "old_contract": "ESH6",
            "new_contract": "ESM6",
            "expected_window_start": datetime(2026, 3, 10, tzinfo=UTC),  # Week before expiry
            "expected_window_end": datetime(2026, 3, 21, tzinfo=UTC),    # After expiry (03-20)
        },
        {
            "base_symbol": "NQ",
            "old_contract": "NQH6",
            "new_contract": "NQM6",
            "expected_window_start": datetime(2026, 3, 10, tzinfo=UTC),
            "expected_window_end": datetime(2026, 3, 21, tzinfo=UTC),
        },
        # Add RTY, YM, etc. as applicable
    ]
    ```

    **Validation functions:**

    1. `async def validate_known_roll(conn, roll: dict) -> tuple[bool, str]`:
       - Query system_events for `event_type='roll_detected'` AND `base_symbol=roll["base_symbol"]`
       - Check detection timestamp falls within expected_window_start..expected_window_end
       - If no event found, check if RollComputeAgent was even running during that period (check systemd journal or just report "no detection event found")
       - Return (passed, message)

    2. `async def validate_no_false_positives(conn, base_symbol: str, start: datetime, end: datetime) -> tuple[bool, str]`:
       - Query system_events for roll_detected events for base_symbol in the 30 days BEFORE the roll window
       - Any events found = false positive = FAIL
       - Return (passed, message)

    3. `async def validate_cooldown(conn, base_symbol: str) -> tuple[bool, str]`:
       - Query system_events for all roll_detected events for base_symbol ordered by detected_at
       - Check no two events within 48 hours of each other (cooldown)
       - Return (passed, message)

    **Main function:**
    ```python
    async def main():
        settings = Settings()
        conn = await asyncpg.connect(settings.database_url)

        passed = 0
        failed = 0
        skipped = 0

        print("=" * 60)
        print("RollComputeAgent Backtest Validation")
        print("=" * 60)

        for roll in KNOWN_ROLLS:
            print(f"\n--- {roll['base_symbol']}: {roll['old_contract']} -> {roll['new_contract']} ---")

            # Check 1: Known roll detected
            ok, msg = await validate_known_roll(conn, roll)
            status = "PASS" if ok else "FAIL"
            if "no detection event" in msg.lower():
                status = "SKIP"
                skipped += 1
            elif ok:
                passed += 1
            else:
                failed += 1
            print(f"  [{status}] Detection in window: {msg}")

            # Check 2: No false positives
            pre_window_start = roll["expected_window_start"] - timedelta(days=30)
            ok, msg = await validate_no_false_positives(
                conn, roll["base_symbol"], pre_window_start, roll["expected_window_start"]
            )
            if ok:
                passed += 1
            else:
                failed += 1
            print(f"  [{'PASS' if ok else 'FAIL'}] No false positives: {msg}")

            # Check 3: Cooldown
            ok, msg = await validate_cooldown(conn, roll["base_symbol"])
            if ok:
                passed += 1
            else:
                failed += 1
            print(f"  [{'PASS' if ok else 'FAIL'}] Cooldown: {msg}")

        await conn.close()

        print(f"\n{'=' * 60}")
        print(f"Results: {passed} passed, {failed} failed, {skipped} skipped")
        print(f"{'=' * 60}")

        if failed > 0:
            print("\nFAILED — do NOT enable RollComputeAgent")
            sys.exit(1)
        elif skipped > 0:
            print("\nSKIPPED checks — RollComputeAgent may not have been running during roll period")
            print("Consider: was the agent running? Check: systemctl status indicagent-roll-compute")
            sys.exit(1)
        else:
            print("\nPASSED — safe to enable RollComputeAgent")
            print("Run: sudo systemctl enable --now indicagent-roll-compute.service")
            sys.exit(0)

    if __name__ == "__main__":
        asyncio.run(main())
    ```
