From src/intelligence/composites/common.py:
```python
def is_num(x: Any) -> bool:
    return isinstance(x, int | float)

def crossover_detect(prev_a, now_a, prev_b, now_b) -> tuple[int, int]:
    """Returns (bullish_cross, bearish_cross) — each 0 or 1."""
    if not all(is_num(v) for v in [prev_a, now_a, prev_b, now_b]):
        return 0, 0
    bullish = 1 if prev_a <= prev_b and now_a > now_b else 0
    bearish = 1 if prev_a >= prev_b and now_a < now_b else 0
    return bullish, bearish
```

```python
# In _state: {"ema5": float | None, "ema3": float | None, "sma9_buf": deque(maxlen=9), "prev_ema3": float | None, "prev_signal": float | None}
# Alpha values:
ALPHA_EMA5 = 2 / (5 + 1)   # = 1/3
ALPHA_EMA3 = 2 / (3 + 1)   # = 1/2

ema5 = ALPHA_EMA5 * rsi + (1 - ALPHA_EMA5) * prev_ema5  # or seed with rsi on first bar
ema3 = ALPHA_EMA3 * ema5 + (1 - ALPHA_EMA3) * prev_ema3 # or seed with ema5 on first bar
sma9_buf.append(ema3)
# Output gate (separate from state update):
if len(sma9_buf) < 9:
    return {}  # not yet warm
signal_line = sum(sma9_buf) / len(sma9_buf)
deriv_osc = ema3 - signal_line
```
