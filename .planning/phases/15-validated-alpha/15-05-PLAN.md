AC Oscillator formula (Bill Williams):
```python
# midpoint = (df["high"] + df["low"]) / 2   (Series)
# AO = midpoint.rolling(5).mean() - midpoint.rolling(34).mean()
# AC = AO - AO.rolling(5).mean()
# Output: ao = float(AO.iloc[-1]), ac = float(AC.iloc[-1])
# Both AO and AC require SMA34 + SMA5 of AO to converge → min_lookback = 40 (34 + 5 + small buffer)
```

Test helper (make_ohlcv-style):
```python
import pandas as pd
import numpy as np

def make_ohlcv(n: int, base: float = 100.0, seed: int = 42) -> pd.DataFrame:
    rng = np.random.default_rng(seed)
    closes = base + np.cumsum(rng.normal(0, 0.5, n))
    highs = closes + rng.uniform(0.1, 0.5, n)
    lows = closes - rng.uniform(0.1, 0.5, n)
    opens = closes + rng.normal(0, 0.2, n)
    vols = rng.uniform(1000, 5000, n)
    return pd.DataFrame({"open": opens, "high": highs, "low": lows, "close": closes, "volume": vols})
```
