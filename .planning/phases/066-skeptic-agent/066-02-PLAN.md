```python
#!/usr/bin/env python3
"""compute_skeptic_baseline.py -- naive baseline failure rate per segment.

Per D-12: Compute per-segment historical failure rate from signal_ledger.
Segments: (regime_type_at_fire, feature_tf, winner_plugin).
The LLM must beat this baseline per segment.

Usage:
    .venv/bin/python scripts/compute_skeptic_baseline.py [--days 90] [--min-sample 30]
"""
from __future__ import annotations

import argparse
import asyncio
import json
import sys
from collections import defaultdict
from pathlib import Path

project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

from src.config.settings import Settings

# Outcomes that count as "failure" for baseline computation
_FAILURE_OUTCOMES = frozenset({
    "never_activated",
    "stopped_at_entry",
    "stopped_in_trade",
    "ttl_expired_ahead",
    "ttl_expired_behind",
})

# Outcomes that count as "win"
_WIN_OUTCOMES = frozenset({
    "target_1",
    "target_1_2",
    "target_full",
})


async def run_baseline(days: int = 90, min_sample: int = 30) -> dict:
    """Compute per-segment failure rate from signal_ledger.

    Returns dict with segments and their failure rates.
    """
    import asyncpg

    settings = Settings()
    conn = await asyncpg.connect(settings.database_url)

    try:
        rows = await conn.fetch(
            """
            SELECT
                regime_type_at_fire,
                feature_tf,
                winner_plugin,
                outcome,
                COUNT(*) as cnt
            FROM signal_ledger
            WHERE timestamp >= NOW() - ($1 || ' days')::INTERVAL
              AND outcome IS NOT NULL
              AND regime_type_at_fire IS NOT NULL
              AND feature_tf IS NOT NULL
              AND winner_plugin IS NOT NULL
            GROUP BY regime_type_at_fire, feature_tf, winner_plugin, outcome
            ORDER BY regime_type_at_fire, feature_tf, winner_plugin
            """,
            str(days),
        )

        # Aggregate per segment
        segments: dict[tuple, dict] = defaultdict(lambda: {"total": 0, "failures": 0, "wins": 0})
        for row in rows:
            key = (row["regime_type_at_fire"], row["feature_tf"], row["winner_plugin"])
            segments[key]["total"] += row["cnt"]
            if row["outcome"] in _FAILURE_OUTCOMES:
                segments[key]["failures"] += row["cnt"]
            if row["outcome"] in _WIN_OUTCOMES:
                segments[key]["wins"] += row["cnt"]

        # Build output
        results = []
        for (regime, tf, plugin), stats in sorted(segments.items()):
            if stats["total"] < min_sample:
                continue
            failure_rate = stats["failures"] / stats["total"] if stats["total"] > 0 else 0.0
            win_rate = stats["wins"] / stats["total"] if stats["total"] > 0 else 0.0
            results.append({
                "regime": regime,
                "tf": tf,
                "plugin": plugin,
                "total": stats["total"],
                "failures": stats["failures"],
                "wins": stats["wins"],
                "failure_rate": round(failure_rate, 4),
                "win_rate": round(win_rate, 4),
            })

        return {
            "days": days,
            "min_sample": min_sample,
            "total_segments": len(results),
            "segments": results,
        }
    finally:
        await conn.close()


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Compute naive baseline failure rates for SkepticAgent validation."
    )
    parser.add_argument("--days", type=int, default=90, help="Days of history to query")
    parser.add_argument("--min-sample", type=int, default=30, help="Minimum sample size per segment")
    parser.add_argument("--json", action="store_true", help="Output raw JSON")
    args = parser.parse_args()

    result = asyncio.run(run_baseline(days=args.days, min_sample=args.min_sample))

    if args.json:
        print(json.dumps(result, indent=2))
    else:
        print(f"Skeptic Naive Baseline ({result['days']}d, min N={result['min_sample']})")
        print(f"Total segments: {result['total_segments']}")
        print("-" * 80)
        print(f"{'Regime':<18} {'TF':<6} {'Plugin':<30} {'N':>5} {'Fail%':>7} {'Win%':>7}")
        print("-" * 80)
        for seg in result["segments"]:
            print(
                f"{seg['regime']:<18} {seg['tf']:<6} {seg['plugin']:<30} "
                f"{seg['total']:>5} {seg['failure_rate']:>6.1%} {seg['win_rate']:>6.1%}"
            )


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""validate_skeptic.py -- statistical validation for SkepticAgent predictions.

Per D-13: JOIN alpha_multiplier_shadow -> signal_ledger on signal_id.
Compute Pearson(skeptic_failure_prob, actual_outcome) per segment.
Per D-14: Graduation gate: per-segment rho >= 0.3 AND p < 0.05 AND N >= 30.
Global gate: overall rho >= 0.2.

Usage:
    .venv/bin/python scripts/validate_skeptic.py [--days 90] [--agent-id skeptic_v1]
"""
from __future__ import annotations

import argparse
import asyncio
import json
import sys
from pathlib import Path

import numpy as np
import structlog
from scipy.stats import pearsonr

project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

from src.config.settings import Settings

logger = structlog.get_logger(__name__)

# Per D-14: graduation thresholds
_SEGMENT_MIN_RHO = 0.3
_SEGMENT_MAX_P = 0.05
_SEGMENT_MIN_N = 30
_GLOBAL_MIN_RHO = 0.2


async def run_validation(
    days: int = 90,
    agent_id: str = "skeptic_v1",
) -> dict:
    """Compute Pearson correlation between skeptic predictions and actual outcomes.

    Per D-13: segment dimensions are (hmm_regime, tf, winner_plugin).
    """
    import asyncpg

    settings = Settings()
    conn = await asyncpg.connect(settings.database_url)

    try:
        # JOIN alpha_multiplier_shadow with signal_ledger on signal_id
        rows = await conn.fetch(
            """
            SELECT
                s.signal_id,
                s.tf,
                s.hmm_regime,
                s.features->>'failure_probability' AS failure_prob,
                s.features->>'prompt_version' AS prompt_version,
                s.predicted_multiplier,
                s.confidence,
                l.outcome,
                l.pnl_r,
                l.winner_plugin,
                l.regime_type_at_fire
            FROM alpha_multiplier_shadow s
            JOIN signal_ledger l ON s.signal_id = l.signal_id
            WHERE s.agent_id = $1
              AND s.ts >= NOW() - ($2 || ' days')::INTERVAL
              AND l.exit_at IS NOT NULL
              AND l.outcome IS NOT NULL
            ORDER BY s.ts
            """,
            agent_id,
            str(days),
        )

        if not rows:
            return {
                "verdict": "INSUFFICIENT_DATA",
                "reason": "No resolved predictions found",
                "total_rows": 0,
                "segments": [],
            }

        # Build binary outcome: win=0 (no failure), loss=1 (failure)
        failure_outcomes = {"never_activated", "stopped_at_entry", "stopped_in_trade", "ttl_expired_ahead", "ttl_expired_behind"}

        # Overall correlation
        all_fp = []
        all_outcomes = []
        for row in rows:
            fp = float(row["failure_prob"]) if row["failure_prob"] else None
            if fp is None:
                continue
            all_fp.append(fp)
            all_outcomes.append(1.0 if row["outcome"] in failure_outcomes else 0.0)

        overall_rho, overall_p = None, None
        if len(all_fp) >= 2:
            rho_val, p_val = pearsonr(np.array(all_fp), np.array(all_outcomes))
            overall_rho = float(rho_val)
            overall_p = float(p_val)

        # Per-segment correlation (hmm_regime, tf, winner_plugin)
        from collections import defaultdict
        segment_data: dict[tuple, list[tuple[float, float]]] = defaultdict(list)
        for row in rows:
            fp = float(row["failure_prob"]) if row["failure_prob"] else None
            if fp is None:
                continue
            key = (
                row["hmm_regime"],
                row["tf"],
                row["winner_plugin"] or "unknown",
            )
            outcome = 1.0 if row["outcome"] in failure_outcomes else 0.0
            segment_data[key].append((fp, outcome))

        segments = []
        for key, pairs in sorted(segment_data.items()):
            n = len(pairs)
            fps = np.array([p[0] for p in pairs])
            outcomes = np.array([p[1] for p in pairs])
            seg_rho, seg_p = None, None
            if n >= 2:
                rho_val, p_val = pearsonr(fps, outcomes)
                seg_rho = float(rho_val)
                seg_p = float(p_val)

            # Per D-14: graduation gate
            graduated = False
            if (
                seg_rho is not None
                and seg_p is not None
                and seg_rho >= _SEGMENT_MIN_RHO
                and seg_p < _SEGMENT_MAX_P
                and n >= _SEGMENT_MIN_N
            ):
                graduated = True

            segments.append({
                "regime": key[0],
                "tf": key[1],
                "plugin": key[2],
                "n": n,
                "rho": round(seg_rho, 4) if seg_rho is not None else None,
                "p_value": round(seg_p, 6) if seg_p is not None else None,
                "graduated": graduated,
                "actual_failure_rate": round(float(outcomes.mean()), 4),
                "mean_predicted_fp": round(float(fps.mean()), 4),
            })

        # Global verdict per D-14
        global_graduated = False
        if overall_rho is not None and overall_rho >= _GLOBAL_MIN_RHO:
            global_graduated = True

        n_graduated = sum(1 for s in segments if s["graduated"])

        return {
            "verdict": "PASS" if n_graduated > 0 else "FAIL",
            "agent_id": agent_id,
            "total_rows": len(rows),
            "overall_rho": round(overall_rho, 4) if overall_rho is not None else None,
            "overall_p_value": round(overall_p, 6) if overall_p is not None else None,
            "global_graduated": global_graduated,
            "n_segments": len(segments),
            "n_graduated_segments": n_graduated,
            "segments": segments,
        }
    finally:
        await conn.close()


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Statistical validation for SkepticAgent predictions."
    )
    parser.add_argument("--days", type=int, default=90, help="Days of history to query")
    parser.add_argument("--agent-id", type=str, default="skeptic_v1", help="Agent ID to validate")
    parser.add_argument("--json", action="store_true", help="Output raw JSON")
    args = parser.parse_args()

    result = asyncio.run(run_validation(days=args.days, agent_id=args.agent_id))

    if args.json:
        print(json.dumps(result, indent=2))
    else:
        print(f"Skeptic Validation: {result['verdict']}")
        print(f"Agent: {result['agent_id']}  |  Rows: {result['total_rows']}")
        if result.get("overall_rho") is not None:
            print(f"Overall: rho={result['overall_rho']:.4f}, p={result['overall_p_value']:.6f}")
            print(f"Global gate (rho >= {_GLOBAL_MIN_RHO}): {'PASS' if result['global_graduated'] else 'FAIL'}")
        print(f"Segments: {result['n_segments']} total, {result['n_graduated_segments']} graduated")
        print("-" * 100)
        print(f"{'Regime':<10} {'TF':<6} {'Plugin':<30} {'N':>5} {'rho':>8} {'p':>10} {'Fail%':>7} {'PredFP':>7} {'Gate':>6}")
        print("-" * 100)
        for seg in result["segments"]:
            rho_str = f"{seg['rho']:.4f}" if seg["rho"] is not None else "N/A"
            p_str = f"{seg['p_value']:.6f}" if seg["p_value"] is not None else "N/A"
            gate = "PASS" if seg["graduated"] else ""
            print(
                f"{seg['regime']:<10} {seg['tf']:<6} {seg['plugin']:<30} "
                f"{seg['n']:>5} {rho_str:>8} {p_str:>10} "
                f"{seg['actual_failure_rate']:>6.1%} {seg['mean_predicted_fp']:>6.1%} {gate:>6}"
            )

    # Exit code: 0 = PASS, 1 = FAIL
    if result["verdict"] != "PASS":
        sys.exit(1)


if __name__ == "__main__":
    main()
```
