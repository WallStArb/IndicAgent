```python
"""test_skeptic_validation.py -- tests for SkepticAgent validation logic.

Tests the pure statistical logic without database dependency.
"""
import numpy as np
import pytest
from scipy.stats import pearsonr

# Per D-14: graduation thresholds (must match validate_skeptic.py)
_SEGMENT_MIN_RHO = 0.3
_SEGMENT_MAX_P = 0.05
_SEGMENT_MIN_N = 30
_GLOBAL_MIN_RHO = 0.2

# Per D-12: failure outcomes (must match compute_skeptic_baseline.py)
FAILURE_OUTCOMES = frozenset({
    "never_activated", "stopped_at_entry", "stopped_in_trade",
    "ttl_expired_ahead", "ttl_expired_behind",
})
WIN_OUTCOMES = frozenset({
    "target_1", "target_1_2", "target_full",
})


def _check_graduation(rho, p_value, n):
    """Replicate graduation gate logic from validate_skeptic.py."""
    return (
        rho is not None
        and p_value is not None
        and rho >= _SEGMENT_MIN_RHO
        and p_value < _SEGMENT_MAX_P
        and n >= _SEGMENT_MIN_N
    )


def _check_global(rho):
    """Replicate global gate logic from validate_skeptic.py."""
    return rho is not None and rho >= _GLOBAL_MIN_RHO


def _compute_failure_rate(failures, total):
    return failures / total if total > 0 else 0.0
```
