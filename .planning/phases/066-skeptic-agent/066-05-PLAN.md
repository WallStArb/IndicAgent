def build_volume_prompt(
    ctx: SwarmContext,
    volume_data: dict[str, Any] | None = None,
) -> str:
    """Build the volume prompt from SwarmContext + volume profile data.

    Per D-01: send all available features as structured context.
    Volume profile fields come from volume_data dict (extracted from I4 JSONB
    by the service layer, since SwarmContext doesn't have these fields).

    Uses ACTIVE_VERSION to select the prompt template from registry.
    """
    template = PROMPT_REGISTRY[ACTIVE_VERSION]
    vd = volume_data or {}

    return template.format(
        symbol=ctx.symbol,
        timeframe=ctx.timeframe,
        winner_plugin=ctx.winner_plugin or "unknown",
        winner_direction_label=_DIRECTION_LABELS.get(ctx.winner_direction or 0, "UNKNOWN"),
        winner_confidence=f"{ctx.winner_confidence:.0%}" if isinstance(ctx.winner_confidence, (int, float)) else "N/A",
        price=f"{ctx.price:.2f}" if isinstance(ctx.price, (int, float)) else "N/A",
        volume=f"{ctx.volume:.0f}" if isinstance(ctx.volume, (int, float)) else "N/A",
        atr=f"{ctx.atr:.2f}" if isinstance(ctx.atr, (int, float)) else "N/A",
        rsi=f"{ctx.rsi:.1f}" if isinstance(ctx.rsi, (int, float)) else "N/A",
        adx=f"{ctx.adx:.1f}" if isinstance(ctx.adx, (int, float)) else "N/A",
        hmm_regime=str(ctx.hmm_regime) if ctx.hmm_regime is not None else "N/A",
        trend_regime=f"{ctx.trend_regime:.2f}" if isinstance(ctx.trend_regime, (int, float)) else "N/A",
        vol_regime=f"{ctx.vol_regime:.2f}" if isinstance(ctx.vol_regime, (int, float)) else "N/A",
        # Session volume profile (from I4)
        poc_price=f"{ctx.poc_price:.2f}" if isinstance(ctx.poc_price, (int, float)) else "N/A",
        vah=_fmt_vol(vd.get("vah")),
        val=_fmt_vol(vd.get("val")),
        # Rolling volume profile (from I4)
        poc_price_rolling=f"{ctx.poc_price_rolling:.2f}" if isinstance(ctx.poc_price_rolling, (int, float)) else "N/A",
        vah_rolling=_fmt_vol(vd.get("vah_rolling")),
        val_rolling=_fmt_vol(vd.get("val_rolling")),
        # Volume dynamics
        price_in_value_area=_fmt_vol(vd.get("price_in_value_area"), precision=3),
        distance_to_vah_atr=_fmt_vol(vd.get("distance_to_vah_atr"), precision=3),
        distance_to_val_atr=_fmt_vol(vd.get("distance_to_val_atr"), precision=3),
        # CTF
        ctf_trend_alignment=f"{ctx.ctf_trend_alignment:.2f}" if isinstance(ctx.ctf_trend_alignment, (int, float)) else "N/A",
        ctf_regime_agreement=f"{ctx.ctf_regime_agreement:.2f}" if isinstance(ctx.ctf_regime_agreement, (int, float)) else "N/A",
    )


def _fmt_vol(val: Any, precision: int = 2) -> str:
    """Format a volume profile value for prompt inclusion."""
    if isinstance(val, (int, float)):
        return f"{val:.{precision}f}"
    return "N/A"

```python
"""volume_agent.py -- VolumeAgentComputeAgent (SwarmBaseAgent subclass).

Per D-07: Standalone agent extending SwarmBaseAgent.
Per D-04: Transfer function: multiplier = (1.0 - failure_probability) * llm_confidence.
Per D-02: LLM returns structured JSON with failure_probability, confidence, risk_factors, reasoning.
Per D-10: shadow_only=True -- production impact from day one via shadow tracking.
"""
from __future__ import annotations

import json
import re
from typing import Any

import structlog

from src.core.llm.chain import LLMProviderChain
from src.core.swarm.base_agent import SwarmBaseAgent
from src.intelligence.schemas import AgentResult
from src.intelligence.swarm.agents.volume_prompts import (
    ACTIVE_VERSION,
    build_volume_prompt,
)
from src.intelligence.swarm.context import SwarmContext

logger = structlog.get_logger(__name__)

_SYSTEM_MESSAGE = (
    "You are a volume profile analyst specializing in detecting fake-out sweeps "
    "vs true volume-supported breakouts. Always respond with valid JSON. "
    '{"failure_probability": float, "confidence": float, '
    '"risk_factors": [str], "reasoning": str}'
)

_JSON_BLOCK_RE = re.compile(r"\{[^{}]*\}", re.DOTALL)


class VolumeAgentComputeAgent(SwarmBaseAgent):
    """Volume profile agent -- detects fake-outs and volume-unsupported moves.

    Per D-07: standalone Kafka consumer (service wrapper handles consumption).
    Per D-11: SafeSwarmWrapper wraps this agent externally for timeout protection.
    This class focuses on _compute() logic only.
    """

    agent_id = "volume_v1"
    path = "llm_swarm"
    shadow_only = True  # Per D-10: never auto-promote
    latency_budget_ms = 5000.0  # 5s LLM timeout

    def __init__(self, llm_chain: LLMProviderChain, **kwargs: Any) -> None:
        super().__init__(name="VolumeAgentComputeAgent", **kwargs)
        self._llm = llm_chain

    async def _compute(self, context: SwarmContext) -> AgentResult:
        """Core computation: build prompt with volume data, call LLM, parse, transfer function.

        The service layer sets _volume_data (dict) on the SwarmContext with
        I4 volume profile fields (vah, val, vah_rolling, val_rolling, etc.)
        that are NOT in SwarmContext's schema.

        Per D-04: multiplier = (1.0 - failure_probability) * llm_confidence.
        Per D-06: raw failure_probability and confidence stored in metadata.
        """
        # Extract volume data from attribute set by service layer
        volume_data: dict[str, Any] | None = None
        if hasattr(context, "_volume_data"):
            volume_data = context._volume_data  # type: ignore[attr-defined]  # noqa: SLF001

        # 1. Build structured prompt from context + volume data
        prompt = build_volume_prompt(context, volume_data=volume_data)

        # 2. Call LLM
        response = await self._llm.generate(
            prompt=prompt,
            system=_SYSTEM_MESSAGE,
            max_tokens=500,
            timeout=self.latency_budget_ms / 1000.0,
        )

        if not response:
            return self._neutral("LLM returned empty response", latency_ms=0.0)

        # 3. Parse JSON response (per D-02)
        parsed = _parse_volume_response(response)
        if parsed is None:
            logger.warning(
                "volume_agent.json_parse_failed",
                agent_id=self.agent_id,
                raw_response=response[:200],
            )
            return self._neutral("JSON parse failed", latency_ms=0.0)

        # 4. Transfer function per D-04: (1.0 - failure_probability) * llm_confidence
        failure_prob = parsed["failure_probability"]
        llm_confidence = parsed["confidence"]
        multiplier = (1.0 - failure_prob) * llm_confidence

        # 5. Return AgentResult per D-06: all intermediate values in metadata
        metadata: dict[str, Any] = {
            "failure_probability": failure_prob,
            "confidence": llm_confidence,
            "risk_factors": parsed["risk_factors"],
            "reasoning": parsed["reasoning"],
            "prompt_version": ACTIVE_VERSION,  # Per D-03
        }
        # Store volume profile snapshot in metadata for audit
        if volume_data:
            for key in ("vah", "val", "vah_rolling", "val_rolling",
                        "price_in_value_area", "distance_to_vah_atr",
                        "distance_to_val_atr"):
                if key in volume_data and volume_data[key] is not None:
                    metadata[f"vp_{key}"] = volume_data[key]

        return AgentResult(
            agent_id=self.agent_id,
            path=self.path,
            multiplier=max(0.0, min(2.0, multiplier)),  # clamp [0, 2]
            confidence=llm_confidence,
            shadow_only=self.shadow_only,
            metadata=metadata,
        )


def _parse_volume_response(raw: str) -> dict[str, Any] | None:
    """Parse structured JSON from LLM response. Same pattern as skeptic_agent."""
    try:
        data = json.loads(raw.strip())
        return _validate_volume_fields(data)
    except json.JSONDecodeError:
        pass

    match = _JSON_BLOCK_RE.search(raw)
    if match:
        try:
            data = json.loads(match.group())
            return _validate_volume_fields(data)
        except json.JSONDecodeError:
            pass

    return None


def _validate_volume_fields(data: dict) -> dict[str, Any] | None:
    """Validate and sanitize the parsed volume response fields."""
    if not isinstance(data, dict):
        return None

    fp = data.get("failure_probability")
    conf = data.get("confidence")

    if not isinstance(fp, (int, float)) or not isinstance(conf, (int, float)):
        return None

    fp = max(0.0, min(1.0, float(fp)))
    conf = max(0.0, min(1.0, float(conf)))

    risk_factors = data.get("risk_factors", [])
    if not isinstance(risk_factors, list):
        risk_factors = [str(risk_factors)]

    reasoning = data.get("reasoning", "")
    if not isinstance(reasoning, str):
        reasoning = str(reasoning)

    return {
        "failure_probability": fp,
        "confidence": conf,
        "risk_factors": risk_factors,
        "reasoning": reasoning,
    }
```

"""volume_agent_service.py -- standalone Kafka consumer service for VolumeAgent.

Per D-07: Standalone Kafka consumer on intelligence.i7.signals.
Per D-08: SwarmContext seeded from DB on startup via SwarmContextCache.seed_from_db_row().
Per D-09: Filters to 5m+ TF only (5m, 15m, 1h, 4h, 1d).
Per D-10: Production impact from day one, all predictions tracked to alpha_multiplier_shadow.

from typing import Any
from uuid import uuid4

from src.config.settings import Settings
from src.core.agent.base import BaseAgent
from src.core.kafka_utils import KafkaConsumerClient, KafkaProducerClient

logger = structlog.get_logger(__name__)

# Per D-09: only process 5m+ timeframes
_ELIGIBLE_TFS = frozenset({"5m", "15m", "1h", "4h", "1d"})

# Volume profile fields to extract from I4 JSONB
# Per CLAUDE.md: poc_price/vah/val = session VP; poc_price_rolling/vah_rolling/val_rolling = rolling VP
_VOLUME_PROFILE_FIELDS = (
    "vah",
    "val",
    "vah_rolling",
    "val_rolling",
    "price_in_value_area",
    "distance_to_vah_atr",

class VolumeAgentService(BaseAgent):
    """Standalone Kafka consumer service wrapping VolumeAgentComputeAgent.

    Dual-loop pattern (same as SwarmOrchestratorComputeAgent and SkepticAgentService):
    - Bar loop: consume market.bars + market.bars.htf -> update SwarmContextCache
    - Signal loop: consume intelligence.i7.signals -> filter TF -> extract volume data -> compute -> record -> publish
    """

    def __init__(self, settings: Settings) -> None:
        super().__init__(name="VolumeAgentService", max_idle_seconds=300)
        self.settings = settings

        # LLM chain for volume calls
        self._llm_chain = LLMProviderChain(
            call_type="volume",
            settings=settings,
            cache_ttl=300.0,  # 5 min cache
        )

        # The compute agent
        self._agent = VolumeAgentComputeAgent(llm_chain=self._llm_chain)

        # Context cache
        self._context_cache = SwarmContextCache()

        # Shadow recorder (initialized in _setup after pool created)
        self._recorder: ShadowRecorder | None = None

        # Kafka clients (initialized in _setup)
        self._bar_consumer: KafkaConsumerClient | None = None
        self._signal_consumer: KafkaConsumerClient | None = None
        self._producer: KafkaProducerClient | None = None

    async def _setup(self) -> None:
        env = self.settings.env_name

        # Bar consumer for context cache warmth
        self._bar_consumer = KafkaConsumerClient(
            topic_market_bars(env),
            topic_market_bars_htf(env),
            bootstrap_servers=self.settings.kafka_bootstrap_servers,
            group_id="volume_agent_bar_consumer",
            auto_offset_reset="latest",
        )
        await self._bar_consumer.start()

        # Signal consumer
        self._signal_consumer = KafkaConsumerClient(
            topic_intelligence_i7_signals(env),
            bootstrap_servers=self.settings.kafka_bootstrap_servers,
            group_id="volume_agent_signal_consumer",
            auto_offset_reset="latest",
        )
        await self._signal_consumer.start()

        # Producer for publishing results
        self._producer = KafkaProducerClient(
            bootstrap_servers=self.settings.kafka_bootstrap_servers,
        )
        await self._producer.start()

        # DB pool + ShadowRecorder
        pool = await asyncpg.create_pool(
            self.settings.database_url, min_size=1, max_size=3
        )
        self._recorder = ShadowRecorder(pool, batch_size=50, flush_interval_s=2.0)

        # Per D-08: seed context cache from DB
        await self._seed_context_cache(pool)

        self.logger.info(
            "volume_agent.started",
            agent_id=self._agent.agent_id,
            version=ACTIVE_VERSION,
        )

    async def _teardown(self) -> None:
        # Flush remaining predictions
        if self._recorder:
            await self._recorder.flush()

        if self._bar_consumer:
            await self._bar_consumer.stop()
        if self._signal_consumer:
            await self._signal_consumer.stop()
        if self._producer:
            await self._producer.stop()

    async def _run(self) -> None:
        bar_task = asyncio.create_task(self._bar_loop())
        signal_task = asyncio.create_task(self._signal_loop())
        try:
            await asyncio.gather(bar_task, signal_task)
        except Exception:
            bar_task.cancel()
            signal_task.cancel()
            raise

    async def _bar_loop(self) -> None:
        """Keep SwarmContextCache warm by consuming bar events."""
        assert self._bar_consumer is not None
        async for _topic, _key, payload in self._bar_consumer.messages():
            if not self.running:
                break
            self._record_message_consumed()
            try:
                self._context_cache.update(payload)
            except Exception as exc:
                self.logger.warning("volume_agent.bar_cache_error", error=str(exc))

    async def _signal_loop(self) -> None:
        """Consume I7 winner signals, filter, compute, record, publish."""
        assert self._signal_consumer is not None
        async for _topic, _key, payload in self._signal_consumer.messages():
            if not self.running:
                break
            self._record_message_consumed()
            try:
                await self._handle_signal(payload)
            except Exception as exc:
                self.logger.exception("volume_agent.signal_error", error=str(exc))

    async def _handle_signal(self, signal: Any) -> None:
        """Process a single I7 winner signal through VolumeAgent."""
        # Extract symbol and tf from payload
        symbol = (
            signal.get("symbol")
            if isinstance(signal, dict)
            else getattr(signal, "symbol", None)
        )
        tf = (
            signal.get("tf")
            if isinstance(signal, dict)
            else getattr(signal, "tf", None)
        )

        # Per D-09: filter to 5m+ TF only
        if tf not in _ELIGIBLE_TFS:
            return

        # Build SwarmContext from cache
        signal_id = uuid4()
        ctx = self._context_cache.build(
            symbol=symbol or "",
            tf=tf or "",
            signal=signal,
            signal_id=signal_id,
        )
        if ctx is None:
            self.logger.warning(
                "volume_agent.no_context",
                symbol=symbol,
                tf=tf,
            )
            return

        # Extract volume profile data from the raw cache event's I4 JSONB
        volume_data = self._extract_volume_data(symbol, tf)

        # Attach volume data to context for agent consumption
        # SwarmContext is frozen (Pydantic ConfigDict frozen=True), so we use
        # object.__setattr__ to add the private _volume_data attribute
        object.__setattr__(ctx, "_volume_data", volume_data)

        # Run VolumeAgent compute (per D-11: timeout handled by SwarmBaseAgent)
        result = await self._agent.compute(ctx)

        # Record to alpha_multiplier_shadow per D-10
        if self._recorder:
            await self._recorder.record(
                signal_id=signal_id,
                agent_id=result.agent_id,
                multiplier=result.multiplier,
                confidence=result.confidence,
                symbol=ctx.symbol,
                tf=ctx.timeframe,
                regime=ctx.hmm_regime,
                path=result.path,
                features=result.metadata,
            )

        # Publish AgentResult to swarm results topic (Path B contributor)
        assert self._producer is not None
        await self._producer.publish(
            topic_swarm_results(self.settings.env_name),
            {
                "signal_id": str(signal_id),
                "agent_id": result.agent_id,
                "symbol": ctx.symbol,
                "tf": ctx.timeframe,
                "multiplier": result.multiplier,
                "confidence": result.confidence,
                "path": result.path,
                "shadow_only": result.shadow_only,
                "hmm_regime": ctx.hmm_regime,
                "features": result.metadata,
                "latency_ms": result.latency_ms,
                "error": result.error,
            },
        )

        if result.error:
            self.logger.info(
                "volume_agent.neutral_result",
                signal_id=str(signal_id),
                symbol=symbol,
                tf=tf,
                error=result.error,
            )
        else:
            self.logger.info(
                "volume_agent.prediction",
                signal_id=str(signal_id),
                symbol=symbol,
                tf=tf,
                failure_prob=result.metadata.get("failure_probability"),
                confidence=result.confidence,
                multiplier=result.multiplier,
            )

    def _extract_volume_data(self, symbol: str | None, tf: str | None) -> dict[str, Any] | None:
        """Extract volume profile fields from the cached I4 JSONB data.

        SwarmContextCache stores raw event proxies (SimpleNamespace) that have
        an .i4 attribute (also SimpleNamespace) unpacked from the I4 JSONB.
        We read the volume profile fields from this .i4 object directly,
        since SwarmContext doesn't include vah/val/etc.
        """
        if not symbol or not tf:
            return None

        key = (symbol, tf)
        entry = self._context_cache._cache.get(key)  # noqa: SLF001
        if entry is None:
            return None

        event_proxy, _ = entry
        i4 = getattr(event_proxy, "i4", None)
        if i4 is None:
            return None

        volume_data = {}
        for field in _VOLUME_PROFILE_FIELDS:
            val = getattr(i4, field, None)
            if val is not None:
                volume_data[field] = val

        return volume_data if volume_data else None

    async def _seed_context_cache(self, pool: asyncpg.Pool) -> None:
        """Per D-08: seed SwarmContextCache from intelligence_features."""
        try:
            async with pool.acquire() as conn:
                rows = await conn.fetch("""
                    SELECT DISTINCT ON (symbol, tf)
                        symbol, tf, ts, bar, i1, i4, i6
                    FROM intelligence_features
                    WHERE ts > NOW() - INTERVAL '7 days'
                      AND i1 IS NOT NULL
                      AND i4 IS NOT NULL
                    ORDER BY symbol, tf, ts DESC
                """)
            loaded = 0
            for row in rows:
                self._context_cache.seed_from_db_row(dict(row))
                loaded += 1
            self.logger.info("volume_agent.cache_seeded", rows=loaded)
        except Exception as exc:
            self.logger.warning("volume_agent.cache_seed_failed", error=str(exc))


def main() -> None:
    settings = Settings()
    service = VolumeAgentService(settings)
    asyncio.run(service.start())


if __name__ == "__main__":
    main()
