EIC-04 gate SQL (from RESEARCH.md Code Examples lines 487-517 — use verbatim with the WHERE bar_ts < oos_start in-sample filter added):
```sql
WITH qualifying AS (
    SELECT symbol, tf, regime FROM alpha_ensemble_ic
    WHERE lookahead = (SELECT config_value FROM config_state WHERE config_key='alpha.ensemble_ic.gate_lookahead')
      AND ic_ci_lower > 0 AND passes_fdr = true AND walk_forward_stable = true
      AND scored_at >= NOW() - INTERVAL '7 days'
    GROUP BY symbol, tf, regime
),
total AS (
    SELECT COUNT(DISTINCT (symbol, tf, regime)) AS n_total FROM alpha_ensemble_ic
    WHERE scored_at >= NOW() - INTERVAL '7 days'
)
SELECT (SELECT COUNT(*) FROM qualifying) AS n_qualifying,
       (SELECT n_total FROM total) AS n_total,
       (SELECT COUNT(*) FROM qualifying)::float / NULLIF((SELECT n_total FROM total), 0) AS fraction;
-- Gate verdict: fraction >= (SELECT config_value::float FROM config_state WHERE config_key='alpha.ensemble_ic.min_qualifying_fraction')
```
