```python
"""Circuit breaker for fault-tolerant stage execution."""

from __future__ import annotations

import asyncio
import time
from dataclasses import dataclass
from enum import Enum


class CircuitState(Enum):
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"


class CircuitOpenError(Exception):
    """Raised when circuit is OPEN and calls are rejected."""


@dataclass
class CircuitBreaker:
    """Circuit breaker with failure threshold and auto-recovery."""
    failure_threshold: int = 5
    timeout_sec: int = 60

    def __post_init__(self):
        self._state = CircuitState.CLOSED
        self._failures = 0
        self._last_failure_time = 0.0
        self._lock = asyncio.Lock()

    async def call(self, func, *args, **kwargs):
        """Execute func with circuit breaker protection."""
        async with self._lock:
            # Check if circuit should transition from OPEN to HALF_OPEN
            if self._state == CircuitState.OPEN:
                if time.time() - self._last_failure_time >= self.timeout_sec:
                    self._state = CircuitState.HALF_OPEN
                else:
                    raise CircuitOpenError(f"Circuit is OPEN (opens after {self.timeout_sec}s)")

            try:
                result = await func(*args, **kwargs) if asyncio.iscoroutinefunction(func) else func(*args, **kwargs)

                # Success: reset failures and close circuit
                self._failures = 0
                if self._state == CircuitState.HALF_OPEN:
                    self._state = CircuitState.CLOSED

                return result

            except Exception as e:
                self._failures += 1
                self._last_failure_time = time.time()

                # Open circuit if threshold exceeded
                if self._failures >= self.failure_threshold:
                    self._state = CircuitState.OPEN

                raise e

    def record_failure(self):
        """Manually record a failure (for async contexts outside call())."""
        self._failures += 1
        self._last_failure_time = time.time()
        if self._failures >= self.failure_threshold:
            self._state = CircuitState.OPEN

    @property
    def state(self) -> CircuitState:
        """Current circuit state."""
        return self._state

    @property
    def failures(self) -> int:
        """Current failure count."""
        return self._failures
```

Key implementation points:
- State machine: CLOSED → OPEN → HALF_OPEN → CLOSED
- Async-safe: uses asyncio.Lock for state transitions
- Works with both sync and async functions via asyncio.iscoroutinefunction check
- record_failure() for manual failure recording (used in try/except blocks)
- Thread-safe for single-process microservice usage

```python
"""Data quality monitoring for pipeline stage validation."""

from __future__ import annotations

import structlog
from typing import Any

from src.intelligence.schemas import IntelligenceEvent

logger = structlog.get_logger(__name__)


class DataQualityMonitor:
    """Validates data integrity at each pipeline stage."""

    # Stage output schemas: type, min, max constraints
    STAGE_SCHEMAS = {
        "intelligence": {
            "symbol": {"type": str, "required": True},
            "timeframe": {"type": str, "required": True},
            "timestamp": {"type": str, "required": True},
        },
        "quality_gated": {
            "confidence": {"type": (int, float), "min": 0.0, "max": 1.0, "required": True},
            "symbol": {"type": str, "required": True},
            "timeframe": {"type": str, "required": True},
        },
        "regime_gated": {
            "confidence": {"type": (int, float), "min": 0.0, "max": 1.0, "required": True},
            "regime_eligible": {"type": bool, "required": True},
        },
        "tod_adjusted": {
            "confidence": {"type": (int, float), "min": 0.0, "max": 1.0, "required": True},
            "tod_multiplier": {"type": (int, float), "min": 0.0, "max": 2.0},
        },
        "calibrated": {
            "confidence": {"type": (int, float), "min": 0.0, "max": 1.0, "required": True},
            "calibrated_confidence": {"type": (int, float), "min": 0.0, "max": 1.0},
        },
        "ranked": {
            "adjusted_rank": {"type": (int, float), "min": 0.0, "max": 1000.0, "required": True},
            "confidence": {"type": (int, float), "min": 0.0, "max": 1.0, "required": True},
        },
        "winner": {
            "confidence": {"type": (int, float), "min": 0.0, "max": 1.0, "required": True},
            "direction": {"type": int, "required": True},
        },
    }

    def __init__(self, stage_name: str):
        self.stage_name = stage_name
        self.schema = self.STAGE_SCHEMAS.get(stage_name, {})

    async def validate_input(self, event: dict | IntelligenceEvent) -> bool:
        """Validate input event against schema."""
        if isinstance(event, IntelligenceEvent):
            data = event.model_dump()
        else:
            data = event

        # Basic presence checks
        if not data:
            logger.error(f"[{self.stage_name}] Empty input event")
            return False

        # Validate against schema if defined
        if self.schema:
            return self._validate_against_schema(data, self.schema, "input")

        return True

    async def validate_output(self, output: dict) -> bool:
        """Validate stage output against schema."""
        if not output:
            logger.error(f"[{self.stage_name}] Empty output")
            return False

        if self.schema:
            return self._validate_against_schema(output, self.schema, "output")

        return True

    def _validate_against_schema(self, data: dict, schema: dict, context: str) -> bool:
        """Validate data dict against schema constraints."""
        for field, constraints in schema.items():
            # Required check
            if constraints.get("required", False) and field not in data:
                logger.error(
                    f"[{self.stage_name}] Missing required field '{field}' in {context}",
                    field=field,
                    context=context
                )
                return False

            # Skip further validation if field not present (optional field)
            if field not in data:
                continue

            value = data[field]

            # Type check
            expected_types = constraints.get("type", ())
            if expected_types and not isinstance(value, expected_types):
                logger.error(
                    f"[{self.stage_name}] Field '{field}' has wrong type in {context}",
                    field=field,
                    expected_type=expected_types,
                    actual_type=type(value),
                    context=context
                )
                return False

            # Min/max checks for numeric fields
            if "min" in constraints and value < constraints["min"]:
                logger.error(
                    f"[{self.stage_name}] Field '{field}' below minimum in {context}",
                    field=field,
                    value=value,
                    min_value=constraints["min"],
                    context=context
                )
                return False

            if "max" in constraints and value > constraints["max"]:
                logger.error(
                    f"[{self.stage_name}] Field '{field}' above maximum in {context}",
                    field=field,
                    value=value,
                    max_value=constraints["max"],
                    context=context
                )
                return False

        return True
```

Key implementation points:
- STAGE_SCHEMAS dict defines validation rules per stage
- validate_input() works with both dict and IntelligenceEvent
- validate_output() checks final stage output
- All validation failures logged with structured context
- Returns bool (True=valid, False=invalid)
- Async methods for consistency with pipeline (though validation is sync)
- Graceful handling of missing fields (optional vs required)

```python
"""Pipeline stage base class for DAG-based signal processing."""

from __future__ import annotations

import asyncio
import json
from abc import ABC, abstractmethod
from typing import Any

import structlog
from pydantic import ValidationError

from src.core.kafka_utils import KafkaConsumerClient, KafkaProducerClient
from src.core.stream_keys import topic_attribution, topic_intelligence
from src.intelligence.monitoring.data_quality_monitor import DataQualityMonitor
from src.intelligence.schemas import IntelligenceEvent
from src.observability.circuit_breaker import CircuitBreaker, CircuitOpenError
from src.observability.metrics import counter, gauge

logger = structlog.get_logger(__name__)


class Stage(ABC):
    """Base class for all pipeline stages.

    Each stage:
    - Subscribes to input topic
    - Processes events through circuit breaker
    - Emits attribution to side topic
    - Publishes output to next stage
    - Validates data at input and output
    - Degrades gracefully on failure (bypass mode)
    """

    def __init__(
        self,
        stage_name: str,
        input_topic_builder: callable,
        output_topic_builder: callable,
        settings: Any,
    ):
        self.stage_name = stage_name
        self.settings = settings

        # Kafka clients
        self.consumer = KafkaConsumerClient(
            bootstrap_servers=settings.kafka_bootstrap_servers,
            group_id=f"{stage_name}_group",
            topic_builder=input_topic_builder,
            auto_offset_reset="latest",
        )
        self.producer = KafkaProducerClient(
            bootstrap_servers=settings.kafka_bootstrap_servers,
            topic_builder=output_topic_builder,
        )
        self.attribution_producer = KafkaProducerClient(
            bootstrap_servers=settings.kafka_bootstrap_servers,
            topic_builder=lambda s, tf: topic_attribution(settings.env_prefix),
        )

        # Fault tolerance and data quality
        self.circuit_breaker = CircuitBreaker(failure_threshold=5, timeout_sec=60)
        self.data_quality_monitor = DataQualityMonitor(stage_name)

        # Metrics
        self.events_consumed = counter(
            f"stage_{stage_name}_events_consumed",
            ["stage", "symbol", "timeframe"]
        )
        self.events_produced = counter(
            f"stage_{stage_name}_events_produced",
            ["stage", "symbol", "timeframe"]
        )
        self.processing_errors = counter(
            f"stage_{stage_name}_errors",
            ["stage", "error_type"]
        )
        self.circuit_state = gauge(
            f"stage_{stage_name}_circuit_state",
            ["stage"],
            # 0=closed, 1=open, 2=half_open
        )

        logger.info(f"[{stage_name}] Stage initialized", stage=stage_name)

    @abstractmethod
    async def process(self, event: IntelligenceEvent) -> dict:
        """Process event through this stage.

        Must be implemented by subclasses.

        Returns
        -------
        dict: Processed event with stage-specific transformations
        """
        raise NotImplementedError(f"{self.__class__.__name__}.process() must be implemented")

    async def emit_attribution(
        self,
        event: IntelligenceEvent,
        result: dict,
        value_added: float = 0.0,
        reason: str = "",
    ) -> None:
        """Emit attribution metadata to side channel."""
        attribution = {
            "signal_id": event.signal_id if hasattr(event, "signal_id") else str(event.symbol + event.timeframe + str(event.timestamp)),
            "symbol": event.symbol,
            "timeframe": event.timeframe,
            "timestamp": event.timestamp.isoformat(),
            "stage": self.stage_name,
            "before": event.features.get("confidence", 0.0) if isinstance(event.features, dict) else 0.0,
            "after": result.get("confidence", 0.0),
            "value_added": round(value_added, 4),
            "reason": reason,
            "inputs": result.get("attribution_inputs", {}),
        }

        await self.attribution_producer.send(
            symbol=event.symbol,
            timeframe=event.timeframe,
            value=attribution,
        )

    async def run(self) -> None:
        """Main processing loop.

        Flow:
        1. Consume event from input topic
        2. Validate input (drop invalid)
        3. Process through circuit breaker
        4. Emit attribution
        5. Publish to output topic
        6. Handle errors gracefully (fault tolerance)
        """
        logger.info(f"[{self.stage_name}] Starting stage run loop", stage=self.stage_name)

        async for msg in self.consumer:
            try:
                # Parse event
                try:
                    if isinstance(msg, dict):
                        raw_event = msg
                    else:
                        raw_event = json.loads(msg.value)

                    event = IntelligenceEvent(**raw_event)

                except (json.JSONDecodeError, ValidationError) as e:
                    logger.error(f"[{self.stage_name}] Failed to parse event", error=str(e))
                    self.processing_errors.labels(
                        stage=self.stage_name,
                        error_type="parse_error"
                    ).inc()
                    continue

                # Update metrics
                self.events_consumed.labels(
                    stage=self.stage_name,
                    symbol=event.symbol,
                    timeframe=event.timeframe,
                ).inc()

                # Data quality check (input)
                if not await self.data_quality_monitor.validate_input(event):
                    logger.warning(
                        f"[{self.stage_name}] Input validation failed",
                        symbol=event.symbol,
                        timeframe=event.timeframe,
                    )
                    self.processing_errors.labels(
                        stage=self.stage_name,
                        error_type="input_validation",
                    ).inc()
                    continue

                # Process with circuit breaker
                try:
                    result = await self.circuit_breaker.call(self.process, event)

                except CircuitOpenError:
                    # Circuit is open: bypass with warning
                    logger.warning(
                        f"[{self.stage_name}] Circuit open, bypassing stage",
                        symbol=event.symbol,
                        timeframe=event.timeframe,
                    )
                    result = event.model_dump()  # Pass through unchanged
                    result["bypassed"] = True
                    result["bypass_reason"] = "circuit_open"

                    self.processing_errors.labels(
                        stage=self.stage_name,
                        error_type="circuit_open",
                    ).inc()

                except Exception as e:
                    # Processing failed: bypass with error details
                    logger.error(
                        f"[{self.stage_name}] Process failed, bypassing",
                        error=str(e),
                        symbol=event.symbol,
                        timeframe=event.timeframe,
                    )
                    result = event.model_dump()
                    result["bypassed"] = True
                    result["bypass_reason"] = "process_error"
                    result["error"] = str(e)

                    self.circuit_breaker.record_failure()
                    self.processing_errors.labels(
                        stage=self.stage_name,
                        error_type="process_error",
                    ).inc()

                # Data quality check (output)
                if not await self.data_quality_monitor.validate_output(result):
                    logger.error(
                        f"[{self.stage_name}] Output validation failed",
                        symbol=event.symbol,
                        timeframe=event.timeframe,
                    )
                    self.processing_errors.labels(
                        stage=self.stage_name,
                        error_type="output_validation",
                    ).inc()
                    continue

                # Emit attribution (only if not bypassed)
                if not result.get("bypassed", False):
                    before_conf = event.features.get("confidence", 0.0) if isinstance(event.features, dict) else 0.0
                    after_conf = result.get("confidence", before_conf)
                    value_added = after_conf - before_conf
                    await self.emit_attribution(
                        event,
                        result,
                        value_added=value_added,
                        reason=result.get("attribution_reason", ""),
                    )

                # Publish to next stage
                await self.producer.send(
                    symbol=event.symbol,
                    timeframe=event.timeframe,
                    value=result,
                )

                self.events_produced.labels(
                    stage=self.stage_name,
                    symbol=event.symbol,
                    timeframe=event.timeframe,
                ).inc()

                # Update circuit state gauge
                self.circuit_state.labels(stage=self.stage_name).set(
                    {"closed": 0, "open": 1, "half_open": 2}[self.circuit_breaker.state.value]
                )

            except Exception as e:
                logger.exception(
                    f"[{self.stage_name}] Unexpected error in run loop",
                    error=str(e),
                )
                self.processing_errors.labels(
                    stage=self.stage_name,
                    error_type="unexpected",
                ).inc()

        logger.info(f"[{self.stage_name}] Stage run loop stopped", stage=self.stage_name)
```
