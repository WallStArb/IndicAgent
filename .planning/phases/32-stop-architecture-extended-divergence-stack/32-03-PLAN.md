       ```python
               "obv_div_score", "cmf_div_score",

               "vol_divergence_age_bars", "obv_divergence_age_bars",
               "cmf_divergence_age_bars",
               "rsi_divergence_magnitude", "macd_divergence_magnitude",
       ```

    3. **compute_full implementation**:

         ```python
         inputs_map = {

             "vol": (features.get("vol_div_bullish", 0), features.get("vol_div_bearish", 0)),
             "obv": (features.get("obv_div_bullish", 0), features.get("obv_div_bearish", 0)),

         }
         ```

       - Compute weighted_score = sum(DIVERGENCE_WEIGHTS[name] * per_input_score for name in inputs)

       ```python
       state_key = (symbol, timeframe)

       for input_name, score in per_input_scores.items():
           age_key = f"{input_name}_age"
           if score > 0:
               self._state[state_key][age_key] = self._state[state_key].get(age_key, 0) + 1
           else:
               self._state[state_key][age_key] = 0
       ```
