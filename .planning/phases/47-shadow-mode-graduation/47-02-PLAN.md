    **Step 2 — Implement get_expiry_date (per D-18):**
    ```python
    from datetime import date, timedelta
    import calendar as cal_mod

    def get_expiry_date(base_symbol: str, expiry_month: int, expiry_year: int) -> date:
        """Approximate expiry date per contract family.
        - Quarterly (equity index, rates, VIX): third Friday of expiry month
        - Monthly energy/metals: last business day of month PRIOR to expiry month
        - Monthly grain: Friday closest to 15th of expiry month
        - Default: 25th of expiry month (conservative)
        """
        if base_symbol in _QUARTERLY_SYMBOLS:
            # Third Friday of expiry month
            first_day = date(expiry_year, expiry_month, 1)
            weekday_of_first = first_day.weekday()  # 0=Mon, 4=Fri
            first_friday = first_day + timedelta(days=(4 - weekday_of_first) % 7)
            return first_friday + timedelta(weeks=2)

        if base_symbol in _ENERGY_METALS_SYMBOLS:
            # Last business day of month PRIOR to expiry month
            prior_month = expiry_month - 1 if expiry_month > 1 else 12
            prior_year = expiry_year if expiry_month > 1 else expiry_year - 1
            last_day = date(prior_year, prior_month, cal_mod.monthrange(prior_year, prior_month)[1])
            # Walk back to weekday
            while last_day.weekday() >= 5:  # Sat=5, Sun=6
                last_day -= timedelta(days=1)
            return last_day

        if base_symbol in _GRAIN_SYMBOLS:
            # Friday closest to 15th of expiry month
            mid = date(expiry_year, expiry_month, 15)
            days_to_friday = (4 - mid.weekday()) % 7
            if days_to_friday > 3:
                days_to_friday -= 7
            return mid + timedelta(days=days_to_friday)

        # Conservative default: 25th of expiry month (per context.md specifics)
        day = min(25, cal_mod.monthrange(expiry_year, expiry_month)[1])
        return date(expiry_year, expiry_month, day)
    ```

    **Step 3 — Implement get_roll_window (per D-18):**
    ```python
    def get_roll_window(base_symbol: str, ref_date: date) -> tuple[date, date] | None:
        """Return (roll_start, roll_end) if a roll is upcoming within 21 days, else None.

        Roll window: starts ~10 trading days before expiry, ends ~2 trading days before.
        Uses calendar days as approximation (14 cal days before to 3 cal days before).
        """
        chain = derive_roll_chain(base_symbol)
        if not chain:
            return None

        for entry in chain:
            expiry = get_expiry_date(
                base_symbol, entry["expiry_month"], entry["expiry_year"]
            )
            roll_start = expiry - timedelta(days=14)  # ~10 trading days before
            roll_end = expiry - timedelta(days=3)  # ~2 trading days before

            if roll_start <= ref_date <= roll_end:
                return (roll_start, roll_end)
            # Check if upcoming within 21 days (look ahead)
            if roll_start > ref_date and (roll_start - ref_date).days <= 21:
                return (roll_start, roll_end)

        return None
    ```

    ```python
    def check_roll(self, base_symbol: str, utc_now: datetime) -> bool:
        from src.config.contracts import get_roll_window
        roll_window = get_roll_window(base_symbol, utc_now.date())
        if roll_window is None:
            self._confirmation_counts[base_symbol] = 0
            return False

        window = self._volume_windows.get(base_symbol)
        if not window or len(window) < 20:
            return False

        arr = np.array(window)
        mean_vol = arr[:-1].mean()
        std_vol = arr[:-1].std()
        if std_vol < 1e-9:
            return False

        current_vol = arr[-1]
        z_score = (current_vol - mean_vol) / std_vol

        if z_score < -2.0:  # Volume DROP below 2 SD
            self._confirmation_counts[base_symbol] = self._confirmation_counts.get(base_symbol, 0) + 1
        else:
            self._confirmation_counts[base_symbol] = 0

        return self._confirmation_counts.get(base_symbol, 0) >= 3
    ```
