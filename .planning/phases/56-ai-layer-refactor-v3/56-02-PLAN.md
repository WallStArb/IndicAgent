```python
"""Narrative prompt building — testable pure functions.

Extracted from ai_narrative_service to enable unit testing
without Kafka/infrastructure dependencies.
"""

from __future__ import annotations

from typing import Any

from src.intelligence.schemas import BarIntelligenceRecord


SYSTEM_PROMPT = (
    "You are a senior trading desk analyst briefing a portfolio manager. "
    "Provide concise, actionable trading narratives based on signal data. "
    "Focus on: (1) signal quality and conviction, (2) market context, "
    "(3) risk-reward setup, (4) key levels and invalidation points."
)


def build_signal_prompt(signal: BarIntelligenceRecord) -> str:
    """Build narrative prompt from single signal.

    Args:
        signal: BarIntelligenceRecord with I1-I7 features

    Returns:
        Formatted prompt string for LLM
    """
    # Extract I7 signal data
    i7_data = signal.i7 or {}
    winner_plugin = i7_data.get("winner_plugin", "Unknown")
    direction = i7_data.get("direction", "HOLD")
    confidence = i7_data.get("calibrated_confidence", 0.0)

    # Extract I4 context
    i4_data = signal.i4 or {}
    atr = i4_data.get("atr", 0.0)
    rsi = i4_data.get("rsi", 50.0)
    volume_profile = i4_data.get("ctx_VolumeProfile", {})

    # Extract I6 confluence
    i6_data = signal.i6 or {}
    ctf_score = i6_data.get("ctf_score", 0.0)
    trend_alignment = i6_data.get("ctf_trend_alignment", 0.0)

    prompt = f"""Signal Analysis Request:

**Signal**: {winner_plugin}
**Direction**: {direction}
**Confidence**: {confidence:.2f}

**Market Context**:
- ATR: ${atr:.2f}
- RSI: {rsi:.1f}
- Volume Profile: POC={volume_profile.get("poc_price", "N/A")}, VAH={volume_profile.get("vah", "N/A")}, VAL={volume_profile.get("val", "N/A")}

**Confluence**:
- CTF Score: {ctf_score:.2f}
- Trend Alignment: {trend_alignment:.2f}

**Task**: Provide a 2-3 sentence narrative explaining:
1. Why this signal is compelling (or not)
2. Key risk factors to monitor
3. Invalidation point (price level that proves the signal wrong)

Be concise. Use plain language. No bullet points.
"""

    return prompt


def build_group_prompt(signals: list[BarIntelligenceRecord]) -> str:
    """Build group synthesis prompt from multiple signals.

    Args:
        signals: List of BarIntelligenceRecord (same symbol/TF)

    Returns:
        Formatted prompt for multi-signal synthesis
    """
    if not signals:
        return "No signals to synthesize."

    # Group by direction
    longs = [s for s in signals if (s.i7 or {}).get("direction") == "LONG"]
    shorts = [s for s in signals if (s.i7 or {}).get("direction") == "SHORT"]

    signal_summary = []
    for sig in signals:
        i7 = sig.i7 or {}
        signal_summary.append(
            f"- {i7.get('winner_plugin', 'Unknown')}: {i7.get('direction', 'HOLD')} @ {i7.get('calibrated_confidence', 0.0):.2f}"
        )

    prompt = f"""Multi-Signal Synthesis Request:

**Symbol**: {signals[0].symbol}
**Timeframe**: {signals[0].timeframe}
**Total Signals**: {len(signals)} ({len(longs)} long, {len(shorts)} short)

**Signals**:
{chr(10).join(signal_summary)}

**Task**: Synthesize these signals into a 2-3 sentence narrative:
1. Overall market bias (bullish/bearish/neutral)
2. Confluence or conflict between signals
3. Recommended stance (aggressive/cautious/stand aside)

Be concise. No bullet points.
"""

    return prompt
```

```python
"""Signal parsing utilities — extract and flatten intelligence data.

Extracted from ai_narrative_service for reusability across
narratives, skeptics, and other AI agents.
"""

from __future__ import annotations

from typing import Any

from src.intelligence.schemas import BarIntelligenceRecord


def parse_signal_data(signal: BarIntelligenceRecord) -> dict[str, Any]:
    """Parse and flatten signal intelligence data.

    Args:
        signal: BarIntelligenceRecord with I1-I7 JSONB fields

    Returns:
        Flattened dict with key signal attributes
    """
    parsed = {
        "symbol": signal.symbol,
        "timeframe": signal.timeframe,
        "timestamp": signal.timestamp,
        "i1": signal.i1 or {},
        "i2": signal.i2 or {},
        "i3": signal.i3 or {},
        "i4": signal.i4 or {},
        "i5": signal.i5 or {},
        "i6": signal.i6 or {},
        "i7": signal.i7 or {},
    }

    return parsed


def extract_signal_fields(signal: BarIntelligenceRecord, fields: list[str]) -> dict[str, Any]:
    """Extract specific fields from signal (nested dict traversal).

    Args:
        signal: BarIntelligenceRecord
        fields: Field paths (e.g., "i7.winner_plugin", "i4.atr")

    Returns:
        Dict with requested fields
    """
    result = {}

    for field_path in fields:
        value = _get_nested_field(signal, field_path)
        result[field_path] = value

    return result


def _get_nested_field(signal: BarIntelligenceRecord, path: str) -> Any:
    """Get field value from nested dict using dot notation.

    Args:
        signal: BarIntelligenceRecord
        path: Dot-notation path (e.g., "i7.winner_plugin")

    Returns:
        Field value or None if not found
    """
    parts = path.split(".")
    current: Any = signal

    for part in parts:
        if isinstance(current, dict):
            current = current.get(part)
        elif hasattr(current, part):
            current = getattr(current, part)
        else:
            return None

    return current
```

```python
"""Narrative orchestrator — single-signal narrative generation.

Uses LLMProviderChain to generate narratives for individual signals.
Publishes to narratives:SYMBOL:TF topic.
"""

from __future__ import annotations

from typing import Any

from src.config.settings import Settings
from src.core.kafka_utils import KafkaProducerClient
from src.core.llm.providers import LLMProviderChain, LLMResponse
from src.core.stream_keys import message_key, topic_narratives
from src.intelligence.schemas import BarIntelligenceRecord
from src.intelligence.narrative.prompts import build_signal_prompt

from .prompts import build_signal_prompt


class NarrativeOrchestrator:
    """Generate narratives for individual trading signals.

    Usage:
        orchestrator = NarrativeOrchestrator(settings, kafka_producer)
        await orchestrator.generate_narrative(signal)
    """

    def __init__(
        self,
        settings: Settings,
        kafka_producer: KafkaProducerClient,
    ):
        """Initialize narrative orchestrator.

        Args:
            settings: Application settings
            kafka_producer: For publishing narratives
        """
        self._settings = settings
        self._kafka = kafka_producer
        self._llm_chain = LLMProviderChain.from_settings(settings, kafka_producer)

    async def generate_narrative(
        self,
        signal: BarIntelligenceRecord,
    ) -> str | None:
        """Generate narrative for single signal.

        Args:
            signal: BarIntelligenceRecord with I1-I7 data

        Returns:
            Narrative text (or None if LLM fails)
        """
        try:
            # Build prompt
            prompt = build_signal_prompt(signal)

            # Call LLM
            response: LLMResponse = await self._llm_chain.generate(
                prompt=prompt,
                call_type="narrative",
            )

            narrative = response.response_text

            # Publish to narratives:SYMBOL:TF topic
            await self._publish_narrative(signal, narrative, response)

            return narrative

        except Exception as e:
            # Log error but don't crash
            print(f"Narrative generation failed: {e}")
            return None

    async def _publish_narrative(
        self,
        signal: BarIntelligenceRecord,
        narrative: str,
        llm_response: LLMResponse,
    ):
        """Publish narrative to Kafka topic.

        Args:
            signal: Source signal
            narrative: Generated narrative text
            llm_response: LLM metadata (model, latency, etc.)
        """
        topic = topic_narratives(self._settings.env_prefix, signal.symbol, signal.timeframe)

        payload = {
            "symbol": signal.symbol,
            "timeframe": signal.timeframe,
            "timestamp": signal.timestamp,
            "narrative": narrative,
            "model": llm_response.model,
            "provider": llm_response.provider,
            "latency_ms": llm_response.latency_ms,
            "call_id": llm_response.call_id,
        }

        key = message_key(signal.symbol, signal.timeframe)

        await self._kafka.publish(
            topic=topic,
            key=key,
            value=payload,
        )
```

```python
class GroupSynthesizer:
    """Synthesize narratives from multiple signals.

    Usage:
        synthesizer = GroupSynthesizer(settings, kafka_producer)
        await synthesizer.synthesize_group(signals)
    """

    def __init__(
        self,
        settings: Settings,
        kafka_producer: KafkaProducerClient,
    ):
        """Initialize group synthesizer.

        Args:
            settings: Application settings
            kafka_producer: For publishing narratives
        """
        self._settings = settings
        self._kafka = kafka_producer
        self._llm_chain = LLMProviderChain.from_settings(settings, kafka_producer)

    async def synthesize_group(
        self,
        signals: list[BarIntelligenceRecord],
    ) -> str | None:
        """Synthesize narrative from multiple signals.

        Args:
            signals: List of BarIntelligenceRecord (same symbol/TF)

        Returns:
            Synthesized narrative text (or None if LLM fails)
        """
        if not signals:
            return None

        try:
            # Build prompt
            prompt = build_group_prompt(signals)

            # Call LLM
            response: LLMResponse = await self._llm_chain.generate(
                prompt=prompt,
                call_type="group_synthesis",
            )

            narrative = response.response_text

            # Publish to narratives:SYMBOL:TF:group topic
            await self._publish_group_narrative(signals[0], narrative, response)

            return narrative

        except Exception as e:
            # Log error but don't crash
            print(f"Group synthesis failed: {e}")
            return None

    async def _publish_group_narrative(
        self,
        signal: BarIntelligenceRecord,
        narrative: str,
        llm_response: LLMResponse,
    ):
        """Publish group narrative to Kafka topic.

        Args:
            signal: Representative signal (for symbol/TF)
            narrative: Generated narrative text
            llm_response: LLM metadata
        """
        topic = topic_narratives_group(self._settings.env_prefix, signal.symbol, signal.timeframe)

        payload = {
            "symbol": signal.symbol,
            "timeframe": signal.timeframe,
            "timestamp": signal.timestamp,
            "narrative": narrative,
            "signal_count": len([signal]),  # TODO: Fix in Task 5
            "model": llm_response.model,
            "provider": llm_response.provider,
            "latency_ms": llm_response.latency_ms,
            "call_id": llm_response.call_id,
        }

        key = message_key(signal.symbol, signal.timeframe)

        await self._kafka.publish(
            topic=topic,
            key=key,
            value=payload,
        )
```
