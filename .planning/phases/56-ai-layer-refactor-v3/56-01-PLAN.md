"""Circuit breaker for LLM providers — prevents cascading failures.

State transitions:
CLOSED → OPEN: after failure_threshold consecutive failures
OPEN → HALF_OPEN: after timeout_seconds in OPEN state
HALF_OPEN → CLOSED: if next call succeeds
HALF_OPEN → OPEN: if next call fails
"""

from __future__ import annotations

import asyncio
from datetime import UTC, datetime
from enum import Enum
from typing import Callable, TypeVar

import structlog

logger = structlog.get_logger(__name__)

T = TypeVar("T")


class CircuitBreakerState(str, Enum):
    """Circuit breaker states."""
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"


class CircuitBreakerError(Exception):
    """Raised when circuit breaker is OPEN."""
    pass


class CircuitBreaker:
    """Async circuit breaker for LLM provider calls.

    Usage:
        breaker = CircuitBreaker(failure_threshold=5, timeout_seconds=60)

        try:
            result = await breaker.call(llm_function, arg1, arg2)
        except CircuitBreakerError:
            # Circuit is OPEN — call rejected
            # Fallback to next provider
    """

    def __init__(
        self,
        failure_threshold: int = 5,
        timeout_seconds: int = 60,
    ):
        """Initialize circuit breaker.

        Args:
            failure_threshold: Consecutive failures before opening
            timeout_seconds: Seconds to wait before transitioning OPEN→HALF_OPEN
        """
        self._failure_threshold = failure_threshold
        self._timeout_seconds = timeout_seconds
        self._state = CircuitBreakerState.CLOSED
        self._failure_count = 0
        self._last_failure_time: datetime | None = None
        self._lock = asyncio.Lock()

    async def call(self, fn: Callable[..., T], *args, **kwargs) -> T:
        """Execute function with circuit breaker protection.

        Raises:
            CircuitBreakerError: If circuit is OPEN
            Exception: If function fails (recorded as failure)
        """
        async with self._lock:
            if self._state == CircuitBreakerState.OPEN:
                if self._should_attempt_reset():
                    self._state = CircuitBreakerState.HALF_OPEN
                    logger.info("Circuit breaker transitioned to HALF_OPEN")
                else:
                    logger.warning("Circuit breaker is OPEN — rejecting call")
                    raise CircuitBreakerError("Circuit breaker is OPEN")

        try:
            result = await fn(*args, **kwargs)
            await self._on_success()
            return result
        except Exception as e:
            await self._on_failure()
            raise e

    def _should_attempt_reset(self) -> bool:
        """Check if enough time has passed to attempt reset."""
        if self._last_failure_time is None:
            return True
        elapsed = (datetime.now(UTC) - self._last_failure_time).total_seconds()
        return elapsed >= self._timeout_seconds

    async def _on_success(self):
        """Handle successful call."""
        async with self._lock:
            if self._state == CircuitBreakerState.HALF_OPEN:
                self._state = CircuitBreakerState.CLOSED
                self._failure_count = 0
                logger.info("Circuit breaker transitioned to CLOSED")

    async def _on_failure(self):
        """Handle failed call."""
        async with self._lock:
            self._failure_count += 1
            self._last_failure_time = datetime.now(UTC)

            if self._failure_count >= self._failure_threshold:
                old_state = self._state
                self._state = CircuitBreakerState.OPEN
                logger.warning(
                    "Circuit breaker opened",
                    failure_count=self._failure_count,
                    threshold=self._failure_threshold,
                    previous_state=old_state,
                )

"""LLM provider chain with fallback and circuit breaker.

Supports multiple providers (Ollama, OpenRouter) with automatic fallback.
Tracks every call to llm.calls topic for outcome analysis.
"""

from __future__ import annotations

import time
from dataclasses import dataclass
from datetime import UTC, datetime
from typing import Any

import httpx

from src.config.settings import Settings
from src.core.kafka_utils import KafkaProducerClient
from src.core.stream_keys import topic_llm_calls
from src.core.llm.circuit_breaker import CircuitBreaker, CircuitBreakerError

from .circuit_breaker import CircuitBreaker, CircuitBreakerError


@dataclass
class LLMResponse:
    """Response from LLM provider."""
    response_text: str
    model: str
    provider: str
    latency_ms: float
    call_id: str


class LLMProvider:
    """Single LLM provider with circuit breaker."""

    def __init__(
        self,
        name: str,
        base_url: str,
        model: str,
        kafka_producer: KafkaProducerClient,
        env_prefix: str,
    ):
        """Initialize LLM provider.

        Args:
            name: Provider name (e.g., "ollama", "openrouter")
            base_url: API base URL
            model: Default model name
            kafka_producer: For publishing call metadata
            env_prefix: Topic prefix (from Settings.env_prefix)
        """
        self.name = name
        self.base_url = base_url.rstrip("/")
        self.model = model
        self._kafka = kafka_producer
        self._env_prefix = env_prefix
        self._breaker = CircuitBreaker(failure_threshold=5, timeout_seconds=60)
        self._client = httpx.AsyncClient(timeout=30.0)

    async def generate(self, prompt: str, call_type: str) -> LLMResponse:
        """Generate LLM response with circuit breaker protection.

        Args:
            prompt: Input prompt
            call_type: Call type (e.g., "narrative", "skeptic")

        Returns:
            LLMResponse with text, model, latency

        Raises:
            CircuitBreakerError: If circuit is OPEN
            httpx.HTTPError: If API call fails
        """
        call_id = f"{self.name}_{int(time.time() * 1000)}"
        start_time = time.time()

        try:
            response = await self._breaker.call(
                self._call_api, prompt
            )
            latency_ms = (time.time() - start_time) * 1000

            llm_response = LLMResponse(
                response_text=response,
                model=self.model,
                provider=self.name,
                latency_ms=latency_ms,
                call_id=call_id,
            )

            # Track call to Kafka
            await self._publish_call_metadata(
                call_id=call_id,
                call_type=call_type,
                prompt=prompt,
                response=response,
                latency_ms=latency_ms,
                success=True,
            )

            return llm_response

        except CircuitBreakerError:
            # Circuit is OPEN — don't track to Kafka (this is expected)
            raise
        except Exception as e:
            latency_ms = (time.time() - start_time) * 1000
            # Track failed call
            await self._publish_call_metadata(
                call_id=call_id,
                call_type=call_type,
                prompt=prompt,
                response=str(e),
                latency_ms=latency_ms,
                success=False,
            )
            raise e

    async def _call_api(self, prompt: str) -> str:
        """Call LLM API (protected by circuit breaker)."""
        if self.name == "ollama":
            return await self._call_ollama(prompt)
        elif self.name == "openrouter":
            return await self._call_openrouter(prompt)
        else:
            raise ValueError(f"Unknown provider: {self.name}")

    async def _call_ollama(self, prompt: str) -> str:
        """Call Ollama /api/generate endpoint."""
        url = f"{self.base_url}/api/generate"
        payload = {
            "model": self.model,
            "prompt": prompt,
            "stream": False,
        }

        response = await self._client.post(url, json=payload)
        response.raise_for_status()
        data = response.json()
        return data.get("response", "")

    async def _call_openrouter(self, prompt: str) -> str:
        """Call OpenRouter /api/v1/chat/completions endpoint."""
        url = f"{self.base_url}/chat/completions"
        payload = {
            "model": self.model,
            "messages": [{"role": "user", "content": prompt}],
        }

        response = await self._client.post(
            url,
            json=payload,
            headers={"Authorization": f"Bearer {self._api_key}"},
        )
        response.raise_for_status()
        data = response.json()
        return data["choices"][0]["message"]["content"]

    async def _publish_call_metadata(
        self,
        call_id: str,
        call_type: str,
        prompt: str,
        response: str,
        latency_ms: float,
        success: bool,
    ):
        """Publish call metadata to llm.calls topic."""
        topic = topic_llm_calls(self._env_prefix)

        metadata = {
            "call_id": call_id,
            "call_type": call_type,
            "provider": self.name,
            "model": self.model,
            "prompt": prompt[:500],  # Truncate long prompts
            "response": response[:500],
            "latency_ms": latency_ms,
            "success": success,
            "timestamp": datetime.now(UTC).isoformat(),
        }

        await self._kafka.publish(
            topic=topic,
            key=call_id.encode(),
            value=metadata,
        )


class LLMProviderChain:
    """Chain of LLM providers with automatic fallback.

    Tries providers in order. If circuit is OPEN or call fails,
    falls back to next provider.
    """

    def __init__(
        self,
        providers: list[LLMProvider],
        kafka_producer: KafkaProducerClient,
        env_prefix: str,
    ):
        """Initialize provider chain.

        Args:
            providers: Ordered list of providers (primary first)
            kafka_producer: For publishing call metadata
            env_prefix: Topic prefix
        """
        if not providers:
            raise ValueError("Provider chain cannot be empty")

        self._providers = providers
        self._kafka = kafka_producer
        self._env_prefix = env_prefix

    async def generate(self, prompt: str, call_type: str = "default") -> LLMResponse:
        """Try each provider in sequence until one succeeds.

        Args:
            prompt: Input prompt
            call_type: Call type for tracking

        Returns:
            LLMResponse from first successful provider

        Raises:
            Exception: If all providers fail
        """
        last_error = None

        for provider in self._providers:
            try:
                return await provider.generate(prompt, call_type)
            except CircuitBreakerError:
                # Circuit is OPEN — try next provider
                continue
            except Exception as e:
                last_error = e
                continue

        # All providers failed
        raise Exception(f"All LLM providers failed. Last error: {last_error}")

    @classmethod
    def from_settings(
        cls,
        settings: Settings,
        kafka_producer: KafkaProducerClient,
    ) -> "LLMProviderChain":
        """Create provider chain from Settings.

        Args:
            settings: Application settings
            kafka_producer: For publishing call metadata

        Returns:
            Configured LLMProviderChain
        """
        providers = []

        # Ollama (primary)
        if settings.ollama_base_url:
            providers.append(
                LLMProvider(
                    name="ollama",
                    base_url=settings.ollama_base_url,
                    model=settings.ollama_default_model,
                    kafka_producer=kafka_producer,
                    env_prefix=settings.env_prefix,
                )
            )

        # OpenRouter (fallback) — if configured
        # TODO: Add openrouter_api_key to Settings
        # if settings.openrouter_api_key:
        #     providers.append(...)

        if not providers:
            raise ValueError("No LLM providers configured in Settings")

        return cls(
            providers=providers,
            kafka_producer=kafka_producer,
            env_prefix=settings.env_prefix,
        )
