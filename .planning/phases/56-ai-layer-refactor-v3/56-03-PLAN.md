```python
#!/usr/bin/env python3
"""
AI Narrative Agent — I8 LLM synthesis of trading signals into human-readable narratives

Subscribes to intelligence.i7.signals stream. For each selected signal,
delegates to NarrativeOrchestrator (single signal) or GroupSynthesizer (multi-signal).

Version: 2.0.0 (Refactored)
Last Updated: 2026-04-08
Status: Production Ready

Refactored from ai_narrative_service.py (1,327 lines → ~200 lines):
- Narrative logic extracted to src/intelligence/narrative/
- LLM infrastructure centralized in src/core/llm/
- Agent is thin coordinator (Kafka I/O + orchestration only)
"""

from __future__ import annotations

import asyncio
import signal
import sys
from pathlib import Path
from typing import Any

project_root = Path(__file__).parent.parent
sys.path.insert(0, str(project_root))

import structlog

from src.config.settings import Settings
from src.core.agent.base_agent import BaseAgent
from src.core.kafka_utils import KafkaConsumerClient, KafkaProducerClient
from src.core.service_utils import setup_service_logging
from src.core.stream_keys import (
    topic_signals_aggregated,
)
from src.intelligence.narrative import (
    NarrativeOrchestrator,
    GroupSynthesizer,
)
from src.observability.metrics import counter, start_metrics_server


class AINarrativeAgent(BaseAgent):
    """AI Narrative Agent — thin coordinator for narrative generation.

    Consumes: development.signals.aggregated (winner signal per bar)
    Publishes: development.narratives:SYMBOL:TF, development.narratives:SYMBOL:TF:group

    Delegates all narrative logic to src.intelligence.narrative module.
    """

    def __init__(self, settings: Settings):
        """Initialize AI Narrative Agent.

        Args:
            settings: Application settings
        """
        super().__init__(settings)
        self._settings = settings

        # Kafka clients
        self._consumer: KafkaConsumerClient | None = None
        self._producer: KafkaProducerClient | None = None

        # Narrative components (from src.intelligence.narrative)
        self._orchestrator: NarrativeOrchestrator | None = None
        self._synthesizer: GroupSynthesizer | None = None

        # Metrics
        self._narratives_generated = counter(
            "ai_narrative_narratives_generated_total",
            "Total narratives generated",
        )
        self._narratives_failed = counter(
            "ai_narrative_narratives_failed_total",
            "Total narrative generation failures",
        )

    async def _setup(self):
        """Initialize agent components."""
        # Setup logging
        setup_service_logging("logs/ai_narrative_agent.log")

        # Initialize Kafka clients
        self._producer = KafkaProducerClient(
            bootstrap_servers=self._settings.kafka_bootstrap_servers,
            client_id="ai_narrative_producer",
        )

        self._consumer = KafkaConsumerClient(
            bootstrap_servers=self._settings.kafka_bootstrap_servers,
            group_id="ai_narrative_consumer",
            topics=[topic_signals_aggregated(self._settings.env_prefix)],
        )

        # Initialize narrative components
        self._orchestrator = NarrativeOrchestrator(
            settings=self._settings,
            kafka_producer=self._producer,
        )

        self._synthesizer = GroupSynthesizer(
            settings=self._settings,
            kafka_producer=self._producer,
        )

        self.logger.info(
            "AI Narrative Agent initialized",
            topics_consumed=self._consumer.topics,
        )

    async def _run(self):
        """Main loop: consume signals and generate narratives."""
        self.logger.info("AI Narrative Agent starting")

        async for message in self._consumer.consume():
            try:
                await self._process_signal(message.value)
            except Exception as e:
                self.logger.error(
                    "Failed to process signal",
                    error=str(e),
                    signal_key=message.key.decode(),
                )
                self._narratives_failed.inc()

    async def _process_signal(self, signal_data: dict[str, Any]):
        """Process single signal and generate narrative.

        Args:
            signal_data: Signal dict from Kafka
        """
        # Extract signal metadata
        symbol = signal_data.get("symbol")
        timeframe = signal_data.get("timeframe")
        timestamp = signal_data.get("timestamp")

        self.logger.debug(
            "Processing signal",
            symbol=symbol,
            timeframe=timeframe,
            timestamp=timestamp,
        )

        # Generate narrative (delegated to orchestrator)
        narrative = await self._orchestrator.generate_narrative(
            signal=signal_data,  # BarIntelligenceRecord
        )

        if narrative:
            self._narratives_generated.inc()
            self.logger.info(
                "Narrative generated",
                symbol=symbol,
                timeframe=timeframe,
                narrative_length=len(narrative),
            )
        else:
            self.logger.warning(
                "Narrative generation returned None",
                symbol=symbol,
                timeframe=timeframe,
            )

    async def _teardown(self):
        """Clean up resources."""
        self.logger.info("AI Narrative Agent shutting down")

        if self._consumer:
            await self._consumer.close()
        if self._producer:
            await self._producer.close()


async def main():
    """Entry point for AI Narrative Agent."""
    settings = Settings()

    # Start metrics server
    start_metrics_server(port=9113)

    # Create and start agent
    agent = AINarrativeAgent(settings)
    await agent.start()


if __name__ == "__main__":
    # Signal handlers
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)

    for sig in (signal.SIGINT, signal.SIGTERM):
        loop.add_signal_handler(sig, lambda: loop.stop())

    try:
        loop.run_until_complete(main())
    finally:
        loop.close()
```
