    ```tsx
    const OutcomeBadge = ({ outcome }: { outcome?: string }) => {
      if (!outcome) return null;

      // Map 8-class outcomes to 5 display labels
      const labelMap: Record<string, string> = {
        never_activated: "EXPIRED",
        stopped_at_entry: "STOPPED",
        stopped_in_trade: "STOPPED",
        target_1: "T1 HIT",
        target_1_2: "T1+T2 HIT",
        target_full: "FULL TARGET",
        ttl_expired_ahead: "EXPIRED",
        ttl_expired_behind: "EXPIRED",
      };

      const label = labelMap[outcome] || outcome.toUpperCase();

      // Color mapping (optional, or use neutral gray)
      const colorClass = label.includes("HIT") || label.includes("TARGET")
        ? "bg-green-600"
        : label === "STOPPED"
        ? "bg-red-600"
        : "bg-gray-600";

      return (
        <div className={`${colorClass} text-white text-xs font-bold px-2 py-1 rounded inline-block`}>
          {label}
        </div>
      );
    };
    ```

    Note: Design doc shows badge above or below entry row — placement is flexible. Key requirement: badge is visible and clearly indicates outcome.

    Also note: Staleness ratio should NOT be shown for resolved signals (signal is definitively closed, not just old). If component has staleness display, condition it on `!signal.resolved`.
