<objective>
Close the two dashboard-side gaps that prevent signals from appearing on refresh and history entries from showing outcome badges.

**Gap 3B:** Add `fetchActiveSignals()` call in the SSE mount effect to seed `signalsByTf` from the REST API on mount. This ensures signals appear immediately on refresh/load without waiting for a new live SSE event.

**Gap 2:** In the `dir===0` terminal event handler, after updating `signalsByTf`, also call `setSignalsHistory` to replace the matching history entry (by `signal_id`) with `resolved:true`, `outcome`, and `exit_price`. This makes `RecentSignalCard`'s `OutcomeBadge` visible for resolved signals.

Purpose: The dashboard currently shows no signals on refresh because `signalsByTf` is only populated by live SSE events — there is no REST seed on mount. The history array is also never updated on resolution, so `RecentSignalCard` never renders an `OutcomeBadge`.

Output: Updated `use-market-stream.ts` with `fetchActiveSignals` + `signalsHistory` resolution update.
</objective>

    The function pattern mirrors `fetchSession`:

    ```typescript
    const fetchActiveSignals = async () => {
      try {
        for (const sym of symbols) {
          const contract = sym; // symbols array contains base symbols; API accepts both
          const url = `${base}/api/signals/${encodeURIComponent(contract)}?limit=10`;
          const res = await fetch(url);
          if (!res.ok) continue;
          const body = await res.json() as {
            signals: Array<{
              signal_id: string;
              symbol: string;
              timeframe: string;
              setup_plugin: string;
              signal_type: string;
              direction: number;
              entry_price: number | null;
              stop_loss: number | null;
              confidence: number | null;
              status: string;
              outcome: string | null;
              exit_price: number | null;
              signal_computed_at: string | null;
              market_price_at_signal: number | null;
              entry_zone_low: number | null;
              entry_zone_high: number | null;
            }>;
          };
          const RESOLVED_STATUSES = new Set([
            "stopped_at_entry", "stopped_in_trade",
            "target_1", "target_1_2", "target_full",
            "ttl_expired_ahead", "ttl_expired_behind",
            "never_activated",
          ]);
          // Group by timeframe — take the most recent per TF
          const byTf: Record<string, SignalData> = {};
          for (const row of body.signals) {
            const tf = row.timeframe;
            if (byTf[tf]) continue; // already have one for this TF (results are DESC)
            const dir = row.direction;
            if (dir === 0) continue; // skip terminal rows
            const isResolved = RESOLVED_STATUSES.has(row.status);
            const signal: SignalData = {
              symbol: sym,
              direction: dir > 0 ? "long" : "short",
              signal_type: row.signal_type || "",
              setup_plugin: row.setup_plugin || "",
              confidence: row.confidence ?? 0,
              entry_price: row.entry_price ?? 0,
              entry_type: "at_close",
              stop_loss: row.stop_loss ?? 0,
              stop_type: "atr",
              framing_method: "atr_fallback",
              risk_reward_ratio: 0,
              regime_context: "",
              timeframe: tf,
              timestamp: row.signal_computed_at || "",
              signal_computed_at: row.signal_computed_at ?? undefined,
              signal_id: row.signal_id,
              market_price_at_signal: row.market_price_at_signal ?? undefined,
              entry_zone_low: row.entry_zone_low ?? undefined,
              entry_zone_high: row.entry_zone_high ?? undefined,
              resolved: isResolved || undefined,
              outcome: row.outcome ?? undefined,
              exit_price: row.exit_price ?? undefined,
            };
            byTf[tf] = signal;
          }
          if (Object.keys(byTf).length === 0) continue;
          setSymbolData((prev) => {
            const old = prev[sym];
            if (!old) return prev;
            // Only seed TFs that have no live signal yet (don't overwrite SSE data)
            const mergedByTf: Record<string, SignalData> = { ...byTf };
            for (const [tf, existing] of Object.entries(old.signalsByTf)) {
              mergedByTf[tf] = existing; // live data wins
            }
            return {
              ...prev,
              [sym]: {
                ...old,
                signalsByTf: mergedByTf,
                lastUpdate: Date.now(),
              },
            };
          });
          // Also seed signalsHistory
          setSignalsHistory((prev) => {
            const existing = prev[sym] || [];
            const existingIds = new Set(existing.map((s) => s.signal_id));
            const newEntries = Object.values(byTf).filter((s) => !existingIds.has(s.signal_id ?? ""));
            if (newEntries.length === 0) return prev;
            const merged = [...newEntries, ...existing].slice(0, HISTORY_LIMIT);
            return { ...prev, [sym]: merged };
          });
        }
      } catch {
        // non-fatal — SSE stream will populate signals as they arrive
      }
    };
    void fetchActiveSignals();
    ```

    Note: `regime_suppressed` is intentionally NOT in `RESOLVED_STATUSES` — those signals should display as live/suppressed, not resolved.
