from pydantic import BaseModel, ConfigDict, Field, field_validator


class SkepticResult(BaseModel):
    """Strict result model for the Pydantic AI Skeptic agent.

    extra="forbid" rejects unexpected keys; ge/le bounds REJECT (not clamp)
    out-of-range values so NativeOutput / validation retry can correct them.
    """

    model_config = ConfigDict(extra="forbid")

    failure_probability: float = Field(ge=0.0, le=1.0)
    confidence: float = Field(ge=0.0, le=1.0)
    risk_factors: list[str] = Field(default_factory=list)
    reasoning: str

    @field_validator("risk_factors", mode="before")
    @classmethod
    def _coerce_risk_factors(cls, v: object) -> list[str]:
        if v is None:
            return []
        if not isinstance(v, list):
            return [str(v)]
        return [str(x) for x in v]
```

Do not modify `ACTIVE_VERSION`, `PROMPT_REGISTRY`, `build_skeptic_prompt`, or `_validate_skeptic_fields`.

from src.intelligence.ai.alpha.skeptic_prompts import SkepticResult  # add to existing import


class SkepticComputeAgentV2(BaseMultiplierAgent):
    """Pydantic AI typed-output shadow proof of the Skeptic agent.

    Thin: sets result_type=SkepticResult and calls inherited _run_typed.
    Same prompt, same transfer function, same AgentOutput shape as v1.
    """

    output_schema: ClassVar[dict] = {
        "failure_probability": float,
        "confidence": float,
        "risk_factors": list,
        "reasoning": str,
    }
    result_type: ClassVar[type[BaseModel] | None] = SkepticResult

    agent_id = "skeptic_pydantic_v1"
    prompt_version = ACTIVE_VERSION
    group = "alpha"
    tiers_needed = frozenset({Tier.I1, Tier.I4, Tier.I6, Tier.I7, Tier.SMC})
    latency_budget_ms = 120000.0
    shadow_only = True  # reference shadow implementation — never live

    def __init__(self, llm_chain: LLMProviderChain, **kwargs: Any) -> None:
        super().__init__(name="SkepticComputeAgentV2", **kwargs)
        self._llm = llm_chain

    async def _compute(self, context: AIContext) -> AgentOutput:
        prompt = build_skeptic_prompt(context)
        result: SkepticResult = await self._run_typed(
            context,
            prompt=prompt,
            system=_SYSTEM_MESSAGE,
            max_tokens=500,
            timeout=self.latency_budget_ms / 1000.0,
        )
        multiplier = (1.0 - result.failure_probability) * result.confidence
        return self._build_multiplier_output(
            context=context,
            multiplier=multiplier,
            confidence=result.confidence,
            payload={
                "failure_probability": result.failure_probability,
                "risk_factors": result.risk_factors,
                "reasoning": result.reasoning,
            },
            prompt_version=ACTIVE_VERSION,
        )
```
