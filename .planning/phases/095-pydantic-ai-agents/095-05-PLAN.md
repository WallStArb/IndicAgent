```python
async def _check_ollama_structured_output_support(self) -> None:
    """Warn (do not crash) if Ollama < 0.5.0 — NativeOutput may silently degrade."""
    try:
        proc = await asyncio.create_subprocess_exec(
            "docker", "exec", "ollama", "ollama", "--version",
            stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
        )
        out, _ = await proc.communicate()
        text = out.decode(errors="ignore")
        m = re.search(r"(\d+)\.(\d+)\.(\d+)", text)
        if m and (int(m.group(1)), int(m.group(2))) < (0, 5):
            self.logger.warning(
                "alpha_swarm.ollama_version_too_old",
                version=m.group(0), required="0.5.0",
                note="Pydantic AI structured output may degrade",
            )
        elif not m:
            self.logger.warning("alpha_swarm.ollama_version_unparsed", raw=text[:80])
    except Exception as exc:
        self.logger.warning("alpha_swarm.ollama_version_check_failed", error=str(exc)[:120])
```
