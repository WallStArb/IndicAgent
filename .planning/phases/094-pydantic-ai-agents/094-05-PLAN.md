    - Confidence delta between old and new agent is measured
    - Parse success rate = 100% (NativeOutput guarantee)
    """

    import pytest
    from decimal import Decimal

    @pytest.mark.integration
    def test_shadow_validation_gate_query():
        """Verify shadow validation gate query returns expected metrics.

        This test documents the query used to validate AGENT-EXEC-04:
        - n >= 100 inferences before promotion
        - Confidence delta measured between old and new agent
        - Parse success = 100% for Pydantic AI agent
        """
        # Query to validate shadow performance
        query = """
            SELECT
                agent_id,
                prompt_version,
                COUNT(*) as n,
                AVG(confidence) as avg_confidence,
                AVG(pnl_r) as avg_pnl,
                SUM(CASE WHEN parse_success = true THEN 1 ELSE 0 END) as parse_successes,
                SUM(CASE WHEN parse_success = false THEN 1 ELSE 0 END) as parse_failures
            FROM llm_calls
            WHERE agent_id IN ('skeptic_v2', 'skeptic_v2_pydantic')
              AND called_at > NOW() - INTERVAL '7 days'
            GROUP BY agent_id, prompt_version
            HAVING COUNT(*) >= 100;
        """
        # In production, run this query to verify shadow gate
        # Expected: skeptic_v2_pydantic parse_failures = 0 (NativeOutput guarantee)
        # Expected: avg_confidence within ±0.05 of skeptic_v2 (calibration check)

    @pytest.mark.integration
    def test_parse_success_validation_query():
        """Verify parse success = 100% for Pydantic AI agent.

        NativeOutput should eliminate parse failures entirely.
        """
        query = """
            SELECT
                agent_id,
                COUNT(*) as total_calls,
                SUM(CASE WHEN parse_success = true THEN 1 ELSE 0 END) as parse_successes,
                SUM(CASE WHEN parse_success = false THEN 1 ELSE 0 END) as parse_failures,
                AVG(parse_success::int) as parse_success_rate
            FROM llm_calls
            WHERE agent_id = 'skeptic_v2_pydantic'
              AND called_at > NOW() - INTERVAL '7 days'
            GROUP BY agent_id;
        """
