    class SkepticComputeAgentPydantic(PydanticAIAdapter):
        """Pydantic AI implementation of Skeptic agent.

        Preserves existing prompt (skeptic_v2) and transfer function.
        Switches transport to NativeOutput for zero parse failures.
        Shadow-only until calibrated (shadow_only=True).
        """

        agent_id = "skeptic_v2_pydantic"
        prompt_version = ACTIVE_VERSION
        group = "alpha"
        tiers_needed = frozenset({Tier.I1, Tier.I4, Tier.I6, Tier.I7, Tier.SMC})
        latency_budget_ms = 120000.0
        shadow_only = True  # Shadow mode per AGENT-EXEC-04

        def __init__(self, llm_chain: LLMProviderChain, **kwargs: Any) -> None:
            # Import Agent and NativeOutput at runtime (not in requirements yet)
            # When pydantic-ai is added to requirements, remove TYPE_CHECKING guard
            from pydantic_ai import Agent, NativeOutput

            # Configure Pydantic AI agent with NativeOutput for structured output
            pydantic_agent = Agent(
                name="skeptic",
                model=llm_chain._llm,  # Reuse existing Ollama model
                deps_type=None,  # No deps needed for basic Skeptic
                output_type=NativeOutput(SkepticResult),
                instructions=(
                    "OUTPUT ONLY RAW JSON. NO PROSE. NO EXPLANATION. NO PREAMBLE. "
                    "Your entire response must be a single JSON object starting with { and ending with }. "
                    "Schema: "
                    '{"failure_probability": float (0-1), "confidence": float (0-1), '
                    '"risk_factors": [str], "reasoning": str (max 100 words)}'
                ),
            )
            super().__init__(pydantic_agent=pydantic_agent, **kwargs)
            self._llm = llm_chain

        async def _to_agent_output(
            self, result: Any, context: AIContext
        ) -> AgentOutput:
            """Convert Pydantic AI result to canonical AgentOutput.

            result.output is guaranteed to be SkepticResult (NativeOutput).
            Applies transfer function: multiplier = (1.0 - fp) * confidence.
            """
            skepticism = result.output  # SkepticResult guaranteed by NativeOutput
            failure_probability = skepticism.failure_probability
            llm_confidence = skepticism.confidence
            multiplier = (1.0 - failure_probability) * llm_confidence

            return self._build_multiplier_output(
                context=context,
                multiplier=multiplier,
                confidence=llm_confidence,
                payload={
                    "failure_probability": failure_probability,
                    "risk_factors": skepticism.risk_factors,
                    "reasoning": skepticism.reasoning,
                },
                prompt_version=self.prompt_version,
            )
