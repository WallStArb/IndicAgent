```python
"""BaseWriterAgent — abstract base for all Kafka-to-DB writer agents.

Provides the standard consume → parse → buffer → flush → commit loop
with write-path reliability guarantees:
- Manual offset commit (only after successful _flush_batch)
- DLQ routing for unparseable payloads
- Bounded buffer with overflow metric
- Buffer depth Prometheus gauge
- Final flush on teardown
"""

from __future__ import annotations

import abc
import time
from typing import Any

import structlog
from prometheus_client import Counter as _Counter, Gauge as _Gauge

from src.core.agent.base import BaseAgent


class BaseWriterAgent(BaseAgent, abc.ABC):
    """Abstract base for writer agents that consume Kafka and write to DB."""

    BATCH_SIZE: int = 100
    FLUSH_INTERVAL_SECS: float = 5.0
    MAX_BUFFER_SIZE: int = 10_000
    BUFFER_ALERT_PCT: float = 0.80

    def __init__(
        self,
        name: str,
        metrics_port: int | None = None,
    ) -> None:
        super().__init__(name=name, metrics_port=metrics_port)
        self._buffer: list[Any] = []
        self._last_flush: float = 0.0
        self._consumer: Any = None  # Set in subclass _setup()

        # Metrics — use prometheus_client directly for label support
        agent_snake = name.lower().replace(" ", "_")
        self._buffer_depth_gauge = _Gauge(
            f"{agent_snake}_buffer_depth",
            f"Current buffer depth for {name}",
        )
        self._buffer_overflow_total = _Counter(
            f"{agent_snake}_buffer_overflow_total",
            f"Rows dropped due to buffer overflow in {name}",
        )

    @abc.abstractmethod
    def _topic_name(self) -> str:
        """Kafka topic to consume from."""

    @property
    @abc.abstractmethod
    def _consumer_group(self) -> str:
        """Kafka consumer group ID."""

    @abc.abstractmethod
    def _parse_payload(self, payload: dict) -> list | None:
        """Parse raw Kafka payload into rows for buffer.

        Return list of parsed rows to buffer, or None to route to DLQ.
        """

    @abc.abstractmethod
    async def _flush_batch(self, batch: list) -> None:
        """Write batch to database. Must NOT clear self._buffer — caller handles that."""

    def _dlq_topic(self) -> str | None:
        """Override to return DLQ topic name. None = log-only (default)."""
        return None

    async def _run(self) -> None:
        """Standard consume-parse-buffer-flush-commit loop."""
        self._last_flush = time.monotonic()

        while not self._stop_event.is_set():
            payload = await self._consumer.consume_one(timeout_ms=500)
            if payload is None:
                # Check time-based flush even with no message
                now = time.monotonic()
                if self._buffer and (now - self._last_flush) >= self.FLUSH_INTERVAL_SECS:
                    await self._do_flush()
                    self._last_flush = now
                continue

            # Parse payload
            rows = self._parse_payload(payload)
            if rows is None:
                # DLQ route
                dlq = self._dlq_topic()
                if dlq and hasattr(self, "_producer") and self._producer:
                    self._producer.produce(dlq, payload)
                else:
                    self.logger.warning("dlq_payload_dropped", payload_keys=list(payload.keys()) if isinstance(payload, dict) else "non-dict")
                continue

            # Buffer rows
            self._buffer.extend(rows)

            # Overflow guard
            if len(self._buffer) > self.MAX_BUFFER_SIZE:
                dropped = len(self._buffer) - self.MAX_BUFFER_SIZE
                self._buffer = self._buffer[-self.MAX_BUFFER_SIZE:]
                self._buffer_overflow_total.inc(dropped)
                self.logger.warning("buffer_overflow", dropped=dropped, max_size=self.MAX_BUFFER_SIZE)

            # Alert on high watermark
            if len(self._buffer) > self.BUFFER_ALERT_PCT * self.MAX_BUFFER_SIZE:
                self.logger.warning("buffer_high_watermark", depth=len(self._buffer), threshold=self.BUFFER_ALERT_PCT)

            self._buffer_depth_gauge.set(len(self._buffer))

            # Flush trigger
            now = time.monotonic()
            if len(self._buffer) >= self.BATCH_SIZE or (now - self._last_flush) >= self.FLUSH_INTERVAL_SECS:
                await self._do_flush()
                self._last_flush = now

    async def _do_flush(self) -> None:
        """Flush buffer and commit offset on success."""
        if not self._buffer:
            return
        batch = self._buffer[:]
        try:
            await self._flush_batch(batch)
            self._buffer.clear()
            self._buffer_depth_gauge.set(0)
            # Commit offset AFTER successful flush
            if self._consumer and hasattr(self._consumer, "commit"):
                await self._consumer.commit()
        except Exception:
            self.logger.exception("flush_failed", batch_size=len(batch))

    async def _teardown(self) -> None:
        """Final flush before shutdown."""
        if self._buffer:
            await self._do_flush()
```

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-68-04 | Tampering | _parse_payload | mitigate | Returns None for unparseable payloads; routes to DLQ rather than crashing or silently dropping |

| T-68-06 | Denial of Service | Unbounded buffer | mitigate | MAX_BUFFER_SIZE=10,000 hard cap; overflow drops oldest entries and increments counter |
| T-68-07 | Denial of Service | Offset commit before flush | mitigate | Manual offset commit ONLY after _flush_batch succeeds; prevents data loss on crash |
</threat_model>
