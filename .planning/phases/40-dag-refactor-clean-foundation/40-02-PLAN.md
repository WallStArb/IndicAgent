"""Quality Gate stage — applies Hurst×Entropy and drift penalty multipliers."""

from __future__ import annotations

from typing import Any

import structlog

from src.core.stream_keys import topic_quality_gated, topic_intelligence
from src.intelligence.schemas import IntelligenceEvent
from src.intelligence.stages.base import Stage
from src.monitoring.ks_drift_monitor import DRIFT_PENALTIES
from src.config.settings import Settings

logger = structlog.get_logger(__name__)


class QualityGateService(Stage):
    """Applies Hurst×Entropy and KS drift penalty multipliers.

    Inputs: intelligence:SYMBOL:TF (I7 plugin outputs)
    Outputs: quality_gated:SYMBOL:TF

    Attribution:
    - value_added: after - before (negative = suppression)
    - inputs: hurst_quality, entropy_quality, drift_penalty
    """

    def __init__(self, settings: Settings):
        super().__init__(
            stage_name="quality_gate",
            input_topic_builder=topic_intelligence,
            output_topic_builder=topic_quality_gated,
            settings=settings,
        )

    async def process(self, event: IntelligenceEvent) -> dict:
        """Apply quality multipliers to signal confidence."""
        result = event.model_dump()

        # Extract base confidence
        features = event.features if isinstance(event.features, dict) else {}
        before_confidence = features.get("confidence", 0.0)

        # Apply Hurst×Entropy quality gate
        hurst_quality = features.get("hurst_trend_quality", 1.0)
        entropy_quality = features.get("entropy_quality", 1.0)
        quality_multiplier = min(hurst_quality, entropy_quality)

        after_confidence = round(before_confidence * quality_multiplier, 4)

        # Apply drift penalty
        drift_penalty = DRIFT_PENALTIES.get(event.symbol, 1.0)
        before_drift = after_confidence
        after_confidence = round(after_confidence * drift_penalty, 4)

        # Update result
        result["features"]["confidence"] = after_confidence

        # Attribution metadata
        result["attribution_reason"] = (
            f"Quality gate:hurst={hurst_quality:.2f},entropy={entropy_quality:.2f},"
            f"drift={drift_penalty:.2f}"
        )
        result["attribution_inputs"] = {
            "hurst_quality": hurst_quality,
            "entropy_quality": entropy_quality,
            "drift_penalty": drift_penalty,
        }

        logger.debug(
            "[quality_gate] Applied multipliers",
            symbol=event.symbol,
            timeframe=event.timeframe,
            before=before_confidence,
            after=after_confidence,
            hurst_quality=hurst_quality,
            entropy_quality=entropy_quality,
            drift_penalty=drift_penalty,
        )

        return result
```

Key implementation points:
- Reads hurst_trend_quality and entropy_quality from event.features
- Multiplies base confidence by min(hurst_quality, entropy_quality)
- Applies symbol-specific drift penalty from DRIFT_PENALTIES
- Stores attribution reason and inputs for emission
- Updates features["confidence"] in place
- Reuses existing DRIFT_PENALTIES from ks_drift_monitor.py

"""Regime Gate stage — suppresses signals based on HMM regime."""

from __future__ import annotations

import structlog

from src.core.stream_keys import topic_quality_gated, topic_regime_gated
from src.intelligence.schemas import IntelligenceEvent
from src.intelligence.stages.base import Stage
from src.config.settings import Settings

logger = structlog.get_logger(__name__)

# Regime type map from aggregator.py
_REGIME_MAP = {
    "trend": [1, 2],
    "mean_reversion": [0],
    "any": [0, 1, 2],
}

_REGIME_PROB_MIN = 0.55
_REGIME_DUR_MIN = 3


class RegimeGateService(Stage):
    """Suppresses signals based on HMM regime.

    Inputs: quality_gated:SYMBOL:TF
    Outputs: regime_gated:SYMBOL:TF

    Attribution:
    - value_added: 0.0 (gate doesn't change confidence, just suppresses)
    - reason: suppression reason if suppressed, or "Regime eligible"
    """

    def __init__(self, settings: Settings):
        super().__init__(
            stage_name="regime_gate",
            input_topic_builder=topic_quality_gated,
            output_topic_builder=topic_regime_gated,
            settings=settings,
        )

    async def process(self, event: IntelligenceEvent) -> dict:
        """Apply regime gating to signal."""
        result = event.model_dump()

        # Extract regime data from features (I4 HMM regime)
        features = event.features if isinstance(event.features, dict) else {}
        regime_data = features.get("regime_data")

        # If no regime data, skip gate (pass through)
        if regime_data is None:
            result["regime_eligible"] = True
            result["suppression_reason"] = None
            result["attribution_reason"] = "No regime data, gate skipped"
            return result

        hmm_regime = regime_data.get("hmm_regime")
        hmm_regime_prob = float(regime_data.get("hmm_regime_prob", 0.0))
        hmm_regime_duration = int(regime_data.get("hmm_regime_duration", 0))

        # Get plugin's regime_type (from I7 setup metadata)
        plugin_regime_type = result.get("regime_type", "any")
        allowed = _REGIME_MAP.get(plugin_regime_type, [0, 1, 2])

        # Gate priority: prob → duration → type
        regime_eligible = True
        suppression_reason = None

        if hmm_regime_prob < _REGIME_PROB_MIN:
            regime_eligible = False
            suppression_reason = "regime_prob"
        elif hmm_regime_duration < _REGIME_DUR_MIN:
            regime_eligible = False
            suppression_reason = "regime_duration"
        elif hmm_regime is not None and int(hmm_regime) not in allowed:
            regime_eligible = False
            suppression_reason = "regime_type"

        result["regime_eligible"] = regime_eligible
        result["suppression_reason"] = suppression_reason
        result["attribution_reason"] = (
            f"Regime gate: {suppression_reason}" if not regime_eligible else "Regime eligible"
        )
        result["attribution_inputs"] = {
            "hmm_regime": hmm_regime,
            "hmm_regime_prob": hmm_regime_prob,
            "hmm_regime_duration": hmm_regime_duration,
            "plugin_regime_type": plugin_regime_type,
        }

        logger.debug(
            "[regime_gate] Applied regime gate",
            symbol=event.symbol,
            timeframe=event.timeframe,
            regime_eligible=regime_eligible,
            suppression_reason=suppression_reason,
        )

        return result
```

Key implementation points:
- Reads regime_data from event.features
- Uses same gate logic as aggregator._regime_gate_signals()
- Sets regime_eligible and suppression_reason flags
- Doesn't modify confidence (gate is binary: suppress/pass)
- Passes through if regime_data is None (graceful degradation)
- Reuses _REGIME_MAP, _REGIME_PROB_MIN, _REGIME_DUR_MIN constants

"""Time-of-Day Adjuster stage — applies TOD multipliers."""

from __future__ import annotations

import zoneinfo
from datetime import datetime, UTC

import numpy as np
import structlog

from src.core.stream_keys import topic_regime_gated, topic_tod_adjusted
from src.intelligence.schemas import IntelligenceEvent
from src.intelligence.stages.base import Stage
from src.config.settings import Settings

logger = structlog.get_logger(__name__)

# TOD constants from signal_generator_service.py
_TOD_SESSION_PRIORS = {
    "regular_trading": {
        0: 0.48,  # Midnight
        1: 0.48,
        # ... (simplified for example, use full table from service)
        9: 0.52,  # 9am ET
        10: 0.54,  # 10am ET
        14: 0.58,  # 2pm ET (bullish window)
        16: 0.50,  # 4pm ET
    },
}
_TOD_ALPHA = 20.0  # Smoothing parameter
_TOD_CLAMP = (0.7, 1.3)  # Min/max multiplier bounds


class TODAdjusterService(Stage):
    """Applies time-of-day win rate multipliers.

    Multipliers grouped by (regime_type, tf, hour_et) — 120 cells.
    Uses Bayesian smoothing with alpha=20.0 and clamps to [0.7, 1.3].

    Inputs: regime_gated:SYMBOL:TF
    Outputs: tod_adjusted:SYMBOL:TF

    Attribution:
    - value_added: after - before (TOD adjustment)
    - inputs: tod_multiplier, regime_type, tf, hour_et
    """

    def __init__(self, settings: Settings):
        super().__init__(
            stage_name="tod_adjuster",
            input_topic_builder=topic_regime_gated,
            output_topic_builder=topic_tod_adjusted,
            settings=settings,
        )
        self.tod_multipliers = {}  # Loaded from DB every 4h
        self._last_multiplier_load = 0.0

    async def process(self, event: IntelligenceEvent) -> dict:
        """Apply TOD multiplier to signal confidence."""
        result = event.model_dump()

        # Extract current state
        features = event.features if isinstance(event.features, dict) else {}
        before_confidence = features.get("confidence", 0.0)

        # Get regime_type, tf, hour_et
        regime_type = result.get("regime_type_at_fire", "any")
        tf = event.timeframe
        timestamp = event.timestamp
        if isinstance(timestamp, str):
            timestamp = datetime.fromisoformat(timestamp.replace("Z", "+00:00"))
        hour_et = timestamp.astimezone(zoneinfo.ZoneInfo("America/New_York")).hour

        # Lookup TOD multiplier
        cell_key = (regime_type, tf, hour_et)
        tod_multiplier = self.tod_multipliers.get(cell_key)

        if tod_multiplier is None:
            # Fallback to session priors if DB multiplier not loaded
            session = "regular_trading" if 9 <= hour_et <= 16 else "outside_rth"
            tod_multiplier = _TOD_SESSION_PRIORS.get(session, {}).get(hour_et, 1.0)

        # Clamp to bounds
        tod_multiplier = max(_TOD_CLAMP[0], min(_TOD_CLAMP[1], tod_multiplier))

        # Apply multiplier
        after_confidence = round(before_confidence * tod_multiplier, 4)

        result["features"]["confidence"] = after_confidence
        result["tod_multiplier"] = tod_multiplier
        result["attribution_reason"] = f"TOD:{regime_type},{tf},{hour_et}h={tod_multiplier:.2f}"
        result["attribution_inputs"] = {
            "tod_multiplier": tod_multiplier,
            "regime_type": regime_type,
            "timeframe": tf,
            "hour_et": hour_et,
        }

        logger.debug(
            "[tod_adjuster] Applied TOD multiplier",
            symbol=event.symbol,
            timeframe=tf,
            regime_type=regime_type,
            hour_et=hour_et,
            tod_multiplier=tod_multiplier,
            before=before_confidence,
            after=after_confidence,
        )

        return result
```

Key implementation points:
- Groups multipliers by (regime_type, tf, hour_et) — 120 cells total
- Falls back to _TOD_SESSION_PRIORS if DB not loaded
- Clamps multiplier to [0.7, 1.3]
- Stores tod_multiplier in result for downstream stages
- Converts UTC timestamp to ET hour
- Reuses TOD constants from signal_generator_service.py

"""Calibrator stage — applies isotonic regression confidence calibration."""

from __future__ import annotations

import numpy as np
import structlog

from src.core.stream_keys import topic_tod_adjusted, topic_calibrated
from src.intelligence.schemas import IntelligenceEvent
from src.intelligence.stages.base import Stage
from src.config.settings import Settings

logger = structlog.get_logger(__name__)


class CalibratorService(Stage):
    """Applies isotonic regression confidence calibration curves.

    Calibration curves stored per (plugin_name, timeframe).
    Uses np.interp to map raw confidence to calibrated confidence.

    Inputs: tod_adjusted:SYMBOL:TF
    Outputs: calibrated:SYMBOL:TF

    Attribution:
    - value_added: after - before (calibration adjustment)
    - inputs: raw_confidence, calibrated_confidence, plugin_name, tf
    """

    def __init__(self, settings: Settings):
        super().__init__(
            stage_name="calibrator",
            input_topic_builder=topic_tod_adjusted,
            output_topic_builder=topic_calibrated,
            settings=settings,
        )
        self.calibration_curves = {}  # {(plugin_name, tf): (breakpoints, values)}
        self._last_curve_load = 0.0

    async def process(self, event: IntelligenceEvent) -> dict:
        """Apply calibration curve to signal confidence."""
        result = event.model_dump()

        # Extract current state
        features = event.features if isinstance(event.features, dict) else {}
        raw_confidence = features.get("confidence", 0.0)

        # Get plugin name and timeframe
        plugin_name = result.get("setup_plugin", "unknown")
        tf = event.timeframe

        # Lookup calibration curve
        curve_key = (plugin_name, tf)
        curve = self.calibration_curves.get(curve_key)

        if curve is None:
            # No calibration curve available: pass through unchanged
            calibrated_confidence = raw_confidence
            result["attribution_reason"] = f"No calibration curve for {plugin_name}:{tf}"
        else:
            # Apply isotonic calibration
            breakpoints, values = curve
            calibrated_confidence = float(np.interp(raw_confidence, breakpoints, values))
            calibrated_confidence = round(calibrated_confidence, 4)

            result["attribution_reason"] = f"Calibrated:{plugin_name}:{tf}"
            result["attribution_inputs"] = {
                "raw_confidence": raw_confidence,
                "calibrated_confidence": calibrated_confidence,
                "plugin_name": plugin_name,
                "timeframe": tf,
            }

        # Update result
        result["features"]["confidence"] = calibrated_confidence
        result["calibrated_confidence"] = calibrated_confidence

        logger.debug(
            "[calibrator] Applied calibration",
            symbol=event.symbol,
            timeframe=tf,
            plugin_name=plugin_name,
            raw_confidence=raw_confidence,
            calibrated_confidence=calibrated_confidence,
        )

        return result
```

Key implementation points:
- Loads calibration curves from confidence_calibration table
- Curves keyed by (plugin_name, timeframe) tuple
- Uses np.interp for fast monotonic curve lookup
- Passes through unchanged if curve not available
- Stores both raw and calibrated confidence
- Reuses calibration curve loading logic from signal_generator_service.py

"""Ranker stage — sorts signals by performance-weighted priority."""

from __future__ import annotations

import structlog

from src.core.stream_keys import topic_calibrated, topic_ranked
from src.intelligence.schemas import IntelligenceEvent
from src.intelligence.stages.base import Stage
from src.config.settings import Settings

logger = structlog.get_logger(__name__)

# Plugin priority from aggregator.py
SETUP_PRIORITY = {
    "trad_MeanReversion": 1,
    "trad_SqueezeExpansion": 2,
    "trad_TrendFollowing": 3,
    "trad_MTFAlignment": 4,
    "trad_LiquiditySweepReclaim": 5,
}


class RankerService(Stage):
    """Sorts signals by performance-weighted priority.

    Computes adjusted_rank = priority × perf_multiplier.
    Sorts all signals by adjusted_rank (lowest = highest priority).

    Inputs: calibrated:SYMBOL:TF (single signal dict)
    Outputs: ranked:SYMBOL:TF (single signal with adjusted_rank)

    Attribution:
    - value_added: 0.0 (ranking doesn't change confidence)
    - inputs: priority, perf_multiplier, adjusted_rank
    """

    def __init__(self, settings: Settings):
        super().__init__(
            stage_name="ranker",
            input_topic_builder=topic_calibrated,
            output_topic_builder=topic_ranked,
            settings=settings,
        )
        self.perf_weights = {}  # {(plugin_name, tf): perf_multiplier}
        self._last_weight_load = 0.0

    async def process(self, event: IntelligenceEvent) -> dict:
        """Compute adjusted_rank for signal."""
        result = event.model_dump()

        # Extract plugin name and timeframe
        plugin_name = result.get("setup_plugin", "unknown")
        tf = event.timeframe

        # Get priority
        priority = SETUP_PRIORITY.get(plugin_name, 999)

        # Get performance multiplier
        weight_key = (plugin_name, tf)
        perf_multiplier = self.perf_weights.get(weight_key, 1.0)

        # Compute adjusted_rank
        adjusted_rank = priority * perf_multiplier

        result["adjusted_rank"] = round(adjusted_rank, 4)
        result["perf_multiplier"] = perf_multiplier
        result["attribution_reason"] = f"Ranked:pri={priority},perf={perf_multiplier:.2f}"
        result["attribution_inputs"] = {
            "priority": priority,
            "perf_multiplier": perf_multiplier,
            "adjusted_rank": adjusted_rank,
        }

        logger.debug(
            "[ranker] Computed adjusted rank",
            symbol=event.symbol,
            timeframe=tf,
            plugin_name=plugin_name,
            priority=priority,
            perf_multiplier=perf_multiplier,
            adjusted_rank=adjusted_rank,
        )

        return result
```

Key implementation points:
- Computes adjusted_rank = priority × perf_multiplier
- Loads perf_weights from setup_performance table every 15 min
- Uses SETUP_PRIORITY dict from aggregator.py
- Lower adjusted_rank = higher priority (for sorting)

"""Winner Selector stage — selects winning signal using CIS or priority/majority."""

from __future__ import annotations

from collections import defaultdict
from typing import Any

import structlog
from pydantic import ValidationError

from src.core.stream_keys import topic_ranked, topic_winner
from src.intelligence.schemas import IntelligenceEvent
from src.intelligence.stages.base import Stage

logger = structlog.get_logger(__name__)

_CONFIDENCE_BOOST_PER_AGREE = 0.05
_REGIME_TIEBREAK_THRESHOLD = 0.4


class WinnerSelectorService(Stage):
    """Selects winning signal using CIS or priority/majority tiebreak.

    Collects all ranked signals for the current bar, applies CIS override
    if available, otherwise picks by priority/majority/regime tiebreak.

    Inputs: ranked:SYMBOL:TF (stream of ranked signals)
    Outputs: winner:SYMBOL:TF (single winner or None)

    Attribution:
    - value_added: 0.0 (selection doesn't change confidence)
    - inputs: resolution_method, num_signals_fired, winner_plugin_name
    """

    def __init__(self, settings: Settings):
        super().__init__(
            stage_name="winner_selector",
            input_topic_builder=topic_ranked,
            output_topic_builder=topic_winner,
            settings=settings,
        )
        self._bar_buffer = defaultdict(list)  # {(symbol, tf): [signals]}

    async def process(self, event: IntelligenceEvent) -> dict:
        """Buffer signals and select winner when bar complete."""
        # Buffer signal
        key = (event.symbol, event.timeframe)
        self._bar_buffer[key].append(event.model_dump())

        # TODO: Detect bar completion (needs bar boundary marker)
        # For now, return immediately (will be fixed in integration)
        result = {"status": "buffered", "signal_count": len(self._bar_buffer[key])}
        return result

    async def _select_winner_for_bar(
        self,
        symbol: str,
        tf: str,
        signals: list[dict],
    ) -> dict | None:
        """Select winner from all signals for a bar."""
        if not signals:
            return None

        # Separate active (regime_eligible=True) from suppressed
        active = [s for s in signals if s.get("regime_eligible", True)]

        if not active:
            return {
                "selected_signal": None,
                "all_ranked": signals,
                "resolution_method": "no_signal",
                "num_signals_fired": 0,
            }

        # Run CIS if features available
        features = signals[0].get("features", {})
        cis_result = None
        if features:
            scorer = CISScorer()
            plugin_outputs = {s["setup_plugin"]: s for s in signals}
            cis_result = scorer.score(features, plugin_outputs)

        # CIS override path
        if cis_result and cis_result.direction != 0:
            return self._aggregate_via_cis(active, cis_result, signals)

        # Fallback path: priority / majority / regime
        return self._aggregate_fallback(active, features, signals)

    def _aggregate_via_cis(
        self,
        active: list[dict],
        cis_result: Any,
        all_ranked: list[dict],
    ) -> dict:
        """Select winner matching CIS direction."""
        cis_direction = cis_result.direction
        matching = [s for s in active if s.get("direction", 0) == cis_direction]

        if not matching:
            # No signals match CIS direction: fallback to priority
            return self._aggregate_fallback(active, {}, all_ranked)

        # Sort by adjusted_rank
        matching = sorted(matching, key=lambda s: s.get("adjusted_rank", 999))
        selected = matching[0]

        # Boost confidence from agreeing plugins
        extra_agreeing = len(matching) - 1
        boosted_confidence = min(
            1.0,
            selected.get("confidence", 0.0) + _CONFIDENCE_BOOST_PER_AGREE * extra_agreeing,
        )
        selected["confidence"] = round(boosted_confidence, 4)

        return {
            "selected_signal": selected,
            "all_ranked": all_ranked,
            "resolution_method": "cis_override",
            "num_signals_fired": len(all_ranked),
            "num_agreeing": len(matching),
        }

    def _aggregate_fallback(
        self,
        active: list[dict],
        features: dict,
        all_ranked: list[dict],
    ) -> dict:
        """Select winner by priority/majority/regime tiebreak."""
        # Group by direction
        by_direction = defaultdict(list)
        for s in active:
            by_direction[s.get("direction", 0)].append(s)

        # Count agreeing vs conflicting
        longs = len(by_direction.get(1, []))
        shorts = len(by_direction.get(-1, []))
        agreeing = max(longs, shorts)
        conflicting = min(longs, shorts)

        # Pick group with majority
        if longs >= shorts:
            group = by_direction[1]
        else:
            group = by_direction[-1]

        # Sort by adjusted_rank
        group = sorted(group, key=lambda s: s.get("adjusted_rank", 999))
        selected = group[0]

        return {
            "selected_signal": selected,
            "all_ranked": all_ranked,
            "resolution_method": "priority_majority",
            "num_signals_fired": len(all_ranked),
            "num_agreeing": agreeing,
            "num_conflicting": conflicting,
        }
```

Key implementation points:

- Runs CIS scorer if features available
- CIS override: picks highest-priority signal matching CIS direction
- Fallback: majority direction, then priority sort
- Returns selected_signal + all_ranked + resolution metadata
- Reuses CIS scoring logic from aggregator.py
- Handles tiebreak cases (no signals, no CIS match, etc.)
