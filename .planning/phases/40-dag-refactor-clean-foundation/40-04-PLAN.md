```python
async def _consume_winner_signals(self):
    """Consume winner signals from DAG pipeline and write to DB.

    Runs alongside existing _process_bar() loop.
    """
    async for msg in self.winner_consumer:
        try:
            # Parse winner event
            if isinstance(msg, dict):
                raw_event = msg
            else:
                raw_event = json.loads(msg.value)

            event = IntelligenceEvent(**raw_event)

            # Extract selected_signal from event
            # Winner stage produces: {selected_signal: dict, all_ranked: list, ...}
            winner_data = raw_event.get("selected_signal")

            if not winner_data:
                # No winner (no signal this bar)
                continue

            # Create LedgerEntry from winner signal
            # CRITICAL: Use SignalStatus enum, not raw string "pending"
            entry = LedgerEntry(
                symbol=event.symbol,
                timeframe=event.timeframe,
                signal_computed_at=event.timestamp,
                status=SignalStatus.PENDING.value,  # Enum, not "pending" string
                outcome=SignalOutcome.PENDING.value,  # Enum, not "pending" string
                # ... other fields from winner_data
            )

            # Write to signal_ledger + signal_features
            await insert_signals_with_features(
                entries=[entry],
                features=event.features,
                conn=self.conn,
            )

            # Publish to signals:SYMBOL:TF:aggregated (for lifecycle service)
            await self.signals_aggregated_producer.send(
                symbol=event.symbol,
                timeframe=event.timeframe,
                value=winner_data,
            )

            logger.debug(
                "[signal_generator] Wrote winner signal to DB",
                symbol=event.symbol,
                timeframe=event.timeframe,
                setup_plugin=winner_data.get("setup_plugin"),
            )

        except Exception as e:
            logger.exception(
                "[signal_generator] Failed to consume winner signal",
                error=str(e),
            )
```

```python
from __future__ import annotations

import asyncio

@pytest.mark.integration
async def test_dag_pipeline_e2e():
    """Test signal flow through all 6 DAG stages.

    Flow:
    1. Publish intelligence event (I7 plugin output)
    2. Verify quality_gated event
    3. Verify regime_gated event
    4. Verify tod_adjusted event
    5. Verify calibrated event
    6. Verify ranked event
    7. Verify winner event
    8. Verify attribution events
    """
    settings = Settings()

    # Create producer for initial event
    producer = KafkaProducerClient(
        bootstrap_servers=settings.kafka_bootstrap_servers,
        topic_builder=topic_intelligence,
    )

    # Create consumer for final output
    from src.core.kafka_utils import KafkaConsumerClient
    winner_consumer = KafkaConsumerClient(
        bootstrap_servers=settings.kafka_bootstrap_servers,
        group_id=f"test_dag_e2e_{uuid4()}",
        topic_builder=topic_winner,
        auto_offset_reset="latest",
    )

    # Attribution consumer
    attribution_consumer = KafkaConsumerClient(
        bootstrap_servers=settings.kafka_bootstrap_servers,
        group_id=f"test_dag_attribution_{uuid4()}",
        topic_builder=topic_attribution,
        auto_offset_reset="latest",
    )

    # Create test intelligence event (I7 plugin output)
    test_event = IntelligenceEvent(
        symbol="ES",
        timeframe="1m",
        timestamp=datetime.now(UTC),
        signal_id=str(uuid4()),
        features={
            "confidence": 0.50,
            "hurst_trend_quality": 0.80,
            "entropy_quality": 0.70,
            "regime_data": {
                "hmm_regime": 1,
                "hmm_regime_prob": 0.75,
                "hmm_regime_duration": 10,
            },
        },
        setup_plugin="trad_TrendFollowing",
        direction=1,
        regime_type="trend",
    )

    # Publish to intelligence topic (DAG input)
    await producer.send(
        symbol=test_event.symbol,
        timeframe=test_event.timeframe,
        value=test_event.model_dump(),
    )

    # Wait for winner event (timeout 10s)
    winner_received = False
    attribution_count = 0

    async for msg in winner_consumer:
        if winner_received and attribution_count >= 6:
            break

        try:
            data = json.loads(msg.value) if isinstance(msg.value, (str, bytes)) else msg

            if data.get("selected_signal"):
                winner_received = True
                print(f"✓ Winner received: {data['selected_signal'].get('setup_plugin')}")

                # Verify winner structure
                assert "selected_signal" in data
                assert "all_ranked" in data
                assert "resolution_method" in data
                assert data["selected_signal"]["setup_plugin"] == "trad_TrendFollowing"

        except Exception as e:
            print(f"Error parsing winner: {e}")

    # Verify attribution events
    async for msg in attribution_consumer:
        if attribution_count >= 6:
            break

        try:
            data = json.loads(msg.value) if isinstance(msg.value, (str, bytes)) else msg

            if data.get("stage"):
                attribution_count += 1
                print(f"✓ Attribution received: {data['stage']}")

                # Verify attribution structure
                assert "stage" in data
                assert "before" in data
                assert "after" in data
                assert "value_added" in data
                assert data["stage"] in [
                    "quality_gate",
                    "regime_gate",
                    "tod_adjuster",
                    "calibrator",
                    "ranker",
                    "winner_selector",
                ]

        except Exception as e:
            print(f"Error parsing attribution: {e}")

    # Assertions
    assert winner_received, "Winner event not received"
    assert attribution_count >= 6, f"Expected 6 attribution events, got {attribution_count}"

    print("✓ DAG pipeline E2E test passed")
```
