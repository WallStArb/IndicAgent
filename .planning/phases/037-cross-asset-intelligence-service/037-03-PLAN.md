```python
elif self._cross_asset_enabled and topic == _cross_asset_topic:
    try:
        import json
        payload_data = json.loads(payload) if isinstance(payload, (str, bytes)) else payload
        # Cross-asset features are group-level, not symbol-level.
        # Store for each group member symbol at this tf.
        tf = payload_data.get("tf", "")
        group = payload_data.get("group", "")
        if not tf or not group:
            return

        # Persist cross_asset features by merging into i7 JSONB
        # for each group member's most recent intelligence_features row
        cross_asset_data = {
            "cross_asset": {
                "es_nq_spread_z": payload_data.get("es_nq_spread_z"),
                "es_rty_spread_z": payload_data.get("es_rty_spread_z"),
                "eq_corr_break": payload_data.get("eq_corr_break"),
                "eq_vol_imbalance": payload_data.get("eq_vol_imbalance"),
                "active_pair": payload_data.get("active_pair"),
                "pairs_confirming": payload_data.get("pairs_confirming"),
                "data_quality_score": payload_data.get("data_quality_score"),
            }
        }
        # Buffer for batch write using existing _buffer mechanism
        # Use the group name as a virtual symbol for the buffer key
        self._buffer.append({
            "type": "cross_asset",
            "tf": tf,
            "group": group,
            "data": cross_asset_data,
            "ts": payload_data.get("ts"),
        })
    except Exception as e:
        self.logger.warning("cross_asset_write_failed", error=str(e))
```
