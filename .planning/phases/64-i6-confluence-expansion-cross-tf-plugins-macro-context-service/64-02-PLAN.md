```python
"""Cross-timeframe support/resistance confluence plugin.

Computes continuous gradient [0, 1] measuring how well S/R levels
from different timeframes align at current price. Multiple TFs agreeing
on a level = strong confluence. Uses threshold_decay() for proximity scoring.

Outputs:
    ctf_sr_confluence: float [0, 1]
        Near 1.0: Price near S/R confluence from multiple TFs (strong signal)
        Near 0.5: Price moderately distant from levels
        Near 0.0: Price far from any S/R levels (weak/conflicting)
    ctf_sr_regime: str
        aligned_confluence: Multiple TFs agree on S/R level
        aligned_divergence: TFs agree price is AWAY from levels
        mixed: Conflicting signals
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

import numpy as np

from ..plugins import InputSpec
from ..utils.gradient_utils import threshold_decay
from .confluence_weights import _TF_MINUTES, get_recency_weight


@dataclass
class CrossTFSRConfluencePlugin:
    """Cross-TF S/R confluence plugin.

    Reads swing_high/low from I3 structure intelligence and computes
    how well S/R levels align across timeframes at current price.
    """

    name: str = "i6_CrossTFSRConfluence"
    outputs: frozenset[str] = frozenset({
        "ctf_sr_confluence",
        "ctf_sr_regime",
    })
    min_lookback: int = 1
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"confluence", "cross_tf", "structure"})
    inputs: list[InputSpec] = ()
    _state: dict = field(default_factory=dict)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """Compute S/R confluence across timeframes."""
        features = frames.get("features") or {}
        current_tf = frames.get("timeframe", "")
        current_price = features.get("close")

        if current_price is None:
            return {
                "ctf_sr_confluence": 0.0,
                "ctf_sr_regime": "mixed",
            }

        # Collect cross-TF structure data
        structure_data: dict[str, dict[str, Any]] = {}
        for key, val in frames.items():
            if key.startswith("intel_") and isinstance(val, dict):
                tf = key[6:]
                # Check if this TF has structure data (swing_high/low)
                if "swing_high" in val or "swing_low" in val:
                    structure_data[tf] = val

        if not structure_data:
            return {
                "ctf_sr_confluence": 0.0,
                "ctf_sr_regime": "mixed",
            }

        # Compute confluence score
        confluence_score = 0.0
        total_weight = 0.0

        for tf, intel in structure_data.items():
            weight = get_recency_weight(frames, tf)
            if weight <= 0:
                continue

            # Get S/R levels from this TF
            swing_high = intel.get("swing_high")
            swing_low = intel.get("swing_low")

            # Skip if no levels
            if swing_high is None and swing_low is None:
                continue

            # Compute proximity to nearest level
            prox_score = 0.0
            atr = features.get("atr", 1.0)  # Default to avoid div/0

            if swing_high is not None:
                # threshold_decay: peaks at swing_high, decays over distance
                dist_high = abs(current_price - swing_high) / atr
                prox_high = threshold_decay(dist_high, center=0, width=2.0)
                prox_score = max(prox_score, prox_high)

            if swing_low is not None:
                dist_low = abs(current_price - swing_low) / atr
                prox_low = threshold_decay(dist_low, center=0, width=2.0)
                prox_score = max(prox_score, prox_low)

            confluence_score += weight * prox_score
            total_weight += weight

        if total_weight == 0:
            return {
                "ctf_sr_confluence": 0.0,
                "ctf_sr_regime": "mixed",
            }

        final_score = confluence_score / total_weight

        # Classify regime
        if final_score > 0.6:
            regime = "aligned_confluence"
        elif final_score < 0.2:
            regime = "aligned_divergence"  # Price far from levels
        else:
            regime = "mixed"

        return {
            "ctf_sr_confluence": round(final_score, 4),
            "ctf_sr_regime": regime,
        }

    def compute_next(self, windows: dict[str, Any]) -> dict[str, Any]:
        return self.compute_full(windows)
```

Key points:
- Uses threshold_decay for proximity-based scoring
- ATR-normalized distance (relative to volatility)
- Weighted by recency (stale intel contributes less)
- Regime classification based on confluence strength

```python
"""Cross-timeframe HMM regime agreement plugin.

Computes continuous gradient [0, 1] measuring HMM regime consistency
across timeframes. All TFs in same regime = high agreement.

Outputs:
    ctf_regime_agreement: float [0, 1]
        1.0: All TFs in same regime (high agreement)
        0.5: Mixed regimes
        0.0: Regimes fully opposed
    ctf_regime_label: str
        trending: All TFs in trending regime (1 or 2)
        ranging: All TFs in ranging regime (0)
        mixed: Conflicting regimes
"""

from __future__ import annotations

from collections import Counter
from dataclasses import dataclass, field
from typing import Any

from ..plugins import InputSpec
from ..utils.gradient_utils import linear_ramp
from .confluence_weights import _TF_MINUTES, get_recency_weight


@dataclass
class CrossTFRegimeAgreementPlugin:
    """Cross-TF HMM regime agreement plugin.

    Reads hmm_regime, hmm_prob_* from I4 context and computes
    how consistent regimes are across timeframes.
    """

    name: str = "i6_CrossTFRegimeAgreement"
    outputs: frozenset[str] = frozenset({
        "ctf_regime_agreement",
        "ctf_regime_label",
    })
    min_lookback: int = 1
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"confluence", "cross_tf", "regime"})
    inputs: list[InputSpec] = ()
    _state: dict = field(default_factory=dict)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """Compute regime agreement across timeframes."""
        features = frames.get("features") or {}
        current_tf = frames.get("timeframe", "")

        # Collect cross-TF regime data
        regime_data: dict[str, dict[str, Any]] = {}
        for key, val in frames.items():
            if key.startswith("intel_") and isinstance(val, dict):
                tf = key[6:]
                if "hmm_regime" in val:
                    regime_data[tf] = val

        if not regime_data:
            return {
                "ctf_regime_agreement": 0.5,
                "ctf_regime_label": "mixed",
            }

        # Weighted regime counts
        regime_weights: dict[int, float] = {}
        total_weight = 0.0

        for tf, intel in regime_data.items():
            weight = get_recency_weight(frames, tf)
            if weight <= 0:
                continue

            regime = intel.get("hmm_regime")
            if regime is None:
                continue

            # Group trending regimes (1 and 2) together
            regime_key = 1 if regime in (1, 2) else 0
            regime_weights[regime_key] = regime_weights.get(regime_key, 0.0) + weight
            total_weight += weight

        if total_weight == 0:
            return {
                "ctf_regime_agreement": 0.5,
                "ctf_regime_label": "mixed",
            }

        # Compute agreement: max weight / total weight
        # 1.0 = all TFs in same regime, 0.5 = split
        max_weight = max(regime_weights.values())
        agreement = max_weight / total_weight

        # Classify overall regime
        # 1 = trending (grouped), 0 = ranging
        trending_weight = regime_weights.get(1, 0.0)
        ranging_weight = regime_weights.get(0, 0.0)

        if agreement > 0.7:
            if trending_weight > ranging_weight:
                label = "trending"
            else:
                label = "ranging"
        else:
            label = "mixed"

        return {
            "ctf_regime_agreement": round(agreement, 4),
            "ctf_regime_label": label,
        }

    def compute_next(self, windows: dict[str, Any]) -> dict[str, Any]:
        return self.compute_full(windows)
```

Key points:
- Groups trending regimes (1 and 2) together for binary agreement
- Weighted by recency
- Agreement = max regime weight / total weight
- Regime label based on which regime dominates

```python
"""Cross-timeframe squeeze/expansion divergence plugin.

Computes continuous gradient [-1, +1] measuring volatility regime
divergence across timeframes. Squeeze in HTF + expansion in LTF
= potential breakout signal.

Outputs:
    ctf_volatility_divergence: float [-1, +1]
        Positive: Aligned (both squeezing or both expanding)
        Negative: Divergent (HTF squeeze, LTF expansion or vice versa)
        Near 0: Weak or conflicting signals
    ctf_volatility_regime: str
        aligned_squeeze: All TFs in squeeze
        aligned_expansion: All TFs in expansion
        squeeze_divergence: HTF squeeze, LTF expansion (bullish setup)
        expansion_divergence: HTF expansion, LTF squeeze (bearish setup)
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from ..plugins import InputSpec
from ..utils.gradient_utils import linear_ramp, sigmoid_score
from .confluence_weights import _TF_MINUTES, get_recency_weight


@dataclass
class SqueezeExpansionDivergencePlugin:
    """Cross-TF squeeze/expansion divergence plugin.

    Reads squeeze_status, volatility from I4 context and computes
    volatility regime divergence across timeframes.
    """

    name: str = "i6_SqueezeExpansionDivergence"
    outputs: frozenset[str] = frozenset({
        "ctf_volatility_divergence",
        "ctf_volatility_regime",
    })
    min_lookback: int = 1
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"confluence", "cross_tf", "volatility"})
    inputs: list[InputSpec] = ()
    _state: dict = field(default_factory=dict)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """Compute squeeze/expansion divergence across timeframes."""
        features = frames.get("features") or {}
        current_tf = frames.get("timeframe", "")

        # Collect cross-TF volatility data
        vol_data: dict[str, dict[str, Any]] = {}
        for key, val in frames.items():
            if key.startswith("intel_") and isinstance(val, dict):
                tf = key[6:]
                # Check for squeeze or volatility data
                if "squeeze_status" in val or "volatility" in val:
                    vol_data[tf] = val

        if not vol_data:
            return {
                "ctf_volatility_divergence": 0.0,
                "ctf_volatility_regime": "mixed",
            }

        # Separate into HTF and LTF groups
        current_tf_min = _TF_MINUTES.get(current_tf, 1)
        htf_score = self._compute_vol_score(
            {tf: d for tf, d in vol_data.items() if _TF_MINUTES.get(tf, 1) > current_tf_min},
            frames,
        )
        ltf_score = self._compute_vol_score(
            {tf: d for tf, d in vol_data.items() if _TF_MINUTES.get(tf, 1) < current_tf_min},
            frames,
        )

        # If no HTF/LTF split possible, return overall score
        if htf_score is None or ltf_score is None:
            # Use overall score if available
            overall_score = self._compute_vol_score(vol_data, frames)
            if overall_score is None:
                return {
                    "ctf_volatility_divergence": 0.0,
                    "ctf_volatility_regime": "mixed",
                }
            # Classify based on overall score
            if overall_score > 0.5:
                return {
                    "ctf_volatility_divergence": overall_score,
                    "ctf_volatility_regime": "aligned_expansion",
                }
            else:
                return {
                    "ctf_volatility_divergence": overall_score,
                    "ctf_volatility_regime": "aligned_squeeze",
                }

        # Compute divergence: product for sign preservation
        # Both squeeze (0) or both expand (1) = positive (aligned)
        # One squeeze (0), one expand (1) = negative (divergent)
        divergence = (htf_score - 0.5) * (ltf_score - 0.5) * 4.0  # Scale to [-1, +1]

        # Classify regime
        htf_squeeze = htf_score < 0.4
        ltf_squeeze = ltf_score < 0.4

        if htf_squeeze and ltf_squeeze:
            regime = "aligned_squeeze"
        elif (not htf_squeeze) and (not ltf_squeeze):
            regime = "aligned_expansion"
        elif htf_squeeze and (not ltf_squeeze):
            regime = "squeeze_divergence"  # HTF squeeze, LTF expansion
        else:
            regime = "expansion_divergence"  # HTF expansion, LTF squeeze

        return {
            "ctf_volatility_divergence": round(divergence, 4),
            "ctf_volatility_regime": regime,
        }

    def _compute_vol_score(
        self, vol_group: dict[str, dict[str, Any]], frames: dict[str, Any]
    ) -> float | None:
        """Compute weighted volatility score for a TF group."""
        if not vol_group:
            return None

        total_weight = 0.0
        weighted_score = 0.0

        for tf, intel in vol_group.items():
            weight = get_recency_weight(frames, tf)
            if weight <= 0:
                continue

            # Get squeeze status if available (0=squeeze, 1=expand)
            squeeze_status = intel.get("squeeze_status")
            if squeeze_status is not None:
                score = float(squeeze_status)
            else:
                # Fall back to volatility (low vol = squeeze)
                vol = intel.get("volatility")
                if vol is None:
                    continue
                # Normalize volatility: low = squeeze (0), high = expand (1)
                # Use sigmoid for smooth transition
                score = sigmoid_score(vol, center=0.02, steepness=100.0)

            weighted_score += weight * score
            total_weight += weight

        if total_weight == 0:
            return None

        return weighted_score / total_weight

    def compute_next(self, windows: dict[str, Any]) -> dict[str, Any]:
        return self.compute_full(windows)
```

Key points:
- Reads squeeze_status (0=squeeze, 1=expand) or volatility
- Uses sigmoid_score to normalize volatility
- Product-based divergence calculation
- Regime classification for 4 states

```python
Computes continuous gradient [-1, +1] measuring orderflow direction
agreement across timeframes using OFI (order flow imbalance) and
CVD (cumulative volume delta). Aligned bullish/bearish flow across
TFs = stronger conviction.

Outputs:
    ctf_orderflow_alignment: float [-1, +1]
        Positive: Bullish flow alignment across TFs
        Negative: Bearish flow alignment across TFs
        Near 0: Conflicting or weak signals
    ctf_orderflow_regime: str
        aligned_bullish: All TFs showing bullish flow
        aligned_bearish: All TFs showing bearish flow
        divergent: Flow directions conflict across TFs
        mixed: Weak or conflicting signals
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from ..plugins import InputSpec
from ..utils.gradient_utils import linear_ramp, z_score_to_score
from .confluence_weights import _TF_MINUTES, get_recency_weight


@dataclass
class CrossTFOrderFlowAlignmentPlugin:
    """Cross-TF orderflow alignment plugin.

    Reads ofi, cvd from I1 indicators and computes orderflow
    direction agreement across timeframes.
    """

    name: str = "i6_CrossTFOrderFlowAlignment"
    outputs: frozenset[str] = frozenset({
        "ctf_orderflow_alignment",
        "ctf_orderflow_regime",
    })
    min_lookback: int = 1
    supports_incremental: bool = False
    capability_tags: frozenset[str] = frozenset({"confluence", "cross_tf", "orderflow"})
    inputs: list[InputSpec] = ()
    _state: dict = field(default_factory=dict)

    def compute_full(self, frames: dict[str, Any]) -> dict[str, Any]:
        """Compute orderflow alignment across timeframes."""
        features = frames.get("features") or {}
        current_tf = frames.get("timeframe", "")

        # Collect cross-TF orderflow data
        of_data: dict[str, dict[str, Any]] = {}
        for key, val in frames.items():
            if key.startswith("intel_") and isinstance(val, dict):
                tf = key[6:]
                # Check for OFI or CVD data
                if "ofi" in val or "cvd" in val:
                    of_data[tf] = val

        if not of_data:
            return {
                "ctf_orderflow_alignment": 0.0,
                "ctf_orderflow_regime": "mixed",
            }

        # Compute weighted orderflow score
        total_weight = 0.0
        weighted_score = 0.0

        for tf, intel in of_data.items():
            weight = get_recency_weight(frames, tf)
            if weight <= 0:
                continue

            # Get orderflow score from OFI or CVD
            ofi = intel.get("ofi")
            cvd = intel.get("cvd")

            if ofi is not None:
                # OFI: positive = bullish, negative = bearish
                # Normalize via z-score for continuous gradient
                score = z_score_to_score(ofi, sigma_scale=2.0)
                # Map [0, 1] to [-1, +1] centered at 0.5
                score = linear_ramp(score, 0.0, 1.0, -1.0, 1.0)
            elif cvd is not None:
                # CVD: positive = bullish, negative = bearish
                score = z_score_to_score(cvd, sigma_scale=2.0)
                score = linear_ramp(score, 0.0, 1.0, -1.0, 1.0)
            else:
                continue

            weighted_score += weight * score
            total_weight += weight

        if total_weight == 0:
            return {
                "ctf_orderflow_alignment": 0.0,
                "ctf_orderflow_regime": "mixed",
            }

        final_score = weighted_score / total_weight

        # Classify regime
        if final_score > 0.5:
            regime = "aligned_bullish"
        elif final_score < -0.5:
            regime = "aligned_bearish"
        else:
            regime = "divergent"  # Weak or conflicting

        return {
            "ctf_orderflow_alignment": round(final_score, 4),
            "ctf_orderflow_regime": regime,
        }

    def compute_next(self, windows: dict[str, Any]) -> dict[str, Any]:
        return self.compute_full(windows)
```

Key points:

- Uses z_score_to_score for normalization
- Weighted average across TFs
- Regime classification based on alignment strength
