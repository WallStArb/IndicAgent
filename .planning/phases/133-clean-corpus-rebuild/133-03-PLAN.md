    ```sql
    -- Migration 149: trade_frames hypertable conversion
    -- Phase 133: convert trade_frames to TimescaleDB hypertable on signal_ts.
    -- Must run while table is empty (or before full corpus rebuild) to avoid chunk migration overhead.
    -- 8-step sequence per .planning/todos/pending/2026-06-16-trade-frames-hypertable-migration.md

    -- Step 1: Drop FK from trade_executions -> trade_frames (UUID-only reference, being replaced)
    ALTER TABLE trade_executions
        DROP CONSTRAINT IF EXISTS fk_trade_executions_frame;

    -- Step 2: Drop existing UUID-only PK (hypertable requires partition key in PK)
    ALTER TABLE trade_frames
        DROP CONSTRAINT IF EXISTS trade_frames_pkey;

    -- Step 3: Create hypertable on signal_ts
    SELECT create_hypertable(
        'trade_frames',
        'signal_ts',
        chunk_time_interval => INTERVAL '7 days',
        if_not_exists => TRUE
    );

    -- Step 4: Recreate PK as (frame_id, signal_ts) — hypertable composite PK requirement
    ALTER TABLE trade_frames
        ADD CONSTRAINT trade_frames_pkey PRIMARY KEY (frame_id, signal_ts);

    -- Step 5: Add signal_ts to trade_executions as FK anchor (denormalization, same pattern as trade_frames.signal_ts)
    ALTER TABLE trade_executions
        ADD COLUMN IF NOT EXISTS signal_ts TIMESTAMPTZ NOT NULL DEFAULT NOW();
    -- Note: default NOW() is a migration convenience; the INSERT path will supply the real value.
    -- Remove the default after migration so new INSERTs are forced to provide the value explicitly.
    ALTER TABLE trade_executions
        ALTER COLUMN signal_ts DROP DEFAULT;

    -- Step 6: Recreate FK using composite (frame_id, signal_ts)
    ALTER TABLE trade_executions
        ADD CONSTRAINT fk_trade_executions_frame
        FOREIGN KEY (frame_id, signal_ts)
        REFERENCES trade_frames (frame_id, signal_ts)
        ON DELETE CASCADE;

    -- Step 7: Enable compression
    ALTER TABLE trade_frames SET (
        timescaledb.compress,
        timescaledb.compress_segmentby = 'symbol,tf',
        timescaledb.compress_orderby = 'signal_ts DESC'
    );

    -- Step 8: Add compression policy
    SELECT add_compression_policy('trade_frames', INTERVAL '7 days', if_not_exists => TRUE);
    ```
