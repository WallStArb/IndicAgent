```python
# src/intelligence/metrics/validator.py
"""Data quality validation for signal_ledger rows before metrics computation.

Four gates applied in order — first failure short-circuits.
NULL pnl_r (zone never activated) passes validation; it is excluded from
performance metrics but counted toward never_activated_pct.

Never modifies raw signal_ledger rows. Invalid rows are logged to
signal_metrics_dq_failures by the caller.
"""
from __future__ import annotations

from dataclasses import dataclass

# ES/NQ minimum tick size in points. Signals with |entry - stop| < this
# produce pnl_r blowups (CVDDivergence -496 Sharpe bug).
MIN_TICK_SIZE: float = 0.25

# Signals with |pnl_r| > this are data anomalies, not market moves.
# 10R in a single trade requires price to move 10× the initial risk — possible
# only with near-zero stops, which Gate 2 should catch first.
MAX_VALID_R: float = 10.0


@dataclass(frozen=True)
class ValidationResult:
    is_valid: bool
    reason_code: str | None  # None when valid


_VALID = ValidationResult(is_valid=True, reason_code=None)


def validate_signal_row(
    direction: int | None,
    entry_price: float | None,
    stop_loss: float | None,
    pnl_r: float | None,
    hmm_regime_at_fire: int | None,
) -> ValidationResult:
    """Apply 4 data quality gates to a signal_ledger row.

    Call with zone-track fields (pnl_r, entry_price, stop_loss) for zone
    validation, or with market-track fields for market validation.

    Args:
        direction:         +1 (long) or -1 (short)
        entry_price:       signal entry price (entry_price column)
        stop_loss:         signal stop loss (stop_loss column)
        pnl_r:             realised P&L in R-multiples; None = never activated
        hmm_regime_at_fire: HMM state at signal fire time (0/1/2)

    Returns:
        ValidationResult(is_valid=True, reason_code=None) when valid.
    """
    # NULL pnl_r = zone never activated. Valid — not a DQ failure.
    if pnl_r is None:
        return _VALID

    # Gate 1: direction must be +1 or -1
    if direction not in (1, -1):
        return ValidationResult(is_valid=False, reason_code="invalid_direction")

    # Gate 2: risk must be at least one tick (prevents pnl_r blowup)
    if entry_price is None or stop_loss is None:
        return ValidationResult(is_valid=False, reason_code="risk_below_min_tick")
    if abs(entry_price - stop_loss) < MIN_TICK_SIZE:
        return ValidationResult(is_valid=False, reason_code="risk_below_min_tick")

    # Gate 3: pnl_r magnitude must be plausible
    if abs(pnl_r) > MAX_VALID_R:
        return ValidationResult(is_valid=False, reason_code="pnl_r_outlier")

    # Gate 4: regime must be known for regime-conditioned segmentation
    if hmm_regime_at_fire is None:
        return ValidationResult(is_valid=False, reason_code="missing_regime")

    return _VALID
```

"""Pure compute functions for signal performance metrics.

No I/O. Called by SignalMetricsComputeAgent after data quality validation.
Two tracks: 'zone' (structural setup quality) and 'market' (tradeable alpha).
"""
from __future__ import annotations

import math
from collections import defaultdict
from dataclasses import dataclass
from datetime import UTC, datetime

from scipy.stats import t as _scipy_t

from src.intelligence.metrics.validator import validate_signal_row
from src.intelligence.ml.information_coefficient import (

# Minimum N gate — FEED-02: setups with fewer samples use multiplier=1.0 (neutral)

# Rolling windows to compute metrics for

# 0 = ranging market (mean-reversion conditions)
# 1 = uptrend, 2 = downtrend (both are 'trend' for metric segmentation)

    2: "trend",

def _p_value(avg_r: float, std_r: float, n: int) -> float | None:
    """Two-sided one-sample t-test: H0 = avg_r == 0."""
    if n < 2 or std_r < 1e-9 or math.isnan(std_r):
        return None
    t_stat = avg_r / (std_r / math.sqrt(n))
    return float(_scipy_t.sf(abs(t_stat), df=n - 1) * 2)


def _stats_from_values(
    pnl_rs: list[float],
    maes: list[float],
    mfes: list[float],
    win_flags: list[bool],
    n_never_activated: int,
    n_total: int,
    n_outliers: int,
    track: str,
    setup_plugin: str,
    tf: str,
    regime_type: str,
    window_days: int,
) -> SignalMetricsRow:
    n = len(pnl_rs)
    now = datetime.now(UTC)
    never_act_pct = round(n_never_activated / n_total, 4) if n_total > 0 else None

    if n == 0:
        return SignalMetricsRow(
            track=track, setup_plugin=setup_plugin, tf=tf,
            regime_type=regime_type, window_days=window_days,
            n=0, n_outliers=n_outliers,
            never_activated_pct=never_act_pct,
            win_rate=None, avg_r=None, std_r=None, sharpe=None,
            p_value=None, avg_mae=None, avg_mfe=None, computed_at=now,
        )

    avg_r = sum(pnl_rs) / n
    variance = sum((r - avg_r) ** 2 for r in pnl_rs) / (n - 1) if n > 1 else 0.0
    std_r = math.sqrt(variance) if variance > 0 else 0.0
    sharpe = round(avg_r / std_r, 4) if std_r > 1e-9 else None
    p_val = _p_value(avg_r, std_r, n)
    win_rate = sum(1 for w in win_flags if w) / len(win_flags) if win_flags else None
    avg_mae = sum(maes) / len(maes) if maes else None
    avg_mfe = sum(mfes) / len(mfes) if mfes else None

    return SignalMetricsRow(
        track=track, setup_plugin=setup_plugin, tf=tf,
        regime_type=regime_type, window_days=window_days,
        n=n, n_outliers=n_outliers,
        never_activated_pct=never_act_pct,
        win_rate=round(win_rate, 4) if win_rate is not None else None,
        avg_r=round(avg_r, 4),
        std_r=round(std_r, 4),
        sharpe=sharpe,
        p_value=round(p_val, 4) if p_val is not None else None,
        avg_mae=round(avg_mae, 4) if avg_mae is not None else None,
        avg_mfe=round(avg_mfe, 4) if avg_mfe is not None else None,
        computed_at=now,
    )


def compute_signal_metrics(
    rows: list[dict],
    track: str,
    window_days: int,
) -> list[SignalMetricsRow]:
    """Compute per-segment metrics for one track and window.

    Applies DataQualityValidator to each row. Invalid rows increment
    n_outliers and are excluded from avg_r/win_rate/sharpe.
    NULL pnl_r (zone never activated) counts toward never_activated_pct.

    Returns one SignalMetricsRow per (setup_plugin, tf, regime_type) group
    with n >= MIN_SAMPLE_SIZE, plus an 'all' rollup row per (setup, tf).

    Args:
        rows:       list of signal_ledger row dicts (already fetched from DB)
        track:      'zone' or 'market'
        window_days: rolling window (7, 30, or 90)
    """
    # Keyed by (setup_plugin, tf, regime_type)
    # Values: (pnl_rs, maes, mfes, win_flags, n_never_activated, n_total, n_outliers)
    groups: dict[tuple, list] = defaultdict(lambda: [[], [], [], [], 0, 0, 0])

    for row in rows:
        plugin = row.get("setup_plugin")
        tf = row.get("tf") or row.get("timeframe")
        hmm = row.get("hmm_regime_at_fire")
        if not plugin or not tf:
            continue

        if track == "zone":
            pnl_r = row.get("pnl_r")
            mae = row.get("mae")
            mfe = row.get("mfe")
            outcome = row.get("outcome")
        else:  # market
            pnl_r = row.get("market_entry_pnl_r")
            mae = row.get("market_entry_mae")
            mfe = row.get("market_entry_mfe")
            outcome = row.get("market_entry_outcome")

        regime_label = HMM_TO_REGIME.get(hmm, "unknown") if hmm is not None else "unknown"
        if regime_label == "unknown":
            continue  # missing regime — should be caught by validator upstream

        key = (plugin, tf, regime_label)
        g = groups[key]

        g[5] += 1  # n_total

        if pnl_r is None:
            g[4] += 1  # n_never_activated
            continue

        # Validate this row
        vr = validate_signal_row(
            direction=row.get("direction"),
            entry_price=row.get("entry_price"),
            stop_loss=row.get("stop_loss"),
            pnl_r=pnl_r,
            hmm_regime_at_fire=hmm,
        )
        if not vr.is_valid:
            g[6] += 1  # n_outliers
            continue

        g[0].append(pnl_r)
        if mae is not None:
            g[2].append(float(mae))
        if mfe is not None:
            g[3].append(float(mfe))  # note: index 3 is win_flags below — fix order
        g[3].append(outcome in WIN_OUTCOMES)  # win_flags at index 3

    # Fix: the above has a bug — maes at index 1, mfes at index 2, win_flags at index 3
    # Rewrite with explicit structure:
    # Actually let me restructure groups to be explicit dicts:

    # Re-do with explicit dict accumulation for clarity
    Acc = dict  # type alias
    accs: dict[tuple, Acc] = defaultdict(lambda: {
        "pnl_rs": [], "maes": [], "mfes": [], "win_flags": [],
        "n_never_activated": 0, "n_total": 0, "n_outliers": 0,
    })

    for row in rows:
        plugin = row.get("setup_plugin")
        tf_val = row.get("tf") or row.get("timeframe")
        hmm = row.get("hmm_regime_at_fire")
        if not plugin or not tf_val:
            continue

        if track == "zone":
            pnl_r = row.get("pnl_r")
            mae = row.get("mae")
            mfe = row.get("mfe")
            outcome = row.get("outcome")
        else:
            pnl_r = row.get("market_entry_pnl_r")
            mae = row.get("market_entry_mae")
            mfe = row.get("market_entry_mfe")
            outcome = row.get("market_entry_outcome")

        regime_label = HMM_TO_REGIME.get(hmm) if hmm is not None else None
        if not regime_label:
            continue

        key = (plugin, tf_val, regime_label)
        acc = accs[key]
        acc["n_total"] += 1

        if pnl_r is None:
            acc["n_never_activated"] += 1
            continue

        vr = validate_signal_row(
            direction=row.get("direction"),
            entry_price=row.get("entry_price"),
            stop_loss=row.get("stop_loss"),
            pnl_r=pnl_r,
            hmm_regime_at_fire=hmm,
        )
        if not vr.is_valid:
            acc["n_outliers"] += 1
            continue

        acc["pnl_rs"].append(float(pnl_r))
        if mae is not None:
            acc["maes"].append(float(mae))
        if mfe is not None:
            acc["mfes"].append(float(mfe))
        acc["win_flags"].append(outcome in WIN_OUTCOMES)

    result: list[SignalMetricsRow] = []

    # Per-regime rows (only if n >= MIN_SAMPLE_SIZE)
    # Also accumulate into 'all' rollup per (plugin, tf)
    all_accs: dict[tuple, Acc] = defaultdict(lambda: {
        "pnl_rs": [], "maes": [], "mfes": [], "win_flags": [],
        "n_never_activated": 0, "n_total": 0, "n_outliers": 0,
    })

    for (plugin, tf_val, regime_label), acc in accs.items():
        # Accumulate into 'all' rollup
        all_key = (plugin, tf_val)
        all_acc = all_accs[all_key]
        all_acc["pnl_rs"].extend(acc["pnl_rs"])
        all_acc["maes"].extend(acc["maes"])
        all_acc["mfes"].extend(acc["mfes"])
        all_acc["win_flags"].extend(acc["win_flags"])
        all_acc["n_never_activated"] += acc["n_never_activated"]
        all_acc["n_total"] += acc["n_total"]
        all_acc["n_outliers"] += acc["n_outliers"]

        n = len(acc["pnl_rs"])
        if n < MIN_SAMPLE_SIZE:
            continue

        result.append(_stats_from_values(
            pnl_rs=acc["pnl_rs"],
            maes=acc["maes"],
            mfes=acc["mfes"],
            win_flags=acc["win_flags"],
            n_never_activated=acc["n_never_activated"],
            n_total=acc["n_total"],
            n_outliers=acc["n_outliers"],
            track=track,
            setup_plugin=plugin,
            tf=tf_val,
            regime_type=regime_label,
            window_days=window_days,
        ))

    # 'all' rollup rows
    for (plugin, tf_val), acc in all_accs.items():
        n = len(acc["pnl_rs"])
        if n < MIN_SAMPLE_SIZE:
            continue
        result.append(_stats_from_values(
            pnl_rs=acc["pnl_rs"],
            maes=acc["maes"],
            mfes=acc["mfes"],
            win_flags=acc["win_flags"],
            n_never_activated=acc["n_never_activated"],
            n_total=acc["n_total"],
            n_outliers=acc["n_outliers"],
            track=track,
            setup_plugin=plugin,
            tf=tf_val,
            regime_type="all",
            window_days=window_days,
        ))

    return result


def compute_ic_metrics(
    rows: list[dict],
    window_days: int,
) -> list[ICMetricsRow]:
    """Compute IC per (setup_plugin, tf, regime_type) group.

    Uses src.intelligence.ml.information_coefficient.compute_ic().
    Returns rows with n >= IC_MIN_SAMPLE_SIZE (30).
    Also emits 'all' regime rollup per (setup, tf).

    Args:
        rows:       signal_ledger row dicts with 'confidence' and 'outcome' fields
        window_days: rolling window
    """
    # Accumulate (confidences, outcomes) per (plugin, tf, regime)
    ic_accs: dict[tuple, tuple[list, list]] = defaultdict(lambda: ([], []))
    all_ic_accs: dict[tuple, tuple[list, list]] = defaultdict(lambda: ([], []))

    for row in rows:
        plugin = row.get("setup_plugin")
        tf_val = row.get("tf") or row.get("timeframe")
        hmm = row.get("hmm_regime_at_fire")
        conf = row.get("confidence")
        outcome = row.get("outcome")

        if not plugin or not tf_val or conf is None or outcome is None:
            continue

        regime_label = HMM_TO_REGIME.get(hmm) if hmm is not None else None
        if not regime_label:
            continue

        key = (plugin, tf_val, regime_label)
        ic_accs[key][0].append(float(conf))
        ic_accs[key][1].append(outcome)

        all_key = (plugin, tf_val)
        all_ic_accs[all_key][0].append(float(conf))
        all_ic_accs[all_key][1].append(outcome)

    result: list[ICMetricsRow] = []
    now = datetime.now(UTC)

    def _make_ic_row(plugin, tf_val, regime_label, confs, outcomes):
        ic_score, p_val, n_used = compute_ic(confs, outcomes)
        if n_used < IC_MIN_SAMPLE_SIZE:
            return None
        sig = is_ic_significant(ic_score, p_val, n_used)
        return ICMetricsRow(
            setup_plugin=plugin, tf=tf_val, regime_type=regime_label,
            window_days=window_days, n=n_used,
            ic=round(ic_score, 4) if ic_score is not None else None,
            p_value=round(p_val, 4) if p_val is not None else None,
            is_significant=sig,
            computed_at=now,
        )

    for (plugin, tf_val, regime_label), (confs, outcomes) in ic_accs.items():
        row = _make_ic_row(plugin, tf_val, regime_label, confs, outcomes)
        if row:
            result.append(row)

    for (plugin, tf_val), (confs, outcomes) in all_ic_accs.items():
        row = _make_ic_row(plugin, tf_val, "all", confs, outcomes)
        if row:
            result.append(row)

    return result
