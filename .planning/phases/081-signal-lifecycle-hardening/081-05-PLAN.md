    ```python
    """SignalReplayAuditorAgent — L9 periodic.

    Phase 81. Every 5 minutes, finds v1 signals with exit_at IS NULL past TTL
    and replays them against market_data_ohlcv to compute outcomes.
    """
    from __future__ import annotations

    import asyncio
    import signal
    import sys
    from datetime import datetime, UTC, timedelta

    import asyncpg
    import structlog

    from src.config.settings import Settings
    from src.core.service_utils import setup_service_logging
    from src.core.stream_keys import TF_SECONDS, topic_lifecycle_transitions
    from src.core.kafka_client import KafkaProducerClient  # adjust to project path
    from src.observability.metrics import (
        SIGNAL_REPLAY_UNRESOLVED_GAUGE,
        SIGNAL_REPLAY_ATTEMPTED_TOTAL,
        SIGNAL_REPLAY_RESOLVED_TOTAL,
        SIGNAL_REPLAY_OHLCV_GAP_TOTAL,
    )
    # Reuse the live tracker's evaluators — do not duplicate.
    from services.lifecycle_tracker import evaluate_signal, evaluate_market_entry

    REPLAY_INTERVAL_SECONDS = 300
    LIVE_HOLD_BUFFER_SECONDS = 0  # already covered by ttl_bars * tf_seconds check


    class SignalReplayAuditorAgent:
        agent_id = "signal_replay_auditor"

        def __init__(self) -> None:
            self._log = structlog.get_logger(self.agent_id)
            self._settings = Settings()
            self._producer: KafkaProducerClient | None = None
            self._pool: asyncpg.Pool | None = None
            self._stop = asyncio.Event()

        async def _setup(self) -> None:
            self._pool = await asyncpg.create_pool(self._settings.database_url)
            self._producer = KafkaProducerClient(...)  # match other services
            await self._producer.start()

        async def _teardown(self) -> None:
            if self._producer:
                await self._producer.stop()
            if self._pool:
                await self._pool.close()

        async def _fetch_unresolved(self) -> list[asyncpg.Record]:
            query = """
                SELECT signal_id, symbol, timeframe, timestamp, entry_price, stop_loss,
                       direction, targets, entry_zone_low, entry_zone_high,
                       market_entry_price, ttl_bars
                FROM signal_ledger
                WHERE exit_at IS NULL
                  AND signal_schema_version = 'v1'
                  AND timestamp < NOW() - INTERVAL '2 minutes'
            """
            async with self._pool.acquire() as conn:
                return await conn.fetch(query)

        async def _fetch_window_bars(
            self, symbol: str, tf: str, start_ts: datetime, end_ts: datetime
        ) -> list[asyncpg.Record]:
            async with self._pool.acquire() as conn:
                return await conn.fetch(
                    """
                    SELECT timestamp, open, high, low, close, volume
                    FROM market_data_ohlcv
                    WHERE symbol = $1 AND timeframe = $2
                      AND timestamp >= $3 AND timestamp <= $4
                    ORDER BY timestamp ASC
                    """,
                    symbol, tf, start_ts, end_ts,
                )

        async def _replay_signal(self, row: asyncpg.Record) -> None:
            tf = row["timeframe"]
            tf_secs = TF_SECONDS.get(tf, 60)
            ttl = int(row["ttl_bars"])
            start_ts = row["timestamp"]
            end_ts = start_ts + timedelta(seconds=ttl * tf_secs)

            now_utc = datetime.now(UTC)
            # Live tracker may still hold this signal — skip if within TTL window.
            if (now_utc - start_ts).total_seconds() <= ttl * tf_secs:
                return

            bars = await self._fetch_window_bars(row["symbol"], tf, start_ts, end_ts)
            if not bars:
                SIGNAL_REPLAY_OHLCV_GAP_TOTAL.labels(
                    symbol=row["symbol"], timeframe=tf
                ).inc()
                return

            # Reuse the live tracker's evaluation logic. Build a minimal signal dict
            # matching the shape evaluate_signal expects (read lifecycle_tracker.py
            # to confirm exact field names — likely the canonical _load_signal output).
            signal_dict = dict(row)
            zone_outcome = await evaluate_signal(signal_dict, bars)
            market_outcome = await evaluate_market_entry(signal_dict, bars)

            env = self._settings.env_prefix
            topic = topic_lifecycle_transitions(env)

            # Publish zone-track LifecycleTransition. Match the field names exactly
            # to whatever LifecycleWriterAgent consumes (see services/lifecycle_writer_agent.py).
            await self._producer.publish(topic, msg=zone_outcome)
            SIGNAL_REPLAY_RESOLVED_TOTAL.labels(
                outcome=zone_outcome.get("outcome", "unknown")
            ).inc()

            # Publish market-entry track if present (independent track).
            if market_outcome is not None:
                await self._producer.publish(topic, msg=market_outcome)

        async def _cycle(self) -> None:
            rows = await self._fetch_unresolved()
            SIGNAL_REPLAY_ATTEMPTED_TOTAL.inc(len(rows))
            unresolved_now = 0
            for row in rows:
                if self._stop.is_set():
                    return
                tf_secs = TF_SECONDS.get(row["timeframe"], 60)
                if (datetime.now(UTC) - row["timestamp"]).total_seconds() <= int(row["ttl_bars"]) * tf_secs:
                    unresolved_now += 1
                    continue
                try:
                    await self._replay_signal(row)
                except Exception as exc:
                    self._log.exception("replay_signal_failed", signal_id=row["signal_id"], error=str(exc))
                    unresolved_now += 1

            # Recompute unresolved gauge AFTER replay batch publishes (writer applies async).
            async with self._pool.acquire() as conn:
                cnt = await conn.fetchval(
                    """
                    SELECT COUNT(*) FROM signal_ledger
                    WHERE exit_at IS NULL
                      AND signal_schema_version = 'v1'
                      AND timestamp < NOW() - INTERVAL '2 minutes'
                    """
                )
            SIGNAL_REPLAY_UNRESOLVED_GAUGE.set(float(cnt or 0))

        async def _run(self) -> None:
            while not self._stop.is_set():
                try:
                    await self._cycle()
                except Exception as exc:
                    self._log.exception("replay_cycle_failed", error=str(exc))
                try:
                    await asyncio.wait_for(self._stop.wait(), timeout=REPLAY_INTERVAL_SECONDS)
                except asyncio.TimeoutError:
                    pass

        def _install_signals(self) -> None:
            loop = asyncio.get_running_loop()
            for sig in (signal.SIGINT, signal.SIGTERM):
                loop.add_signal_handler(sig, self._stop.set)

        async def main(self) -> int:
            setup_service_logging("logs/signal_replay_auditor_agent.log")
            self._install_signals()
            await self._setup()
            try:
                await self._run()
            finally:
                await self._teardown()
            return 0


    if __name__ == "__main__":
        sys.exit(asyncio.run(SignalReplayAuditorAgent().main()))
    ```
