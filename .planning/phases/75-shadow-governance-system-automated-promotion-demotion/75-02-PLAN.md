```python
"""Statistical utility functions shared across intelligence services.

Extracted from weight_updater._bootstrap_ci_lower (Phase 75).
Both ShadowAuditorAgent and weight_updater import from this module.
"""
from __future__ import annotations

import numpy as np


def bootstrap_ci_lower(
    pnl_r_values: list[float],
    alpha: float = 0.05,
    n_boot: int = 1000,
) -> float:
    """Bootstrap lower confidence bound on E[PnL_R].

    Returns float('-inf') if fewer than 10 samples (insufficient for reliable estimate).
    Uses a fixed RNG seed (42) for reproducibility in tests.

    Args:
        pnl_r_values: PnL in R-multiples for resolved signals.
        alpha: Significance level. Default 0.05 gives 95% CI lower bound.
        n_boot: Number of bootstrap resamples. Default 1000.

    Returns:
        Lower bound float, or -inf on insufficient data.
    """
    if len(pnl_r_values) < 10:
        return float("-inf")
    rng = np.random.default_rng(42)
    arr = np.array(pnl_r_values)
    boot_means = np.array([
        rng.choice(arr, size=len(arr), replace=True).mean()
        for _ in range(n_boot)
    ])
    return float(np.percentile(boot_means, alpha / 2 * 100))
```

NOTE: The public signature differs from weight_updater's private copy — argument order is
`(pnl_r_values, alpha, n_boot)` matching D-07. The n_boot default changes from 2000 to 1000
per D-07. The internal logic (fixed seed 42, same percentile formula) is identical.

**Step A — Create `services/shadow_auditor_agent.py`:**

Follow the ml_data_quality_agent.py pattern exactly: async main, DatabaseManager, structlog,
asyncio.run(_amain()) entry point, setup_service_logging with full log path.

The agent's `_amain()` function must:
1. Connect to DB (async with DatabaseManager)
2. Load all rows from shadow_registry: `SELECT component_name, component_type, is_shadow, min_n, min_ev_r, ci_alpha, demotion_lookback_days, demotion_threshold_ev_r, demotion_min_evaluations, demotion_consecutive_count FROM shadow_registry`
3. For each shadow component (is_shadow = TRUE): run promotion check
   - Query: `SELECT outcome, pnl_r, signal_computed_at FROM signal_ledger WHERE setup_plugin = $1 AND outcome IS NOT NULL AND outcome NOT IN ('never_activated', 'ttl_expired_behind')`
   - Compute: `n = len(rows)`, `pnl_r_values = [float(r['pnl_r']) for r in rows if r['pnl_r'] is not None]`
   - Compute: `ev_r = sum(pnl_r_values)/len(pnl_r_values) if pnl_r_values else 0.0`
   - Compute: `win_rate = sum(1 for r in rows if r['outcome'] in {'target_1','target_1_2','target_full'})/n if n > 0 else 0.0`
   - Compute: `ci_lower = bootstrap_ci_lower(pnl_r_values, alpha=row['ci_alpha'])`
   - Update last_eval columns: `UPDATE shadow_registry SET last_eval_n=$1, last_eval_ev_r=$2, last_eval_ci_lower=$3, last_eval_win_rate=$4, last_eval_at=NOW() WHERE component_name=$5`
   - Promotion gate (per D-05): if `n >= row['min_n'] AND ci_lower > row['min_ev_r']`:
     - `UPDATE shadow_registry SET is_shadow=FALSE, promoted_at=NOW(), demotion_consecutive_count=0 WHERE component_name=$1`
     - `INSERT INTO shadow_transition_log (component_name, component_type, from_state, to_state, trigger_reason, n, ev_r, ci_lower, win_rate) VALUES ($1,$2,'shadow','live','promotion_gate_cleared',$3,$4,$5,$6)`
     - Build and publish ShadowTransitionEvent to topic_shadow_transitions
     - Emit SHADOW_PROMOTION_READY.labels(plugin=component_name).set(1)
   - If not promoted: emit SHADOW_PROMOTION_READY.labels(plugin=component_name).set(0)
   - Always emit: SHADOW_N_RESOLVED, SHADOW_WIN_RATE, SHADOW_EV_R, SHADOW_EV_CI_LOWER gauges
4. For each live component (is_shadow = FALSE): run demotion check (per D-06)
   - Query: `SELECT pnl_r FROM signal_ledger WHERE setup_plugin = $1 AND outcome IS NOT NULL AND outcome NOT IN ('never_activated', 'ttl_expired_behind') AND signal_computed_at > NOW() - INTERVAL '1 day' * $2` (bind demotion_lookback_days)
   - Compute rolling `ev_r` for the lookback window
   - If `ev_r < row['demotion_threshold_ev_r']`: increment consecutive count
     - `UPDATE shadow_registry SET demotion_consecutive_count = demotion_consecutive_count + 1, last_eval_at=NOW() WHERE component_name=$1`
     - If `new_count >= row['demotion_min_evaluations']`: demote
       - `UPDATE shadow_registry SET is_shadow=TRUE, demoted_at=NOW(), demotion_consecutive_count=0 WHERE component_name=$1`
       - INSERT shadow_transition_log with from_state='live', to_state='shadow', trigger_reason='demotion_ev_r_degraded'
       - Publish ShadowTransitionEvent
   - Else: reset consecutive count
     - `UPDATE shadow_registry SET demotion_consecutive_count=0, last_eval_at=NOW() WHERE component_name=$1`
5. Publish ShadowTransitionEvent via KafkaProducerClient: serialize to JSON with `dataclasses.asdict()`, publish to `topic_shadow_transitions(settings.env_name)` with key=component_name

Use structlog logger with fields: `component_name`, `component_type`, `from_state`, `to_state`, `trigger_reason` on each transition log message.

from __future__ import annotations
import math

import pytest

from src.core.stats_utils import bootstrap_ci_lower


def test_empty_list_returns_neg_inf():
    assert bootstrap_ci_lower([]) == float("-inf")


def test_fewer_than_10_samples_returns_neg_inf():
    assert bootstrap_ci_lower([0.1] * 9) == float("-inf")


def test_exactly_10_samples_returns_finite():
    result = bootstrap_ci_lower([0.1] * 10)
    assert math.isfinite(result)


def test_positive_pnl_returns_positive_ci():
    """100 positive signals: CI lower bound should be > 0."""
    result = bootstrap_ci_lower([1.0] * 100)
    assert result > 0.0


def test_negative_pnl_returns_negative_ci():
    """100 losing signals: CI lower bound should be < 0."""
    result = bootstrap_ci_lower([-1.0] * 100)
    assert result < 0.0


def test_ci_lower_le_mean():
    """CI lower bound must be <= mean for symmetric distributions."""
    values = [0.5] * 100
    mean = 0.5
    result = bootstrap_ci_lower(values)
    assert result <= mean


def test_custom_alpha():
    """alpha=0.01 (99% CI) produces a lower CI lower bound than alpha=0.05 (95% CI)."""
    values = list(range(1, 101))  # 100 values
    ci_95 = bootstrap_ci_lower(values, alpha=0.05)
    ci_99 = bootstrap_ci_lower(values, alpha=0.01)
    assert ci_99 <= ci_95


def test_custom_n_boot():
    """Custom n_boot does not cause errors."""
    result = bootstrap_ci_lower([0.1] * 50, n_boot=500)
    assert math.isfinite(result)


def test_reproducible_with_seed():
    """Two calls with same input produce same result (fixed seed=42)."""
    values = [float(i) for i in range(20)]
    r1 = bootstrap_ci_lower(values)
    r2 = bootstrap_ci_lower(values)
    assert r1 == r2
