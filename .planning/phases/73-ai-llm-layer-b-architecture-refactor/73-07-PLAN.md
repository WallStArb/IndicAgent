    ```python
    """NarrativeComputeAgent — generates 2-3 sentence market context for 5m+ bars.

    Per CONTEXT.md narrative TF gate: rejects 1m bars before any LLM call.
    Extends BaseAIAgent (Phase 73 B+ architecture).
    """
    from __future__ import annotations

    import structlog
    from src.core.ai.base_agent import BaseAIAgent
    from src.core.ai.context import AIContext, Tier
    from src.core.ai.output import AgentOutput
    from src.core.llm.chain import LLMProviderChain
    from src.intelligence.ai.narrative.prompts import build_narrative_prompt
    from src.intelligence.ai.narrative.parsers import parse_narrative_response

    logger = structlog.get_logger(__name__)

    _NARRATIVE_TFS = frozenset({"5m", "15m", "1h", "4h", "1d"})


    class NarrativeComputeAgent(BaseAIAgent):
        agent_id = "narrative_v1"
        group = "narrative"
        tiers_needed = frozenset({Tier.I1, Tier.I4, Tier.I6, Tier.I7})  # confirm against current orchestrator usage
        shadow_only = True
        latency_budget_ms = <value from 73-LATENCY-PROFILE.md narrative recommendation; fallback 60000.0>

        def __init__(self, llm_chain: LLMProviderChain) -> None:
            self._chain = llm_chain

        async def _compute(self, context: AIContext) -> AgentOutput:
            # TF gate — hard reject 1m and any TF outside the allowed set
            if context.timeframe not in _NARRATIVE_TFS:
                return AgentOutput(
                    agent_id=self.agent_id, group=self.group,
                    signal_id=context.signal_id, symbol=context.symbol,
                    timeframe=context.timeframe, ts=context.ts,
                    output_type="neutral",
                    payload={},
                    shadow_only=self.shadow_only,
                    latency_ms=0.0,
                    error=f"tf_gate:{context.timeframe}",
                )

            prompt = build_narrative_prompt(context)
            text = await self._chain.generate(
                prompt=prompt,
                system=_NARRATIVE_SYSTEM_PROMPT,
                max_tokens=512,
                timeout=self.latency_budget_ms / 1000.0,
                audit_context={
                    "signal_id": str(context.signal_id),
                    "agent_id": self.agent_id,
                    "symbol": context.symbol,
                    "timeframe": context.timeframe,
                },
            )

            if text is None:
                return self._neutral(context, error="llm_none", latency_ms=0.0)

            parsed = parse_narrative_response(text)

            return AgentOutput(
                agent_id=self.agent_id, group=self.group,
                signal_id=context.signal_id, symbol=context.symbol,
                timeframe=context.timeframe, ts=context.ts,
                output_type="narrative",
                payload={"text": parsed.text, **parsed.extra_fields},
                shadow_only=self.shadow_only,
                latency_ms=0.0,
            )
    ```
