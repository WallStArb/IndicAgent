    ```python
    async def _evaluate_graduation_once(self) -> None:
        """Query signal_transform_log and evaluate graduation gates for each agent.

        Direct asyncpg read is a deliberate exception to the DAG rule (CONTEXT.md).
        graduation_loop is a background task (15-min interval), NOT on the bar dispatch hot path.
        """
        from src.intelligence.swarm import graduation  # lazy import — keeps src/core/ai/ boundary clean

        if self._pool is None:
            self.logger.warning("graduation_loop.no_pool", group_id=self.group_id)
            return

        for agent in self.agents:
            try:
                async with self._pool.acquire() as conn:
                    rows = await conn.fetch(
                        """
                        SELECT multiplier, pnl_r, ts
                        FROM signal_transform_log
                        WHERE agent_id = $1
                          AND pnl_r IS NOT NULL
                          AND multiplier IS NOT NULL
                          AND ts >= NOW() - INTERVAL '90 days'
                        ORDER BY ts ASC
                        """,
                        agent.agent_id,
                    )

                if not rows:
                    self.logger.debug(
                        "graduation_loop.no_data",
                        agent_id=agent.agent_id,
                        group_id=self.group_id,
                    )
                    continue

                import pandas as pd
                df = pd.DataFrame([dict(r) for r in rows], columns=["multiplier", "pnl_r", "ts"])
                result = graduation.evaluate_all(df, transform_id=agent.agent_id)

                is_graduated = bool(result.get("is_graduated", False))
                was_shadow = bool(agent.shadow_only)

                if is_graduated and was_shadow:
                    agent.shadow_only = False
                    self.logger.info(
                        "graduation_loop.promoted",
                        agent_id=agent.agent_id,
                        spearman_rho=result.get("spearman_rho"),
                        n=result.get("n"),
                    )
                elif not is_graduated and not was_shadow:
                    agent.shadow_only = True
                    self.logger.warning(
                        "graduation_loop.demoted",
                        agent_id=agent.agent_id,
                        spearman_rho=result.get("spearman_rho"),
                        n=result.get("n"),
                    )
                else:
                    self.logger.debug(
                        "graduation_loop.evaluated",
                        agent_id=agent.agent_id,
                        is_graduated=is_graduated,
                        shadow_only=was_shadow,
                        spearman_rho=result.get("spearman_rho"),
                        n=result.get("n"),
                    )

                # Publish graduation event to topic_swarm_graduation
                if self._producer:
                    from src.core.stream_keys import topic_swarm_graduation
                    await self._producer.send(
                        topic_swarm_graduation(self.settings.env_name),
                        key=f"{agent.agent_id}:{self.group_id}",
                        value={
                            **result,
                            "agent_id": agent.agent_id,
                            "group_id": self.group_id,
                            "shadow_only": bool(agent.shadow_only),
                        },
                    )

            except Exception:
                self.logger.exception(
                    "graduation_loop.agent_error",
                    agent_id=getattr(agent, "agent_id", "unknown"),
                    group_id=self.group_id,
                )
                # Continue to next agent — one agent failure must not abort the loop
    ```
