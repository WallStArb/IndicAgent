    ```python
    """narrative_agent.py -- NarrativeComputeAgent extending BaseAIAgent.

    Per Phase 73 CONTEXT.md: narrative TF gate in _compute (5m+ only).
    infrastructure (Kafka, DB) owned by NarrativeGroupComputeAgent in services/.
    """
    from __future__ import annotations

    from typing import Any

    import structlog

    from src.core.ai.base_agent import BaseAIAgent
    from src.core.ai.context import AIContext, Tier
    from src.core.ai.output import AgentOutput
    from src.core.llm.chain import LLMProviderChain
    from src.intelligence.narrative.orchestrator import NarrativeOrchestrator

    logger = structlog.get_logger(__name__)

    _ALLOWED_TFS = frozenset({"5m", "15m", "1h", "4h", "1d"})


    class NarrativeComputeAgent(BaseAIAgent):
        """LLM narrative generation agent.

        Rejects non-allowed TFs before any LLM call.
        shadow_only=True until graduation gate passes.
        """

        agent_id = "narrative_v1"
        group = "narrative"
        tiers_needed: frozenset[Tier] = frozenset({Tier.I4, Tier.I7})
        shadow_only = True
        latency_budget_ms = 60_000.0  # narrative calls can take up to 60s (Ollama)

        def __init__(self, llm_chain: LLMProviderChain, **kwargs: Any) -> None:
            super().__init__(name=self.__class__.__name__)
            self._orchestrator = NarrativeOrchestrator(
                chain=llm_chain, max_tokens=200, timeout=60.0,
            )

        async def _compute(self, context: AIContext) -> AgentOutput:
            """Generate market narrative. Returns neutral AgentOutput for disallowed TFs."""
            # TF gate — must be BEFORE any LLM call (per CONTEXT.md locked decision)
            if context.timeframe not in _ALLOWED_TFS:
                return AgentOutput(
                    agent_id=self.agent_id,
                    group=self.group,
                    signal_id=context.signal_id,
                    symbol=context.symbol,
                    timeframe=context.timeframe,
                    ts=context.ts,
                    output_type="neutral",
                    payload={"reason": f"tf_rejected:{context.timeframe}"},
                    shadow_only=self.shadow_only,
                )

            # Pass the full context as the record; NarrativeOrchestrator uses attribute access
            text = await self._orchestrator.generate(context)
            if text is None:
                return self._neutral(context, "orchestrator returned None", latency_ms=0.0)

            return AgentOutput(
                agent_id=self.agent_id,
                group=self.group,
                signal_id=context.signal_id,
                symbol=context.symbol,
                timeframe=context.timeframe,
                ts=context.ts,
                output_type="narrative",
                payload={"text": text},
                shadow_only=self.shadow_only,
            )
    ```
