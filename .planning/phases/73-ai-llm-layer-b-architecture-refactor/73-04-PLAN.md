       ```python
       """narrative_agent.py -- NarrativeComputeAgent (pure compute, no infrastructure).

       Wraps NarrativeOrchestrator for dispatch by NarrativeGroupComputeAgent.
       Per Phase 73 D-03: module reorganized to src/intelligence/ai/narrative/.
       Base class migration to BaseAIAgent happens in Plan 05.
       """
       from __future__ import annotations

       from typing import Any

       import structlog

       from src.core.llm.chain import LLMProviderChain
       from src.intelligence.narrative.orchestrator import NarrativeOrchestrator

       logger = structlog.get_logger(__name__)

       _ALLOWED_TFS = frozenset({"5m", "15m", "1h", "4h", "1d"})


       class NarrativeComputeAgent:
           """Pure compute wrapper around NarrativeOrchestrator.

           Exposes async generate(record) interface; filters to allowed TFs.
           infrastructure (Kafka, DB) is owned by NarrativeGroupComputeAgent (Plan 05).
           """

           agent_id = "narrative_v1"
           group = "narrative"
           shadow_only = True

           def __init__(self, llm_chain: LLMProviderChain, **kwargs: Any) -> None:
               self._orchestrator = NarrativeOrchestrator(
                   chain=llm_chain, max_tokens=200, timeout=60.0,
               )

           async def generate(self, record: Any, tf: str | None = None) -> str | None:
               """Generate narrative for a bar record. Returns None for disallowed TFs.

               TF gate (per CONTEXT.md): rejects timeframes not in {"5m", "15m", "1h", "4h", "1d"}.
               """
               resolved_tf = tf or (
                   record.get("tf") if isinstance(record, dict)
                   else getattr(record, "tf", None)
               )
               if resolved_tf not in _ALLOWED_TFS:
                   logger.debug(
                       "narrative_agent.tf_rejected",
                       tf=resolved_tf,
                       allowed=list(_ALLOWED_TFS),
                   )
                   return None
               return await self._orchestrator.generate(record)
       ```
