    ```python
    """BaseGroupService — abstract dispatcher shared by all AI agent groups.

    Owns Kafka consumers/producer, AIContextCache, optional graduation loop,
    optional aggregator. Subclasses declare 4 abstract properties.

    Per Phase 73 CONTEXT.md:
    - ShadowRecorder is Kafka-publish-only (topic_shadow_recordings)
    - graduation_loop reads signal_transform_log via asyncpg directly (acceptable
      background-task exception to the DAG rule)
    - All agents shadow_only=True initially; graduation_loop owns mutations
    """
    from __future__ import annotations

    import abc
    import asyncio
    from typing import Any

    import asyncpg
    import structlog

    from src.config.settings import Settings
    from src.core.agent.base import BaseAgent
    from src.core.ai.base_agent import BaseAIAgent
    from src.core.ai.context import AIContext, AIContextCache
    from src.core.ai.output import AgentOutput
    from src.core.ai.safe_wrapper import SafeAgentWrapper
    from src.core.kafka_utils import KafkaConsumerClient, KafkaProducerClient
    from src.core.llm.chain import LLMProviderChain
    from src.core.stream_keys import topic_shadow_recordings, topic_swarm_graduation
    from src.intelligence.schemas import IntelligenceEvent
    from src.intelligence.swarm import graduation

    logger = structlog.get_logger(__name__)

    _GRADUATION_INTERVAL_S = 900  # 15 min
    _GRADUATION_RHO_GATE = 0.15
    _GRADUATION_N_GATE = 30
    _GRADUATION_P_GATE = 0.05


    class ShadowRecorder:
        """Publishes AgentOutput to topic_shadow_recordings. Kafka-first DAG."""

        def __init__(self, producer: KafkaProducerClient, env_name: str) -> None:
            self._producer = producer
            self._topic = topic_shadow_recordings(env_name)

        async def record(self, output: AgentOutput) -> None:
            try:
                await self._producer.publish(self._topic, output.model_dump(mode="json"))
            except Exception:
                logger.exception("shadow_recorder.publish_failed", agent_id=output.agent_id)


    class BaseGroupService(BaseAgent, abc.ABC):
        # Abstract properties subclasses MUST override
        @property
        @abc.abstractmethod
        def group_id(self) -> str: ...

        @property
        @abc.abstractmethod
        def agents(self) -> list[BaseAIAgent]: ...

        @property
        @abc.abstractmethod
        def trigger_topics(self) -> list[str]: ...

        @property
        @abc.abstractmethod
        def output_topic(self) -> str: ...

        # Optional overrides
        has_graduation: bool = True
        aggregate_topic: str | None = None  # if set, aggregator runs
        bar_topic: str | None = None  # for AIContextCache update

        def __init__(self, name: str, settings: Settings | None = None) -> None:
            super().__init__(name=name, max_idle_seconds=300)
            self.settings = settings or Settings()
            self._llm_chain: LLMProviderChain | None = None
            self._context_cache = AIContextCache()
            self._recorder: ShadowRecorder | None = None
            self._bar_consumer: KafkaConsumerClient | None = None
            self._trigger_consumer: KafkaConsumerClient | None = None
            self._producer: KafkaProducerClient | None = None
            self._pool: asyncpg.Pool | None = None

        async def _setup(self) -> None:
            self._producer = KafkaProducerClient(bootstrap_servers=self.settings.kafka_bootstrap_servers)
            await self._producer.start()
            self._llm_chain = LLMProviderChain(
                call_type=self.group_id,
                settings=self.settings,
                producer=self._producer,
                cache_ttl=300.0,
            )
            self._recorder = ShadowRecorder(self._producer, self.settings.env_name)
            # Bar consumer (cache updates)
            if self.bar_topic:
                self._bar_consumer = KafkaConsumerClient(
                    self.bar_topic,
                    bootstrap_servers=self.settings.kafka_bootstrap_servers,
                    group_id=f"{self.group_id}_bar_consumer",
                    auto_offset_reset="latest",
                )
                await self._bar_consumer.start()
            # Trigger consumer
            self._trigger_consumer = KafkaConsumerClient(
                ",".join(self.trigger_topics),  # multi-topic if needed
                bootstrap_servers=self.settings.kafka_bootstrap_servers,
                group_id=f"{self.group_id}_consumer",
                auto_offset_reset="latest",
            )
            await self._trigger_consumer.start()
            # DB pool for graduation_loop and cache seeding
            self._pool = await asyncpg.create_pool(self.settings.database_url, min_size=2, max_size=5)
            await self._seed_cache_from_db()

        async def _seed_cache_from_db(self) -> None:
            assert self._pool is not None
            async with self._pool.acquire() as conn:
                rows = await conn.fetch(
                    """
                    SELECT DISTINCT ON (symbol, tf)
                        symbol, tf, ts, bar, i1, i4, i6
                    FROM intelligence_features
                    WHERE ts > NOW() - INTERVAL '7 days'
                      AND i1 IS NOT NULL AND i4 IS NOT NULL
                    ORDER BY symbol, tf, ts DESC
                    """
                )
            for row in rows:
                self._context_cache.seed_from_db_row(dict(row))
            self.logger.info("base_group_service.cache_seeded", count=len(rows), group_id=self.group_id)

        async def _run(self) -> None:
            tasks = []
            if self._bar_consumer:
                tasks.append(asyncio.create_task(self._bar_loop()))
            tasks.append(asyncio.create_task(self._trigger_loop()))
            if self.has_graduation:
                tasks.append(asyncio.create_task(self._graduation_loop()))
            try:
                await asyncio.gather(*tasks)
            except Exception:
                for t in tasks:
                    t.cancel()
                raise

        async def _bar_loop(self) -> None:
            assert self._bar_consumer is not None
            async for _topic, _key, payload in self._bar_consumer.messages():
                if not self.running:
                    break
                self._record_message_consumed()
                try:
                    event = IntelligenceEvent.model_validate(payload)
                    self._context_cache.update(event)
                except Exception as exc:
                    self.logger.warning("base_group_service.bar_cache_error", error=str(exc))

        async def _trigger_loop(self) -> None:
            assert self._trigger_consumer is not None
            async for _topic, _key, payload in self._trigger_consumer.messages():
                if not self.running:
                    break
                self._record_message_consumed()
                try:
                    event = IntelligenceEvent.model_validate(payload)
                    await self._handle_trigger(event)
                except Exception as exc:
                    self.logger.exception("base_group_service.trigger_error", error=str(exc))

        async def _handle_trigger(self, event: IntelligenceEvent) -> None:
            assert self._producer is not None and self._recorder is not None
            wrapped: list[tuple[BaseAIAgent, SafeAgentWrapper, AIContext]] = []
            for agent in self.agents:
                ctx = self._context_cache.build(event.symbol, event.tf, agent.tiers_needed)
                if ctx is None:
                    continue
                wrapped.append((agent, SafeAgentWrapper(agent), ctx))
            if not wrapped:
                return
            outputs: list[AgentOutput] = await asyncio.gather(
                *[wrapper.compute(ctx) for _, wrapper, ctx in wrapped]
            )
            # Fan-out per-agent
            for output in outputs:
                await self._producer.publish(self.output_topic, output.model_dump(mode="json"))
                await self._recorder.record(output)
            # Aggregate (subclass hook)
            if self.aggregate_topic is not None:
                aggregate = await self._aggregate(outputs)
                if aggregate is not None:
                    await self._producer.publish(self.aggregate_topic, aggregate)

        async def _aggregate(self, outputs: list[AgentOutput]) -> dict | None:
            """Override in subclass when aggregate_topic is set. Default: no aggregation."""
            return None

        async def _graduation_loop(self) -> None:
            assert self._pool is not None
            while self.running:
                try:
                    await asyncio.sleep(_GRADUATION_INTERVAL_S)
                    for agent in self.agents:
                        await self._evaluate_and_maybe_flip(agent)
                except asyncio.CancelledError:
                    raise
                except Exception:
                    self.logger.exception("graduation_loop.error", group_id=self.group_id)

        async def _evaluate_and_maybe_flip(self, agent: BaseAIAgent) -> None:
            assert self._pool is not None
            async with self._pool.acquire() as conn:
                rows = await conn.fetch(
                    """
                    SELECT multiplier, pnl_r, regime_type, n_resolved
                    FROM signal_transform_log
                    WHERE agent_id = $1 AND pnl_r IS NOT NULL
                    """,
                    agent.agent_id,
                )
            if not rows:
                return
            result = graduation.evaluate_all([dict(r) for r in rows])
            self.logger.info(
                "graduation.evaluated",
                agent_id=agent.agent_id,
                spearman_rho=result.get("spearman_rho"),
                n=result.get("n"),
                p_value=result.get("p_value"),
                is_graduated=result.get("is_graduated"),
            )
            assert self._producer is not None
            if result.get("is_graduated") and agent.shadow_only:
                agent.shadow_only = False
                self.logger.info("graduation.promoted", agent_id=agent.agent_id)
                await self._producer.publish(
                    topic_swarm_graduation(self.settings.env_name),
                    {"agent_id": agent.agent_id, "shadow_only": False, **result},
                )
            elif (not result.get("is_graduated")) and (not agent.shadow_only) and result.get("degraded"):
                agent.shadow_only = True
                self.logger.warning("graduation.demoted", agent_id=agent.agent_id)
                await self._producer.publish(
                    topic_swarm_graduation(self.settings.env_name),
                    {"agent_id": agent.agent_id, "shadow_only": True, **result},
                )

        async def _teardown(self) -> None:
            for c in (self._bar_consumer, self._trigger_consumer):
                if c is not None:
                    await c.stop()
            if self._producer is not None:
                await self._producer.stop()
            if self._pool is not None:
                await self._pool.close()
    ```
