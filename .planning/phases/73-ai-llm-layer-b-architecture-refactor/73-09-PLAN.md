    ```python
    """ai_narrative_agent.py — NarrativeGroupComputeAgent extending BaseGroupService.

    Per Phase 73 B+ architecture: narrative group, single agent, no graduation.
    Preserves 10-min staleness gate from prior implementation (skip stale bars
    before triggering NarrativeComputeAgent — a service-level concern).
    """
    from __future__ import annotations

    import asyncio
    from datetime import UTC, datetime, timedelta

    from src.config.settings import Settings
    from src.core.ai.base_agent import BaseAIAgent
    from src.core.ai.base_group_service import BaseGroupService
    from src.core.stream_keys import (
        topic_intelligence_journal,
        topic_narratives,
    )
    from src.intelligence.ai.narrative import NarrativeComputeAgent
    from src.intelligence.schemas import IntelligenceEvent

    _STALENESS_LIMIT = timedelta(minutes=10)


    class NarrativeGroupComputeAgent(BaseGroupService):
        @property
        def group_id(self) -> str:
            return "narrative"

        @property
        def trigger_topics(self) -> list[str]:
            return [topic_intelligence_journal(self.settings.env_name)]

        @property
        def output_topic(self) -> str:
            return topic_narratives(self.settings.env_name)

        # Narrative is informational — no graduation, no aggregate
        has_graduation = False
        aggregate_topic = None
        bar_topic = None  # narrative builds context from the trigger event itself

        def __init__(self, settings: Settings | None = None) -> None:
            super().__init__(name="NarrativeGroupComputeAgent", settings=settings)
            self._agents: list[BaseAIAgent] = []

        async def _setup(self) -> None:
            await super()._setup()
            assert self._llm_chain is not None
            self._agents = [NarrativeComputeAgent(llm_chain=self._llm_chain)]

        @property
        def agents(self) -> list[BaseAIAgent]:
            return self._agents

        async def _handle_trigger(self, event: IntelligenceEvent) -> None:
            # Staleness gate — preserve from prior implementation (10-min limit)
            try:
                bar_ts_raw = event.ts
                if isinstance(bar_ts_raw, str):
                    bar_ts = datetime.fromisoformat(bar_ts_raw)
                else:
                    bar_ts = bar_ts_raw
                if bar_ts.tzinfo is None:
                    bar_ts = bar_ts.replace(tzinfo=UTC)
                age = datetime.now(UTC) - bar_ts
                if age > _STALENESS_LIMIT:
                    self.logger.info(
                        "ai_narrative_agent.skipped_stale_bar",
                        symbol=event.symbol,
                        timeframe=event.tf,
                        age_seconds=int(age.total_seconds()),
                    )
                    return
            except Exception as exc:
                self.logger.warning("ai_narrative_agent.staleness_check_failed", error=str(exc))

            # Hand off to BaseGroupService dispatch — TF gate enforced by NarrativeComputeAgent itself
            await super()._handle_trigger(event)


    if __name__ == "__main__":
        asyncio.run(NarrativeGroupComputeAgent().run())
    ```
