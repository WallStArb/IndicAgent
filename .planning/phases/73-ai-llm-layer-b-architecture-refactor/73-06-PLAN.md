    Run a timing script against the local Ollama instance to measure actual P95 latency. This informs whether the 3000ms and 60000ms budgets are appropriate.

    ```bash
    python3 -c "
    import asyncio, time, statistics
    from src.config.settings import Settings
    from src.core.llm.chain import LLMProviderChain

    async def measure():
        settings = Settings()
        chain = LLMProviderChain(call_type='swarm', settings=settings)
        latencies = []
        prompt = 'Rate this signal: ES, 5m, trend_following, confidence 0.72, hmm_regime=1. JSON: {\"failure_probability\": float, \"confidence\": float, \"risk_factors\": [str], \"reasoning\": str}'
        print('Measuring 5 Ollama calls (gemma4:e4b fallback)...')
        for i in range(5):
            t0 = time.monotonic()
            try:
                result = await chain.generate(prompt=prompt, system='You are a trading analyst. Always respond with valid JSON.', max_tokens=200, timeout=90.0)
                latency_ms = (time.monotonic() - t0) * 1000
                latencies.append(latency_ms)
                print(f'  call {i+1}: {latency_ms:.0f}ms, got={bool(result)}')
            except Exception as e:
                print(f'  call {i+1}: ERROR {e}')
        if latencies:
            print(f'P50={statistics.median(latencies):.0f}ms P95={sorted(latencies)[int(len(latencies)*0.95)]:.0f}ms max={max(latencies):.0f}ms')
            print(f'Recommended alpha budget: {max(latencies)*2:.0f}ms, narrative budget: 120000ms')
        else:
            print('No successful calls — Ollama may be offline')

    asyncio.run(measure())
    " 2>/dev/null
    ```
