    Create tests/integration/test_pipeline_flow.py - a smoke test for the hot path:

    ```python
    """
    Phase 107 Regression Smoke Test.
    Runs BEFORE Wave 1 (baseline) and AFTER each wave.
    Fails if hot path is broken.
    """
    import asyncio
    import asyncpg
    from src.config.settings import Settings

    DB_URL = Settings().database_url

    async def get_connection():
        return await asyncpg.connect(DB_URL)

    async def test_timescaledb_connectable():
        """TimescaleDB is reachable and responsive."""
        conn = await get_connection()
        try:
            result = await conn.fetchval("SELECT 1")
            assert result == 1
        finally:
            await conn.close()

    async def test_intelligence_features_has_recent_data():
        """intelligence_features hypertable has data from last hour."""
        conn = await get_connection()
        try:
            count = await conn.fetchval("""
                SELECT COUNT(*) FROM intelligence_features
                WHERE ts > NOW() - INTERVAL '1 hour'
            """)
            assert count > 0, f"No intelligence_features data in last hour (count={count})"
        finally:
            await conn.close()

    async def test_signal_ledger_has_recent_data():
        """signal_ledger has data from last hour."""
        conn = await get_connection()
        try:
            count = await conn.fetchval("""
                SELECT COUNT(*) FROM signal_ledger
                WHERE timestamp > NOW() - INTERVAL '1 hour'
            """)
            assert count >= 0  # May be 0 during off-hours, but query must not error
        finally:
            await conn.close()

    async def test_market_data_ohlcv_has_recent_data():
        """market_data_ohlcv has data from last hour (live bars flowing)."""
        conn = await get_connection()
        try:
            count = await conn.fetchval("""
                SELECT COUNT(*) FROM market_data_ohlcv
                WHERE timestamp > NOW() - INTERVAL '1 hour'
            """)
            assert count > 0, f"No market_data_ohlcv data in last hour (count={count})"
        finally:
            await conn.close()

    async def test_jsonb_not_double_serialized():
        """JSONB columns contain structured data, not escaped strings."""
        conn = await get_connection()
        try:
            row = await conn.fetchrow("""
                SELECT i1 FROM intelligence_features
                WHERE i1 IS NOT NULL
                ORDER BY ts DESC LIMIT 1
            """)
            if row and row['i1']:
                val = row['i1']
                if isinstance(val, str):
                    assert not val.startswith('"'), f"JSONB double-serialized: {val[:50]}"
                assert isinstance(val, dict), f"Expected dict, got {type(val)}"
        finally:
            await conn.close()

    def test_pipeline_flow():
        """Main regression test - runs all checks."""
        asyncio.run(test_timescaledb_connectable())
        asyncio.run(test_intelligence_features_has_recent_data())
        asyncio.run(test_signal_ledger_has_recent_data())
        asyncio.run(test_market_data_ohlcv_has_recent_data())
        asyncio.run(test_jsonb_not_double_serialized())
    ```

    Note: During off-market hours, the recent data tests may need relaxed assertions. Mark them with pytest.mark.skipif or use count >= 0 with a warning.
