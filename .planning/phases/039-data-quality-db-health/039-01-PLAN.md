```sql
-- Migration: signal_ledger schema hardening
-- Phase 39 — DB foundation for ML training data quality
-- Idempotent: all ADD COLUMN and ADD CONSTRAINT use IF NOT EXISTS / DO NOTHING guards

BEGIN;

-- 1. Generated column: effective_ts
-- COALESCE(signal_computed_at, feature_ts) — the canonical "when this signal was produced"
-- Replaces ad-hoc COALESCE in queries; enables O(1) index-based ordering.
-- Hypertable note: generated columns on TimescaleDB hypertables require PG 14+
-- If generated columns are unsupported on this version, use a regular column + trigger
ALTER TABLE signal_ledger
  ADD COLUMN IF NOT EXISTS effective_ts timestamptz
  GENERATED ALWAYS AS (COALESCE(signal_computed_at, feature_ts)) STORED;

-- 2. Generated column: pipeline_lag_ms
-- EXTRACT(EPOCH FROM (signal_computed_at - feature_ts)) * 1000
-- Instruments pipeline latency at the row level — "instrument everything"
-- NULL when signal_computed_at IS NULL (pending signals not yet processed by aggregator)
ALTER TABLE signal_ledger
  ADD COLUMN IF NOT EXISTS pipeline_lag_ms double precision
  GENERATED ALWAYS AS (
    CASE
      WHEN signal_computed_at IS NOT NULL AND feature_ts IS NOT NULL
      THEN EXTRACT(EPOCH FROM (signal_computed_at - feature_ts)) * 1000.0
      ELSE NULL
    END
  ) STORED;

-- 3. CHECK constraint: status must be a known value
-- Complements the code-level SignalStatus enum from Phase 39.1
-- outcome CHECK constraint was added by Phase 39.1 Plan 05 — not duplicated here
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conrelid = 'signal_ledger'::regclass
      AND conname = 'chk_signal_ledger_status'
  ) THEN
    ALTER TABLE signal_ledger
      ADD CONSTRAINT chk_signal_ledger_status
      CHECK (status IN ('pending', 'active', 'regime_suppressed'));
  END IF;
END $$;

-- 4. CHECK constraint: direction must be LONG, SHORT, or NULL
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conrelid = 'signal_ledger'::regclass
      AND conname = 'chk_signal_ledger_direction'
  ) THEN
    ALTER TABLE signal_ledger
      ADD CONSTRAINT chk_signal_ledger_direction
      CHECK (direction IS NULL OR direction IN ('LONG', 'SHORT'));
  END IF;
END $$;

COMMIT;
```

**TimescaleDB gotcha**: Generated columns on hypertables may not be supported in all versions. If the ALTER TABLE fails with `generated columns not supported`, use this fallback instead for effective_ts and pipeline_lag_ms — add as regular columns and create a trigger to populate them. Document which approach was used.

```sql
-- Materialized view: signal_stats_daily
-- Aggregates signal outcomes by (symbol, timeframe, setup_plugin, date)
-- Used by Plan 05 IC calculation and dashboard win-rate queries
-- Refresh: after each lifecycle batch (Plan 06 monitoring job triggers this)

CREATE MATERIALIZED VIEW IF NOT EXISTS signal_stats_daily AS
SELECT
    sl.symbol,
    sl.timeframe,
    sl.setup_plugin,
    date_trunc('day', sl.effective_ts) AS signal_date,
    COUNT(*) AS total_signals,
    COUNT(*) FILTER (WHERE sl.outcome IN (
        'target_1', 'target_1_2', 'target_full'
    )) AS wins,
    AVG(sl.pnl_r) FILTER (WHERE sl.pnl_r IS NOT NULL) AS avg_pnl_r,
    AVG(sl.calibrated_confidence) AS avg_confidence,
    -- Correlation between confidence and outcome (IC numerator components)
    AVG(sl.calibrated_confidence *
        CASE WHEN sl.outcome IN ('target_1','target_1_2','target_full') THEN 1.0
             WHEN sl.outcome IS NOT NULL THEN -1.0
             ELSE 0.0 END
    ) AS conf_outcome_cov
FROM signal_ledger sl
WHERE sl.outcome IS NOT NULL
  AND sl.effective_ts IS NOT NULL
GROUP BY 1, 2, 3, 4
WITH DATA;

-- Index for fast IC queries by date range
CREATE UNIQUE INDEX IF NOT EXISTS idx_signal_stats_daily_pk
  ON signal_stats_daily (symbol, timeframe, setup_plugin, signal_date);

CREATE INDEX IF NOT EXISTS idx_signal_stats_daily_date
  ON signal_stats_daily (signal_date DESC);
```
