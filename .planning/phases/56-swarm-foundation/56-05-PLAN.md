```python
"""AINarrativeComputeAgent — thin coordinator for LLM narrative generation.

Subscribes to intelligence record topic, generates narratives via
NarrativeOrchestrator, publishes to narratives Kafka topic.

~200 lines — all narrative logic lives in src/intelligence/narrative/.
"""
from __future__ import annotations

import asyncio
import json
import sys
from pathlib import Path

import structlog

# Add project root to path (same pattern as other services)
sys.path.insert(0, str(Path(__file__).parent.parent))

from src.config.settings import Settings
from src.core.agent.base import BaseAgent
from src.core.kafka_utils import KafkaConsumerClient, KafkaProducerClient
from src.core.llm import LLMChain, OllamaProvider, OpenRouterProvider
from src.core.service_utils import setup_service_logging
from src.core.stream_keys import topic_intelligence_journal, topic_narratives
from src.intelligence.narrative import NarrativeOrchestrator
from src.intelligence.schemas import BarIntelligenceRecord
from src.observability.metrics import counter

logger = structlog.get_logger(__name__)

CONSUMER_GROUP = "ai_narrative_consumer"
MAX_TOKENS = 200
LLM_TIMEOUT = 30.0


class AINarrativeComputeAgent(BaseAgent):
    """I8: Consumes BarIntelligenceRecord → generates LLM narrative → publishes."""

    def __init__(self) -> None:
        super().__init__(
            name="ai_narrative_agent",
            metrics_port=9120,
            max_idle_seconds=600,
        )
        self._settings = Settings()
        self._consumer: KafkaConsumerClient | None = None
        self._producer: KafkaProducerClient | None = None
        self._orchestrator: NarrativeOrchestrator | None = None

        self._narratives_generated = counter(
            "ai_narrative_generated_total",
            "Narrative strings successfully generated",
        )
        self._llm_failures = counter(
            "ai_narrative_llm_failures_total",
            "LLM calls that returned None (all providers failed)",
        )
        self._records_consumed = counter(
            "ai_narrative_records_consumed_total",
            "BarIntelligenceRecord messages consumed",
        )

    async def _setup(self) -> None:
        settings = self._settings

        # Build LLM provider chain: OpenRouter (free) → Ollama (offline fallback)
        providers = []
        if settings.openrouter_api_key:
            providers.append(
                OpenRouterProvider(
                    model=getattr(settings, "openrouter_model", "meta-llama/llama-3.3-70b-instruct:free"),
                    api_key=settings.openrouter_api_key,
                )
            )
        providers.append(
            OllamaProvider(
                model=getattr(settings, "ollama_model", "gemma4:e4b"),
                base_url=getattr(settings, "ollama_base_url", "http://localhost:11434"),
            )
        )

        self._producer = KafkaProducerClient(
            bootstrap_servers=settings.kafka_bootstrap_servers,
        )
        await self._producer.start()

        chain = LLMChain(providers=providers, producer=self._producer, settings=settings)
        self._orchestrator = NarrativeOrchestrator(chain=chain, max_tokens=MAX_TOKENS, timeout=LLM_TIMEOUT)

        topic_in = topic_intelligence_journal(settings.env_name)
        self._consumer = KafkaConsumerClient(
            topic_in,
            bootstrap_servers=settings.kafka_bootstrap_servers,
            group_id=CONSUMER_GROUP,
            auto_offset_reset="latest",
        )
        await self._consumer.start()
        self.logger.info("ai_narrative_agent.started", topic=topic_in)

    async def _run(self) -> None:
        async for _topic, _key, payload in self._consumer.messages():
            if not isinstance(payload, dict):
                continue
            self._record_message_consumed()
            self._records_consumed.inc()
            await self._process_record(payload)

    async def _process_record(self, payload: dict) -> None:
        """Parse payload → generate narrative → publish."""
        try:
            record = BarIntelligenceRecord.model_validate(payload)
        except Exception as exc:
            self.logger.warning("ai_narrative.parse_error", error=str(exc))
            return

        narrative = await self._orchestrator.generate(record, call_type="narrative")

        if narrative is None:
            self._llm_failures.inc()
            self.logger.warning(
                "ai_narrative.llm_failed",
                symbol=record.intelligence.symbol,
                tf=record.intelligence.tf,
            )
            return

        self._narratives_generated.inc()

        topic_out = topic_narratives(self._settings.env_name)
        await self._producer.publish(
            topic_out,
            {
                "symbol": record.intelligence.symbol,
                "timeframe": record.intelligence.tf,
                "narrative": narrative,
                "plugin": record.winner_plugin,
                "direction": record.winner_direction,
                "confidence": record.winner_confidence,
                "ts": record.intelligence.ts.isoformat(),
            },
            key=f"{record.intelligence.symbol}:{record.intelligence.tf}",
        )
        self.logger.info(
            "ai_narrative.published",
            symbol=record.intelligence.symbol,
            tf=record.intelligence.tf,
        )

    async def _teardown(self) -> None:
        if self._consumer:
            await self._consumer.stop()
        if self._producer:
            await self._producer.stop()


if __name__ == "__main__":
    setup_service_logging("logs/ai_narrative_agent.log")
    asyncio.run(AINarrativeComputeAgent().start())
```
