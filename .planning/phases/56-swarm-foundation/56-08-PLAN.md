# Plan 56-08: PromotionAuditorAgent — Automated Shadow Validation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the manual `scripts/alpha_promotion.py` approach with an automated timer-triggered `PromotionAuditorAgent` that runs every 6 hours, computes Pearson(confidence, win) per (agent_id, symbol, tf, hmm_regime) segment, writes results to `agent_promotion_stats`, and emits Prometheus metrics. No human should need to run a script to find out if an agent is ready for promotion.

**Architecture:** One-shot `BaseAgent` — runs once and exits (systemd timer restarts every 6h). Reads `alpha_multiplier_shadow` JOIN `signal_ledger`, groups by segment, computes stats, upserts to `agent_promotion_stats`. Gate: `rho >= 0.4 AND n >= 100 AND p_value < 0.05` per segment. Requires Plan 56-06 (`alpha_multiplier_shadow` table exists). Can be written and deployed before 56-07 produces any data — it simply finds 0 eligible rows until data flows.

**Renaissance Principle:** "Let the system run." Promotion validation must be automatic, not a human memory task.

**Tech Stack:** Python 3.11, asyncpg, scipy.stats, structlog, prometheus_client, systemd timer

```sql
-- 059_agent_promotion_stats.sql
-- Phase 56-08: Agent promotion eligibility tracking.
-- Written by PromotionAuditorAgent every 6h run.
-- Each row = one segment computation (agent_id, symbol, tf, hmm_regime) at a point in time.
-- Use latest computed_at row per segment for current eligibility status.

CREATE TABLE IF NOT EXISTS agent_promotion_stats (
    computed_at     TIMESTAMPTZ     NOT NULL,
    agent_id        TEXT            NOT NULL,
    symbol          TEXT            NOT NULL,
    timeframe       TEXT            NOT NULL,
    hmm_regime      INT             NOT NULL,   -- 0=ranging, 1=trending_up, 2=trending_down
    n               INT             NOT NULL,   -- sample count in segment
    pearson_r       FLOAT           NOT NULL,   -- Pearson(confidence, win_flag)
    p_value         FLOAT           NOT NULL,
    eligible        BOOLEAN         NOT NULL,   -- rho>=0.4 AND n>=100 AND p<0.05
    PRIMARY KEY (computed_at, agent_id, symbol, timeframe, hmm_regime)
);

SELECT create_hypertable('agent_promotion_stats', 'computed_at', if_not_exists => TRUE);

-- Fast lookup: is this segment currently eligible?
CREATE INDEX IF NOT EXISTS idx_promotion_stats_latest
    ON agent_promotion_stats (agent_id, symbol, timeframe, hmm_regime, computed_at DESC);

COMMENT ON TABLE agent_promotion_stats IS
    'Automated shadow validation results per (agent_id, symbol, tf, hmm_regime) segment. '
    'Written every 6h by PromotionAuditorAgent. '
    'eligible=TRUE when: pearson_r >= 0.4, n >= 100, p_value < 0.05. '
    'Use MAX(computed_at) row per segment for current status.';
```

```python
"""PromotionAuditorAgent — automated shadow validation for swarm agents.

Timer-triggered (every 6h via systemd timer). On each run:
  1. Query alpha_multiplier_shadow JOIN signal_ledger for resolved outcomes
  2. Group by (agent_id, symbol, timeframe, hmm_regime)
  3. Compute Pearson(confidence, win_flag) per segment (min n=30)
  4. Upsert results to agent_promotion_stats
  5. Log eligible segments + update Prometheus gauge
  6. Exit (systemd timer restarts next run)

Gate for promotion: rho >= 0.4 AND n >= 100 AND p_value < 0.05 per segment.
Global rho alone does NOT qualify — segment relentlessly (Renaissance Principle).
"""
from __future__ import annotations

import asyncio
import sys
from datetime import UTC, datetime
from itertools import groupby
from pathlib import Path
from typing import Any

import structlog

sys.path.insert(0, str(Path(__file__).parent.parent))

from src.config.settings import Settings
from src.core.agent.base import BaseAgent
from src.core.database_manager import DatabaseManager
from src.core.service_utils import setup_service_logging
from src.observability.metrics import counter, gauge

logger = structlog.get_logger(__name__)

MIN_SEGMENT_N = 30       # minimum samples to compute stats
PROMOTION_N = 100        # minimum n for promotion eligibility
PROMOTION_RHO = 0.4      # minimum Pearson rho
PROMOTION_P = 0.05       # maximum p-value

_QUERY = """
    SELECT
        a.agent_id,
        a.symbol,
        a.timeframe,
        a.hmm_regime,
        a.confidence,
        CASE WHEN s.pnl_r > 0 THEN 1.0 ELSE -1.0 END AS pnl_r
    FROM alpha_multiplier_shadow a
    JOIN signal_ledger s
        ON s.signal_id::text = a.signal_id::text
    WHERE a.error IS NULL
      AND s.pnl_r IS NOT NULL
      AND a.hmm_regime IS NOT NULL
    ORDER BY a.agent_id, a.symbol, a.timeframe, a.hmm_regime
"""

_UPSERT = """
    INSERT INTO agent_promotion_stats
        (computed_at, agent_id, symbol, timeframe, hmm_regime, n, pearson_r, p_value, eligible)
    VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
    ON CONFLICT (computed_at, agent_id, symbol, timeframe, hmm_regime) DO NOTHING
"""


def _compute_segment_stats(rows: list[dict]) -> dict[str, Any] | None:
    """Compute Pearson stats for one segment. Returns None if n < MIN_SEGMENT_N."""
    if len(rows) < MIN_SEGMENT_N:
        return None
    try:
        from scipy.stats import pearsonr
    except ImportError:
        # Fallback: numpy correlation
        import numpy as np
        confs = [r["confidence"] for r in rows]
        wins = [r["pnl_r"] for r in rows]
        rho = float(np.corrcoef(confs, wins)[0, 1])
        p_value = 1.0  # can't compute p without scipy — mark ineligible
        return {
            "n": len(rows),
            "pearson_r": round(rho, 4),
            "p_value": p_value,
            "eligible": False,
        }

    confs = [r["confidence"] for r in rows]
    wins = [r["pnl_r"] for r in rows]
    rho, p_value = pearsonr(confs, wins)
    eligible = (
        abs(rho) >= PROMOTION_RHO
        and len(rows) >= PROMOTION_N
        and p_value < PROMOTION_P
    )
    return {
        "n": len(rows),
        "pearson_r": round(float(rho), 4),
        "p_value": round(float(p_value), 4),
        "eligible": eligible,
    }


class PromotionAuditorAgent(BaseAgent):
    """One-shot auditor: runs once, writes stats, exits. Systemd timer triggers every 6h."""

    def __init__(self) -> None:
        super().__init__(name="promotion_auditor_agent", metrics_port=9129)
        self._settings = Settings()
        self._db: DatabaseManager | None = None
        self._segments_computed = counter(
            "promotion_auditor_segments_total", "Segments with sufficient data"
        )
        self._eligible_segments = gauge(
            "promotion_auditor_eligible_segments",
            "Segments currently meeting promotion gate",
        )

    async def _setup(self) -> None:
        self._db = DatabaseManager(self._settings.database_url)
        await self._db.initialize()
        self.logger.info("promotion_auditor.started")

    async def _run(self) -> None:
        await self._run_audit()
        # One-shot: exit after audit completes

    async def _run_audit(self) -> None:
        computed_at = datetime.now(UTC)
        async with self._db.pool.acquire() as conn:
            rows = await conn.fetch(_QUERY)

        if not rows:
            self.logger.info("promotion_auditor.no_data")
            self._eligible_segments.set(0)
            return

        rows = [dict(r) for r in rows]
        key_fn = lambda r: (r["agent_id"], r["symbol"], r["timeframe"], r["hmm_regime"])
        results = []
        for key, group in groupby(sorted(rows, key=key_fn), key=key_fn):
            agent_id, symbol, tf, regime = key
            stats = _compute_segment_stats(list(group))
            if stats is None:
                continue
            segment = {"agent_id": agent_id, "symbol": symbol, "timeframe": tf,
                       "hmm_regime": regime, **stats}
            results.append(segment)
            self._segments_computed.inc()
            self._log_segment_result(segment)

        if results:
            async with self._db.pool.acquire() as conn:
                await conn.executemany(_UPSERT, [
                    (computed_at, r["agent_id"], r["symbol"], r["timeframe"],
                     r["hmm_regime"], r["n"], r["pearson_r"], r["p_value"], r["eligible"])
                    for r in results
                ])

        eligible_count = sum(1 for r in results if r["eligible"])
        self._eligible_segments.set(eligible_count)
        self.logger.info(
            "promotion_auditor.complete",
            segments=len(results),
            eligible=eligible_count,
        )

    def _log_segment_result(self, segment: dict) -> None:
        log_fn = self.logger.info if segment["eligible"] else self.logger.debug
        log_fn(
            "promotion_auditor.segment",
            agent_id=segment["agent_id"],
            symbol=segment["symbol"],
            timeframe=segment["timeframe"],
            hmm_regime=segment["hmm_regime"],
            n=segment["n"],
            pearson_r=segment["pearson_r"],
            p_value=segment["p_value"],
            eligible=segment["eligible"],
        )

    async def _teardown(self) -> None:
        if self._db:
            await self._db.close()


if __name__ == "__main__":
    setup_service_logging("logs/promotion_auditor_agent.log")
    asyncio.run(PromotionAuditorAgent().start())
```
