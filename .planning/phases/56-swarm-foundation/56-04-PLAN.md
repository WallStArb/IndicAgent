"""SafeSwarmWrapper — defensive guardrail for all swarm agents.

Wraps an IAlphaContributor instance. Enforces:
- asyncio.wait_for timeout (from contributor.latency_budget_ms)
- Per-agent PluginCircuitBreaker
- Pydantic schema validation (already enforced by AgentResult construction)
- Multiplier bounds check [0.0, 2.0]
- Prometheus violation metrics
- Neutral fallback on any failure (multiplier=1.0, confidence=0.0)
"""
from __future__ import annotations

import asyncio
import time
from typing import TYPE_CHECKING

import structlog
from pydantic import ValidationError

from src.core.plugin_circuit_breaker import CircuitBreakerConfig, PluginCircuitBreaker
from src.intelligence.schemas import MAX_MULTIPLIER, MIN_MULTIPLIER, AgentResult
from src.intelligence.swarm.metrics import SWARM_SAFETY_VIOLATIONS

if TYPE_CHECKING:
    from src.intelligence.swarm.context import SwarmContext
    from src.intelligence.swarm.interface import IAlphaContributor

logger = structlog.get_logger(__name__)


class SafeSwarmWrapper:
    """Wraps IAlphaContributor with safety guardrails and observability."""

    def __init__(self, contributor: "IAlphaContributor") -> None:
        self._contributor = contributor
        self._circuit_breaker = PluginCircuitBreaker(
            config=CircuitBreakerConfig(
                failure_threshold=3,
                recovery_timeout=60,
                success_threshold=2,
                max_half_open_calls=2,
                failure_window=60,
                performance_threshold_ms=contributor.latency_budget_ms * 2,
            )
        )

    async def compute(self, context: "SwarmContext") -> AgentResult:
        """Execute agent with timeout, circuit breaker, and bounds check."""
        start = time.monotonic()
        agent_id = self._contributor.agent_id

        try:
            result = await asyncio.wait_for(
                self._contributor.compute(context),
                timeout=self._contributor.latency_budget_ms / 1000.0,
            )
            latency_ms = (time.monotonic() - start) * 1000

            # Bounds check — Pydantic already validates on AgentResult construction,
            # but double-check in case contributor bypasses model validation
            if not (MIN_MULTIPLIER <= result.multiplier <= MAX_MULTIPLIER):
                SWARM_SAFETY_VIOLATIONS.labels(
                    agent_id=agent_id, violation="out_of_bounds"
                ).inc()
                logger.error(
                    "swarm.safety_violation",
                    agent_id=agent_id,
                    violation="out_of_bounds",
                    multiplier=result.multiplier,
                )
                return self._neutral("out_of_bounds")

            return result.model_copy(update={"latency_ms": latency_ms})

        except asyncio.TimeoutError:
            SWARM_SAFETY_VIOLATIONS.labels(agent_id=agent_id, violation="timeout").inc()
            logger.warning("swarm.agent_timeout", agent_id=agent_id,
                           budget_ms=self._contributor.latency_budget_ms)
            return self._neutral("timeout")

        except ValidationError as ve:
            SWARM_SAFETY_VIOLATIONS.labels(agent_id=agent_id, violation="schema_error").inc()
            logger.error("swarm.schema_error", agent_id=agent_id, error=str(ve))
            return self._neutral("schema_error")

        except Exception as exc:
            SWARM_SAFETY_VIOLATIONS.labels(agent_id=agent_id, violation="execution_error").inc()
            logger.error("swarm.execution_error", agent_id=agent_id, error=str(exc), exc_info=True)
            return self._neutral(f"execution_error: {exc}")

    def _neutral(self, error: str) -> AgentResult:
        """Fallback neutral result — multiplier=1.0 (no effect), confidence=0.0 (ignored in weighting)."""
        return AgentResult(
            agent_id=self._contributor.agent_id,
            path=self._contributor.path,  # use contributor's actual path, not hardcoded
            multiplier=1.0,
            confidence=0.0,
            shadow_only=True,
            error=error,
        )
```

"""SwarmAggregator — assembles AgentResult list → AlphaMultiplier.

Confidence-weighted average with Path B discount (0.3).
production_multiplier is clamped to [0.7, 1.3] for safety.
"""
from __future__ import annotations

from datetime import datetime
from typing import Any
from uuid import UUID

from src.intelligence.schemas import AgentResult, AlphaMultiplier

_PATH_B_DISCOUNT = 0.3
_PRODUCTION_CLAMP_LOW = 0.7
_PRODUCTION_CLAMP_HIGH = 1.3


class SwarmAggregator:
    """Assembles Path A + Path B AgentResult lists into a final AlphaMultiplier."""

    def aggregate(
        self,
        signal_id: UUID,
        symbol: str,
        timeframe: str,
        ts: datetime,
        path_a_results: list[AgentResult],
        path_b_results: list[AgentResult],
    ) -> AlphaMultiplier:
        """Compute confidence-weighted multiplier from both paths.

        Path B confidence is discounted by PATH_B_DISCOUNT (0.3) to weight
        deterministic results higher until LLM agents are shadow-validated.
        production_multiplier is always clamped to [0.7, 1.3].
        """
        total_weight = 0.0
        weighted_sum = 0.0

        for r in path_a_results:
            w = r.confidence
            weighted_sum += r.multiplier * w
            total_weight += w

        for r in path_b_results:
            w = r.confidence * _PATH_B_DISCOUNT
            weighted_sum += r.multiplier * w
            total_weight += w

        raw = weighted_sum / total_weight if total_weight > 0.0 else 1.0
        production = max(_PRODUCTION_CLAMP_LOW, min(_PRODUCTION_CLAMP_HIGH, raw))

        all_results = path_a_results + path_b_results
        shadow_only = any(r.shadow_only for r in all_results) if all_results else True
        contributors = {r.agent_id: r for r in all_results}

        path_a_mult = (
            sum(r.multiplier * r.confidence for r in path_a_results) /
            max(sum(r.confidence for r in path_a_results), 1e-9)
            if path_a_results else None
        )
        path_b_mult = (
            sum(r.multiplier * r.confidence for r in path_b_results) /
            max(sum(r.confidence for r in path_b_results), 1e-9)
            if path_b_results else None
        )

        return AlphaMultiplier(
            signal_id=signal_id,
            symbol=symbol,
            timeframe=timeframe,
            ts=ts,
            path_a_multiplier=path_a_mult,
            path_b_multiplier=path_b_mult,
            contributors=contributors,
            final_alpha_multiplier=round(raw, 4),
            production_multiplier=round(production, 4),
            shadow_only=shadow_only,
        )
```

"""PromptRegistry — bounded template lookup for LLM swarm agents.

All LLM agents MUST use get_prompt() — never build prompts with raw
f-strings from market data (prevents prompt injection via symbol names,
plugin names, or other field values that originate from external data).
"""
from __future__ import annotations

import re

_REGISTRY: dict[str, str] = {
    # Placeholder templates — Phase 57 (SkepticAgent) will populate real versions
    "skeptic_agent_v1": (
        "/no_think\n\n"
        "Signal: {symbol}/{timeframe} {direction} (confidence {confidence:.0%})\n"
        "Plugin: {plugin}\n"
        "Regime: {regime}\n\n"
        "Given this market state, what is the probability this signal fails? "
        "Respond with: FAIL_PROB: <0.0-1.0> and one sentence reason."
    ),
    "narrative_signal_v2": (
        "/no_think\n\n"
        "Symbol: {symbol}/{timeframe} — {direction} (confidence {confidence:.0%})\n"
        "Structure: {plugin}\n"
        "Write exactly 2 sentences: market context + execution guidance."
    ),
}

_MAX_VALUE_LEN = 500
_INJECTION_RE = re.compile(r"[\x00-\x1f\x7f]")


def get_prompt(template_id: str, **kwargs: object) -> str:
    """Look up a prompt template by ID and render with sanitized kwargs.

    Raises ValueError for unknown template_id — fail loud, not silent.
    All kwarg values are sanitized: control chars stripped, truncated at 500 chars.
    """
    template = _REGISTRY.get(template_id)
    if template is None:
        raise ValueError(
            f"Unknown prompt template: {template_id!r}. "
            f"Valid IDs: {sorted(_REGISTRY.keys())}"
        )
    safe_kwargs = {k: _sanitize(str(v)) for k, v in kwargs.items()}
    return template.format_map(safe_kwargs)


def _sanitize(value: str) -> str:
    """Strip control characters and truncate to prevent prompt injection."""
    cleaned = _INJECTION_RE.sub("", value)
    return cleaned[:_MAX_VALUE_LEN]


def register_template(template_id: str, template: str) -> None:
    """Register a new template at runtime (for Phase 57+ agent development)."""
    _REGISTRY[template_id] = template
```

"""SwarmRegistry — load and validate IAlphaContributor instances from config.

Validates that each loaded class implements IAlphaContributor protocol
(duck-typed via isinstance check with @runtime_checkable).
"""
from __future__ import annotations

import importlib
import json

import structlog

from src.intelligence.swarm.interface import IAlphaContributor
from src.intelligence.swarm.safety import SafeSwarmWrapper

logger = structlog.get_logger(__name__)


def load_contributors(
    config_path: str = "config/intelligence_contributors.json",
) -> list[SafeSwarmWrapper]:
    """Load IAlphaContributor instances from config, wrap each in SafeSwarmWrapper.

    Returns empty list if config file does not exist (swarm is optional).
    Each entry needs: module, class, path, latency_budget_ms, shadow_only.
    """
    try:
        with open(config_path) as f:
            config = json.load(f)
    except FileNotFoundError:
        logger.info("swarm_registry.config_missing", path=config_path)
        return []
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON in contributors config: {e}")

    wrappers: list[SafeSwarmWrapper] = []
    for entry in config.get("contributors", []):
        module_name = entry.get("module")
        class_name = entry.get("class")
        if not module_name or not class_name:
            logger.warning("swarm_registry.entry_missing_fields", entry=entry)
            continue
        try:
            module = importlib.import_module(module_name)
            klass = getattr(module, class_name)
            instance = klass()

            # Validate protocol compliance
            if not isinstance(instance, IAlphaContributor):
                logger.error(
                    "swarm_registry.protocol_violation",
                    class_name=class_name,
                    missing="does not implement IAlphaContributor",
                )
                continue

            wrappers.append(SafeSwarmWrapper(instance))
            logger.info(
                "swarm_registry.loaded",
                agent_id=instance.agent_id,
                path=instance.path,
                shadow_only=instance.shadow_only,
            )

        except Exception as e:
            logger.warning(
                "swarm_registry.load_failed",
                module=module_name,
                class_name=class_name,
                error=str(e),
            )

    return wrappers
```

"""SwarmBaseAgent — abstract base for all IAlphaContributor implementations.

Extends BaseAgent with swarm-specific instrumentation baked in:
- OTel span around every compute() call (agent_id, shadow_only, signal_id as attributes)
- Prometheus: latency histogram, violation counter, prediction counter (all per agent_id)
- shadow_only flag enforced at base level — never promoted to production without segment ρ ≥ 0.4
- Latency budget: asyncio.wait_for enforced by SafeSwarmWrapper; budget declared here

Usage (Phase 66+):
    class SkepticAgent(SwarmBaseAgent):
        agent_id = "skeptic"
        path = "llm_swarm"
        shadow_only = True
        latency_budget_ms = 8_000.0

        async def compute(self, ctx: SwarmContext) -> AgentResult:
            ...
"""
from __future__ import annotations

import abc
import time
from typing import TYPE_CHECKING

from src.core.agent.base import BaseAgent
from src.intelligence.schemas import AgentResult
from src.intelligence.swarm.metrics import (
    SWARM_AGENT_CALLS,
    SWARM_AGENT_LATENCY,
    SWARM_SAFETY_VIOLATIONS,
)
from src.observability.otel import get_tracer

if TYPE_CHECKING:
    from src.intelligence.swarm.context import SwarmContext


class SwarmBaseAgent(BaseAgent, abc.ABC):
    """Abstract base for all swarm intelligence agents.

    Subclasses declare class-level attributes and implement compute().
    All observability (OTel + Prometheus) is handled here — no boilerplate in agents.
    """

    agent_id: str          # e.g. "skeptic", "trend_vol"
    path: str              # "deterministic" or "llm_swarm"
    shadow_only: bool = True            # stays True until segment-level ρ ≥ 0.4 validated
    latency_budget_ms: float = 5_000.0  # asyncio.wait_for budget in SafeSwarmWrapper

    def __init__(self) -> None:
        super().__init__(name=self.__class__.__name__)
        self._swarm_tracer = get_tracer(f"swarm.{self.agent_id}")

    @abc.abstractmethod
    async def compute(self, ctx: "SwarmContext") -> AgentResult:
        """Compute alpha contribution. Must not raise — return neutral on any error."""
        ...

    async def _timed_compute(self, ctx: "SwarmContext") -> AgentResult:
        """Internal wrapper: OTel span + Prometheus latency around compute().

        Called by SwarmOrchestratorAgent instead of compute() directly.
        """
        start = time.monotonic()
        with self._swarm_tracer.start_as_current_span(
            f"{self.agent_id}.compute",
            attributes={
                "agent_id": self.agent_id,
                "shadow_only": self.shadow_only,
                "symbol": ctx.symbol,
                "timeframe": ctx.timeframe,
            },
        ):
            result = await self.compute(ctx)

        latency_ms = (time.monotonic() - start) * 1000
        SWARM_AGENT_LATENCY.labels(agent_id=self.agent_id, path=self.path).observe(latency_ms)
        SWARM_AGENT_CALLS.labels(agent_id=self.agent_id, path=self.path).inc()
        return result

    async def warm_up(self) -> None:
        """Pre-load models or validate dependencies. No-op by default."""

    def health_check(self) -> dict:
        return {"agent_id": self.agent_id, "shadow_only": self.shadow_only, "status": "ok"}

    async def _run(self) -> None:
        """SwarmBaseAgent is not a long-running service — _run is a no-op."""
