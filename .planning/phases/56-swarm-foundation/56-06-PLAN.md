def topic_swarm_results(env_name: str) -> str:
    """Per-agent AlphaMultiplier result fan-out (Renaissance DAG persistence topic).

    Published by SwarmOrchestratorAgent once per AgentResult (fan-out, not assembled).
    Consumed by SwarmWriterAgent — one DB row written per message.
    Key format: {symbol}:{tf}:{agent_id}

    This is the primary persistence stream for alpha_multiplier_shadow.
    Keeps WriterAgent simple: one message → one row (no contributor expansion needed).
    """
    return f"{env_prefix(env_name)}intelligence.swarm"


def topic_swarm_alpha_path_a(env_name: str) -> str:
    """Path A (deterministic) AlphaMultiplier results.

    Published by SwarmOrchestratorAgent immediately after Path A agents complete.
    Consumed by SwarmWriterAgent for shadow persistence.
    Key format: {symbol}:{tf} (e.g. 'ESM6:1m')
    """
    return f"{env_prefix(env_name)}swarm.alpha.path_a"


def topic_swarm_alpha_path_b(env_name: str) -> str:
    """Path B (llm_swarm) AlphaMultiplier results.

    Published by SwarmOrchestratorAgent asynchronously (up to 60s after signal).
    Consumed by SwarmWriterAgent for shadow persistence only.
    Key format: {symbol}:{tf}
    """
    return f"{env_prefix(env_name)}swarm.alpha.path_b"


def topic_swarm_world_state(env_name: str) -> str:
    """Compacted topic for SwarmOrchestratorAgent world state checkpoints.

    Key format: {symbol}:{tf} — compacted, so latest value per key is retained.
    Topic config: cleanup.policy=compact, min.cleanable.dirty.ratio=0.1,
                  segment.ms=3600000 — set on topic creation, not in code.

    On restart, SwarmOrchestratorAgent seeks to latest offset on this topic
    to rebuild SwarmContextCache without hitting the database.
    """
    return f"{env_prefix(env_name)}swarm.world_state"


def topic_swarm_orchestrator_dlq(env_name: str) -> str:
    """DLQ for SwarmOrchestratorAgent unprocessable payloads.

    Written when: signal event arrives but SwarmContext cannot be built
    (no cache entry, stale cache, schema parse failure).
    Payload: original message + error context.
    """
    return f"{env_prefix(env_name)}swarm.orchestrator.dlq"


def topic_swarm_writer_dlq(env_name: str) -> str:
    """DLQ for SwarmWriterAgent unprocessable payloads.

    Written when: AlphaMultiplier schema parse fails or DB insert fails
    after retries.
    Payload: original message + error context.
    """
    return f"{env_prefix(env_name)}swarm.writer.dlq"
```

-- 058_alpha_multiplier_shadow.sql
-- Phase 56: Alpha Multiplier Shadow Table
-- Stores swarm agent predictions for shadow validation before production promotion.
-- alpha_promotion.py JOINs this table against signal_ledger on signal_id to
-- compute Pearson(agent_confidence, realized_pnl_r) segmented by (symbol, tf, hmm_regime).

CREATE TABLE IF NOT EXISTS alpha_multiplier_shadow (
    ts              TIMESTAMPTZ     NOT NULL,
    signal_id       UUID            NOT NULL,
    symbol          TEXT            NOT NULL,
    timeframe       TEXT            NOT NULL,
    agent_id        TEXT            NOT NULL,
    path            TEXT            NOT NULL CHECK (path IN ('deterministic', 'llm_swarm')),
    multiplier      FLOAT           NOT NULL CHECK (multiplier >= 0.0 AND multiplier <= 2.0),
    confidence      FLOAT           NOT NULL CHECK (confidence >= 0.0 AND confidence <= 1.0),
    shadow_only     BOOLEAN         NOT NULL DEFAULT TRUE,
    hmm_regime      INT,            -- 0=ranging, 1=trending_up, 2=trending_down (from SwarmContext)
    metadata        JSONB,
    latency_ms      FLOAT,
    error           TEXT,           -- non-null means fallback result, not real prediction

    -- Back-filled by alpha_promotion.py after signal_ledger outcome resolves
    realized_pnl_r  FLOAT,
    outcome         TEXT,
    resolved_at     TIMESTAMPTZ,    -- when back-fill occurred

    -- TimescaleDB requires partition column (ts) in PRIMARY KEY
    -- Logical key: one row per (signal_id, agent_id) — enforced here + ON CONFLICT in writer
    PRIMARY KEY (ts, signal_id, agent_id)
);

-- Convert to TimescaleDB hypertable (partition by time)
SELECT create_hypertable('alpha_multiplier_shadow', 'ts', if_not_exists => TRUE);

-- Index for JOIN with signal_ledger in alpha_promotion.py
-- (signal_id, agent_id) uniqueness enforced by PRIMARY KEY above; add covering index for JOIN performance
CREATE INDEX IF NOT EXISTS idx_alpha_shadow_signal_agent
    ON alpha_multiplier_shadow (signal_id, agent_id)
    INCLUDE (confidence, realized_pnl_r, hmm_regime, error);

-- Index for regime-segmented Pearson queries:
-- SELECT ... WHERE symbol=? AND timeframe=? AND hmm_regime=?
CREATE INDEX IF NOT EXISTS idx_alpha_shadow_segment
    ON alpha_multiplier_shadow (symbol, timeframe, hmm_regime)
    WHERE error IS NULL;  -- partial index: only real predictions, not fallbacks

COMMENT ON TABLE alpha_multiplier_shadow IS
    'Shadow validation table for swarm agent alpha multiplier predictions. '
    'PRIMARY KEY (ts, signal_id, agent_id) — one row per agent per signal. '
    'Published by SwarmOrchestratorAgent fan-out to intelligence.swarm topic; '
    'persisted by SwarmWriterAgent (one message → one row). '
    'shadow_only=TRUE until alpha_promotion.py confirms rho >= 0.4, n >= 100, p < 0.05 '
    'per (symbol, timeframe, hmm_regime) segment. '
    'Rows with error IS NOT NULL are fallback/neutral results — exclude from Pearson analysis. '
    'resolved_at/realized_pnl_r/outcome back-filled by alpha_promotion.py.';

COMMENT ON COLUMN alpha_multiplier_shadow.hmm_regime IS
    '0=ranging, 1=trending_up, 2=trending_down. '
    'Promotion requires segment-level significance — global rho alone does not qualify.';
```
