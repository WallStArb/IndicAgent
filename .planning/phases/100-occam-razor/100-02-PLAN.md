```python
def _bootstrap_sharpe_delta(
    self,
    complex_returns: list[float],
    baseline_returns: list[float],
    n_boot: int = 1000,
) -> tuple[float, float, float]:
    """Bootstrap 95% CI on Sharpe ratio delta.

    Returns:
        (delta_mean, ci_lower, ci_upper) where delta = sharpe_complex - sharpe_baseline

    Raises:
        ValueError: if either list has < 30 samples
    """
    if len(complex_returns) < 30 or len(baseline_returns) < 30:
        raise ValueError(f"Bootstrap requires n >= 30, got {len(complex_returns)}, {len(baseline_returns)}")

    import numpy as np
    rng = np.random.default_rng(42)

    arr_complex = np.array(complex_returns)
    arr_baseline = np.array(baseline_returns)

    def sharpe(returns: np.ndarray) -> float:
        if returns.std() < 1e-9:
            return 0.0
        return returns.mean() / returns.std() * np.sqrt(252)  # Annualized

    boot_deltas = []
    for _ in range(n_boot):
        boot_complex = rng.choice(arr_complex, size=len(arr_complex), replace=True)
        boot_baseline = rng.choice(arr_baseline, size=len(arr_baseline), replace=True)
        delta = sharpe(boot_complex) - sharpe(boot_baseline)
        boot_deltas.append(delta)

    boot_arr = np.array(boot_deltas)
    ci_lower = float(np.percentile(boot_arr, 2.5))
    ci_upper = float(np.percentile(boot_arr, 97.5))
    return float(boot_arr.mean()), ci_lower, ci_upper
```

**Acceptance:**
- Raises ValueError for n < 30 (D-07 gate)
- Uses fixed RNG seed (42) for reproducibility
- Zero-variance guard returns sharpe=0.0 (not NaN)
- Unit test verifies CI width > 0 for noisy data

def _compute_complexity_score(
    self,
    n_params: int,
    latency_ms: float,
    train_time_ms: float,
) -> float:
    """Compute log-transformed complexity score.

    Formula: log(params + 1) × (1 + latency_ms / 1000) × (1 + train_time_ms / 60000)
    """
    import numpy as np

    if n_params < 0:
        raise ValueError(f"n_params must be non-negative, got {n_params}")

    log_params = np.log(n_params + 1)
    latency_factor = 1.0 + latency_ms / 1000.0
    train_factor = 1.0 + train_time_ms / 60000.0

    return float(log_params * latency_factor * train_factor)
```

**Acceptance:**
- Raises ValueError for negative n_params (fail-closed per D-06)
- Returns 0.0 for n_params=0 (log(1) = 0)
- Unit test verifies monotonic increase with each input

def __init__(self, *, dependencies: AgentDependencies, **kwargs: Any) -> None:
    super().__init__(**kwargs)
    self._penalty_weight = float(
        dependencies.settings.get("OCCAM_PENALTY_WEIGHT", "0.5")
    )

def _compare_models(
    self,
    complex_score: ModelScore,
    baseline_score: ModelScore,
) -> OccamResult:
    """Statistical comparison with complexity penalty.

    Decision rule (D-03):
    - ci_lower > 0 → complex wins
    - ci_upper < 0 → baseline wins
    - CI spans 0 → tie (prefer simpler)

    Penalty applied to raw Sharpe delta before CI computation.
    """
    import numpy as np

    # Compute complexity penalty (D-02)
    if baseline_score.complexity_score <= 0:
        complexity_ratio = 1.0
    else:
        complexity_ratio = complex_score.complexity_score / baseline_score.complexity_score

    penalty = np.log(complexity_ratio) if complexity_ratio > 1 else 0.0

    # Bootstrap CI on penalty-adjusted delta
    # NOTE: Bootstrap operates on RETURNS, not Sharpe. This is a simplified
    # version that assumes Sharpe delta is approximately normally distributed.
    # Full implementation would resample returns and recompute Sharpe each iteration.
    raw_delta = complex_score.sharpe - baseline_score.sharpe
    adjusted_delta = raw_delta - self._penalty_weight * penalty

    # Simplified CI: assume normality of delta distribution
    # TODO: Replace with full bootstrap in Task 2 integration
    se = (complex_score.pnl_r.std() / np.sqrt(complex_score.n)
          if complex_score.n > 0 else 1.0)
    ci_width = 1.96 * se  # 95% CI
    ci_lower = adjusted_delta - ci_width
    ci_upper = adjusted_delta + ci_width

    if ci_lower > 0:
        winner, recommendation = "complex", "promote"
    elif ci_upper < 0:
        winner, recommendation = "baseline", "reject"
    else:
        winner, recommendation = "tie", "reject"

    return OccamResult(
        complex_model=complex_score,
        baseline=baseline_score,
        winner=winner,
        confidence=ci_upper - ci_lower,
        recommendation=recommendation,
    )
```

**Acceptance:**
- Penalty weight read from settings (default 0.5)
- Ties default to rejection (fail-closed)
- Unit test verifies all three decision branches

    3. Apply statistical test with complexity penalty
    4. Fail-closed: reject if baseline wins or ties

    shadow_only=True: never affects live promotion directly.
    Updates shadow_registry via rejection flow.

    group = "alpha"
    tiers_needed = frozenset()
    latency_budget_ms = 1000.0  # Not time-critical

    def __init__(self, *, dependencies: AgentDependencies, **kwargs: Any) -> None:
        super().__init__(**kwargs)
        if dependencies.pool is None:
            raise ValueError("OccamRazorEvaluator requires dependencies.pool")
        self._pool = dependencies.pool
        self._penalty_weight = float(
            dependencies.settings.get("OCCAM_PENALTY_WEIGHT", "0.5")

        """Run Occam's Razor test on all shadow ML agents.

        Returns neutral with metadata — ORE runs in background cycle,

            error="occam_razor_runs_in_graduation_cycle_only",

    evaluator = OccamRazorEvaluator(dependencies=mock_deps)
