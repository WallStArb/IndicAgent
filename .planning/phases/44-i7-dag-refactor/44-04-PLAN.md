    ```python
    # After plugin.compute_full(frames) returns a result dict:
    if result.get("signal_type") == "none" or result.get("direction") == 0:
        continue  # No signal from this plugin

    # Construct via factory (per D-29)
    try:
        signal = make_signal(
            symbol=symbol,
            timeframe=timeframe,
            timestamp=timestamp,
            signal_type=result["signal_type"],
            setup_plugin=plugin.name,
            direction=result["direction"],
            entry_price=result.get("entry_price", close_price),
            stop_loss=result["stop_loss"],

            invalidation_conditions=result.get("invalidation_conditions", []),
            ttl_bars=result.get("ttl_bars", self._get_ttl_bars(timeframe)),
        )
    except (KeyError, TypeError) as e:
        logger.error("make_signal_construction_failed",
                      plugin=plugin.name, error=str(e),
                      result_keys=list(result.keys()))
        SIGNAL_VALIDATION_FAILURES.labels(plugin=plugin.name).inc()
    ```
