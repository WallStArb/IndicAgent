    **`plugin_utils.py`** (per D-01, D-03, D-04, D-11):
    ```python
    from __future__ import annotations
    from typing import Any
    import numpy as np

    def no_signal() -> dict[str, Any]:
        """Return canonical no-signal dict."""
        return {"signal_type": "none", "direction": 0, "confidence": 0.0}

    def extract_ohlcv(frames: dict[str, Any], min_bars: int) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray] | None:
        """Extract (open, high, low, close) arrays from frames. Returns None if insufficient data."""
        df = frames.get("main")
        if df is None or len(df) < min_bars:
            return None
        return (
            df["open"].to_numpy(dtype=float),
            df["high"].to_numpy(dtype=float),
            df["low"].to_numpy(dtype=float),
            df["close"].to_numpy(dtype=float),
        )

    def default_compute_next(plugin: Any, windows: dict[str, Any]) -> dict[str, Any]:
        """Default compute_next delegation — calls plugin.compute_full(windows)."""
        return plugin.compute_full(windows)

    def signal_type_for_direction(prefix: str, direction: int) -> str:
        """Return '{prefix}_long' or '{prefix}_short'."""
        return f"{prefix}_long" if direction == 1 else f"{prefix}_short"
    ```

    **`atr_utils.py`** (per D-05, D-06, D-07):
    ```python
    from __future__ import annotations
    from typing import Any

    def get_atr(features: dict[str, Any]) -> float | None:
        """Return atr_14 from I1 features, or None if unavailable/invalid.
        ATR is computed once in I1 — no recomputation here (D-07).
        """
        val = features.get("atr_14")
        if val is None:
            return None
        try:
            f = float(val)
        except (TypeError, ValueError):
            return None
        return f if f > 0 else None
    ```

    **`confidence_utils.py`** (per D-12, D-13, D-14):
    ```python
    from __future__ import annotations

    CONF_FLOOR: float = 0.10
    CONF_CEIL: float = 0.95

    def compose_confidence(raw: float) -> float:
        """Clamp raw confidence to the system contract [CONF_FLOOR, CONF_CEIL].
        All I7 plugins route through this function. Zero inline min()/max() clamping.
        """
        return round(max(CONF_FLOOR, min(CONF_CEIL, raw)), 4)
    ```
