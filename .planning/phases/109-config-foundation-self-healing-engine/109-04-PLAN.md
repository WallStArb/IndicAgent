    );

    CREATE INDEX IF NOT EXISTS idx_remediation_alert ON remediation_ledger (alert_id, timestamp);

    SELECT add_retention_policy('remediation_ledger', INTERVAL '90 days', if_not_exists => TRUE);
    SELECT add_compression_policy('remediation_ledger', INTERVAL '7 days', if_not_exists => TRUE);

    -- Success rate materialized view with REFRESH CONCURRENTLY support
    CREATE MATERIALIZED VIEW IF NOT EXISTS remediation_success_rates AS
    SELECT
      action,
      COUNT(*) AS attempt_count,
      SUM(CASE WHEN outcome = 'success' THEN 1 ELSE 0 END) AS success_count,
      SUM(CASE WHEN outcome = 'success' THEN 1 ELSE 0 END)::FLOAT / COUNT(*) AS success_rate
    FROM remediation_ledger
    WHERE timestamp > NOW() - INTERVAL '30 days'
    GROUP BY action
    WITH DATA;

    CREATE UNIQUE INDEX IF NOT EXISTS idx_remediation_success_rates_action ON remediation_success_rates (action);

    3. @dataclass class RemediationResult:
       remediation_id: str
       status: str  # success, failed, no_action, no_strategy
       pre_value: float | None
       post_value: float | None
       duration_ms: int
       error: str | None

    4. class SelfHealingEngine:
       def __init__(self, pool: asyncpg.Pool):
           self._pool = pool
           self._ledger = RemediationLedger(pool)
           self._http_session: aiohttp.ClientSession | None = None
           self._processed_alerts: set[str] = set()  # In-memory idempotency tracking
           self._webhook_secret = os.getenv("CONFIG_WEBHOOK_SHARED_SECRET")

       async def initialize(self):
           self._http_session = aiohttp.ClientSession()

       async def close(self):
           if self._http_session:
               await self._http_session.close()

       async def handle_webhook(self, payload: dict[str, Any]) -> RemediationResult:
           """
           Handle Alertmanager webhook with authentication.

           Validates webhook shared secret if configured, then processes alert.
           Returns 401 if auth fails.
           """
           # 1. Verify webhook shared secret
           if self._webhook_secret:
               webhook_secret = payload.get("webhook_secret") or payload.get("shared_secret")
               if webhook_secret != self._webhook_secret:
                   WEBHOOK_AUTH_FAILED_TOTAL.add(1, {"reason": "invalid_secret"})
                   raise HTTPException(status_code=401, detail="Invalid webhook shared secret")

           # 2. Validate payload
           try:
               alert = AlertRequest(
                   alert_id=payload.get("alert_id", str(uuid.uuid4())),
                   fired_at=datetime.fromisoformat(payload.get("fired_at", datetime.now(UTC).isoformat())),
                   severity=payload.get("severity", "WARNING"),
                   state_variable=payload.get("state_variable", ""),
                   current_value=float(payload.get("current_value", 0)),
                   threshold=float(payload.get("threshold", 0)),
                   labels=payload.get("labels", {}),
                   reason=payload.get("reason")
               )
           except (ValueError, TypeError) as exc:
               WEBHOOK_VALIDATION_FAILED_TOTAL.add(1, {"reason": "invalid_payload"})
               raise HTTPException(status_code=400, detail=f"Invalid payload: {exc}")

           WEBHOOK_RECEIVED_TOTAL.add(1, {"severity": alert.severity, "state_variable": alert.state_variable})

           # 3. Check idempotency
           if alert.alert_id in self._processed_alerts:
               return RemediationResult(remediation_id="", status="no_action", pre_value=None, post_value=None, duration_ms=0, error="Already processed")

           # 4. Process remediation
           result = await self.execute_remediation(alert)
           self._processed_alerts.add(alert.alert_id)
           return result

       async def execute_remediation(self, alert: AlertRequest) -> RemediationResult:
           """
           Execute remediation using control loop pattern.

           Maps alert.state_variable to strategy key via state_variable_to_strategy_key().
           """
           a. Measure pre_value = await self._measure_state(alert.state_variable)
           b. Calculate error = pre_value - alert.threshold
           c. if error <= 0: return RemediationResult(status="no_action", remediation_id="", pre_value=pre_value, post_value=pre_value, duration_ms=0, error=None)
           d. # Map state_variable to strategy key
           strategy_key = state_variable_to_strategy_key(alert.state_variable)
           e. if not strategy_key:
               return RemediationResult(status="no_strategy", remediation_id="", pre_value=pre_value, post_value=None, duration_ms=0, error=f"No strategy for state_variable={alert.state_variable}")
           f. strategy = REMEDIATION_STRATEGIES.get(strategy_key)
           g. if not strategy or not strategy.enabled:
               return RemediationResult(status="no_strategy", remediation_id="", pre_value=pre_value, post_value=None, duration_ms=0, error=f"Strategy {strategy_key} not enabled")
           h. # Check rate limit
           if not check_rate_limit(strategy_key, strategy.max_per_hour):
               return RemediationResult(status="no_action", remediation_id="", pre_value=pre_value, post_value=None, duration_ms=0, error="Rate limit exceeded")
           i. start = time.monotonic()
           j. action_name = strategy.action
           k. try:
               record_execution(strategy_key)
               async with asyncio.timeout(strategy.timeout_seconds):
                   await self._execute_action(strategy, alert)
               post_value = await self._measure_state(alert.state_variable)
               duration_ms = int((time.monotonic() - start) * 1000)
               outcome = "success" if post_value < alert.threshold else "partial"
               remediation_id = str(uuid.uuid4())
               await self._ledger.record(RemediationRecord(..., outcome=outcome, duration_ms=duration_ms, action=action_name, ...))
               return RemediationResult(remediation_id=remediation_id, status=outcome, pre_value=pre_value, post_value=post_value, duration_ms=duration_ms, error=None)
           except Exception as exc:
               duration_ms = int((time.monotonic() - start) * 1000)
               await self._ledger.record(RemediationRecord(..., outcome="failed", duration_ms=duration_ms, error_message=str(exc), action=action_name, ...))
               # Auto-disable strategy if success rate < 80%
               success_rate = await self._ledger.get_success_rate(action_name)
               if success_rate < 0.8:
                   disable_strategy(strategy_key)
               return RemediationResult(status="failed", remediation_id="", error=str(exc), pre_value=pre_value, post_value=None, duration_ms=duration_ms)

       async def _measure_state(self, state_variable: str) -> float:
           # Query Prometheus for current state
           try:
               async with self._http_session.get(f"http://localhost:9090/api/v1/query", params={"query": state_variable}) as resp:
                   data = await resp.json()
                   return float(data["data"]["result"][0]["value"][1])
           except (IndexError, KeyError, ValueError):
               structlog.get_logger().warning("prometheus.query_failed", state_variable=state_variable)
               return 0.0

       async def _execute_action(self, strategy: RemediationStrategy, alert: AlertRequest):
           if strategy.action == "delete_old_logs":
               mountpoint = alert.labels.get("mountpoint", "/var/log")
               # Validate mountpoint - only allow safe paths
               allowed_mountpoints = {"/var/log", "/tmp", "/home/bg/dev/indicagent/logs"}
               if mountpoint not in allowed_mountpoints:
                   raise ValueError(f"Mountpoint {mountpoint} not in allowlist")
               await self._delete_old_logs(mountpoint)
           elif strategy.action == "restart_consumer":
               unit_name = alert.labels.get("service")
               # Validate unit name - only allow indicagent services
               if not unit_name or not unit_name.startswith("indicagent-"):
                   raise ValueError(f"Invalid unit name: {unit_name}")
               await self._restart_service(unit_name)
           elif strategy.action == "flush_connection_pool":
               await self._flush_db_pools()

       async def _delete_old_logs(self, mountpoint: str):
           import subprocess
           # Use subprocess.run with bounded path, no shell injection
           result = subprocess.run(
               ["find", mountpoint, "-name", "*.log", "-mtime", "+7", "-delete"],
               check=True,
               capture_output=True,
               timeout=30
           )
           return result

       async def _restart_service(self, unit_name: str):
           # Use service_auditor's systemd restart endpoint
           async with self._http_session.post(f"http://localhost:12224/restart", json={"unit": unit_name}) as resp:
               resp.raise_for_status()

       async def _flush_db_pools(self):
           # Signal services to flush via topic (or just wait for natural flush)
           pass

Imports: asyncio, time, uuid, aiohttp; from fastapi import HTTPException; from src.self_healing.strategies import REMEDIATION_STRATEGIES, RemediationStrategy, check_rate_limit, record_execution, disable_strategy, state_variable_to_strategy_key; from src.self_healing.ledger import RemediationLedger, RemediationRecord; from src.observability.metrics import WEBHOOK_RECEIVED_TOTAL, WEBHOOK_VALIDATION_FAILED_TOTAL, WEBHOOK_AUTH_FAILED_TOTAL, REMEDIATION_ATTEMPT_TOTAL, REMEDIATION_DURATION_SECONDS
