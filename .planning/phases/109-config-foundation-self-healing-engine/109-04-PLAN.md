    CREATE TABLE IF NOT EXISTS remediation_ledger (
      timestamp TIMESTAMPTZ NOT NULL,
      remediation_id TEXT NOT NULL,
      alert_id TEXT NOT NULL,
      state_variable TEXT NOT NULL,
      pre_value FLOAT,
      post_value FLOAT,
      target_value FLOAT,
      action TEXT NOT NULL,
      outcome TEXT NOT NULL,
      duration_ms INT,
      error_message TEXT,
      changed_by TEXT NOT NULL,
      reason TEXT,
      PRIMARY KEY (timestamp, remediation_id)
    );

    SELECT create_hypertable('remediation_ledger', 'timestamp', if_not_exists => TRUE);

    CREATE INDEX IF NOT EXISTS idx_remediation_alert ON remediation_ledger (alert_id, timestamp);

    SELECT add_retention_policy('remediation_ledger', INTERVAL '90 days', if_not_exists => TRUE);
    SELECT add_compression_policy('remediation_ledger', INTERVAL '7 days', if_not_exists => TRUE);

    -- Success rate materialized view
    CREATE MATERIALIZED VIEW IF NOT EXISTS remediation_success_rates AS
    SELECT
      action,
      COUNT(*) AS attempt_count,
      SUM(CASE WHEN outcome = 'success' THEN 1 ELSE 0 END) AS success_count,
      SUM(CASE WHEN outcome = 'success' THEN 1 ELSE 0 END)::FLOAT / COUNT(*) AS success_rate
    FROM remediation_ledger
    WHERE timestamp > NOW() - INTERVAL '30 days'
    GROUP BY action;

    CREATE UNIQUE INDEX IF NOT EXISTS idx_remediation_success_rates_action ON remediation_success_rates (action);

    Follow exact schema from CONTEXT.md. Use IF NOT EXISTS for idempotency. Materialized view for success_rate aggregation.

    3. @dataclass class RemediationResult:
       remediation_id: str
       status: str  # success, failed, no_action, no_strategy
       pre_value: float | None
       post_value: float | None
       duration_ms: int
       error: str | None

    4. class SelfHealingEngine:
       def __init__(self, pool: asyncpg.Pool):
           self._pool = pool
           self._ledger = RemediationLedger(pool)
           self._http_session: aiohttp.ClientSession | None = None

       async def initialize(self):
           self._http_session = aiohttp.ClientSession()

       async def close(self):
           if self._http_session:
               await self._http_session.close()

       async def handle_webhook(self, payload: dict[str, Any]) -> RemediationResult:
           a. Validate payload, return error if invalid (WEBHOOK_VALIDATION_FAILED_TOTAL)
           b. Parse AlertRequest from payload
           c. Check idempotency (alert_id already processed?)
           d. Look up strategy from REMEDIATION_STRATEGIES
           e. Check rate limit via check_rate_limit()
           f. Execute remediation via execute_remediation()
           g. Return result

       async def execute_remediation(self, alert: AlertRequest) -> RemediationResult:
           a. Measure pre_value = await self._measure_state(alert.state_variable)
           b. Calculate error = pre_value - alert.threshold
           c. if error <= 0: return RemediationResult(status="no_action", ...)
           d. strategy = REMEDIATION_STRATEGIES.get(alert.state_variable)
           e. if not strategy or not strategy.enabled: return RemediationResult(status="no_strategy", ...)
           f. start = time.monotonic()
           g. action_name = strategy.action
           h. try:
               async with asyncio.timeout(strategy.timeout_seconds):
                   await self._execute_action(strategy, alert)
               post_value = await self._measure_state(alert.state_variable)
               duration_ms = int((time.monotonic() - start) * 1000)
               outcome = "success" if post_value < alert.threshold else "partial"
               remediation_id = str(uuid.uuid4())
               await self._ledger.record(RemediationRecord(..., outcome=outcome, duration_ms=duration_ms, action=action_name, ...))
               return RemediationResult(remediation_id=remediation_id, status=outcome, pre_value=pre_value, post_value=post_value, duration_ms=duration_ms, error=None)
           except Exception as exc:
               duration_ms = int((time.monotonic() - start) * 1000)
               await self._ledger.record(RemediationRecord(..., outcome="failed", duration_ms=duration_ms, error_message=str(exc), action=action_name, ...))
               i. # Auto-disable strategy if success rate < 80%
               success_rate = await self._ledger.get_success_rate(action_name)
               if success_rate < 0.8:
                   from src.self_healing.strategies import disable_strategy
                   disable_strategy(alert.state_variable)
               return RemediationResult(status="failed", error=str(exc), ...)

       async def _measure_state(self, state_variable: str) -> float:
           # Query Prometheus for current state
           async with self._http_session.get(f"http://localhost:9090/api/v1/query", params={"query": state_variable}) as resp:
               data = await resp.json()
               return float(data["data"]["result"][0]["value"][1])

       async def _execute_action(self, strategy: RemediationStrategy, alert: AlertRequest):
           if strategy.action == "delete_old_logs":
               await self._delete_old_logs(alert.labels.get("mountpoint", "/var/log"))
           elif strategy.action == "restart_consumer":
               await self._restart_service(alert.labels.get("service"))
           elif strategy.action == "flush_connection_pool":
               await self._flush_db_pools()

       async def _delete_old_logs(self, mountpoint: str):
           import subprocess
           subprocess.run(["find", mountpoint, "-name", "*.log", "-mtime", "+7", "-delete"], check=True)

       async def _restart_service(self, unit_name: str):
           async with self._http_session.post(f"http://localhost:12224/restart", json={"unit": unit_name}) as resp:
               resp.raise_for_status()

       async def _flush_db_pools(self):
           # Signal services to flush via topic (or just wait for natural flush)
           pass
