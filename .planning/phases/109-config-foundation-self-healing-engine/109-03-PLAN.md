    2. async def _pre_setup_config_load(self) -> None:
       """
       Phase A of config integration: load OPS snapshot BEFORE service _setup().

       Called from BaseAgent.start() BEFORE await self._setup() so that _setup() can
       read OPS config values (e.g., to initialize thresholds, feature flags). This
       addresses Codex HIGH finding: "BaseAgent lifecycle ordering may be wrong".

       NON-FATAL: on DB failure, leave _config_cache untouched (empty on first start;
       last-known-good on restart with valid prior cache - but cache is in-memory only,
       so first start with DB down means empty cache and service uses code defaults).

       Emits CONFIG_STALE_TOTAL on failure.
       """
       try:
           svc = ConfigService(self.settings.database_url)
           await svc.initialize()
           snapshot = await svc.list()  # OPS only
           # Filter by prefixes if set; otherwise accept all
           if self._config_prefixes:
               self._config_cache = {k: v for k, v in snapshot.items() if k.startswith(self._config_prefixes)}
           else:
               self._config_cache = dict(snapshot)
           self._config_loaded = True
           CONFIG_LAST_RELOAD_TIMESTAMP_SECONDS.set(time.time(), {"agent_id": self.name})
           structlog.get_logger().info("config.snapshot.loaded", agent=self.name, keys=len(self._config_cache))
       except Exception as exc:
           CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": type(exc).__name__})
           structlog.get_logger().warning("config.snapshot.load_failed", agent=self.name, error=str(exc))
           # Do NOT re-raise

    3. async def _setup_config_consumer(self) -> None:
       """
       Phase B of config integration: subscribe to Kafka AFTER service _setup() completes.

       Hot-reload via Kafka requires Kafka to be reachable. Service _setup() should not
       depend on Kafka for config; if it needs OPS values they came from the snapshot.

       NON-FATAL. If self._config_layer != "OPS", skip subscription (INFRA/STRUCT services
       don't hot-reload, and the outbox dispatcher itself skips to avoid circular dependency).

       IMPORTANT: KafkaConsumerClient takes the topic(s) as its FIRST positional argument
       at construction time. There is NO .subscribe() method - topics are bound at __init__.
       The correct Settings attribute is `kafka_bootstrap_servers` (NOT `kafka_brokers`);
       see services/alerting_agent.py line 51 for the canonical usage.
       """
       if self._config_layer != "OPS":
           return
       try:
           self._config_consumer = KafkaConsumerClient(
               topic_config_updates(self.env_name),
               bootstrap_servers=self.settings.kafka_bootstrap_servers,
               group_id=f"{self.name}_config_consumer",
           )
           await self._config_consumer.start()
           # NOTE: no .subscribe() call - topic was bound at construction (above).
           self._config_reload_task = asyncio.create_task(self._reload_config_loop())
       except Exception as exc:
           CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": "kafka_subscribe_failed"})
           structlog.get_logger().warning("config.consumer.setup_failed", agent=self.name, error=str(exc))

    4. async def _reload_config_loop(self) -> None:
       """
       Kafka message loop with prefix filtering to avoid reload storms.

       Per consensus LOW finding: every agent ignores keys outside its _config_prefixes
       allowlist cheaply (dict lookup; no parse, no work, no metric noise).

       Records CONFIG_RELOAD_LATENCY_SECONDS = (now - changed_at) for each accepted key
       (per Gemini suggestion: detect agents falling behind the update stream).
       """
       try:
           async for _topic, _key, payload in self._config_consumer.messages():
               try:
                   config_key = payload.get("config_key")
                   if not config_key:
                       continue
                   # Prefix filter: skip irrelevant keys cheaply
                   if self._config_prefixes and not config_key.startswith(self._config_prefixes):
                       continue
                   value_type = payload.get("value_type", "string")
                   parsed = self._parse_config_value(payload.get("config_value"), value_type)
                   self._config_cache[config_key] = parsed
                   self._config_loaded = True
                   # Latency: changed_at -> now
                   changed_at_iso = payload.get("changed_at")
                   if changed_at_iso:
                       try:
                           changed_at = datetime.fromisoformat(changed_at_iso.replace("Z", "+00:00"))
                           lag_s = (datetime.now(UTC) - changed_at).total_seconds()
                           CONFIG_RELOAD_LATENCY_SECONDS.record(lag_s, {"agent_id": self.name})
                       except Exception:
                           pass
                   CONFIG_RELOAD_TOTAL.add(1, {"agent_id": self.name, "key": config_key})
                   CONFIG_LAST_RELOAD_TIMESTAMP_SECONDS.set(time.time(), {"agent_id": self.name})
                   # Phase 109: subclass hook for post-reload reactions (e.g., ServiceAuditor reloading lag thresholds).
                   # No-op by default; subclasses override _on_config_message_received(key, value).
                   try:
                       await self._on_config_message_received(config_key, parsed)
                   except Exception as hook_exc:
                       structlog.get_logger().warning("config.hook_failed", agent=self.name, error=str(hook_exc))
               except Exception as exc:
                   CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": "parse_failed"})
                   structlog.get_logger().warning("config.parse_failed", agent=self.name, error=str(exc))
               if not self.running:
                   break
       except Exception as exc:
           CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": "consumer_loop_failed"})
           structlog.get_logger().error("config.consumer.loop_failed", agent=self.name, error=str(exc))

    5. async def _teardown_config_consumer(self) -> None:
       a. Cancel self._config_reload_task if exists
       b. Stop self._config_consumer if exists

    6. def get_config(self, key: str, default: Any = None) -> Any:
       return self._config_cache.get(key, default)

    7. def _parse_config_value(self, value: str, value_type: str) -> Any:
       a. int: int(value); float: float(value); bool: value.lower() in ('true','1','yes'); json: json.loads(value); else str(value)

    7b. async def _on_config_message_received(self, key: str, value: Any) -> None:
       """No-op hook called after each accepted Kafka config message is cached.

       Subclasses (e.g., ServiceAuditorAgent in Plan 05 Task 3) override this to react
       to specific config_key updates (e.g., reload lag-threshold map on alert.lag.* keys)
       WITHOUT having to override the whole _reload_config_loop. Exceptions raised by the
       hook are caught and logged by the loop; they do NOT break consumption.
       """
       return None
