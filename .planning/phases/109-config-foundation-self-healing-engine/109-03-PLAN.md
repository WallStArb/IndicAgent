    7. async def _load_config_snapshot(self) -> None:
       """
       Load config snapshot from DB with NON-FATAL failure mode.

       If DB unavailable, uses existing cache or service defaults.
       Emits CONFIG_STALE_TOTAL if load fails.
       Service continues running with last-known-good or defaults.
       """
       try:
           config_service = ConfigService(self.settings.database_url)
           await config_service.initialize()
           self._config_cache = await config_service.list(category=None)  # Only OPS stored
           self._config_loaded = True
           CONFIG_LAST_RELOAD_TIMESTAMP_SECONDS.set(time.time(), {"agent_id": self.name})
           structlog.get_logger().info("config.loaded", agent=self.name, keys=len(self._config_cache))
       except Exception as exc:
           CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": type(exc).__name__})
           structlog.get_logger().warning("config.load_failed", agent=self.name, error=str(exc), using="cached_or_defaults")
           # DO NOT re-raise - service continues with existing cache/defaults

    8. async def _setup_config_consumer(self) -> None:
       """
       Setup Kafka config consumer with NON-FATAL failure mode.

       If Kafka unavailable, config reload is disabled but service continues.
       Subsequent retries happen on next Kafka message if connection recovers.
       """
       if self._config_layer != "OPS":  # INFRA/STRUCT don't hot-reload
           return
       try:
           self._config_consumer = KafkaConsumerClient(
               bootstrap_servers=self.settings.kafka_brokers,
               group_id=f"{self.name}_config_consumer"
           )
           await self._config_consumer.start()
           await self._config_consumer.subscribe([topic_config_updates(self.env_name)])
           self._config_reload_task = asyncio.create_task(self._reload_config_loop())
           structlog.get_logger().info("config.consumer.started", agent=self.name, topic=topic_config_updates(self.env_name))
       except Exception as exc:
           CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": "kafka_subscribe_failed"})
           structlog.get_logger().warning("config.consumer.setup_failed", agent=self.name, error=str(exc))
           # DO NOT re-raise - service continues without hot-reload

    9. async def _reload_config_loop(self) -> None:
       """
       Kafka message loop for config hot-reload.

       NON-FATAL: Logs and continues on message processing errors.
       """
       try:
           async for _topic, _key, payload in self._config_consumer.messages():
               try:
                   config_key = payload.get("config_key")
                   config_value = payload.get("config_value")
                   version = payload.get("version")
                   if not config_key:
                       continue
                   parsed_value = self._parse_config_value(config_value, payload.get("value_type", "string"))
                   self._config_cache[config_key] = parsed_value
                   self._config_loaded = True
                   CONFIG_RELOAD_TOTAL.add(1, {"agent_id": self.name, "key": config_key})
                   CONFIG_LAST_RELOAD_TIMESTAMP_SECONDS.set(time.time(), {"agent_id": self.name})
                   structlog.get_logger().info("config.reloaded", agent=self.name, key=config_key, version=version)
               except Exception as exc:
                   CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": "parse_failed"})
                   structlog.get_logger().warning("config.parse_failed", agent=self.name, payload=payload, error=str(exc))
               if not self.running:
                   break
       except Exception as exc:
           CONFIG_STALE_TOTAL.add(1, {"agent_id": self.name, "reason": "consumer_loop_failed"})
           structlog.get_logger().error("config.consumer.loop_failed", agent=self.name, error=str(exc))

    10. async def _teardown_config_consumer(self) -> None:
       a. Cancel self._config_reload_task if exists
       b. Stop self._config_consumer if exists

    11. def get_config(self, key: str, default: Any = None) -> Any:
       """
       Get config value from cache.

       Returns cached value if loaded, else default.
       Services should provide sensible defaults for critical config.
       """
       return self._config_cache.get(key, default)

    12. def _parse_config_value(self, value: str, value_type: str) -> Any:
       a. If value_type='int': int(value)
       b. If value_type='float': float(value)
       c. If value_type='bool': value.lower() in ('true', '1', 'yes')
       d. If value_type='json': json.loads(value)
       e. Else: str(value)
