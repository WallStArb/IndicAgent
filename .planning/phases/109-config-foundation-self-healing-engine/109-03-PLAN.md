    1. ConfigConsumerMixin class with:

    2. _config_cache: dict[str, Any] = {} class attribute

    3. _config_layer: str = "OPS" class attribute (INFRA/STRUCT/OPS)

    4. _config_consumer: KafkaConsumerClient | None = None

    5. _config_reload_task: asyncio.Task | None = None

    6. async def _load_config_snapshot(self) -> None: create ConfigService with self.settings.database_url, call await config_service.initialize(), set self._config_cache = await config_service.list(category=self._config_layer)

    7. async def _setup_config_consumer(self) -> None:
       a. If self._config_layer != "OPS": return (INFRA/STRUCT don't hot-reload)
       b. Create KafkaConsumerClient for topic_config_updates(self.env_name) with group_id f"{self.name}_config_consumer"
       c. Call await self._consumer.start()
       d. Create asyncio task: self._config_reload_task = asyncio.create_task(self._reload_config_loop())

    8. async def _reload_config_loop(self) -> None:
       a. async for _topic, _key, payload in self._config_consumer.messages():
       b. Extract config_key, config_value, version from payload
       c. Update self._config_cache[config_key] = parsed_value
       d. Emit CONFIG_RELOAD_TOTAL.add(1, {"agent_id": self.name, "key": config_key})
       e. Log config.reloaded with key and version
       f. Break if not self.running

    9. async def _teardown_config_consumer(self) -> None:
       a. Cancel self._config_reload_task if exists
       b. Stop self._config_consumer if exists

    10. def get_config(self, key: str, default: Any = None) -> Any: return self._config_cache.get(key, default)

    11. _parse_config_value(self, value: str, value_type: str) -> Any: parse by type (int, float, bool, json, string)

    12. Import from src.config.config_service import ConfigService

    13. Import from src.core.kafka_utils import KafkaConsumerClient

    14. Import from src.core.stream_keys import topic_config_updates

    15. Import from src.observability.metrics import CONFIG_RELOAD_TOTAL

    Follow BaseAgent lifecycle pattern. Use same consumer pattern as alerting_agent.py. Parse JSON values via json.loads().
