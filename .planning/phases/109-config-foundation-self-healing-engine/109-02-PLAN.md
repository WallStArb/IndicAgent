    async def verify_auth(request: Request) -> None:
        """Verify Bearer token if CONFIG_API_TOKEN is set.

        If CONFIG_API_TOKEN is configured, requests without valid token are rejected with 401.
        If CONFIG_API_TOKEN is not set, auth is disabled (dev mode).
        """
        if CONFIG_API_TOKEN:
            auth_header = request.headers.get("Authorization", "")
            if not auth_header.startswith("Bearer "):
                CONFIG_AUTH_FAILED_TOTAL.add(1, {"reason": "missing_header"})
                raise HTTPException(status_code=401, detail="Missing Bearer token")
            token = auth_header[7:]  # Remove "Bearer " prefix
            if token != CONFIG_API_TOKEN:
                CONFIG_AUTH_FAILED_TOTAL.add(1, {"reason": "invalid_token"})
                raise HTTPException(status_code=401, detail="Invalid Bearer token")
