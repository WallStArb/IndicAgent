  <action>
    Create src/config/config_service.py with:

    1. ConfigService class with __init__(self, database_url: str, pool: asyncpg.Pool | None = None) that stores database_url and creates/receives pool

    2. async def initialize(self) -> None: creates pool if not exists via DatabaseManager.create_pool()

    3. In-memory cache: self._cache: dict[str, Any] = {}

    4. def _validate_key_domain(self, key: str) -> None:
       """
       Validate key belongs to OPS domain (hot-reloadable).

       Renaissance three-layer invariant:
       - INFRA (DB URLs, secrets): .env files ONLY, changes require restart
       - STRUCT (plugin tiers, DAG order): code/YAML ONLY, changes require deployment
       - OPS (feature flags, thresholds): DB + Kafka ONLY, hot-reloadable

       OPS keys start with: regime., swarm., alert., ai., feature., threshold.

       Raises:
           ConfigValidationError: If key doesn't match OPS prefix pattern
       """
       OPS_PREFIXES = ("regime.", "swarm.", "alert.", "ai.", "feature.", "threshold.")
       if not key.startswith(OPS_PREFIXES):
           raise ConfigValidationError(
               f"Key '{key}' is not a hot-reloadable OPS config. "
               f"INFRA keys (DATABASE_URL, KAFKA_BROKERS) must be set via .env. "
               f"STRUCT keys (plugin tiers, DAG order) must be changed via code deployment."
           )

    5. async def get(self, key: str, default: Any = None) -> Any: checks cache first, if miss SELECT FROM config_state WHERE config_key=$1, updates cache, returns value or default

    6. async def get_at(self, key: str, t: datetime) -> Any: time-travel query SELECT config_value FROM config_history WHERE config_key=$1 AND timestamp < $2 ORDER BY timestamp DESC LIMIT 1

    7. async def list(self, category: str | None = None) -> dict[str, Any]: SELECT * FROM config_state, returns {config_key: parsed_value}. Note: category param ignored for now - only OPS stored

    8. async def set(self, key: str, value: Any, changed_by: str = "system", expected_version: int | None = None, reason: str | None = None) -> ConfigChange:
       a. Call self._validate_key_domain(key) FIRST - raise ConfigValidationError if not OPS key
       b. Fetch schema from config_schema WHERE config_key=$1, raise ConfigValidationError if not found
       c. Validate value via validate_value(), raise ConfigValidationError if invalid
       d. Fetch current state for version check, if expected_version and current.version != expected_version, raise ConfigVersionConflict
       e. Calculate new_version = current.version + 1 or 1 if new key
       f. Start transaction: INSERT config_history, UPSERT config_state ON CONFLICT, INSERT config_outbox status='pending'
       g. Commit transaction
       h. Update cache
       i. Return ConfigChange object

    9. async def revert_key(self, key: str, version: int, changed_by: str = "system", reason: str | None = None) -> dict: fetch config_value from config_history WHERE config_key=$1 AND version=$2, call set() with that value

    10. Parse value by type: if value_type='int', int(value); 'float', float(value); 'bool', value.lower() in ('true', '1', 'yes'); 'json', json.loads(value); else str(value)

    11. Metrics: import config metrics (add later in separate task), emit CONFIG_SET_TOTAL.add(1, {"key": key, "changed_by": changed_by, "outcome": "success"})

    12. For is_secret keys, log "**REDACTED**" instead of actual value in any logging

    Follow exact transaction pattern: async with conn.transaction(): await conn.execute(...), await conn.execute(...), await conn.execute(...). Use FOR UPDATE on config_state read for version check.
