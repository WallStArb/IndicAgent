Note: create_pool is NOT a classmethod on DatabaseManager - it is a module-level function in src/core/database_manager.py.

    2. In-memory cache: self._cache: dict[str, Any] = {}

    3. OPS_PREFIXES tuple as a class-level constant:
       OPS_PREFIXES: ClassVar[tuple[str, ...]] = ("regime.", "swarm.", "alert.", "ai.", "feature.", "threshold.", "roll.", "cross_asset.", "macro.")

    4. def _validate_key_domain(self, key: str) -> None: raise ConfigValidationError if key does not start with any OPS_PREFIXES entry. Message includes guidance: "INFRA keys (DATABASE_URL, KAFKA_BROKERS) must be set via .env. STRUCT keys (plugin tiers, DAG order) must be changed via code deployment."

    5. async def get(self, key: str, default: Any = None) -> Any: check cache, then SELECT FROM config_state, parse by schema value_type, populate cache, return.

    6. async def get_at(self, key: str, t: datetime) -> Any: SELECT config_value FROM config_history WHERE config_key=$1 AND timestamp < $2 ORDER BY timestamp DESC LIMIT 1.

    7. async def list(self) -> dict[str, Any]: SELECT * FROM config_state; return parsed dict. (No category param - OPS only.)

    8. async def set(self, key: str, value: Any, changed_by: str = "system", expected_version: int | None = None, reason: str | None = None) -> ConfigChange:
       a. self._validate_key_domain(key)  # raises ConfigValidationError
       b. Fetch schema row WHERE config_key=$1; raise ConfigValidationError("unknown key") if missing
       c. Validate via validate_value(); raise ConfigValidationError(result.reason) if invalid
       d. async with conn.transaction():
            e. row = SELECT * FROM config_state WHERE config_key=$1 FOR UPDATE  (INSIDE transaction)
            f. if row and expected_version is not None and row.version != expected_version: raise ConfigVersionConflict(expected_version, row.version)
            g. new_version = (row.version + 1) if row else 1
            h. INSERT INTO config_history (timestamp, config_key, version, config_value, changed_by, reason) VALUES (NOW(), $1, $2, $3, $4, $5)
            i. INSERT INTO config_state (config_key, config_value, version) VALUES ($1, $2, $3) ON CONFLICT (config_key) DO UPDATE SET config_value=EXCLUDED.config_value, version=EXCLUDED.version, updated_at=NOW()
            j. INSERT INTO config_outbox (config_key, config_value, version, status, retry_count, next_attempt_at) VALUES ($1, $2, $3, 'pending', 0, NOW())
       k. Commit (exit transaction context)
       l. Update self._cache[key] = parsed_value
       m. Try emit CONFIG_SET_TOTAL.add(1, {"key": key, "changed_by": changed_by, "outcome": "success"}) inside try/except ImportError (metric defined in 109-02; first run of 109-01 stand-alone tests should not fail if metric missing).
       n. Logging: if schema.is_secret, log redacted_value="**REDACTED**" instead of actual value in any structlog event; otherwise log value.
       o. Return ConfigChange(key=key, value=parsed_value if not schema.is_secret else "**REDACTED**", version=new_version, changed_at=NOW(), changed_by=changed_by, reason=reason)

    9. async def revert_key(self, key: str, version: int, changed_by: str, reason: str | None) -> ConfigChange:
       a. row = SELECT config_value FROM config_history WHERE config_key=$1 AND version=$2 LIMIT 1
       b. if not row: raise ConfigValidationError(f"No history for {key}@{version}")
       c. return await self.set(key, row["config_value"], changed_by=changed_by, reason=f"revert to v{version}: {reason}")

    10. _parse_value(value: str, value_type: str) -> Any: int/float/bool/json/str dispatcher.

    11. Imports: asyncpg, json, structlog; from datetime import UTC, datetime; from typing import Any, ClassVar; from src.core.database_manager import create_pool; from src.config.config_schema import (ConfigChange, ConfigSchemaEntry, ConfigValidationError, ConfigVersionConflict, validate_value).
