    1. #!/usr/bin/env python3
    """SelfHealingAgent — receives Alertmanager webhooks, executes remediation strategies.

    Control-theory-based self-healing:
    - Sensor: measure pre-state from Prometheus
    - Setpoint: compare to threshold
    - Actuator: execute remediation action
    - Feedback: measure post-state, record outcome

    Conservative Phase 1 strategies: disk cleanup, consumer restart, pool flush.

    Webhook authentication requires CONFIG_WEBHOOK_SHARED_SECRET (optional in dev, required in prod).
    """

    2. from fastapi import FastAPI, HTTPException, Request
    from fastapi.responses import JSONResponse

    3. app = FastAPI(title="IndicAgent Self-Healing Engine", version="1.0.0")

    4. self_healing_engine: SelfHealingEngine | None = None

    5. @app.on_startup("startup")
    async def startup():
       from src.config.settings import get_settings
       from src.core.database_manager import create_pool
       settings = get_settings()
       pool = await create_pool(settings.database_url, pool_name="self_healing")
       global self_healing_engine
       self_healing_engine = SelfHealingEngine(pool)
       await self_healing_engine.initialize()

    6. @app.post("/webhook/alertmanager")
    async def webhook_alertmanager(request: Request):
       """
       Alertmanager webhook endpoint for self-healing triggers.

       Requires CONFIG_WEBHOOK_SHARED_SECRET header for authentication.
       Returns 401 if secret missing or invalid.
       """
       if not self_healing_engine:
           raise HTTPException(status_code=503, detail="Engine not initialized")

       payload = await request.json()

       # Verify webhook secret - engine's handle_webhook checks this too, but validate early
       webhook_secret = payload.get("webhook_secret") or payload.get("shared_secret")
       expected_secret = self_healing_engine._webhook_secret
       if expected_secret and webhook_secret != expected_secret:
           WEBHOOK_AUTH_FAILED_TOTAL.add(1, {"reason": "invalid_secret"})
           raise HTTPException(status_code=401, detail="Invalid webhook shared secret")

       WEBHOOK_RECEIVED_TOTAL.add(1, {"severity": payload.get("severity", "unknown")})
       try:
           result = await self_healing_engine.handle_webhook(payload)
           return JSONResponse(content={"remediation_id": result.remediation_id, "status": result.status}, status_code=200)
       except HTTPException:
           raise
       except Exception as exc:
           WEBHOOK_VALIDATION_FAILED_TOTAL.add(1, {"reason": type(exc).__name__})
           raise HTTPException(status_code=400, detail=str(exc))

    7. if __name__ == "__main__":
       import uvicorn
       uvicorn.run(app, host="0.0.0.0", port=9002)
