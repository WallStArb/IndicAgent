# LLMChain already tries providers sequentially:
```python
class LLMChain:
    async def generate(
        self, prompt: str, system: str, max_tokens: int, timeout: float
    ) -> str | None:
        for provider in self.providers:
            result = await provider.generate(prompt, system, max_tokens, timeout)
            if result is not None:
                self.last_provider_id = provider.provider_id
                return result
        self.last_provider_id = None
        return None
```

       ```python
       async def _call_llm_with_circuit_breaker(
           provider_id: str,
           call_coro,
       ) -> str | None:
           """Call LLM provider with circuit breaker tracking.

           Args:
               provider_id: Unique provider identifier (e.g., "zai:default")
               call_coro: Coroutine that performs the LLM call (already wrapped in to_thread)

           Returns:
               LLM response string or None on failure

           Note: call_coro is already an async coroutine (from asyncio.to_thread),
           so we pass it directly to retry_with_backoff which awaits it.
           """
           from datetime import datetime
           from src.core.plugin_circuit_breaker import CircuitState

           plugin_state = _llm_circuit_breaker.plugin_states[provider_id]
           previous_state = plugin_state.state

           try:
               # Call the coroutine with retry logic
               result = await retry_with_backoff(
                   call_coro,
                   max_attempts=3,
                   base_delay=1.0,
                   max_delay=10.0,
               )

               # Record success in circuit breaker
               plugin_state.success_count += 1
               plugin_state.last_success_time = datetime.now()
               plugin_state.total_calls += 1

               return result

           except Exception as exc:
               # Record failure in circuit breaker
               plugin_state.failure_count += 1
               plugin_state.last_failure_time = datetime.now()
               plugin_state.total_calls += 1

               logger.warning(
                   "LLM provider call failed",
                   provider=provider_id,
                   error=str(exc),
               )
               return None
       ```

    Note: retry_with_backoff does NOT support a retry_tracker callback.
    Retry attempts are tracked implicitly via circuit breaker failure_count.
    The key insight: to_thread returns a coroutine, which is what retry_with_backoff expects.
