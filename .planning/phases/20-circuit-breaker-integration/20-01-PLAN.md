       ```python
       import asyncio
       import random
       from collections.abc import Callable
       from typing import Any

       def exponential_backoff_with_jitter(
           attempt: int,
           base_delay: float = 1.0,
           max_delay: float = 30.0,
           jitter_factor: float = 0.5,
       ) -> float:
           """Calculate exponential backoff delay with jitter.

           Args:
               attempt: Current retry attempt (0-indexed)
               base_delay: Base delay in seconds (default: 1.0s)
               max_delay: Maximum delay cap (default: 30.0s)
               jitter_factor: Jitter as percentage of delay (default: 0.5 = 50%)

           Returns:
               Delay in seconds with jitter applied

           Jitter prevents thundering herd problem when multiple processes
           retry simultaneously after external API failure.
           """
           # Exponential backoff: base_delay * (2 ^ attempt)
           delay = base_delay * (2 ** attempt)

           # Cap at max_delay
           delay = min(delay, max_delay)

           # Add jitter: +/- jitter_factor * delay
           jitter = delay * jitter_factor * (random.random() * 2 - 1)
           return max(0.0, delay + jitter)
       ```

    3. Implement async retry_with_backoff wrapper:
       ```python
       async def retry_with_backoff(
           coro_fn: Callable,
           *args,
           max_attempts: int = 3,
           base_delay: float = 1.0,
           max_delay: float = 30.0,
           jitter_factor: float = 0.5,
           retry_on: tuple[type[Exception], ...] = (Exception,),
           **kwargs,
       ) -> Any:
           """Async retry wrapper with exponential backoff and jitter.

           Args:
               coro_fn: Async function to retry
               *args: Positional args for coro_fn
               max_attempts: Maximum retry attempts (default: 3)
               base_delay: Base delay in seconds (default: 1.0s)
               max_delay: Maximum delay cap (default: 30.0s)
               jitter_factor: Jitter as percentage (default: 0.5)
               retry_on: Exception types to catch for retry (default: all Exception)
               **kwargs: Keyword args for coro_fn

           Returns:
               Result from successful call

           Raises:
               Last exception after max_attempts exhausted

           Jitter prevents thundering herd: concurrent retries spread across
           time window instead of hitting external API simultaneously.
           """
           last_exception = None

           for attempt in range(max_attempts):
               try:
                   return await coro_fn(*args, **kwargs)
               except retry_on as e:
                   last_exception = e

                   if attempt < max_attempts - 1:
                       delay = exponential_backoff_with_jitter(
                           attempt, base_delay, max_delay, jitter_factor
                       )
                       await asyncio.sleep(delay)
                   else:
                       raise

           # Should never reach here, but handle type checker
           if last_exception:
               raise last_exception
           raise RuntimeError("retry_with_backoff failed without exception")
       ```

