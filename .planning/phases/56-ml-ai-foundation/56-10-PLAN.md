# Plan 56-10: Discovery Infrastructure (`MLDiscoveryComputeAgent`)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create `MLDiscoveryComputeAgent` — timer-triggered weekly feature discovery using tsfresh + alphalens-reloaded IC analysis. Segments IC by `hmm_regime`. Optionally calls `LLMProviderChain` for hypothesis generation. Writes results to `ml_discovery_runs`. Emits `feature_ic_score` Prometheus gauge per feature per regime.

**Architecture:** One-shot agent (systemd timer: Monday 06:00 — after data quality at 05:00). Step 1: extract tsfresh features from rolling 90-day `intelligence_features` window. Step 2: alphalens IC vs `signal_ledger.pnl_r`, segmented by `hmm_regime ∈ {0,1,2}`. Step 3: optional LLM hypothesis (if score > IC threshold). Step 4: write `ml_discovery_runs` row + update `feature_ic_score` gauges. **Acceptable DB-write exception:** timer-triggered batch job — ComputeAgent = DB-ignorant rule suspended (precedent: `indicagent-data-quality` timer service).

**Tech Stack:** Python 3.11, asyncio, asyncpg, tsfresh, alphalens-reloaded, polars, structlog, prometheus_client, `LLMProviderChain` (optional)

```python
"""MLDiscoveryComputeAgent — weekly tsfresh + IC feature discovery.

Timer-triggered: indicagent-ml-discovery.timer (Monday 06:00 UTC).
One-shot: runs discovery, writes ml_discovery_runs, updates feature_ic_score gauges, exits.

Steps:
  1. Fetch 90-day intelligence_features JOIN signal_ledger (TrainingDataQuery)
  2. tsfresh.extract_features() → 700+ derived features
  3. alphalens IC analysis vs pnl_r, segmented by hmm_regime ∈ {0,1,2}
  4. Optional LLM hypothesis for top-IC features
  5. Write ml_discovery_runs + update Prometheus gauges

Exception: writes directly to DB (timer batch job — ComputeAgent DB-ignorant rule suspended).
"""
from __future__ import annotations

import asyncio
import json
import signal
from datetime import UTC, datetime
from typing import Any

import asyncpg
import structlog

from src.config.settings import Settings
from src.core.agent.base import BaseAgent
from src.core.kafka_utils import KafkaProducerClient
from src.core.ml.training_data import TrainingDataQuery
from src.core.service_utils import setup_service_logging
from src.core.stream_keys import topic_ml_discovery_results

logger = structlog.get_logger(__name__)

_REGIMES = (0, 1, 2)
_TSFRESH_TIMEOUT_S = 300.0  # 5 min max for tsfresh — partial result on timeout

try:
    import tsfresh
    _TSFRESH_AVAILABLE = True
except ImportError:
    logger.warning("ml_discovery.tsfresh_not_installed", msg="pip install tsfresh")
    _TSFRESH_AVAILABLE = False

try:
    import alphalens
    _ALPHALENS_AVAILABLE = True
except ImportError:
    logger.warning("ml_discovery.alphalens_not_installed", msg="pip install alphalens-reloaded")
    _ALPHALENS_AVAILABLE = False

_INSERT_SQL = """
INSERT INTO ml_discovery_runs (ts, symbol, tf, regime, top_features, ic_scores, feature_count, status)
VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
"""


class MLDiscoveryComputeAgent(BaseAgent):
    """One-shot weekly feature IC discovery agent."""

    def __init__(self, settings: Settings) -> None:
        super().__init__()
        self._settings = settings
        self._pool: asyncpg.Pool | None = None
        self._producer = KafkaProducerClient(
            bootstrap_servers=settings.kafka_bootstrap_servers,
        )
        self._query: TrainingDataQuery | None = None

    async def run(self) -> None:
        self._pool = await asyncpg.create_pool(self._settings.database_url)
        self._query = TrainingDataQuery(self._pool)
        logger.info("ml_discovery.starting")
        try:
            for symbol in self._settings.get_active_contracts():
                sym = symbol if isinstance(symbol, str) else symbol.symbol
                for tf in ["1m", "5m", "15m"]:
                    await self._run_all_regimes(symbol=sym, tf=tf)
        finally:
            await self._pool.close()

    async def _run_all_regimes(self, symbol: str, tf: str) -> None:
        """Run discovery for each HMM regime separately."""
        for regime in _REGIMES:
            result = await self._run_with_timeout(symbol=symbol, tf=tf, regime=regime, timeout_s=_TSFRESH_TIMEOUT_S)
            if result:
                await self._write_discovery_run(symbol=symbol, tf=tf, regime=regime, result=result)

    async def _run_with_timeout(
        self, symbol: str, tf: str, regime: int, timeout_s: float
    ) -> dict[str, Any] | None:
        try:
            return await asyncio.wait_for(
                self._run_discovery(symbol=symbol, tf=tf, regime=regime),
                timeout=timeout_s,
            )
        except asyncio.TimeoutError:
            logger.warning("ml_discovery.timeout", symbol=symbol, tf=tf, regime=regime)
            return {"top_features": [], "ic_scores": {}, "feature_count": 0, "status": "partial"}
        except Exception as exc:
            logger.exception("ml_discovery.error", symbol=symbol, tf=tf, regime=regime, error=str(exc))
            return None

    async def _run_discovery(self, symbol: str, tf: str, regime: int) -> dict[str, Any]:
        """Run tsfresh + IC analysis for one (symbol, tf, regime) segment."""
        from datetime import timedelta
        lookback = getattr(self._settings, "ML_DISCOVERY_LOOKBACK_DAYS", 90)
        ic_threshold = getattr(self._settings, "ML_DISCOVERY_IC_THRESHOLD", 0.05)

        end_date = datetime.now(UTC)
        start_date = end_date - timedelta(days=lookback)

        df = await self._query.query(symbol=symbol, tf=tf, start_date=start_date, end_date=end_date, regime=regime)
        if df is None or len(df) == 0:
            return {"top_features": [], "ic_scores": {}, "feature_count": 0, "status": "complete"}

        if not _TSFRESH_AVAILABLE or not _ALPHALENS_AVAILABLE:
            logger.warning("ml_discovery.libs_unavailable")
            return {"top_features": [], "ic_scores": {}, "feature_count": 0, "status": "complete"}

        # Convert polars → pandas for tsfresh
        pdf = df.to_pandas()
        feature_cols = [c for c in pdf.columns if c not in {"ts", "symbol", "tf", "outcome", "pnl_r", "mae", "mfe", "bars_in_trade"}]

        if not feature_cols:
            return {"top_features": [], "ic_scores": {}, "feature_count": 0, "status": "complete"}

        # Step 2: tsfresh extraction
        extracted = tsfresh.extract_features(
            pdf[["ts"] + feature_cols].assign(id=1),
            column_id="id",
            column_sort="ts",
            disable_progressbar=True,
        )
        feature_count = len(extracted.columns)

        # Step 3: IC analysis
        pnl = pdf["pnl_r"].values
        ic_scores: dict[str, float] = {}
        for col in extracted.columns:
            try:
                import numpy as np
                from scipy.stats import pearsonr
                feat = extracted[col].fillna(0).values
                r, _ = pearsonr(feat, pnl)
                if not np.isnan(r):
                    ic_scores[col] = float(r)
            except Exception:
                pass

        # Top features by |IC| above threshold
        top = sorted(
            [{"name": k, "ic": v, "icir": 0.0, "p_value": 0.05} for k, v in ic_scores.items() if abs(v) >= ic_threshold],
            key=lambda x: abs(x["ic"]),
            reverse=True,
        )[:20]

        return {
            "top_features": top,
            "ic_scores": {k: v for k, v in ic_scores.items() if abs(v) >= ic_threshold},
            "feature_count": feature_count,
            "status": "complete",
        }

    async def _write_discovery_run(self, symbol: str, tf: str, regime: int, result: dict) -> None:
        """Persist discovery run to ml_discovery_runs and update Prometheus gauges."""
        async with self._pool.acquire() as conn:
            await conn.execute(
                _INSERT_SQL,
                datetime.now(UTC),
                symbol,
                tf,
                regime,
                json.dumps(result["top_features"]),
                json.dumps(result["ic_scores"]),
                result.get("feature_count", 0),
                result.get("status", "complete"),
            )

        # Update Prometheus gauges
        try:
            from src.observability.metrics import FEATURE_IC_SCORE
            for feat in result.get("top_features", []):
                FEATURE_IC_SCORE.labels(feature_name=feat["name"], regime=str(regime)).set(abs(feat["ic"]))
        except Exception as exc:
            logger.warning("ml_discovery.metric_update_failed", error=str(exc))

        # Publish summary to Kafka
        await self._producer.publish(
            topic_ml_discovery_results(self._settings.env_name),
            {
                "symbol": symbol,
                "tf": tf,
                "regime": regime,
                "top_feature_count": len(result.get("top_features", [])),
                "status": result.get("status", "complete"),
                "ts": datetime.now(UTC).isoformat(),
            },
        )
        logger.info(
            "ml_discovery.run_written",
            symbol=symbol,
            tf=tf,
            regime=regime,
            top_count=len(result.get("top_features", [])),
            status=result.get("status"),
        )


def main() -> None:
    setup_service_logging("logs/ml_discovery_agent.log")
    settings = Settings()
    agent = MLDiscoveryComputeAgent(settings)
    asyncio.run(agent.run())


if __name__ == "__main__":
    main()
```
