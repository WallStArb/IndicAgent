```python
def _build_all_ranked(
    fired: list[dict],
    perf_weights: dict[str, float] | None = None,
) -> list[dict]:
    """Build ranked list. perf_weights maps setup_plugin → perf_multiplier [0.5, 1.5].
    composite_rank is assigned 1-based for every fired plugin by SETUP_PRIORITY sort
    (unknown plugins receive SETUP_PRIORITY.get() = 0, so they sort after known ones,
    but still receive a valid 1-based composite_rank from enumerate).
    adjusted_rank = composite_rank * perf_multiplier (higher = higher priority).
    Setups absent from perf_weights get multiplier=1.0 (neutral).
    Sort by adjusted_rank DESCENDING: highest adjusted_rank wins.
    """
    # 1. Sort by SETUP_PRIORITY descending to assign initial composite_rank
    priority_sorted = sorted(fired, key=lambda s: SETUP_PRIORITY.get(s.get("setup_plugin", ""), 0), reverse=True)
    # 2. Assign composite_rank 1-based for ALL plugins (enumerate gives 0-based → +1)
    with_ranks = [{**sig, "composite_rank": i + 1} for i, sig in enumerate(priority_sorted)]
    # 3. Apply perf_multiplier to get adjusted_rank
    weights = perf_weights or {}
    for sig in with_ranks:
        multiplier = weights.get(sig.get("setup_plugin", ""), 1.0)
        sig["adjusted_rank"] = round(sig["composite_rank"] * multiplier, 4)
    # 4. Re-sort by adjusted_rank DESCENDING (higher adjusted_rank = higher priority)
    return sorted(with_ranks, key=lambda s: s["adjusted_rank"], reverse=True)

def aggregate(
    signals: list[dict],
    *,
    trend_regime: float = 0.0,
    features: dict[str, Any] | None = None,

) -> AggregatedResult:
```

```python
async def _perf_weights_refresh_loop(self) -> None:
    _REFRESH_INTERVAL = 3600  # 60 minutes
    while self.running and not self.shutdown_requested:
        try:
            try:
                await asyncio.wait_for(self.shutdown_event.wait(), timeout=_REFRESH_INTERVAL)
                break
            except TimeoutError:
                pass
            if self.shutdown_requested:
                break
            await self._load_perf_weights()
        except asyncio.CancelledError:
            break
        except Exception as e:
            self.logger.warning("Perf weights refresh error", error=str(e))
            await asyncio.sleep(30)
```
